 187   /** 
 188    * Creates a scope with all types declared. Declares newly discovered types 
 189    * and type properties in the type registry. 
 190    */ 
 191   @Override 
 192   public Scope createScope(Node root, Scope parent) { 
 193     // Constructing the global scope is very different than constructing 
 194     // inner scopes, because only global scopes can contain named classes that 
 195     // show up in the type registry. 
 196     Scope newScope = null; 
 197     AbstractScopeBuilder scopeBuilder = null; 
 198     if (parent == null) { 
 199       // Run a first-order analysis over the syntax tree. 
 200       (new FirstOrderFunctionAnalyzer(compiler, functionAnalysisResults)) 
 201           .process(root.getFirstChild(), root.getLastChild()); 
 202  
 203       // Find all the classes in the global scope. 
 204       newScope = createInitialScope(root); 
 205  
 206       GlobalScopeBuilder globalScopeBuilder = new GlobalScopeBuilder(newScope); 
 207       scopeBuilder = globalScopeBuilder; 
 208       NodeTraversal.traverse(compiler, root, scopeBuilder); 
 209     } else { 
 210       newScope = new Scope(parent, root); 
 211       LocalScopeBuilder localScopeBuilder = new LocalScopeBuilder(newScope); 
 212       scopeBuilder = localScopeBuilder; 
 213       localScopeBuilder.build(); 
 214     } 
 215  
 216     scopeBuilder.resolveStubDeclarations(); 
 217     scopeBuilder.resolveTypes(); 
 218  
 219     // Gather the properties in each function that we found in the 
 220     // global scope, if that function has a @this type that we can 
 221     // build properties on. 
 222     for (Node functionNode : scopeBuilder.nonExternFunctions) { 
 223       JSType type = functionNode.getJSType(); 
 224       if (type != null && type.isFunctionType()) { 
 225         FunctionType fnType = type.toMaybeFunctionType(); 
 226         ObjectType fnThisType = fnType.getTypeOfThis(); 
 227         if (!fnThisType.isUnknownType()) { 
 228           NodeTraversal.traverse(compiler, functionNode.getLastChild(), 
 229               scopeBuilder.new CollectProperties(fnThisType)); 
 230         } 
 231       } 
 232     } 
 233  
 234     if (parent == null) { 
 235       codingConvention.defineDelegateProxyPrototypeProperties( 
 236           typeRegistry, newScope, delegateProxyPrototypes, 
 237           delegateCallingConventions); 
 238     } 
 239     return newScope; 
 240   }

 242   /** 
 243    * Patches a given global scope by removing variables previously declared in 
 244    * a script and re-traversing a new version of that script. 
 245    * 
 246    * @param globalScope The global scope generated by {@code createScope}. 
 247    * @param scriptRoot The script that is modified. 
 248    */ 
 249   void patchGlobalScope(Scope globalScope, Node scriptRoot) { 
 250     // Preconditions: This is supposed to be called only on (named) SCRIPT nodes 
 251     // and a global typed scope should have been generated already. 
 252     Preconditions.checkState(scriptRoot.isScript()); 
 253     Preconditions.checkNotNull(globalScope); 
 254     Preconditions.checkState(globalScope.isGlobal()); 
 255  
 256     String scriptName = NodeUtil.getSourceName(scriptRoot); 
 257     Preconditions.checkNotNull(scriptName); 
 258     for (Node node : ImmutableList.copyOf(functionAnalysisResults.keySet())) { 
 259       if (scriptName.equals(NodeUtil.getSourceName(node))) { 
 260         functionAnalysisResults.remove(node); 
 261       } 
 262     } 
 263  
 264     (new FirstOrderFunctionAnalyzer( 
 265         compiler, functionAnalysisResults)).process(null, scriptRoot); 
 266  
 267     // TODO(bashir): Variable declaration is not the only side effect of last 
 268     // global scope generation but here we only wipe that part off! 
 269  
 270     // Remove all variables that were previously declared in this scripts. 
 271     // First find all vars to remove then remove them because of iterator! 
 272     Iterator<Var> varIter = globalScope.getVars(); 
 273     List<Var> varsToRemove = Lists.newArrayList(); 
 274     while (varIter.hasNext()) { 
 275       Var oldVar = varIter.next(); 
 276       if (scriptName.equals(oldVar.getInputName())) { 
 277         varsToRemove.add(oldVar); 
 278       } 
 279     } 
 280     for (Var var : varsToRemove) { 
 281       globalScope.undeclare(var); 
 282       globalScope.getTypeOfThis().removeProperty(var.getName()); 
 283     } 
 284  
 285     // Now re-traverse the given script. 
 286     GlobalScopeBuilder scopeBuilder = new GlobalScopeBuilder(globalScope); 
 287     NodeTraversal.traverse(compiler, scriptRoot, scopeBuilder); 
 288   }

 350     @Override 
 351     public void visit(NodeTraversal t, Node node, Node parent) { 
 352       Node nameNode = null; 
 353       switch (node.getType()) { 
 354         case Token.VAR: 
 355           for (Node child = node.getFirstChild(); 
 356                child != null; child = child.getNext()) { 
 357             identifyNameNode( 
 358                 child, child.getFirstChild(), 
 359                 NodeUtil.getBestJSDocInfo(child)); 
 360           } 
 361           break; 
 362         case Token.EXPR_RESULT: 
 363           Node firstChild = node.getFirstChild(); 
 364           if (firstChild.isAssign()) { 
 365             identifyNameNode( 
 366                 firstChild.getFirstChild(), firstChild.getLastChild(), 
 367                 firstChild.getJSDocInfo()); 
 368           } else { 
 369             identifyNameNode( 
 370                 firstChild, null, firstChild.getJSDocInfo()); 
 371           } 
 372           break; 
 373       } 
 374     }

 377         Node nameNode, Node valueNode, JSDocInfo info) {

 382           } else if (info.hasTypedefType()) { 
 383             registry.identifyNonNullableName(nameNode.getQualifiedName()); 
 384           } 
 385         }

 395       implements NodeTraversal.Callback { 
 396

 398      * The scope that we're building. 
 399      */ 
 400     final Scope scope; 
 401  
 402     private final List<DeferredSetType> deferredSetTypes =

 404  
 405     /** 
 406      * Functions that we found in the global scope and not in externs. 
 407      */ 
 408     private final List<Node> nonExternFunctions = Lists.newArrayList(); 
 409  
 410     /** 
 411      * Object literals with a @lends annotation aren't analyzed until we 
 412      * reach the root of the statement they're defined in. 
 413      * 
 414      * This ensures that if there are any @lends annotations on the object 
 415      * literals, the type on the @lends annotation resolves correctly. 
 416      * 
 417      * For more information, see 
 418      * http://code.google.com/p/closure-compiler/issues/detail?id=314 
 419      */

 426      * stubs, then we should declare UNKNOWN types.

 429         Lists.newArrayList();

 432      * The current source file that we're in. 
 433      */

 445     void setDeferredType(Node node, JSType type) { 
 446       deferredSetTypes.add(new DeferredSetType(node, type));

 449     void resolveTypes() { 
 450       // Resolve types and attach them to nodes. 
 451       for (DeferredSetType deferred : deferredSetTypes) { 
 452         deferred.resolve(scope); 
 453       } 
 454  
 455       // Resolve types and attach them to scope slots. 
 456       Iterator<Var> vars = scope.getVars(); 
 457       while (vars.hasNext()) { 
 458         vars.next().resolveType(typeParsingErrorReporter); 
 459       } 
 460  
 461       // Tell the type registry that any remaining types 
 462       // are unknown. 
 463       typeRegistry.resolveTypesInScope(scope); 
 464     }

 466     @Override 
 467     public final boolean shouldTraverse(NodeTraversal t, Node n, 
 468         Node parent) { 
 469       inputId = t.getInputId(); 
 470       if (n.isFunction() || 
 471           n.isScript()) { 
 472         Preconditions.checkNotNull(inputId); 
 473         sourceName = NodeUtil.getSourceName(n); 
 474       } 
 475  
 476       // We do want to traverse the name of a named function, but we don't 
 477       // want to traverse the arguments or body. 
 478       boolean descend = parent == null || !parent.isFunction() || 
 479           n == parent.getFirstChild() || parent == scope.getRootNode(); 
 480  
 481       if (descend) { 
 482         // Handle hoisted functions on pre-order traversal, so that they 
 483         // get hit before other things in the scope. 
 484         if (NodeUtil.isStatementParent(n)) { 
 485           for (Node child = n.getFirstChild(); 
 486                child != null; 
 487                child = child.getNext()) { 
 488             if (NodeUtil.isHoistedFunctionDeclaration(child)) { 
 489               defineFunctionLiteral(child, n); 
 490             } 
 491           } 
 492         } 
 493       } 
 494  
 495       return descend; 
 496     }

 498     @Override 
 499     public void visit(NodeTraversal t, Node n, Node parent) { 
 500       inputId = t.getInputId(); 
 501       attachLiteralTypes(t, n); 
 502  
 503       switch (n.getType()) { 
 504         case Token.CALL: 
 505           checkForClassDefiningCalls(t, n, parent); 
 506           checkForCallingConventionDefiningCalls(n, delegateCallingConventions); 
 507           break; 
 508  
 509         case Token.FUNCTION: 
 510           if (t.getInput() == null || !t.getInput().isExtern()) { 
 511             nonExternFunctions.add(n); 
 512           } 
 513  
 514           // Hoisted functions are handled during pre-traversal. 
 515           if (!NodeUtil.isHoistedFunctionDeclaration(n)) { 
 516             defineFunctionLiteral(n, parent); 
 517           } 
 518           break; 
 519  
 520         case Token.ASSIGN: 
 521           // Handle initialization of properties. 
 522           Node firstChild = n.getFirstChild(); 
 523           if (firstChild.isGetProp() && 
 524               firstChild.isQualifiedName()) { 
 525             maybeDeclareQualifiedName(t, n.getJSDocInfo(), 
 526                 firstChild, n, firstChild.getNext()); 
 527           } 
 528           break; 
 529  
 530         case Token.CATCH: 
 531           defineCatch(n, parent); 
 532           break; 
 533  
 534         case Token.VAR: 
 535           defineVar(n, parent); 
 536           break; 
 537  
 538         case Token.GETPROP: 
 539           // Handle stubbed properties. 
 540           if (parent.isExprResult() && 
 541               n.isQualifiedName()) { 
 542             maybeDeclareQualifiedName(t, n.getJSDocInfo(), n, parent, null); 
 543           } 
 544           break; 
 545       } 
 546  
 547       // Analyze any @lends object literals in this statement. 
 548       if (n.getParent() != null && NodeUtil.isStatement(n) && 
 549           lentObjectLiterals != null) { 
 550         for (Node objLit : lentObjectLiterals) { 
 551           defineObjectLiteral(objLit); 
 552         } 
 553         lentObjectLiterals.clear(); 
 554       } 
 555     }

 557     private void attachLiteralTypes(NodeTraversal t, Node n) { 
 558       switch (n.getType()) { 
 559         case Token.NULL: 
 560           n.setJSType(getNativeType(NULL_TYPE)); 
 561           break; 
 562  
 563         case Token.VOID: 
 564           n.setJSType(getNativeType(VOID_TYPE)); 
 565           break; 
 566  
 567         case Token.STRING: 
 568           n.setJSType(getNativeType(STRING_TYPE)); 
 569           break; 
 570  
 571         case Token.NUMBER: 
 572           n.setJSType(getNativeType(NUMBER_TYPE)); 
 573           break; 
 574  
 575         case Token.TRUE: 
 576         case Token.FALSE: 
 577           n.setJSType(getNativeType(BOOLEAN_TYPE)); 
 578           break; 
 579  
 580         case Token.REGEXP: 
 581           n.setJSType(getNativeType(REGEXP_TYPE)); 
 582           break; 
 583  
 584         case Token.OBJECTLIT: 
 585           JSDocInfo info = n.getJSDocInfo(); 
 586           if (info != null && 
 587               info.getLendsName() != null) { 
 588             if (lentObjectLiterals == null) { 
 589               lentObjectLiterals = Lists.newArrayList(); 
 590             } 
 591             lentObjectLiterals.add(n); 
 592           } else { 
 593             defineObjectLiteral(n); 
 594           } 
 595           break; 
 596  
 597           // NOTE(nicksantos): If we ever support Array tuples, 
 598           // we will need to put ARRAYLIT here as well. 
 599       } 
 600     }

 602     private void defineObjectLiteral(Node objectLit) { 
 603       // Handle the @lends annotation. 
 604       JSType type = null; 
 605       JSDocInfo info = objectLit.getJSDocInfo(); 
 606       if (info != null && 
 607           info.getLendsName() != null) { 
 608         String lendsName = info.getLendsName(); 
 609         Var lendsVar = scope.getVar(lendsName); 
 610         if (lendsVar == null) { 
 611           compiler.report( 
 612               JSError.make(sourceName, objectLit, UNKNOWN_LENDS, lendsName)); 
 613         } else { 
 614           type = lendsVar.getType(); 
 615           if (type == null) { 
 616             type = typeRegistry.getNativeType(UNKNOWN_TYPE); 
 617           } 
 618           if (!type.isSubtype(typeRegistry.getNativeType(OBJECT_TYPE))) { 
 619             compiler.report( 
 620                 JSError.make(sourceName, objectLit, LENDS_ON_NON_OBJECT, 
 621                     lendsName, type.toString())); 
 622             type = null; 
 623           } else { 
 624             objectLit.setJSType(type); 
 625           } 
 626         } 
 627       } 
 628  
 629       info = NodeUtil.getBestJSDocInfo(objectLit); 
 630       Node lValue = NodeUtil.getBestLValue(objectLit); 
 631       String lValueName = NodeUtil.getBestLValueName(lValue); 
 632       boolean createdEnumType = false; 
 633       if (info != null && info.hasEnumParameterType()) { 
 634         type = createEnumTypeFromNodes(objectLit, lValueName, info, lValue); 
 635         createdEnumType = true; 
 636       } 
 637  
 638       if (type == null) { 
 639         type = typeRegistry.createAnonymousObjectType(); 
 640       } 
 641  
 642       setDeferredType(objectLit, type); 
 643  
 644       // If this is an enum, the properties were already taken care of above. 
 645       processObjectLitProperties( 
 646           objectLit, ObjectType.cast(objectLit.getJSType()), !createdEnumType); 
 647     }

 651      * @param objLit The OBJECTLIT node. 
 652      * @param objLitType The type of the OBJECTLIT node. This might be a named 
 653      *     type, because of the lends annotation. 
 654      * @param declareOnOwner If true, declare properties on the objLitType as 
 655      *     well. If false, the caller should take care of this. 
 656      */ 
 657     void processObjectLitProperties( 
 658         Node objLit, ObjectType objLitType, 
 659         boolean declareOnOwner) { 
 660       for (Node keyNode = objLit.getFirstChild(); keyNode != null; 
 661            keyNode = keyNode.getNext()) {

 663         String memberName = NodeUtil.getObjectLitKeyName(keyNode);

 666             getDeclaredType(keyNode.getSourceFileName(), info, keyNode, value);

 668             objLitType.toMaybeEnumType().getElementsType() : 
 669             NodeUtil.getObjectLitKeyTypeFromValueType(keyNode, valueType);

 671         // Try to declare this property in the current scope if it 
 672         // has an authoritative name. 
 673         String qualifiedName = NodeUtil.getBestLValueName(keyNode); 
 674         if (qualifiedName != null) { 
 675           boolean inferred = keyType == null; 
 676           defineSlot(keyNode, objLit, qualifiedName, keyType, inferred); 
 677         } else if (keyType != null) { 
 678           setDeferredType(keyNode, keyType); 
 679         } 
 680  
 681         if (keyType != null && objLitType != null && declareOnOwner) { 
 682           // Declare this property on its object literal. 
 683           boolean isExtern = keyNode.isFromExterns(); 
 684           objLitType.defineDeclaredProperty(memberName, keyType, keyNode); 
 685         } 
 686       }

 819     /** 
 820      * Creates a new function type, based on the given nodes. 
 821      * 
 822      * This handles two cases that are semantically very different, but 
 823      * are not mutually exclusive: 
 824      * - A function literal that needs a type attached to it. 
 825      * - An assignment expression with function-type info in the JsDoc. 
 826      * 
 827      * All parameters are optional, and we will do the best we can to create 
 828      * a function type. 
 829      * 
 830      * This function will always create a function type, so only call it if 
 831      * you're sure that's what you want. 
 832      * 
 833      * @param rValue The function node. 
 834      * @param name the function's name 
 835      * @param info the {@link JSDocInfo} attached to the function definition 
 836      * @param lvalueNode The node where this function is being 
 837      *     assigned. For example, {@code A.prototype.foo = ...} would be used to 
 838      *     determine that this function is a method of A.prototype. May be 
 839      *     null to indicate that this is not being assigned to a qualified name. 
 840      */ 
 841     private FunctionType createFunctionTypeFromNodes( 
 842         @Nullable Node rValue, 
 843         @Nullable String name, 
 844         @Nullable JSDocInfo info, 
 845         @Nullable Node lvalueNode) { 
 846  
 847       FunctionType functionType = null; 
 848  
 849       // Global ctor aliases should be registered with the type registry. 
 850       if (rValue != null && rValue.isQualifiedName() && scope.isGlobal()) { 
 851         Var var = scope.getVar(rValue.getQualifiedName()); 
 852         if (var != null && var.getType() != null && 
 853             var.getType().isFunctionType()) { 
 854           FunctionType aliasedType  = var.getType().toMaybeFunctionType(); 
 855           if ((aliasedType.isConstructor() || aliasedType.isInterface()) && 
 856               !aliasedType.isNativeObjectType()) { 
 857             functionType = aliasedType; 
 858  
 859             if (name != null && scope.isGlobal()) { 
 860               typeRegistry.declareType(name, functionType.getInstanceType()); 
 861             } 
 862           } 
 863         } 
 864       } 
 865  
 866       if (functionType == null) { 
 867         Node errorRoot = rValue == null ? lvalueNode : rValue; 
 868         boolean isFnLiteral = 
 869             rValue != null && rValue.isFunction(); 
 870         Node fnRoot = isFnLiteral ? rValue : null; 
 871         Node parametersNode = isFnLiteral ? 
 872             rValue.getFirstChild().getNext() : null; 
 873         Node fnBlock = isFnLiteral ? parametersNode.getNext() : null; 
 874  
 875         if (info != null && info.hasType()) { 
 876           JSType type = info.getType().evaluate(scope, typeRegistry); 
 877  
 878           // Known to be not null since we have the FUNCTION token there. 
 879           type = type.restrictByNotNullOrUndefined(); 
 880           if (type.isFunctionType()) { 
 881             functionType = type.toMaybeFunctionType(); 
 882             functionType.setJSDocInfo(info); 
 883           } 
 884         } 
 885  
 886         if (functionType == null) { 
 887           // Find the type of any overridden function. 
 888           Node ownerNode = NodeUtil.getBestLValueOwner(lvalueNode); 
 889           String ownerName = NodeUtil.getBestLValueName(ownerNode); 
 890           Var ownerVar = null; 
 891           String propName = null; 
 892           ObjectType ownerType = null; 
 893           if (ownerName != null) { 
 894             ownerVar = scope.getVar(ownerName); 
 895             if (ownerVar != null) { 
 896               ownerType = ObjectType.cast(ownerVar.getType()); 
 897             } 
 898             if (name != null) { 
 899               propName = name.substring(ownerName.length() + 1); 
 900             } 
 901           } 
 902  
 903           FunctionType overriddenPropType = null; 
 904           if (ownerType != null && propName != null) { 
 905             overriddenPropType = 
 906                 findOverriddenFunction(ownerType, propName); 
 907           } 
 908  
 909           FunctionTypeBuilder builder = 
 910               new FunctionTypeBuilder(name, compiler, errorRoot, sourceName, 
 911                   scope) 
 912               .setContents(getFunctionAnalysisResults(fnRoot)) 
 913               .inferFromOverriddenFunction(overriddenPropType, parametersNode) 
 914               .inferTemplateTypeName(info) 
 915               .inferReturnType(info) 
 916               .inferInheritance(info); 
 917  
 918  
 919           // Infer the context type. 
 920           boolean searchedForThisType = false; 
 921           if (ownerType != null && ownerType.isFunctionPrototypeType()) { 
 922             builder.inferThisType( 
 923                 info, ownerType.getOwnerFunction().getInstanceType()); 
 924             searchedForThisType = true; 
 925           } else if (ownerNode != null && ownerNode.isThis()) { 
 926             // If 'this' has a type, use that instead. 
 927             // This is a hack, necessary because CollectProperties (below) 
 928             // doesn't run with the scope that it's building, 
 929             // so scope.getTypeOfThis() will be wrong. 
 930             JSType injectedThisType = ownerNode.getJSType(); 
 931             builder.inferThisType( 
 932                 info, 
 933                 injectedThisType == null ? 
 934                 scope.getTypeOfThis() : injectedThisType); 
 935             searchedForThisType = true; 
 936           } 
 937  
 938           if (!searchedForThisType) { 
 939             builder.inferThisType(info); 
 940           } 
 941  
 942           functionType = builder 
 943               .inferParameterTypes(parametersNode, info) 
 944               .buildAndRegister(); 
 945         } 
 946       } 
 947  
 948       // all done 
 949       return functionType; 
 950     }

 952     /** 
 953      * Find the function that's being overridden on this type, if any. 
 954      */ 
 955     private FunctionType findOverriddenFunction( 
 956         ObjectType ownerType, String propName) { 
 957       // First, check to see if the property is implemented 
 958       // on a superclass. 
 959       JSType propType = ownerType.getPropertyType(propName); 
 960       if (propType != null && propType.isFunctionType()) { 
 961         return propType.toMaybeFunctionType(); 
 962       } else { 
 963         // If it's not, then check to see if it's implemented 
 964         // on an implemented interface. 
 965         for (ObjectType iface : 
 966                  ownerType.getCtorImplementedInterfaces()) { 
 967           propType = iface.getPropertyType(propName); 
 968           if (propType != null && propType.isFunctionType()) { 
 969             return propType.toMaybeFunctionType(); 
 970           } 
 971         } 
 972       } 
 973  
 974       return null; 
 975     }

 977     /** 
 978      * Creates a new enum type, based on the given nodes. 
 979      * 
 980      * This handles two cases that are semantically very different, but 
 981      * are not mutually exclusive: 
 982      * - An object literal that needs an enum type attached to it. 
 983      * - An assignment expression with an enum tag in the JsDoc. 
 984      * 
 985      * This function will always create an enum type, so only call it if 
 986      * you're sure that's what you want. 
 987      * 
 988      * @param rValue The node of the enum. 
 989      * @param name The enum's name 
 990      * @param info The {@link JSDocInfo} attached to the enum definition. 
 991      * @param lValueNode The node where this function is being 
 992      *     assigned. 
 993      */ 
 994     private EnumType createEnumTypeFromNodes(Node rValue, String name, 
 995         JSDocInfo info, Node lValueNode) { 
 996       Preconditions.checkNotNull(info); 
 997       Preconditions.checkState(info.hasEnumParameterType()); 
 998  
 999       EnumType enumType = null; 
1000       if (rValue != null && rValue.isQualifiedName()) { 
1001         // Handle an aliased enum. 
1002         Var var = scope.getVar(rValue.getQualifiedName()); 
1003         if (var != null && var.getType() instanceof EnumType) { 
1004           enumType = (EnumType) var.getType(); 
1005         } 
1006       } 
1007  
1008       if (enumType == null) { 
1009         JSType elementsType = 
1010             info.getEnumParameterType().evaluate(scope, typeRegistry); 
1011         enumType = typeRegistry.createEnumType(name, rValue, elementsType); 
1012  
1013         if (rValue != null && rValue.isObjectLit()) { 
1014           // collect enum elements 
1015           Node key = rValue.getFirstChild(); 
1016           while (key != null) { 
1017             String keyName = NodeUtil.getStringValue(key); 
1018             if (keyName == null) { 
1019               // GET and SET don't have a String value; 
1020               compiler.report( 
1021                   JSError.make(sourceName, key, ENUM_NOT_CONSTANT, keyName)); 
1022             } else if (!codingConvention.isValidEnumKey(keyName)) { 
1023               compiler.report( 
1024                   JSError.make(sourceName, key, ENUM_NOT_CONSTANT, keyName)); 
1025             } else { 
1026               enumType.defineElement(keyName, key); 
1027             } 
1028             key = key.getNext(); 
1029           } 
1030         } 
1031       } 
1032  
1033       if (name != null && scope.isGlobal()) { 
1034         typeRegistry.declareType(name, enumType.getElementsType()); 
1035       } 
1036  
1037       return enumType; 
1038     }

1085     /** 
1086      * Defines a symbol in the current scope. 
1087      * 
1088      * @param n the defining NAME or GETPROP or object literal key node. 
1089      * @param parent the {@code n}'s parent. 
1090      * @param variableName The name that this should be known by. 
1091      * @param type the variable's type. It may be {@code null} if 
1092      *     {@code inferred} is {@code true}. 
1093      * @param inferred Whether the type is inferred or declared. 
1094      */ 
1095     void defineSlot(Node n, Node parent, String variableName, 
1096         JSType type, boolean inferred) { 
1097       Preconditions.checkArgument(!variableName.isEmpty()); 
1098  
1099       boolean isGlobalVar = n.isName() && scope.isGlobal(); 
1100       boolean shouldDeclareOnGlobalThis = 
1101           isGlobalVar && 
1102           (parent.isVar() || 
1103            parent.isFunction()); 
1104  
1105       // If n is a property, then we should really declare it in the 
1106       // scope where the root object appears. This helps out people 
1107       // who declare "global" names in an anonymous namespace. 
1108       Scope scopeToDeclareIn = scope; 
1109       if (n.isGetProp() && !scope.isGlobal() && 
1110           isQnameRootedInGlobalScope(n)) { 
1111         Scope globalScope = scope.getGlobalScope(); 
1112  
1113         // don't try to declare in the global scope if there's 
1114         // already a symbol there with this name. 
1115         if (!globalScope.isDeclared(variableName, false)) { 
1116           scopeToDeclareIn = scope.getGlobalScope(); 
1117         } 
1118       } 
1119  
1120       // The input may be null if we are working with a AST snippet. So read 
1121       // the extern info from the node. 
1122       boolean isExtern = n.isFromExterns(); 
1123       Var newVar = null; 
1124  
1125       // declared in closest scope? 
1126       CompilerInput input = compiler.getInput(inputId); 
1127       if (scopeToDeclareIn.isDeclared(variableName, false)) { 
1128         Var oldVar = scopeToDeclareIn.getVar(variableName); 
1129         newVar = validator.expectUndeclaredVariable( 
1130             sourceName, input, n, parent, oldVar, variableName, type); 
1131       } else { 
1132         if (!inferred) { 
1133           setDeferredType(n, type); 
1134         } 
1135  
1136         newVar = 
1137           scopeToDeclareIn.declare(variableName, n, type, input, inferred); 
1138  
1139         if (type instanceof EnumType) { 
1140           Node initialValue = newVar.getInitialValue(); 
1141           boolean isValidValue = initialValue != null && 
1142               (initialValue.isObjectLit() || 
1143                initialValue.isQualifiedName()); 
1144           if (!isValidValue) { 
1145             compiler.report(JSError.make(sourceName, n, ENUM_INITIALIZER)); 
1146           } 
1147         } 
1148       } 
1149  
1150       // We need to do some additional work for constructors and interfaces. 
1151       FunctionType fnType = JSType.toMaybeFunctionType(type); 
1152       if (fnType != null && 
1153           // We don't want to look at empty function types. 
1154           !type.isEmptyType()) { 
1155         if ((fnType.isConstructor() || fnType.isInterface()) && 
1156             !fnType.equals(getNativeType(U2U_CONSTRUCTOR_TYPE))) { 
1157           // Declare var.prototype in the scope chain. 
1158           FunctionType superClassCtor = fnType.getSuperClassConstructor(); 
1159           ObjectType.Property prototypeSlot = fnType.getSlot("prototype"); 
1160  
1161           // When we declare the function prototype implicitly, we 
1162           // want to make sure that the function and its prototype 
1163           // are declared at the same node. We also want to make sure 
1164           // that the if a symbol has both a Var and a JSType, they have 
1165           // the same node. 
1166           // 
1167           // This consistency is helpful to users of SymbolTable, 
1168           // because everything gets declared at the same place. 
1169           prototypeSlot.setNode(n); 
1170  
1171           String prototypeName = variableName + ".prototype"; 
1172  
1173           // There are some rare cases where the prototype will already 
1174           // be declared. See TypedScopeCreatorTest#testBogusPrototypeInit. 
1175           // Fortunately, other warnings will complain if this happens. 
1176           Var prototypeVar = scopeToDeclareIn.getVar(prototypeName); 
1177           if (prototypeVar != null && prototypeVar.scope == scopeToDeclareIn) { 
1178             scopeToDeclareIn.undeclare(prototypeVar); 
1179           } 
1180  
1181           scopeToDeclareIn.declare(prototypeName, 
1182               n, prototypeSlot.getType(), input, 
1183               /* declared iff there's an explicit supertype */ 
1184               superClassCtor == null || 
1185               superClassCtor.getInstanceType().equals( 
1186                   getNativeType(OBJECT_TYPE))); 
1187  
1188           // Make sure the variable is initialized to something if 
1189           // it constructs itself. 
1190           if (newVar.getInitialValue() == null && 
1191               !isExtern && 
1192               // We want to make sure that when we declare a new instance 
1193               // type (with @constructor) that there's actually a ctor for it. 
1194               // This doesn't apply to structural constructors 
1195               // (like function(new:Array). Checking the constructed 
1196               // type against the variable name is a sufficient check for 
1197               // this. 
1198               variableName.equals( 
1199                   fnType.getInstanceType().getReferenceName())) { 
1200             compiler.report( 
1201                 JSError.make(sourceName, n, 
1202                     fnType.isConstructor() ? 
1203                     CTOR_INITIALIZER : IFACE_INITIALIZER, 
1204                     variableName)); 
1205           } 
1206         } 
1207       } 
1208  
1209       if (shouldDeclareOnGlobalThis) { 
1210         ObjectType globalThis = 
1211             typeRegistry.getNativeObjectType(GLOBAL_THIS); 
1212         if (inferred) { 
1213           globalThis.defineInferredProperty(variableName, 
1214               type == null ? 
1215               getNativeType(JSTypeNative.NO_TYPE) : 
1216               type, 
1217               n); 
1218         } else { 
1219           globalThis.defineDeclaredProperty(variableName, type, n); 
1220         } 
1221       } 
1222  
1223       if (isGlobalVar && "Window".equals(variableName) 
1224           && type != null 
1225           && type.isFunctionType() 
1226           && type.isConstructor()) { 
1227         FunctionType globalThisCtor = 
1228             typeRegistry.getNativeObjectType(GLOBAL_THIS).getConstructor(); 
1229         globalThisCtor.getInstanceType().clearCachedValues(); 
1230         globalThisCtor.getPrototype().clearCachedValues(); 
1231         globalThisCtor 
1232             .setPrototypeBasedOn((type.toMaybeFunctionType()).getInstanceType()); 
1233       } 
1234     }

1258     /** 
1259      * Look for a type declaration on a property assignment 
1260      * (in an ASSIGN or an object literal key). 
1261      * 
1262      * @param info The doc info for this property. 
1263      * @param lValue The l-value node. 
1264      * @param rValue The node that {@code n} is being initialized to, 
1265      *     or {@code null} if this is a stub declaration. 
1266      */ 
1267     private JSType getDeclaredType(String sourceName, JSDocInfo info, 
1268         Node lValue, @Nullable Node rValue) { 
1269       if (info != null && info.hasType()) { 
1270         return getDeclaredTypeInAnnotation(sourceName, lValue, info); 
1271       } else if (rValue != null && rValue.isFunction() && 
1272           shouldUseFunctionLiteralType( 
1273               JSType.toMaybeFunctionType(rValue.getJSType()), info, lValue)) { 
1274         return rValue.getJSType(); 
1275       } else if (info != null) { 
1276         if (info.hasEnumParameterType()) { 
1277           if (rValue != null && rValue.isObjectLit()) { 
1278             return rValue.getJSType(); 
1279           } else { 
1280             return createEnumTypeFromNodes( 
1281                 rValue, lValue.getQualifiedName(), info, lValue); 
1282           } 
1283         } else if (info.isConstructor() || info.isInterface()) { 
1284           return createFunctionTypeFromNodes( 
1285               rValue, lValue.getQualifiedName(), info, lValue); 
1286         } else { 
1287           // Check if this is constant, and if it has a known type. 
1288           if (info.isConstant()) { 
1289             JSType knownType = null; 
1290             if (rValue != null) { 
1291               if (rValue.getJSType() != null && !rValue.getJSType().isUnknownType()) { 
1292                 // If rValue has a type-cast, we use the type in the type-cast. 
1293                 // If rValue's type was already computed during scope creation, 
1294                 // then we can safely use that. 
1295                 return rValue.getJSType(); 
1296               } else if (rValue.isOr()) { 
1297                 // Check for a very specific JS idiom: 
1298                 // var x = x || TYPE; 
1299                 // This is used by Closure's base namespace for esoteric 
1300                 // reasons. 
1301                 Node firstClause = rValue.getFirstChild(); 
1302                 Node secondClause = firstClause.getNext(); 
1303                 boolean namesMatch = firstClause.isName() 
1304                     && lValue.isName() 
1305                     && firstClause.getString().equals(lValue.getString()); 
1306                 if (namesMatch && secondClause.getJSType() != null 
1307                     && !secondClause.getJSType().isUnknownType()) { 
1308                   return secondClause.getJSType(); 
1309                 } 
1310               } 
1311             } 
1312           } 
1313         } 
1314       } 
1315  
1316       return getDeclaredTypeInAnnotation(sourceName, lValue, info); 
1317     }

1335      * Look for class-defining calls. 
1336      * Because JS has no 'native' syntax for defining classes, 
1337      * this is often very coding-convention dependent and business-logic heavy. 
1338      */ 
1339     private void checkForClassDefiningCalls( 
1340         NodeTraversal t, Node n, Node parent) { 
1341       SubclassRelationship relationship = 
1342           codingConvention.getClassesDefinedByCall(n); 
1343       if (relationship != null) { 
1344         FunctionType superCtor = getFunctionType( 
1345             scope.getVar(relationship.superclassName)); 
1346         FunctionType subCtor = getFunctionType( 
1347             scope.getVar(relationship.subclassName)); 
1348         if (superCtor != null && superCtor.isConstructor() && 
1349             subCtor != null && subCtor.isConstructor()) { 
1350           ObjectType superClass = superCtor.getInstanceType(); 
1351           ObjectType subClass = subCtor.getInstanceType(); 
1352  
1353           // superCtor and subCtor might be structural constructors 
1354           // (like {function(new:Object)}) so we need to resolve them back 
1355           // to the original ctor objects. 
1356           superCtor = superClass.getConstructor(); 
1357           subCtor = subClass.getConstructor(); 
1358  
1359           if (relationship.type == SubclassType.INHERITS && 
1360               !superClass.isEmptyType() && !subClass.isEmptyType()) { 
1361             validator.expectSuperType(t, n, superClass, subClass); 
1362           } 
1363  
1364           if (superCtor != null && subCtor != null) { 
1365             codingConvention.applySubclassRelationship( 
1366                 superCtor, subCtor, relationship.type); 
1367           } 
1368         } 
1369       } 
1370  
1371       String singletonGetterClassName = 
1372           codingConvention.getSingletonGetterClassName(n); 
1373       if (singletonGetterClassName != null) { 
1374         ObjectType objectType = ObjectType.cast( 
1375             typeRegistry.getType(singletonGetterClassName)); 
1376         if (objectType != null) { 
1377           FunctionType functionType = objectType.getConstructor(); 
1378  
1379           if (functionType != null) { 
1380             FunctionType getterType = 
1381                 typeRegistry.createFunctionType(objectType); 
1382             codingConvention.applySingletonGetter(functionType, getterType, 
1383                 objectType); 
1384           } 
1385         } 
1386       } 
1387  
1388       DelegateRelationship delegateRelationship = 
1389           codingConvention.getDelegateRelationship(n); 
1390       if (delegateRelationship != null) { 
1391         applyDelegateRelationship(delegateRelationship); 
1392       } 
1393  
1394       ObjectLiteralCast objectLiteralCast = 
1395           codingConvention.getObjectLiteralCast(n); 
1396       if (objectLiteralCast != null) { 
1397         if (objectLiteralCast.diagnosticType == null) { 
1398           ObjectType type = ObjectType.cast( 
1399               typeRegistry.getType(objectLiteralCast.typeName)); 
1400           if (type != null && type.getConstructor() != null) { 
1401             setDeferredType(objectLiteralCast.objectNode, type); 
1402           } else { 
1403             compiler.report(JSError.make(t.getSourceName(), n, 
1404                     CONSTRUCTOR_EXPECTED)); 
1405           } 
1406         } else { 
1407           compiler.report(JSError.make(t.getSourceName(), n, 
1408                   objectLiteralCast.diagnosticType)); 
1409         } 
1410       }

1454     /** 
1455      * Declare the symbol for a qualified name in the global scope. 
1456      * 
1457      * @param info The doc info for this property. 
1458      * @param n A top-level GETPROP node (it should not be contained inside 
1459      *     another GETPROP). 
1460      * @param parent The parent of {@code n}. 
1461      * @param rhsValue The node that {@code n} is being initialized to, 
1462      *     or {@code null} if this is a stub declaration. 
1463      */ 
1464     void maybeDeclareQualifiedName(NodeTraversal t, JSDocInfo info, 
1465         Node n, Node parent, Node rhsValue) { 
1466       Node ownerNode = n.getFirstChild(); 
1467       String ownerName = ownerNode.getQualifiedName(); 
1468       String qName = n.getQualifiedName(); 
1469       String propName = n.getLastChild().getString(); 
1470       Preconditions.checkArgument(qName != null && ownerName != null); 
1471  
1472       // Precedence of type information on GETPROPs: 
1473       // 1) @type annotation / @enum annotation 
1474       // 2) ASSIGN to FUNCTION literal 
1475       // 3) @param/@return annotation (with no function literal) 
1476       // 4) ASSIGN to something marked @const 
1477       // 5) ASSIGN to anything else 
1478       // 
1479       // 1, 3, and 4 are declarations, 5 is inferred, and 2 is a declaration iff 
1480       // the function has JsDoc or has not been declared before. 
1481       // 
1482       // FUNCTION literals are special because TypedScopeCreator is very smart 
1483       // about getting as much type information as possible for them. 
1484  
1485       // Determining type for #1 + #2 + #3 + #4 
1486       JSType valueType = getDeclaredType(t.getSourceName(), info, n, rhsValue); 
1487       if (valueType == null && rhsValue != null) { 
1488         // Determining type for #5 
1489         valueType = rhsValue.getJSType(); 
1490       } 
1491  
1492       // Function prototypes are special. 
1493       // It's a common JS idiom to do: 
1494       // F.prototype = { ... }; 
1495       // So if F does not have an explicitly declared super type, 
1496       // allow F.prototype to be redefined arbitrarily. 
1497       if ("prototype".equals(propName)) { 
1498         Var qVar = scope.getVar(qName); 
1499         if (qVar != null) { 
1500           // If the programmer has declared that F inherits from Super, 
1501           // and they assign F.prototype to an object literal, 
1502           // then they are responsible for making sure that the object literal's 
1503           // implicit prototype is set up appropriately. We just obey 
1504           // the @extends tag. 
1505           ObjectType qVarType = ObjectType.cast(qVar.getType()); 
1506           if (qVarType != null && 
1507               rhsValue != null && 
1508               rhsValue.isObjectLit()) { 
1509             typeRegistry.resetImplicitPrototype( 
1510                 rhsValue.getJSType(), qVarType.getImplicitPrototype()); 
1511           } else if (!qVar.isTypeInferred()) { 
1512             // If the programmer has declared that F inherits from Super, 
1513             // and they assign F.prototype to some arbitrary expression, 
1514             // there's not much we can do. We just ignore the expression, 
1515             // and hope they've annotated their code in a way to tell us 
1516             // what props are going to be on that prototype. 
1517             return; 
1518           } 
1519           if (qVar.getScope() == scope) { 
1520             scope.undeclare(qVar); 
1521           } 
1522         } 
1523       } 
1524  
1525       if (valueType == null) { 
1526         if (parent.isExprResult()) { 
1527           stubDeclarations.add(new StubDeclaration( 
1528               n, 
1529               t.getInput() != null && t.getInput().isExtern(), 
1530               ownerName)); 
1531         } 
1532  
1533         return; 
1534       } 
1535  
1536       boolean inferred = isQualifiedNameInferred( 
1537           qName, n, info, rhsValue, valueType); 
1538       if (!inferred) { 
1539         ObjectType ownerType = getObjectSlot(ownerName); 
1540         if (ownerType != null) { 
1541           // Only declare this as an official property if it has not been 
1542           // declared yet. 
1543           boolean isExtern = t.getInput() != null && t.getInput().isExtern(); 
1544           if ((!ownerType.hasOwnProperty(propName) || 
1545                ownerType.isPropertyTypeInferred(propName)) && 
1546               ((isExtern && !ownerType.isNativeObjectType()) || 
1547                !ownerType.isInstanceType())) { 
1548             // If the property is undeclared or inferred, declare it now. 
1549             ownerType.defineDeclaredProperty(propName, valueType, n); 
1550           } 
1551         } 
1552  
1553         // If the property is already declared, the error will be 
1554         // caught when we try to declare it in the current scope. 
1555         defineSlot(n, parent, valueType, inferred); 
1556       } else if (rhsValue != null && rhsValue.isTrue()) { 
1557         // We declare these for delegate proxy method properties. 
1558         FunctionType ownerType = 
1559             JSType.toMaybeFunctionType(getObjectSlot(ownerName)); 
1560         if (ownerType != null) { 
1561           JSType ownerTypeOfThis = ownerType.getTypeOfThis(); 
1562           String delegateName = codingConvention.getDelegateSuperclassName(); 
1563           JSType delegateType = delegateName == null ? 
1564               null : typeRegistry.getType(delegateName); 
1565           if (delegateType != null && 
1566               ownerTypeOfThis.isSubtype(delegateType)) { 
1567             defineSlot(n, parent, getNativeType(BOOLEAN_TYPE), true); 
1568           } 
1569         } 
1570       } 
1571     }

1573     /** 
1574      * Determines whether a qualified name is inferred. 
1575      * NOTE(nicksantos): Determining whether a property is declared or not 
1576      * is really really obnoxious. 
1577      * 
1578      * The problem is that there are two (equally valid) coding styles: 
1579      * 
1580      * (function() { 
1581      *   /* The authoritative definition of goog.bar. / 
1582      *   goog.bar = function() {}; 
1583      * })(); 
1584      * 
1585      * function f() { 
1586      *   goog.bar(); 
1587      *   /* Reset goog.bar to a no-op. / 
1588      *   goog.bar = function() {}; 
1589      * } 
1590      * 
1591      * In a dynamic language with first-class functions, it's very difficult 
1592      * to know which one the user intended without looking at lots of 
1593      * contextual information (the second example demonstrates a small case 
1594      * of this, but there are some really pathological cases as well). 
1595      * 
1596      * The current algorithm checks if either the declaration has 
1597      * JsDoc type information, or @const with a known type, 
1598      * or a function literal with a name we haven't seen before. 
1599      */ 
1600     private boolean isQualifiedNameInferred( 
1601         String qName, Node n, JSDocInfo info, 
1602         Node rhsValue, JSType valueType) { 
1603       if (valueType == null) { 
1604         return true; 
1605       } 
1606  
1607       boolean inferred = true; 
1608       if (info != null) { 
1609         inferred = !(info.hasType() 
1610             || info.hasEnumParameterType() 
1611             || (info.isConstant() && valueType != null 
1612                 && !valueType.isUnknownType()) 
1613             || FunctionTypeBuilder.isFunctionTypeDeclaration(info)); 
1614       } 
1615  
1616       if (inferred && rhsValue != null && rhsValue.isFunction()) { 
1617         if (info != null) { 
1618           return false; 
1619         } else if (!scope.isDeclared(qName, false) && 
1620             n.isUnscopedQualifiedName()) { 
1621  
1622           // Check if this is in a conditional block. 
1623           // Functions assigned in conditional blocks are inferred. 
1624           for (Node current = n.getParent(); 
1625                !(current.isScript() || current.isFunction()); 
1626                current = current.getParent()) { 
1627             if (NodeUtil.isControlStructure(current)) { 
1628               return true; 
1629             } 
1630           } 
1631  
1632           // Check if this is assigned in an inner scope. 
1633           // Functions assigned in inner scopes are inferred. 
1634           AstFunctionContents contents = 
1635               getFunctionAnalysisResults(scope.getRootNode()); 
1636           if (contents == null || 
1637               !contents.getEscapedQualifiedNames().contains(qName)) { 
1638             return false; 
1639           } 
1640         } 
1641       } 
1642       return inferred; 
1643     }

1891     /** Handle bleeding functions and function parameters. */ 
1892     private void handleFunctionInputs(Node fnNode) { 
1893       // Handle bleeding functions. 
1894       Node fnNameNode = fnNode.getFirstChild(); 
1895       String fnName = fnNameNode.getString(); 
1896       if (!fnName.isEmpty()) { 
1897         Scope.Var fnVar = scope.getVar(fnName); 
1898         if (fnVar == null || 
1899             // Make sure we're not touching a native function. Native 
1900             // functions aren't bleeding, but may not have a declaration 
1901             // node. 
1902             (fnVar.getNameNode() != null && 
1903                 // Make sure that the function is actually bleeding by checking 
1904                 // if has already been declared. 
1905                 fnVar.getInitialValue() != fnNode)) { 
1906           defineSlot(fnNameNode, fnNode, fnNode.getJSType(), false); 
1907         } 
1908       } 
1909  
1910       declareArguments(fnNode); 
1911     }

1913     /** 
1914      * Declares all of a function's arguments. 
1915      */ 
1916     private void declareArguments(Node functionNode) { 
1917       Node astParameters = functionNode.getFirstChild().getNext(); 
1918       Node body = astParameters.getNext(); 
1919       FunctionType functionType = 
1920           JSType.toMaybeFunctionType(functionNode.getJSType()); 
1921       if (functionType != null) { 
1922         Node jsDocParameters = functionType.getParametersNode(); 
1923         if (jsDocParameters != null) { 
1924           Node jsDocParameter = jsDocParameters.getFirstChild(); 
1925           for (Node astParameter : astParameters.children()) { 
1926             if (jsDocParameter != null) { 
1927               defineSlot(astParameter, functionNode, 
1928                   jsDocParameter.getJSType(), false); 
1929               jsDocParameter = jsDocParameter.getNext(); 
1930             } else { 
1931               defineSlot(astParameter, functionNode, null, true); 
1932             } 
1933           } 
1934         } 
1935       } 
1936     } // end declareArguments

1939   /** 
1940    * Does a first-order function analysis that just looks at simple things 
1941    * like what variables are escaped, and whether 'this' is used. 
1942    */ 
1943   private static class FirstOrderFunctionAnalyzer 
1944       extends AbstractScopedCallback implements CompilerPass { 
1945     private final AbstractCompiler compiler; 
1946     private final Map<Node, AstFunctionContents> data; 
1947  
1948     FirstOrderFunctionAnalyzer( 
1949         AbstractCompiler compiler, Map<Node, AstFunctionContents> outParam) { 
1950       this.compiler = compiler; 
1951       this.data = outParam; 
1952     } 
1953  
1954     @Override public void process(Node externs, Node root) { 
1955       if (externs == null) { 
1956         NodeTraversal.traverse(compiler, root, this); 
1957       } else { 
1958         NodeTraversal.traverseRoots( 
1959             compiler, ImmutableList.of(externs, root), this); 
1960       } 
1961     } 
1962  
1963     @Override public void enterScope(NodeTraversal t) { 
1964       if (!t.inGlobalScope()) { 
1965         Node n = t.getScopeRoot(); 
1966         data.put(n, new AstFunctionContents(n)); 
1967       } 
1968     } 
1969  
1970     @Override public void visit(NodeTraversal t, Node n, Node parent) { 
1971       if (t.inGlobalScope()) { 
1972         return; 
1973       } 
1974  
1975       if (n.isReturn() && n.getFirstChild() != null) { 
1976         data.get(t.getScopeRoot()).recordNonEmptyReturn(); 
1977       } 
1978  
1979       if (t.getScopeDepth() <= 2) { 
1980         // We only need to worry about escaped variables at depth 3. 
1981         // An variable escaped at depth 2 is, by definition, a global variable. 
1982         // We treat all global variables as escaped by default, so there's 
1983         // no reason to do this extra computation for them. 
1984         return; 
1985       } 
1986  
1987       if (n.isName() && NodeUtil.isLValue(n)) { 
1988         String name = n.getString(); 
1989         Scope scope = t.getScope(); 
1990         Var var = scope.getVar(name); 
1991         if (var != null) { 
1992           Scope ownerScope = var.getScope(); 
1993           if (scope != ownerScope && ownerScope.isLocal()) { 
1994             data.get(ownerScope.getRootNode()).recordEscapedVarName(name); 
1995           } 
1996         } 
1997       } else if (n.isGetProp() && n.isUnscopedQualifiedName() && 
1998           NodeUtil.isLValue(n)) { 
1999         String name = NodeUtil.getRootOfQualifiedName(n).getString(); 
2000         Scope scope = t.getScope(); 
2001         Var var = scope.getVar(name); 
2002         if (var != null) { 
2003           Scope ownerScope = var.getScope(); 
2004           if (scope != ownerScope && ownerScope.isLocal()) { 
2005             data.get(ownerScope.getRootNode()) 
2006                 .recordEscapedQualifiedName(n.getQualifiedName()); 
2007           } 
2008         } 
2009       } 
2010     } 
2011   }

 188    * Creates a scope with all types declared. Declares newly discovered types 
 189    * and type properties in the type registry. 
 190    */ 
 191   @Override 
 192   public Scope createScope(Node root, Scope parent) { 
 193     // Constructing the global scope is very different than constructing 
 194     // inner scopes, because only global scopes can contain named classes that 
 195     // show up in the type registry. 
 196     Scope newScope = null; 
 197     AbstractScopeBuilder scopeBuilder = null; 
 198     if (parent == null) { 
 199       // Run a first-order analysis over the syntax tree. 
 200       (new FirstOrderFunctionAnalyzer(compiler, functionAnalysisResults)) 
 201           .process(root.getFirstChild(), root.getLastChild()); 
 202  
 203       // Find all the classes in the global scope. 
 204       newScope = createInitialScope(root); 
 205  
 206       GlobalScopeBuilder globalScopeBuilder = new GlobalScopeBuilder(newScope); 
 207       scopeBuilder = globalScopeBuilder; 
 208       NodeTraversal.traverse(compiler, root, scopeBuilder); 
 209     } else { 
 210       newScope = new Scope(parent, root); 
 211       LocalScopeBuilder localScopeBuilder = new LocalScopeBuilder(newScope); 
 212       scopeBuilder = localScopeBuilder; 
 213       localScopeBuilder.build(); 
 214     } 
 215  
 216     scopeBuilder.resolveStubDeclarations(); 
 217     scopeBuilder.resolveTypes(); 
 218  
 219     // Gather the properties in each function that we found in the 
 220     // global scope, if that function has a @this type that we can 
 221     // build properties on. 
 222     for (Node functionNode : scopeBuilder.nonExternFunctions) { 
 223       JSType type = functionNode.getJSType(); 
 224       if (type != null && type.isFunctionType()) { 
 225         FunctionType fnType = type.toMaybeFunctionType(); 
 226         ObjectType fnThisType = fnType.getTypeOfThis(); 
 227         if (!fnThisType.isUnknownType()) { 
 228           NodeTraversal.traverse(compiler, functionNode.getLastChild(), 
 229               scopeBuilder.new CollectProperties(fnThisType)); 
 230         } 
 231       } 
 232     } 
 233  
 234     if (parent == null) { 
 235       codingConvention.defineDelegateProxyPrototypeProperties( 
 236           typeRegistry, newScope, delegateProxyPrototypes, 
 237           delegateCallingConventions); 
 238     } 
 239     return newScope; 
 240   }

 242   /** 
 243    * Patches a given global scope by removing variables previously declared in 
 244    * a script and re-traversing a new version of that script. 
 245    * 
 246    * @param globalScope The global scope generated by {@code createScope}. 
 247    * @param scriptRoot The script that is modified. 
 248    */ 
 249   void patchGlobalScope(Scope globalScope, Node scriptRoot) { 
 250     // Preconditions: This is supposed to be called only on (named) SCRIPT nodes 
 251     // and a global typed scope should have been generated already. 
 252     Preconditions.checkState(scriptRoot.isScript()); 
 253     Preconditions.checkNotNull(globalScope); 
 254     Preconditions.checkState(globalScope.isGlobal()); 
 255  
 256     String scriptName = NodeUtil.getSourceName(scriptRoot); 
 257     Preconditions.checkNotNull(scriptName); 
 258     for (Node node : ImmutableList.copyOf(functionAnalysisResults.keySet())) { 
 259       if (scriptName.equals(NodeUtil.getSourceName(node))) { 
 260         functionAnalysisResults.remove(node); 
 261       } 
 262     } 
 263  
 264     (new FirstOrderFunctionAnalyzer( 
 265         compiler, functionAnalysisResults)).process(null, scriptRoot); 
 266  
 267     // TODO(bashir): Variable declaration is not the only side effect of last 
 268     // global scope generation but here we only wipe that part off! 
 269  
 270     // Remove all variables that were previously declared in this scripts. 
 271     // First find all vars to remove then remove them because of iterator! 
 272     Iterator<Var> varIter = globalScope.getVars(); 
 273     List<Var> varsToRemove = Lists.newArrayList(); 
 274     while (varIter.hasNext()) { 
 275       Var oldVar = varIter.next(); 
 276       if (scriptName.equals(oldVar.getInputName())) { 
 277         varsToRemove.add(oldVar); 
 278       } 
 279     } 
 280     for (Var var : varsToRemove) { 
 281       globalScope.undeclare(var); 
 282       globalScope.getTypeOfThis().removeProperty(var.getName()); 
 283     } 
 284  
 285     // Now re-traverse the given script. 
 286     GlobalScopeBuilder scopeBuilder = new GlobalScopeBuilder(globalScope); 
 287     NodeTraversal.traverse(compiler, scriptRoot, scopeBuilder); 
 288   }

 398      * The scope that we're building. 
 399      */ 
 400     final Scope scope;

 402     private final List<DeferredSetType> deferredSetTypes = 
 403         Lists.newArrayList();

 406      * Functions that we found in the global scope and not in externs. 
 407      */ 
 408     private final List<Node> nonExternFunctions = Lists.newArrayList();

 411      * Object literals with a @lends annotation aren't analyzed until we 
 412      * reach the root of the statement they're defined in. 
 413      * 
 414      * This ensures that if there are any @lends annotations on the object 
 415      * literals, the type on the @lends annotation resolves correctly. 
 416      * 
 417      * For more information, see 
 418      * http://code.google.com/p/closure-compiler/issues/detail?id=314 
 419      */

 426      * stubs, then we should declare UNKNOWN types.

 432      * The current source file that we're in. 
 433      */ 
 434     private String sourceName = null;

 442       this.scope = scope;

 445     void setDeferredType(Node node, JSType type) { 
 446       deferredSetTypes.add(new DeferredSetType(node, type)); 
 447     }

 449     void resolveTypes() { 
 450       // Resolve types and attach them to nodes. 
 451       for (DeferredSetType deferred : deferredSetTypes) { 
 452         deferred.resolve(scope); 
 453       } 
 454  
 455       // Resolve types and attach them to scope slots. 
 456       Iterator<Var> vars = scope.getVars(); 
 457       while (vars.hasNext()) { 
 458         vars.next().resolveType(typeParsingErrorReporter); 
 459       } 
 460  
 461       // Tell the type registry that any remaining types 
 462       // are unknown. 
 463       typeRegistry.resolveTypesInScope(scope); 
 464     }

 466     @Override 
 467     public final boolean shouldTraverse(NodeTraversal t, Node n, 
 468         Node parent) { 
 469       inputId = t.getInputId(); 
 470       if (n.isFunction() || 
 471           n.isScript()) { 
 472         Preconditions.checkNotNull(inputId); 
 473         sourceName = NodeUtil.getSourceName(n); 
 474       } 
 475  
 476       // We do want to traverse the name of a named function, but we don't 
 477       // want to traverse the arguments or body. 
 478       boolean descend = parent == null || !parent.isFunction() || 
 479           n == parent.getFirstChild() || parent == scope.getRootNode(); 
 480  
 481       if (descend) { 
 482         // Handle hoisted functions on pre-order traversal, so that they 
 483         // get hit before other things in the scope. 
 484         if (NodeUtil.isStatementParent(n)) { 
 485           for (Node child = n.getFirstChild(); 
 486                child != null; 
 487                child = child.getNext()) { 
 488             if (NodeUtil.isHoistedFunctionDeclaration(child)) { 
 489               defineFunctionLiteral(child, n); 
 490             } 
 491           } 
 492         } 
 493       } 
 494  
 495       return descend; 
 496     }

 499     public void visit(NodeTraversal t, Node n, Node parent) { 
 500       inputId = t.getInputId(); 
 501       attachLiteralTypes(t, n); 
 502  
 503       switch (n.getType()) { 
 504         case Token.CALL: 
 505           checkForClassDefiningCalls(t, n, parent); 
 506           checkForCallingConventionDefiningCalls(n, delegateCallingConventions); 
 507           break; 
 508  
 509         case Token.FUNCTION: 
 510           if (t.getInput() == null || !t.getInput().isExtern()) { 
 511             nonExternFunctions.add(n); 
 512           } 
 513  
 514           // Hoisted functions are handled during pre-traversal. 
 515           if (!NodeUtil.isHoistedFunctionDeclaration(n)) { 
 516             defineFunctionLiteral(n, parent); 
 517           } 
 518           break; 
 519  
 520         case Token.ASSIGN: 
 521           // Handle initialization of properties. 
 522           Node firstChild = n.getFirstChild(); 
 523           if (firstChild.isGetProp() && 
 524               firstChild.isQualifiedName()) { 
 525             maybeDeclareQualifiedName(t, n.getJSDocInfo(), 
 526                 firstChild, n, firstChild.getNext()); 
 527           } 
 528           break; 
 529  
 530         case Token.CATCH: 
 531           defineCatch(n, parent); 
 532           break; 
 533  
 534         case Token.VAR: 
 535           defineVar(n, parent); 
 536           break; 
 537  
 538         case Token.GETPROP: 
 539           // Handle stubbed properties. 
 540           if (parent.isExprResult() && 
 541               n.isQualifiedName()) { 
 542             maybeDeclareQualifiedName(t, n.getJSDocInfo(), n, parent, null); 
 543           } 
 544           break; 
 545       } 
 546  
 547       // Analyze any @lends object literals in this statement. 
 548       if (n.getParent() != null && NodeUtil.isStatement(n) && 
 549           lentObjectLiterals != null) { 
 550         for (Node objLit : lentObjectLiterals) { 
 551           defineObjectLiteral(objLit); 
 552         } 
 553         lentObjectLiterals.clear(); 
 554       } 
 555     }

 557     private void attachLiteralTypes(NodeTraversal t, Node n) { 
 558       switch (n.getType()) { 
 559         case Token.NULL: 
 560           n.setJSType(getNativeType(NULL_TYPE)); 
 561           break; 
 562  
 563         case Token.VOID: 
 564           n.setJSType(getNativeType(VOID_TYPE)); 
 565           break; 
 566  
 567         case Token.STRING: 
 568           n.setJSType(getNativeType(STRING_TYPE)); 
 569           break; 
 570  
 571         case Token.NUMBER: 
 572           n.setJSType(getNativeType(NUMBER_TYPE)); 
 573           break; 
 574  
 575         case Token.TRUE: 
 576         case Token.FALSE: 
 577           n.setJSType(getNativeType(BOOLEAN_TYPE)); 
 578           break; 
 579  
 580         case Token.REGEXP: 
 581           n.setJSType(getNativeType(REGEXP_TYPE)); 
 582           break; 
 583  
 584         case Token.OBJECTLIT: 
 585           JSDocInfo info = n.getJSDocInfo(); 
 586           if (info != null && 
 587               info.getLendsName() != null) { 
 588             if (lentObjectLiterals == null) { 
 589               lentObjectLiterals = Lists.newArrayList(); 
 590             } 
 591             lentObjectLiterals.add(n); 
 592           } else { 
 593             defineObjectLiteral(n); 
 594           } 
 595           break; 
 596  
 597           // NOTE(nicksantos): If we ever support Array tuples, 
 598           // we will need to put ARRAYLIT here as well. 
 599       } 
 600     }

 602     private void defineObjectLiteral(Node objectLit) { 
 603       // Handle the @lends annotation. 
 604       JSType type = null; 
 605       JSDocInfo info = objectLit.getJSDocInfo(); 
 606       if (info != null && 
 607           info.getLendsName() != null) { 
 608         String lendsName = info.getLendsName(); 
 609         Var lendsVar = scope.getVar(lendsName); 
 610         if (lendsVar == null) { 
 611           compiler.report( 
 612               JSError.make(sourceName, objectLit, UNKNOWN_LENDS, lendsName)); 
 613         } else { 
 614           type = lendsVar.getType(); 
 615           if (type == null) { 
 616             type = typeRegistry.getNativeType(UNKNOWN_TYPE); 
 617           } 
 618           if (!type.isSubtype(typeRegistry.getNativeType(OBJECT_TYPE))) { 
 619             compiler.report( 
 620                 JSError.make(sourceName, objectLit, LENDS_ON_NON_OBJECT, 
 621                     lendsName, type.toString())); 
 622             type = null; 
 623           } else { 
 624             objectLit.setJSType(type); 
 625           } 
 626         } 
 627       } 
 628  
 629       info = NodeUtil.getBestJSDocInfo(objectLit); 
 630       Node lValue = NodeUtil.getBestLValue(objectLit); 
 631       String lValueName = NodeUtil.getBestLValueName(lValue); 
 632       boolean createdEnumType = false; 
 633       if (info != null && info.hasEnumParameterType()) { 
 634         type = createEnumTypeFromNodes(objectLit, lValueName, info, lValue); 
 635         createdEnumType = true; 
 636       } 
 637  
 638       if (type == null) { 
 639         type = typeRegistry.createAnonymousObjectType(); 
 640       } 
 641  
 642       setDeferredType(objectLit, type); 
 643  
 644       // If this is an enum, the properties were already taken care of above. 
 645       processObjectLitProperties( 
 646           objectLit, ObjectType.cast(objectLit.getJSType()), !createdEnumType); 
 647     }

 649     /** 
 650      * Process an object literal and all the types on it. 
 651      * @param objLit The OBJECTLIT node. 
 652      * @param objLitType The type of the OBJECTLIT node. This might be a named 
 653      *     type, because of the lends annotation. 
 654      * @param declareOnOwner If true, declare properties on the objLitType as 
 655      *     well. If false, the caller should take care of this. 
 656      */ 
 657     void processObjectLitProperties( 
 658         Node objLit, ObjectType objLitType, 
 659         boolean declareOnOwner) { 
 660       for (Node keyNode = objLit.getFirstChild(); keyNode != null; 
 661            keyNode = keyNode.getNext()) { 
 662         Node value = keyNode.getFirstChild(); 
 663         String memberName = NodeUtil.getObjectLitKeyName(keyNode); 
 664         JSDocInfo info = keyNode.getJSDocInfo(); 
 665         JSType valueType = 
 666             getDeclaredType(keyNode.getSourceFileName(), info, keyNode, value); 
 667         JSType keyType =  objLitType.isEnumType() ? 
 668             objLitType.toMaybeEnumType().getElementsType() : 
 669             NodeUtil.getObjectLitKeyTypeFromValueType(keyNode, valueType); 
 670  
 671         // Try to declare this property in the current scope if it 
 672         // has an authoritative name. 
 673         String qualifiedName = NodeUtil.getBestLValueName(keyNode); 
 674         if (qualifiedName != null) { 
 675           boolean inferred = keyType == null; 
 676           defineSlot(keyNode, objLit, qualifiedName, keyType, inferred); 
 677         } else if (keyType != null) { 
 678           setDeferredType(keyNode, keyType); 
 679         } 
 680  
 681         if (keyType != null && objLitType != null && declareOnOwner) { 
 682           // Declare this property on its object literal. 
 683           boolean isExtern = keyNode.isFromExterns(); 
 684           objLitType.defineDeclaredProperty(memberName, keyType, keyNode); 
 685         } 
 686       } 
 687     }

 691      * 
 692      * Extracts type information from either the {@code @type} tag or from 
 693      * the {@code @return} and {@code @param} tags. 
 694      */ 
 695     private JSType getDeclaredTypeInAnnotation(String sourceName, 
 696         Node node, JSDocInfo info) { 
 697       JSType jsType = null; 
 698       Node objNode = 
 699           node.isGetProp() ? node.getFirstChild() : 
 700           NodeUtil.isObjectLitKey(node, node.getParent()) ? node.getParent() :

 702       if (info != null) { 
 703         if (info.hasType()) {

 707           jsType = createFunctionTypeFromNodes( 
 708               null, fnName, info, node); 
 709         } 
 710       } 
 711       return jsType; 
 712     }

 757       assertDefinitionNode(n, Token.FUNCTION); 
 758  
 759       // Determine the name and JSDocInfo and l-value for the function.

 771       // declaration. Otherwise, the declaration will happen in other 
 772       // code paths. 
 773       if (NodeUtil.isFunctionDeclaration(n)) { 
 774         defineSlot(n.getFirstChild(), n, functionType); 
 775       }

 819     /** 
 820      * Creates a new function type, based on the given nodes. 
 821      * 
 822      * This handles two cases that are semantically very different, but 
 823      * are not mutually exclusive: 
 824      * - A function literal that needs a type attached to it. 
 825      * - An assignment expression with function-type info in the JsDoc. 
 826      * 
 827      * All parameters are optional, and we will do the best we can to create 
 828      * a function type. 
 829      * 
 830      * This function will always create a function type, so only call it if 
 831      * you're sure that's what you want. 
 832      * 
 833      * @param rValue The function node. 
 834      * @param name the function's name 
 835      * @param info the {@link JSDocInfo} attached to the function definition 
 836      * @param lvalueNode The node where this function is being 
 837      *     assigned. For example, {@code A.prototype.foo = ...} would be used to 
 838      *     determine that this function is a method of A.prototype. May be 
 839      *     null to indicate that this is not being assigned to a qualified name. 
 840      */ 
 841     private FunctionType createFunctionTypeFromNodes( 
 842         @Nullable Node rValue, 
 843         @Nullable String name, 
 844         @Nullable JSDocInfo info, 
 845         @Nullable Node lvalueNode) { 
 846  
 847       FunctionType functionType = null; 
 848  
 849       // Global ctor aliases should be registered with the type registry. 
 850       if (rValue != null && rValue.isQualifiedName() && scope.isGlobal()) { 
 851         Var var = scope.getVar(rValue.getQualifiedName()); 
 852         if (var != null && var.getType() != null && 
 853             var.getType().isFunctionType()) { 
 854           FunctionType aliasedType  = var.getType().toMaybeFunctionType(); 
 855           if ((aliasedType.isConstructor() || aliasedType.isInterface()) && 
 856               !aliasedType.isNativeObjectType()) { 
 857             functionType = aliasedType; 
 858  
 859             if (name != null && scope.isGlobal()) { 
 860               typeRegistry.declareType(name, functionType.getInstanceType()); 
 861             } 
 862           } 
 863         } 
 864       } 
 865  
 866       if (functionType == null) { 
 867         Node errorRoot = rValue == null ? lvalueNode : rValue; 
 868         boolean isFnLiteral = 
 869             rValue != null && rValue.isFunction(); 
 870         Node fnRoot = isFnLiteral ? rValue : null; 
 871         Node parametersNode = isFnLiteral ? 
 872             rValue.getFirstChild().getNext() : null; 
 873         Node fnBlock = isFnLiteral ? parametersNode.getNext() : null; 
 874  
 875         if (info != null && info.hasType()) { 
 876           JSType type = info.getType().evaluate(scope, typeRegistry); 
 877  
 878           // Known to be not null since we have the FUNCTION token there. 
 879           type = type.restrictByNotNullOrUndefined(); 
 880           if (type.isFunctionType()) { 
 881             functionType = type.toMaybeFunctionType(); 
 882             functionType.setJSDocInfo(info); 
 883           } 
 884         } 
 885  
 886         if (functionType == null) { 
 887           // Find the type of any overridden function. 
 888           Node ownerNode = NodeUtil.getBestLValueOwner(lvalueNode); 
 889           String ownerName = NodeUtil.getBestLValueName(ownerNode); 
 890           Var ownerVar = null; 
 891           String propName = null; 
 892           ObjectType ownerType = null; 
 893           if (ownerName != null) { 
 894             ownerVar = scope.getVar(ownerName); 
 895             if (ownerVar != null) { 
 896               ownerType = ObjectType.cast(ownerVar.getType()); 
 897             } 
 898             if (name != null) { 
 899               propName = name.substring(ownerName.length() + 1); 
 900             } 
 901           } 
 902  
 903           FunctionType overriddenPropType = null; 
 904           if (ownerType != null && propName != null) { 
 905             overriddenPropType = 
 906                 findOverriddenFunction(ownerType, propName); 
 907           } 
 908  
 909           FunctionTypeBuilder builder = 
 910               new FunctionTypeBuilder(name, compiler, errorRoot, sourceName, 
 911                   scope) 
 912               .setContents(getFunctionAnalysisResults(fnRoot)) 
 913               .inferFromOverriddenFunction(overriddenPropType, parametersNode) 
 914               .inferTemplateTypeName(info) 
 915               .inferReturnType(info) 
 916               .inferInheritance(info); 
 917  
 918  
 919           // Infer the context type. 
 920           boolean searchedForThisType = false; 
 921           if (ownerType != null && ownerType.isFunctionPrototypeType()) { 
 922             builder.inferThisType( 
 923                 info, ownerType.getOwnerFunction().getInstanceType()); 
 924             searchedForThisType = true; 
 925           } else if (ownerNode != null && ownerNode.isThis()) { 
 926             // If 'this' has a type, use that instead. 
 927             // This is a hack, necessary because CollectProperties (below) 
 928             // doesn't run with the scope that it's building, 
 929             // so scope.getTypeOfThis() will be wrong. 
 930             JSType injectedThisType = ownerNode.getJSType(); 
 931             builder.inferThisType( 
 932                 info, 
 933                 injectedThisType == null ? 
 934                 scope.getTypeOfThis() : injectedThisType); 
 935             searchedForThisType = true; 
 936           } 
 937  
 938           if (!searchedForThisType) { 
 939             builder.inferThisType(info); 
 940           } 
 941  
 942           functionType = builder 
 943               .inferParameterTypes(parametersNode, info) 
 944               .buildAndRegister(); 
 945         } 
 946       } 
 947  
 948       // all done 
 949       return functionType; 
 950     }

 952     /** 
 953      * Find the function that's being overridden on this type, if any. 
 954      */ 
 955     private FunctionType findOverriddenFunction( 
 956         ObjectType ownerType, String propName) { 
 957       // First, check to see if the property is implemented 
 958       // on a superclass. 
 959       JSType propType = ownerType.getPropertyType(propName); 
 960       if (propType != null && propType.isFunctionType()) { 
 961         return propType.toMaybeFunctionType(); 
 962       } else { 
 963         // If it's not, then check to see if it's implemented 
 964         // on an implemented interface. 
 965         for (ObjectType iface : 
 966                  ownerType.getCtorImplementedInterfaces()) { 
 967           propType = iface.getPropertyType(propName); 
 968           if (propType != null && propType.isFunctionType()) { 
 969             return propType.toMaybeFunctionType(); 
 970           } 
 971         } 
 972       } 
 973  
 974       return null; 
 975     }

 977     /** 
 978      * Creates a new enum type, based on the given nodes. 
 979      * 
 980      * This handles two cases that are semantically very different, but 
 981      * are not mutually exclusive: 
 982      * - An object literal that needs an enum type attached to it. 
 983      * - An assignment expression with an enum tag in the JsDoc. 
 984      * 
 985      * This function will always create an enum type, so only call it if 
 986      * you're sure that's what you want. 
 987      * 
 988      * @param rValue The node of the enum. 
 989      * @param name The enum's name 
 990      * @param info The {@link JSDocInfo} attached to the enum definition. 
 991      * @param lValueNode The node where this function is being 
 992      *     assigned. 
 993      */ 
 994     private EnumType createEnumTypeFromNodes(Node rValue, String name, 
 995         JSDocInfo info, Node lValueNode) { 
 996       Preconditions.checkNotNull(info); 
 997       Preconditions.checkState(info.hasEnumParameterType()); 
 998  
 999       EnumType enumType = null; 
1000       if (rValue != null && rValue.isQualifiedName()) { 
1001         // Handle an aliased enum. 
1002         Var var = scope.getVar(rValue.getQualifiedName()); 
1003         if (var != null && var.getType() instanceof EnumType) { 
1004           enumType = (EnumType) var.getType(); 
1005         } 
1006       } 
1007  
1008       if (enumType == null) { 
1009         JSType elementsType = 
1010             info.getEnumParameterType().evaluate(scope, typeRegistry); 
1011         enumType = typeRegistry.createEnumType(name, rValue, elementsType); 
1012  
1013         if (rValue != null && rValue.isObjectLit()) { 
1014           // collect enum elements 
1015           Node key = rValue.getFirstChild(); 
1016           while (key != null) { 
1017             String keyName = NodeUtil.getStringValue(key); 
1018             if (keyName == null) { 
1019               // GET and SET don't have a String value; 
1020               compiler.report( 
1021                   JSError.make(sourceName, key, ENUM_NOT_CONSTANT, keyName)); 
1022             } else if (!codingConvention.isValidEnumKey(keyName)) { 
1023               compiler.report( 
1024                   JSError.make(sourceName, key, ENUM_NOT_CONSTANT, keyName)); 
1025             } else { 
1026               enumType.defineElement(keyName, key); 
1027             } 
1028             key = key.getNext(); 
1029           } 
1030         } 
1031       } 
1032  
1033       if (name != null && scope.isGlobal()) { 
1034         typeRegistry.declareType(name, enumType.getElementsType()); 
1035       } 
1036  
1037       return enumType; 
1038     }

1054      * variable's type of {@link JSTypeNative#UNKNOWN_TYPE} if its type is

1085     /** 
1086      * Defines a symbol in the current scope. 
1087      * 
1088      * @param n the defining NAME or GETPROP or object literal key node. 
1089      * @param parent the {@code n}'s parent. 
1090      * @param variableName The name that this should be known by. 
1091      * @param type the variable's type. It may be {@code null} if 
1092      *     {@code inferred} is {@code true}. 
1093      * @param inferred Whether the type is inferred or declared. 
1094      */ 
1095     void defineSlot(Node n, Node parent, String variableName, 
1096         JSType type, boolean inferred) { 
1097       Preconditions.checkArgument(!variableName.isEmpty()); 
1098  
1099       boolean isGlobalVar = n.isName() && scope.isGlobal(); 
1100       boolean shouldDeclareOnGlobalThis = 
1101           isGlobalVar && 
1102           (parent.isVar() || 
1103            parent.isFunction()); 
1104  
1105       // If n is a property, then we should really declare it in the 
1106       // scope where the root object appears. This helps out people 
1107       // who declare "global" names in an anonymous namespace. 
1108       Scope scopeToDeclareIn = scope; 
1109       if (n.isGetProp() && !scope.isGlobal() && 
1110           isQnameRootedInGlobalScope(n)) { 
1111         Scope globalScope = scope.getGlobalScope(); 
1112  
1113         // don't try to declare in the global scope if there's 
1114         // already a symbol there with this name. 
1115         if (!globalScope.isDeclared(variableName, false)) { 
1116           scopeToDeclareIn = scope.getGlobalScope(); 
1117         } 
1118       } 
1119  
1120       // The input may be null if we are working with a AST snippet. So read 
1121       // the extern info from the node. 
1122       boolean isExtern = n.isFromExterns(); 
1123       Var newVar = null; 
1124  
1125       // declared in closest scope? 
1126       CompilerInput input = compiler.getInput(inputId); 
1127       if (scopeToDeclareIn.isDeclared(variableName, false)) { 
1128         Var oldVar = scopeToDeclareIn.getVar(variableName); 
1129         newVar = validator.expectUndeclaredVariable( 
1130             sourceName, input, n, parent, oldVar, variableName, type); 
1131       } else { 
1132         if (!inferred) { 
1133           setDeferredType(n, type); 
1134         } 
1135  
1136         newVar = 
1137           scopeToDeclareIn.declare(variableName, n, type, input, inferred); 
1138  
1139         if (type instanceof EnumType) { 
1140           Node initialValue = newVar.getInitialValue(); 
1141           boolean isValidValue = initialValue != null && 
1142               (initialValue.isObjectLit() || 
1143                initialValue.isQualifiedName()); 
1144           if (!isValidValue) { 
1145             compiler.report(JSError.make(sourceName, n, ENUM_INITIALIZER)); 
1146           } 
1147         } 
1148       } 
1149  
1150       // We need to do some additional work for constructors and interfaces. 
1151       FunctionType fnType = JSType.toMaybeFunctionType(type); 
1152       if (fnType != null && 
1153           // We don't want to look at empty function types. 
1154           !type.isEmptyType()) { 
1155         if ((fnType.isConstructor() || fnType.isInterface()) && 
1156             !fnType.equals(getNativeType(U2U_CONSTRUCTOR_TYPE))) { 
1157           // Declare var.prototype in the scope chain. 
1158           FunctionType superClassCtor = fnType.getSuperClassConstructor(); 
1159           ObjectType.Property prototypeSlot = fnType.getSlot("prototype"); 
1160  
1161           // When we declare the function prototype implicitly, we 
1162           // want to make sure that the function and its prototype 
1163           // are declared at the same node. We also want to make sure 
1164           // that the if a symbol has both a Var and a JSType, they have 
1165           // the same node. 
1166           // 
1167           // This consistency is helpful to users of SymbolTable, 
1168           // because everything gets declared at the same place. 
1169           prototypeSlot.setNode(n); 
1170  
1171           String prototypeName = variableName + ".prototype"; 
1172  
1173           // There are some rare cases where the prototype will already 
1174           // be declared. See TypedScopeCreatorTest#testBogusPrototypeInit. 
1175           // Fortunately, other warnings will complain if this happens. 
1176           Var prototypeVar = scopeToDeclareIn.getVar(prototypeName); 
1177           if (prototypeVar != null && prototypeVar.scope == scopeToDeclareIn) { 
1178             scopeToDeclareIn.undeclare(prototypeVar); 
1179           } 
1180  
1181           scopeToDeclareIn.declare(prototypeName, 
1182               n, prototypeSlot.getType(), input, 
1183               /* declared iff there's an explicit supertype */ 
1184               superClassCtor == null || 
1185               superClassCtor.getInstanceType().equals( 
1186                   getNativeType(OBJECT_TYPE))); 
1187  
1188           // Make sure the variable is initialized to something if 
1189           // it constructs itself. 
1190           if (newVar.getInitialValue() == null && 
1191               !isExtern && 
1192               // We want to make sure that when we declare a new instance 
1193               // type (with @constructor) that there's actually a ctor for it. 
1194               // This doesn't apply to structural constructors 
1195               // (like function(new:Array). Checking the constructed 
1196               // type against the variable name is a sufficient check for 
1197               // this. 
1198               variableName.equals( 
1199                   fnType.getInstanceType().getReferenceName())) { 
1200             compiler.report( 
1201                 JSError.make(sourceName, n, 
1202                     fnType.isConstructor() ? 
1203                     CTOR_INITIALIZER : IFACE_INITIALIZER, 
1204                     variableName)); 
1205           } 
1206         } 
1207       } 
1208  
1209       if (shouldDeclareOnGlobalThis) { 
1210         ObjectType globalThis = 
1211             typeRegistry.getNativeObjectType(GLOBAL_THIS); 
1212         if (inferred) { 
1213           globalThis.defineInferredProperty(variableName, 
1214               type == null ? 
1215               getNativeType(JSTypeNative.NO_TYPE) : 
1216               type, 
1217               n); 
1218         } else { 
1219           globalThis.defineDeclaredProperty(variableName, type, n); 
1220         } 
1221       } 
1222  
1223       if (isGlobalVar && "Window".equals(variableName) 
1224           && type != null 
1225           && type.isFunctionType() 
1226           && type.isConstructor()) { 
1227         FunctionType globalThisCtor = 
1228             typeRegistry.getNativeObjectType(GLOBAL_THIS).getConstructor(); 
1229         globalThisCtor.getInstanceType().clearCachedValues(); 
1230         globalThisCtor.getPrototype().clearCachedValues(); 
1231         globalThisCtor 
1232             .setPrototypeBasedOn((type.toMaybeFunctionType()).getInstanceType()); 
1233       } 
1234     }

1258     /** 
1259      * Look for a type declaration on a property assignment 
1260      * (in an ASSIGN or an object literal key). 
1261      * 
1262      * @param info The doc info for this property. 
1263      * @param lValue The l-value node. 
1264      * @param rValue The node that {@code n} is being initialized to, 
1265      *     or {@code null} if this is a stub declaration. 
1266      */ 
1267     private JSType getDeclaredType(String sourceName, JSDocInfo info, 
1268         Node lValue, @Nullable Node rValue) { 
1269       if (info != null && info.hasType()) { 
1270         return getDeclaredTypeInAnnotation(sourceName, lValue, info); 
1271       } else if (rValue != null && rValue.isFunction() && 
1272           shouldUseFunctionLiteralType( 
1273               JSType.toMaybeFunctionType(rValue.getJSType()), info, lValue)) { 
1274         return rValue.getJSType(); 
1275       } else if (info != null) { 
1276         if (info.hasEnumParameterType()) { 
1277           if (rValue != null && rValue.isObjectLit()) { 
1278             return rValue.getJSType(); 
1279           } else { 
1280             return createEnumTypeFromNodes( 
1281                 rValue, lValue.getQualifiedName(), info, lValue); 
1282           } 
1283         } else if (info.isConstructor() || info.isInterface()) { 
1284           return createFunctionTypeFromNodes( 
1285               rValue, lValue.getQualifiedName(), info, lValue); 
1286         } else { 
1287           // Check if this is constant, and if it has a known type. 
1288           if (info.isConstant()) { 
1289             JSType knownType = null; 
1290             if (rValue != null) { 
1291               JSDocInfo rValueInfo = rValue.getJSDocInfo(); 
1292               if (rValueInfo != null && rValueInfo.hasType()) { 
1293                 // If rValue has a type-cast, we use the type in the type-cast. 
1294                 return rValueInfo.getType().evaluate(scope, typeRegistry); 
1295               } else if (rValue.getJSType() != null 
1296                   && !rValue.getJSType().isUnknownType()) { 
1297                 // If rValue's type was already computed during scope creation, 
1298                 // then we can safely use that. 
1299                 return rValue.getJSType(); 
1300               } else if (rValue.isOr()) { 
1301                 // Check for a very specific JS idiom: 
1302                 // var x = x || TYPE; 
1303                 // This is used by Closure's base namespace for esoteric 
1304                 // reasons. 
1305                 Node firstClause = rValue.getFirstChild(); 
1306                 Node secondClause = firstClause.getNext(); 
1307                 boolean namesMatch = firstClause.isName() 
1308                     && lValue.isName() 
1309                     && firstClause.getString().equals(lValue.getString()); 
1310                 if (namesMatch && secondClause.getJSType() != null 
1311                     && !secondClause.getJSType().isUnknownType()) { 
1312                   return secondClause.getJSType(); 
1313                 } 
1314               } 
1315             } 
1316           } 
1317         } 
1318       } 
1319  
1320       return getDeclaredTypeInAnnotation(sourceName, lValue, info); 
1321     }

1340      * Because JS has no 'native' syntax for defining classes, 
1341      * this is often very coding-convention dependent and business-logic heavy. 
1342      */ 
1343     private void checkForClassDefiningCalls( 
1344         NodeTraversal t, Node n, Node parent) { 
1345       SubclassRelationship relationship = 
1346           codingConvention.getClassesDefinedByCall(n); 
1347       if (relationship != null) { 
1348         FunctionType superCtor = getFunctionType( 
1349             scope.getVar(relationship.superclassName)); 
1350         FunctionType subCtor = getFunctionType( 
1351             scope.getVar(relationship.subclassName)); 
1352         if (superCtor != null && superCtor.isConstructor() && 
1353             subCtor != null && subCtor.isConstructor()) { 
1354           ObjectType superClass = superCtor.getInstanceType(); 
1355           ObjectType subClass = subCtor.getInstanceType(); 
1356  
1357           // superCtor and subCtor might be structural constructors 
1358           // (like {function(new:Object)}) so we need to resolve them back 
1359           // to the original ctor objects. 
1360           superCtor = superClass.getConstructor(); 
1361           subCtor = subClass.getConstructor();

1364               !superClass.isEmptyType() && !subClass.isEmptyType()) { 
1365             validator.expectSuperType(t, n, superClass, subClass); 
1366           } 
1367

1369             codingConvention.applySubclassRelationship( 
1370                 superCtor, subCtor, relationship.type);

1372         } 
1373       }

1375       String singletonGetterClassName = 
1376           codingConvention.getSingletonGetterClassName(n); 
1377       if (singletonGetterClassName != null) { 
1378         ObjectType objectType = ObjectType.cast( 
1379             typeRegistry.getType(singletonGetterClassName)); 
1380         if (objectType != null) { 
1381           FunctionType functionType = objectType.getConstructor(); 
1382  
1383           if (functionType != null) { 
1384             FunctionType getterType = 
1385                 typeRegistry.createFunctionType(objectType); 
1386             codingConvention.applySingletonGetter(functionType, getterType, 
1387                 objectType); 
1388           } 
1389         } 
1390       }

1392       DelegateRelationship delegateRelationship = 
1393           codingConvention.getDelegateRelationship(n); 
1394       if (delegateRelationship != null) { 
1395         applyDelegateRelationship(delegateRelationship); 
1396       }

1398       ObjectLiteralCast objectLiteralCast = 
1399           codingConvention.getObjectLiteralCast(n); 
1400       if (objectLiteralCast != null) { 
1401         if (objectLiteralCast.diagnosticType == null) { 
1402           ObjectType type = ObjectType.cast( 
1403               typeRegistry.getType(objectLiteralCast.typeName)); 
1404           if (type != null && type.getConstructor() != null) { 
1405             setDeferredType(objectLiteralCast.objectNode, type); 
1406           } else { 
1407             compiler.report(JSError.make(t.getSourceName(), n, 
1408                     CONSTRUCTOR_EXPECTED)); 
1409           } 
1410         } else {

1412                   objectLiteralCast.diagnosticType)); 
1413         } 
1414       }

1458     /** 
1459      * Declare the symbol for a qualified name in the global scope. 
1460      * 
1461      * @param info The doc info for this property. 
1462      * @param n A top-level GETPROP node (it should not be contained inside 
1463      *     another GETPROP). 
1464      * @param parent The parent of {@code n}. 
1465      * @param rhsValue The node that {@code n} is being initialized to, 
1466      *     or {@code null} if this is a stub declaration. 
1467      */ 
1468     void maybeDeclareQualifiedName(NodeTraversal t, JSDocInfo info, 
1469         Node n, Node parent, Node rhsValue) { 
1470       Node ownerNode = n.getFirstChild(); 
1471       String ownerName = ownerNode.getQualifiedName(); 
1472       String qName = n.getQualifiedName(); 
1473       String propName = n.getLastChild().getString(); 
1474       Preconditions.checkArgument(qName != null && ownerName != null); 
1475  
1476       // Precedence of type information on GETPROPs: 
1477       // 1) @type annotation / @enum annotation 
1478       // 2) ASSIGN to FUNCTION literal 
1479       // 3) @param/@return annotation (with no function literal) 
1480       // 4) ASSIGN to something marked @const 
1481       // 5) ASSIGN to anything else 
1482       // 
1483       // 1, 3, and 4 are declarations, 5 is inferred, and 2 is a declaration iff 
1484       // the function has JsDoc or has not been declared before. 
1485       // 
1486       // FUNCTION literals are special because TypedScopeCreator is very smart 
1487       // about getting as much type information as possible for them. 
1488  
1489       // Determining type for #1 + #2 + #3 + #4 
1490       JSType valueType = getDeclaredType(t.getSourceName(), info, n, rhsValue); 
1491       if (valueType == null && rhsValue != null) { 
1492         // Determining type for #5 
1493         valueType = rhsValue.getJSType(); 
1494       } 
1495  
1496       // Function prototypes are special. 
1497       // It's a common JS idiom to do: 
1498       // F.prototype = { ... }; 
1499       // So if F does not have an explicitly declared super type, 
1500       // allow F.prototype to be redefined arbitrarily. 
1501       if ("prototype".equals(propName)) { 
1502         Var qVar = scope.getVar(qName); 
1503         if (qVar != null) { 
1504           // If the programmer has declared that F inherits from Super, 
1505           // and they assign F.prototype to an object literal, 
1506           // then they are responsible for making sure that the object literal's 
1507           // implicit prototype is set up appropriately. We just obey 
1508           // the @extends tag. 
1509           ObjectType qVarType = ObjectType.cast(qVar.getType()); 
1510           if (qVarType != null && 
1511               rhsValue != null && 
1512               rhsValue.isObjectLit()) { 
1513             typeRegistry.resetImplicitPrototype( 
1514                 rhsValue.getJSType(), qVarType.getImplicitPrototype()); 
1515           } else if (!qVar.isTypeInferred()) { 
1516             // If the programmer has declared that F inherits from Super, 
1517             // and they assign F.prototype to some arbitrary expression, 
1518             // there's not much we can do. We just ignore the expression, 
1519             // and hope they've annotated their code in a way to tell us 
1520             // what props are going to be on that prototype. 
1521             return; 
1522           } 
1523           if (qVar.getScope() == scope) { 
1524             scope.undeclare(qVar); 
1525           } 
1526         } 
1527       } 
1528  
1529       if (valueType == null) { 
1530         if (parent.isExprResult()) { 
1531           stubDeclarations.add(new StubDeclaration( 
1532               n, 
1533               t.getInput() != null && t.getInput().isExtern(), 
1534               ownerName)); 
1535         } 
1536  
1537         return; 
1538       } 
1539  
1540       boolean inferred = isQualifiedNameInferred( 
1541           qName, n, info, rhsValue, valueType); 
1542       if (!inferred) { 
1543         ObjectType ownerType = getObjectSlot(ownerName); 
1544         if (ownerType != null) { 
1545           // Only declare this as an official property if it has not been 
1546           // declared yet. 
1547           boolean isExtern = t.getInput() != null && t.getInput().isExtern(); 
1548           if ((!ownerType.hasOwnProperty(propName) || 
1549                ownerType.isPropertyTypeInferred(propName)) && 
1550               ((isExtern && !ownerType.isNativeObjectType()) || 
1551                !ownerType.isInstanceType())) { 
1552             // If the property is undeclared or inferred, declare it now. 
1553             ownerType.defineDeclaredProperty(propName, valueType, n); 
1554           } 
1555         } 
1556  
1557         // If the property is already declared, the error will be 
1558         // caught when we try to declare it in the current scope. 
1559         defineSlot(n, parent, valueType, inferred); 
1560       } else if (rhsValue != null && rhsValue.isTrue()) { 
1561         // We declare these for delegate proxy method properties. 
1562         FunctionType ownerType = 
1563             JSType.toMaybeFunctionType(getObjectSlot(ownerName)); 
1564         if (ownerType != null) { 
1565           JSType ownerTypeOfThis = ownerType.getTypeOfThis(); 
1566           String delegateName = codingConvention.getDelegateSuperclassName(); 
1567           JSType delegateType = delegateName == null ? 
1568               null : typeRegistry.getType(delegateName); 
1569           if (delegateType != null && 
1570               ownerTypeOfThis.isSubtype(delegateType)) { 
1571             defineSlot(n, parent, getNativeType(BOOLEAN_TYPE), true); 
1572           } 
1573         } 
1574       } 
1575     }

1577     /** 
1578      * Determines whether a qualified name is inferred. 
1579      * NOTE(nicksantos): Determining whether a property is declared or not 
1580      * is really really obnoxious. 
1581      * 
1582      * The problem is that there are two (equally valid) coding styles: 
1583      * 
1584      * (function() { 
1585      *   /* The authoritative definition of goog.bar. / 
1586      *   goog.bar = function() {}; 
1587      * })(); 
1588      * 
1589      * function f() { 
1590      *   goog.bar(); 
1591      *   /* Reset goog.bar to a no-op. / 
1592      *   goog.bar = function() {}; 
1593      * } 
1594      * 
1595      * In a dynamic language with first-class functions, it's very difficult 
1596      * to know which one the user intended without looking at lots of 
1597      * contextual information (the second example demonstrates a small case 
1598      * of this, but there are some really pathological cases as well). 
1599      * 
1600      * The current algorithm checks if either the declaration has 
1601      * JsDoc type information, or @const with a known type, 
1602      * or a function literal with a name we haven't seen before. 
1603      */ 
1604     private boolean isQualifiedNameInferred( 
1605         String qName, Node n, JSDocInfo info, 
1606         Node rhsValue, JSType valueType) { 
1607       if (valueType == null) { 
1608         return true; 
1609       } 
1610  
1611       boolean inferred = true; 
1612       if (info != null) { 
1613         inferred = !(info.hasType() 
1614             || info.hasEnumParameterType() 
1615             || (info.isConstant() && valueType != null 
1616                 && !valueType.isUnknownType()) 
1617             || FunctionTypeBuilder.isFunctionTypeDeclaration(info)); 
1618       } 
1619  
1620       if (inferred && rhsValue != null && rhsValue.isFunction()) { 
1621         if (info != null) { 
1622           return false; 
1623         } else if (!scope.isDeclared(qName, false) && 
1624             n.isUnscopedQualifiedName()) { 
1625  
1626           // Check if this is in a conditional block. 
1627           // Functions assigned in conditional blocks are inferred. 
1628           for (Node current = n.getParent(); 
1629                !(current.isScript() || current.isFunction()); 
1630                current = current.getParent()) { 
1631             if (NodeUtil.isControlStructure(current)) { 
1632               return true; 
1633             } 
1634           } 
1635  
1636           // Check if this is assigned in an inner scope. 
1637           // Functions assigned in inner scopes are inferred. 
1638           AstFunctionContents contents = 
1639               getFunctionAnalysisResults(scope.getRootNode()); 
1640           if (contents == null || 
1641               !contents.getEscapedQualifiedNames().contains(qName)) { 
1642             return false; 
1643           } 
1644         } 
1645       } 
1646       return inferred; 
1647     }

1770   /** 
1771    * A shallow traversal of the global scope to build up all classes, 
1772    * functions, and methods. 
1773    */ 
1774   private final class GlobalScopeBuilder extends AbstractScopeBuilder { 
1775  
1776     private GlobalScopeBuilder(Scope scope) { 
1777       super(scope); 
1778     } 
1779  
1780     /** 
1781      * Visit a node in the global scope, and add anything it declares to the 
1782      * global symbol table. 
1783      * 
1784      * @param t The current traversal. 
1785      * @param n The node being visited. 
1786      * @param parent The parent of n 
1787      */ 
1788     @Override public void visit(NodeTraversal t, Node n, Node parent) { 
1789       super.visit(t, n, parent); 
1790  
1791       switch (n.getType()) { 
1792  
1793         case Token.VAR: 
1794           // Handle typedefs. 
1795           if (n.hasOneChild()) { 
1796             checkForTypedef(t, n.getFirstChild(), n.getJSDocInfo()); 
1797           } 
1798           break; 
1799       } 
1800     } 
1801  
1802     @Override 
1803     void maybeDeclareQualifiedName( 
1804         NodeTraversal t, JSDocInfo info, 
1805         Node n, Node parent, Node rhsValue) { 
1806       checkForTypedef(t, n, info); 
1807       super.maybeDeclareQualifiedName(t, info, n, parent, rhsValue); 
1808     } 
1809  
1810     /** 
1811      * Handle typedefs. 
1812      * @param t The current traversal. 
1813      * @param candidate A qualified name node. 
1814      * @param info JSDoc comments. 
1815      */ 
1816     private void checkForTypedef( 
1817         NodeTraversal t, Node candidate, JSDocInfo info) { 
1818       if (info == null || !info.hasTypedefType()) { 
1819         return; 
1820       } 
1821  
1822       String typedef = candidate.getQualifiedName(); 
1823       if (typedef == null) { 
1824         return; 
1825       } 
1826  
1827       // TODO(nicksantos|user): This is a terrible, terrible hack 
1828       // to bail out on recursive typedefs. We'll eventually need 
1829       // to handle these properly. 
1830       typeRegistry.declareType(typedef, getNativeType(UNKNOWN_TYPE)); 
1831  
1832       JSType realType = info.getTypedefType().evaluate(scope, typeRegistry); 
1833       if (realType == null) { 
1834         compiler.report( 
1835             JSError.make( 
1836                 t.getSourceName(), candidate, MALFORMED_TYPEDEF, typedef)); 
1837       } 
1838  
1839       typeRegistry.overwriteDeclaredType(typedef, realType); 
1840       if (candidate.isGetProp()) { 
1841         defineSlot(candidate, candidate.getParent(), 
1842             getNativeType(NO_TYPE), false); 
1843       } 
1844     } 
1845   } // end GlobalScopeBuilder

1847   /** 
1848    * A shallow traversal of a local scope to find all arguments and 
1849    * local variables. 
1850    */ 
1851   private final class LocalScopeBuilder extends AbstractScopeBuilder { 
1852     /** 
1853      * @param scope The scope that we're building. 
1854      */ 
1855     private LocalScopeBuilder(Scope scope) { 
1856       super(scope); 
1857     } 
1858  
1859     /** 
1860      * Traverse the scope root and build it. 
1861      */ 
1862     void build() { 
1863       NodeTraversal.traverse(compiler, scope.getRootNode(), this); 
1864  
1865       AstFunctionContents contents = 
1866           getFunctionAnalysisResults(scope.getRootNode()); 
1867       if (contents != null) { 
1868         for (String varName : contents.getEscapedVarNames()) { 
1869           Var v = scope.getVar(varName); 
1870           Preconditions.checkState(v.getScope() == scope); 
1871           v.markEscaped(); 
1872         } 
1873       } 
1874     } 
1875  
1876     /** 
1877      * Visit a node in a local scope, and add any local variables or catch 
1878      * parameters into the local symbol table. 
1879      * 
1880      * @param t The node traversal. 
1881      * @param n The node being visited. 
1882      * @param parent The parent of n 
1883      */ 
1884     @Override public void visit(NodeTraversal t, Node n, Node parent) { 
1885       if (n == scope.getRootNode()) return; 
1886  
1887       if (n.isParamList() && parent == scope.getRootNode()) { 
1888         handleFunctionInputs(parent); 
1889         return; 
1890       } 
1891  
1892       super.visit(t, n, parent); 
1893     } 
1894  
1895     /** Handle bleeding functions and function parameters. */ 
1896     private void handleFunctionInputs(Node fnNode) { 
1897       // Handle bleeding functions. 
1898       Node fnNameNode = fnNode.getFirstChild(); 
1899       String fnName = fnNameNode.getString(); 
1900       if (!fnName.isEmpty()) { 
1901         Scope.Var fnVar = scope.getVar(fnName); 
1902         if (fnVar == null || 
1903             // Make sure we're not touching a native function. Native 
1904             // functions aren't bleeding, but may not have a declaration 
1905             // node. 
1906             (fnVar.getNameNode() != null && 
1907                 // Make sure that the function is actually bleeding by checking 
1908                 // if has already been declared. 
1909                 fnVar.getInitialValue() != fnNode)) { 
1910           defineSlot(fnNameNode, fnNode, fnNode.getJSType(), false); 
1911         } 
1912       } 
1913  
1914       declareArguments(fnNode); 
1915     } 
1916  
1917     /** 
1918      * Declares all of a function's arguments. 
1919      */ 
1920     private void declareArguments(Node functionNode) { 
1921       Node astParameters = functionNode.getFirstChild().getNext(); 
1922       Node body = astParameters.getNext(); 
1923       FunctionType functionType = 
1924           JSType.toMaybeFunctionType(functionNode.getJSType()); 
1925       if (functionType != null) { 
1926         Node jsDocParameters = functionType.getParametersNode(); 
1927         if (jsDocParameters != null) { 
1928           Node jsDocParameter = jsDocParameters.getFirstChild(); 
1929           for (Node astParameter : astParameters.children()) { 
1930             if (jsDocParameter != null) { 
1931               defineSlot(astParameter, functionNode, 
1932                   jsDocParameter.getJSType(), false); 
1933               jsDocParameter = jsDocParameter.getNext(); 
1934             } else { 
1935               defineSlot(astParameter, functionNode, null, true); 
1936             } 
1937           } 
1938         } 
1939       } 
1940     } // end declareArguments 
1941   } // end LocalScopeBuilder

1943   /** 
1944    * Does a first-order function analysis that just looks at simple things 
1945    * like what variables are escaped, and whether 'this' is used. 
1946    */ 
1947   private static class FirstOrderFunctionAnalyzer 
1948       extends AbstractScopedCallback implements CompilerPass { 
1949     private final AbstractCompiler compiler; 
1950     private final Map<Node, AstFunctionContents> data; 
1951  
1952     FirstOrderFunctionAnalyzer( 
1953         AbstractCompiler compiler, Map<Node, AstFunctionContents> outParam) { 
1954       this.compiler = compiler; 
1955       this.data = outParam; 
1956     } 
1957  
1958     @Override public void process(Node externs, Node root) { 
1959       if (externs == null) { 
1960         NodeTraversal.traverse(compiler, root, this); 
1961       } else { 
1962         NodeTraversal.traverseRoots( 
1963             compiler, ImmutableList.of(externs, root), this); 
1964       } 
1965     } 
1966  
1967     @Override public void enterScope(NodeTraversal t) { 
1968       if (!t.inGlobalScope()) { 
1969         Node n = t.getScopeRoot(); 
1970         data.put(n, new AstFunctionContents(n)); 
1971       } 
1972     } 
1973  
1974     @Override public void visit(NodeTraversal t, Node n, Node parent) { 
1975       if (t.inGlobalScope()) { 
1976         return; 
1977       } 
1978  
1979       if (n.isReturn() && n.getFirstChild() != null) { 
1980         data.get(t.getScopeRoot()).recordNonEmptyReturn(); 
1981       } 
1982  
1983       if (t.getScopeDepth() <= 2) { 
1984         // We only need to worry about escaped variables at depth 3. 
1985         // An variable escaped at depth 2 is, by definition, a global variable. 
1986         // We treat all global variables as escaped by default, so there's 
1987         // no reason to do this extra computation for them. 
1988         return; 
1989       } 
1990  
1991       if (n.isName() && NodeUtil.isLValue(n)) { 
1992         String name = n.getString(); 
1993         Scope scope = t.getScope(); 
1994         Var var = scope.getVar(name); 
1995         if (var != null) { 
1996           Scope ownerScope = var.getScope(); 
1997           if (scope != ownerScope && ownerScope.isLocal()) { 
1998             data.get(ownerScope.getRootNode()).recordEscapedVarName(name); 
1999           } 
2000         } 
2001       } else if (n.isGetProp() && n.isUnscopedQualifiedName() && 
2002           NodeUtil.isLValue(n)) { 
2003         String name = NodeUtil.getRootOfQualifiedName(n).getString(); 
2004         Scope scope = t.getScope(); 
2005         Var var = scope.getVar(name); 
2006         if (var != null) { 
2007           Scope ownerScope = var.getScope(); 
2008           if (scope != ownerScope && ownerScope.isLocal()) { 
2009             data.get(ownerScope.getRootNode()) 
2010                 .recordEscapedQualifiedName(n.getQualifiedName()); 
2011           } 
2012         } 
2013       } 
2014     } 
2015   }

2017   private AstFunctionContents getFunctionAnalysisResults(@Nullable Node n) { 
2018     if (n == null) { 
2019       return null; 
2020     } 
2021  
2022     // Sometimes this will return null in things like 
2023     // NameReferenceGraphConstruction that build partial scopes. 
2024     return functionAnalysisResults.get(n);
