 337   /** 
 338    * Main entry point for this phase of processing. This follows the pattern for 
 339    * JSCompiler phases. 
 340    * 
 341    * @param externsRoot The root of the externs parse tree. 
 342    * @param jsRoot The root of the input parse tree to be checked. 
 343    */ 
 344   @Override 
 345   public void process(Node externsRoot, Node jsRoot) { 
 346     Preconditions.checkNotNull(scopeCreator); 
 347     Preconditions.checkNotNull(topScope); 
 348  
 349     Node externsAndJs = jsRoot.getParent(); 
 350     Preconditions.checkState(externsAndJs != null); 
 351     Preconditions.checkState( 
 352         externsRoot == null || externsAndJs.hasChild(externsRoot)); 
 353  
 354     if (externsRoot != null) { 
 355       check(externsRoot, true); 
 356     } 
 357     check(jsRoot, false); 
 358   }

 360   /** Main entry point of this phase for testing code. */ 
 361   public Scope processForTesting(Node externsRoot, Node jsRoot) { 
 362     Preconditions.checkState(scopeCreator == null); 
 363     Preconditions.checkState(topScope == null); 
 364  
 365     Preconditions.checkState(jsRoot.getParent() != null); 
 366     Node externsAndJsRoot = jsRoot.getParent(); 
 367  
 368     scopeCreator = new MemoizedScopeCreator(new TypedScopeCreator(compiler)); 
 369     topScope = scopeCreator.createScope(externsAndJsRoot, null); 
 370  
 371     TypeInferencePass inference = new TypeInferencePass(compiler, 
 372         reverseInterpreter, topScope, scopeCreator); 
 373  
 374     inference.process(externsRoot, jsRoot); 
 375     process(externsRoot, jsRoot); 
 376  
 377     return topScope; 
 378   }

 381   public void check(Node node, boolean externs) { 
 382     Preconditions.checkNotNull(node); 
 383  
 384     NodeTraversal t = new NodeTraversal(compiler, this, scopeCreator); 
 385     inExterns = externs; 
 386     t.traverseWithScope(node, topScope); 
 387     if (externs) { 
 388       inferJSDocInfo.process(node, null); 
 389     } else { 
 390       inferJSDocInfo.process(null, node); 
 391     } 
 392   }

 395   private void checkNoTypeCheckSection(Node n, boolean enterSection) { 
 396     switch (n.getType()) { 
 397       case Token.SCRIPT: 
 398       case Token.BLOCK: 
 399       case Token.VAR: 
 400       case Token.FUNCTION: 
 401       case Token.ASSIGN: 
 402         JSDocInfo info = n.getJSDocInfo(); 
 403         if (info != null && info.isNoTypeCheck()) { 
 404           if (enterSection) { 
 405             noTypeCheckSection++; 
 406           } else { 
 407             noTypeCheckSection--; 
 408           } 
 409         } 
 410         validator.setShouldReport(noTypeCheckSection == 0); 
 411         break; 
 412     } 
 413   }

 418       t.report(n, diagnosticType, arguments); 
 419     }

 422   @Override 
 423   public boolean shouldTraverse( 
 424       NodeTraversal t, Node n, Node parent) { 
 425     checkNoTypeCheckSection(n, true); 
 426     switch (n.getType()) { 
 427       case Token.FUNCTION: 
 428         // normal type checking 
 429         final Scope outerScope = t.getScope(); 
 430         final String functionPrivateName = n.getFirstChild().getString(); 
 431         if (functionPrivateName != null && functionPrivateName.length() > 0 && 
 432             outerScope.isDeclared(functionPrivateName, false) && 
 433             // Ideally, we would want to check whether the type in the scope 
 434             // differs from the type being defined, but then the extern 
 435             // redeclarations of built-in types generates spurious warnings. 
 436             !(outerScope.getVar( 
 437                 functionPrivateName).getType() instanceof FunctionType)) { 
 438           report(t, n, FUNCTION_MASKS_VARIABLE, functionPrivateName); 
 439         } 
 440  
 441         // TODO(user): Only traverse the function's body. The function's 
 442         // name and arguments are traversed by the scope creator, and ideally 
 443         // should not be traversed by the type checker. 
 444         break; 
 445     } 
 446     return true; 
 447   }

 449   /** 
 450    * This is the meat of the type checking.  It is basically one big switch, 
 451    * with each case representing one type of parse tree node.  The individual 
 452    * cases are usually pretty straightforward. 
 453    * 
 454    * @param t The node traversal object that supplies context, such as the 
 455    * scope chain to use in name lookups as well as error reporting. 
 456    * @param n The node being visited. 
 457    * @param parent The parent of the node n. 
 458    */ 
 459   @Override 
 460   public void visit(NodeTraversal t, Node n, Node parent) { 
 461     JSType childType; 
 462     JSType leftType, rightType; 
 463     Node left, right; 
 464     // To be explicitly set to false if the node is not typeable. 
 465     boolean typeable = true; 
 466  
 467     switch (n.getType()) { 
 468       case Token.NAME: 
 469         typeable = visitName(t, n, parent); 
 470         break; 
 471  
 472       case Token.PARAM_LIST: 
 473         typeable = false; 
 474         break; 
 475  
 476       case Token.COMMA: 
 477         ensureTyped(t, n, getJSType(n.getLastChild())); 
 478         break; 
 479  
 480       case Token.TRUE: 
 481       case Token.FALSE: 
 482         ensureTyped(t, n, BOOLEAN_TYPE); 
 483         break; 
 484  
 485       case Token.THIS: 
 486         ensureTyped(t, n, t.getScope().getTypeOfThis()); 
 487         break; 
 488  
 489       case Token.NULL: 
 490         ensureTyped(t, n, NULL_TYPE); 
 491         break; 
 492  
 493       case Token.NUMBER: 
 494         ensureTyped(t, n, NUMBER_TYPE); 
 495         break; 
 496  
 497       case Token.STRING: 
 498         ensureTyped(t, n, STRING_TYPE); 
 499         break; 
 500  
 501       case Token.STRING_KEY: 
 502         typeable = false; 
 503         break; 
 504  
 505       case Token.GETTER_DEF: 
 506       case Token.SETTER_DEF: 
 507         // Object literal keys are handled with OBJECTLIT 
 508         break; 
 509  
 510       case Token.ARRAYLIT: 
 511         ensureTyped(t, n, ARRAY_TYPE); 
 512         break; 
 513  
 514       case Token.REGEXP: 
 515         ensureTyped(t, n, REGEXP_TYPE); 
 516         break; 
 517  
 518       case Token.GETPROP: 
 519         visitGetProp(t, n, parent); 
 520         typeable = !(parent.isAssign() && 
 521                      parent.getFirstChild() == n); 
 522         break; 
 523  
 524       case Token.GETELEM: 
 525         visitGetElem(t, n); 
 526         // The type of GETELEM is always unknown, so no point counting that. 
 527         // If that unknown leaks elsewhere (say by an assignment to another 
 528         // variable), then it will be counted. 
 529         typeable = false; 
 530         break; 
 531  
 532       case Token.VAR: 
 533         visitVar(t, n); 
 534         typeable = false; 
 535         break; 
 536  
 537       case Token.NEW: 
 538         visitNew(t, n); 
 539         break; 
 540  
 541       case Token.CALL: 
 542         visitCall(t, n); 
 543         typeable = !parent.isExprResult(); 
 544         break; 
 545  
 546       case Token.RETURN: 
 547         visitReturn(t, n); 
 548         typeable = false; 
 549         break; 
 550  
 551       case Token.DEC: 
 552       case Token.INC: 
 553         left = n.getFirstChild(); 
 554         validator.expectNumber( 
 555             t, left, getJSType(left), "increment/decrement"); 
 556         ensureTyped(t, n, NUMBER_TYPE); 
 557         break; 
 558  
 559       case Token.NOT: 
 560         ensureTyped(t, n, BOOLEAN_TYPE); 
 561         break; 
 562  
 563       case Token.VOID: 
 564         ensureTyped(t, n, VOID_TYPE); 
 565         break; 
 566  
 567       case Token.TYPEOF: 
 568         ensureTyped(t, n, STRING_TYPE); 
 569         break; 
 570  
 571       case Token.BITNOT: 
 572         childType = getJSType(n.getFirstChild()); 
 573         if (!childType.matchesInt32Context()) { 
 574           report(t, n, BIT_OPERATION, NodeUtil.opToStr(n.getType()), 
 575               childType.toString()); 
 576         } 
 577         ensureTyped(t, n, NUMBER_TYPE); 
 578         break; 
 579  
 580       case Token.POS: 
 581       case Token.NEG: 
 582         left = n.getFirstChild(); 
 583         validator.expectNumber(t, left, getJSType(left), "sign operator"); 
 584         ensureTyped(t, n, NUMBER_TYPE); 
 585         break; 
 586  
 587       case Token.EQ: 
 588       case Token.NE: 
 589       case Token.SHEQ: 
 590       case Token.SHNE: { 
 591         left = n.getFirstChild(); 
 592         right = n.getLastChild(); 
 593  
 594         if (left.isTypeOf()) { 
 595           if (right.isString()) { 
 596             checkTypeofString(t, right, right.getString()); 
 597           } 
 598         } else if (right.isTypeOf() && left.isString()) { 
 599           checkTypeofString(t, left, left.getString()); 
 600         } 
 601  
 602         leftType = getJSType(left); 
 603         rightType = getJSType(right); 
 604  
 605         // We do not want to warn about explicit comparisons to VOID. People 
 606         // often do this if they think their type annotations screwed up. 
 607         // 
 608         // We do want to warn about cases where people compare things like 
 609         // (Array|null) == (Function|null) 
 610         // because it probably means they screwed up. 
 611         // 
 612         // This heuristic here is not perfect, but should catch cases we 
 613         // care about without too many false negatives. 
 614         JSType leftTypeRestricted = leftType.restrictByNotNullOrUndefined(); 
 615         JSType rightTypeRestricted = rightType.restrictByNotNullOrUndefined(); 
 616  
 617         TernaryValue result = TernaryValue.UNKNOWN; 
 618         if (n.getType() == Token.EQ || n.getType() == Token.NE) { 
 619           result = leftTypeRestricted.testForEquality(rightTypeRestricted); 
 620           if (n.isNE()) { 
 621             result = result.not(); 
 622           } 
 623         } else { 
 624           // SHEQ or SHNE 
 625           if (!leftTypeRestricted.canTestForShallowEqualityWith( 
 626                   rightTypeRestricted)) { 
 627             result = n.getType() == Token.SHEQ ? 
 628                 TernaryValue.FALSE : TernaryValue.TRUE; 
 629           } 
 630         } 
 631  
 632         if (result != TernaryValue.UNKNOWN) { 
 633           report(t, n, DETERMINISTIC_TEST, leftType.toString(), 
 634               rightType.toString(), result.toString()); 
 635         } 
 636         ensureTyped(t, n, BOOLEAN_TYPE); 
 637         break; 
 638       } 
 639  
 640       case Token.LT: 
 641       case Token.LE: 
 642       case Token.GT: 
 643       case Token.GE: 
 644         leftType = getJSType(n.getFirstChild()); 
 645         rightType = getJSType(n.getLastChild()); 
 646         if (rightType.isNumber()) { 
 647           validator.expectNumber( 
 648               t, n, leftType, "left side of numeric comparison"); 
 649         } else if (leftType.isNumber()) { 
 650           validator.expectNumber( 
 651               t, n, rightType, "right side of numeric comparison"); 
 652         } else if (leftType.matchesNumberContext() && 
 653                    rightType.matchesNumberContext()) { 
 654           // OK. 
 655         } else { 
 656           // Whether the comparison is numeric will be determined at runtime 
 657           // each time the expression is evaluated. Regardless, both operands 
 658           // should match a string context. 
 659           String message = "left side of comparison"; 
 660           validator.expectString(t, n, leftType, message); 
 661           validator.expectNotNullOrUndefined( 
 662               t, n, leftType, message, getNativeType(STRING_TYPE)); 
 663           message = "right side of comparison"; 
 664           validator.expectString(t, n, rightType, message); 
 665           validator.expectNotNullOrUndefined( 
 666               t, n, rightType, message, getNativeType(STRING_TYPE)); 
 667         } 
 668         ensureTyped(t, n, BOOLEAN_TYPE); 
 669         break; 
 670  
 671       case Token.IN: 
 672         validator.expectObject(t, n, getJSType(n.getLastChild()), 
 673                                "'in' requires an object"); 
 674         left = n.getFirstChild(); 
 675         validator.expectString(t, left, getJSType(left), "left side of 'in'"); 
 676         ensureTyped(t, n, BOOLEAN_TYPE); 
 677         break; 
 678  
 679       case Token.INSTANCEOF: 
 680         left = n.getFirstChild(); 
 681         right = n.getLastChild(); 
 682         rightType = getJSType(right).restrictByNotNullOrUndefined(); 
 683         validator.expectAnyObject( 
 684             t, left, getJSType(left), "deterministic instanceof yields false"); 
 685         validator.expectActualObject( 
 686             t, right, rightType, "instanceof requires an object"); 
 687         ensureTyped(t, n, BOOLEAN_TYPE); 
 688         break; 
 689  
 690       case Token.ASSIGN: 
 691         visitAssign(t, n); 
 692         typeable = false; 
 693         break; 
 694  
 695       case Token.ASSIGN_LSH: 
 696       case Token.ASSIGN_RSH: 
 697       case Token.ASSIGN_URSH: 
 698       case Token.ASSIGN_DIV: 
 699       case Token.ASSIGN_MOD: 
 700       case Token.ASSIGN_BITOR: 
 701       case Token.ASSIGN_BITXOR: 
 702       case Token.ASSIGN_BITAND: 
 703       case Token.ASSIGN_SUB: 
 704       case Token.ASSIGN_ADD: 
 705       case Token.ASSIGN_MUL: 
 706       case Token.LSH: 
 707       case Token.RSH: 
 708       case Token.URSH: 
 709       case Token.DIV: 
 710       case Token.MOD: 
 711       case Token.BITOR: 
 712       case Token.BITXOR: 
 713       case Token.BITAND: 
 714       case Token.SUB: 
 715       case Token.ADD: 
 716       case Token.MUL: 
 717         visitBinaryOperator(n.getType(), t, n); 
 718         break; 
 719  
 720       case Token.DELPROP: 
 721         ensureTyped(t, n, BOOLEAN_TYPE); 
 722         break; 
 723  
 724       case Token.CASE: 
 725         JSType switchType = getJSType(parent.getFirstChild()); 
 726         JSType caseType = getJSType(n.getFirstChild()); 
 727         validator.expectSwitchMatchesCase(t, n, switchType, caseType); 
 728         typeable = false; 
 729         break; 
 730  
 731       case Token.WITH: { 
 732         Node child = n.getFirstChild(); 
 733         childType = getJSType(child); 
 734         validator.expectObject(t, child, childType, "with requires an object"); 
 735         typeable = false; 
 736         break; 
 737       } 
 738  
 739       case Token.FUNCTION: 
 740         visitFunction(t, n); 
 741         break; 
 742  
 743       // These nodes have no interesting type behavior. 
 744       case Token.LABEL: 
 745       case Token.LABEL_NAME: 
 746       case Token.SWITCH: 
 747       case Token.BREAK: 
 748       case Token.CATCH: 
 749       case Token.TRY: 
 750       case Token.SCRIPT: 
 751       case Token.EXPR_RESULT: 
 752       case Token.BLOCK: 
 753       case Token.EMPTY: 
 754       case Token.DEFAULT_CASE: 
 755       case Token.CONTINUE: 
 756       case Token.DEBUGGER: 
 757       case Token.THROW: 
 758         typeable = false; 
 759         break; 
 760  
 761       // These nodes require data flow analysis. 
 762       case Token.DO: 
 763       case Token.FOR: 
 764       case Token.IF: 
 765       case Token.WHILE: 
 766         typeable = false; 
 767         break; 
 768  
 769       // These nodes are typed during the type inference. 
 770       case Token.AND: 
 771       case Token.HOOK: 
 772       case Token.OBJECTLIT: 
 773       case Token.OR: 
 774         if (n.getJSType() != null) { // If we didn't run type inference. 
 775           ensureTyped(t, n); 
 776         } else { 
 777           // If this is an enum, then give that type to the objectlit as well. 
 778           if ((n.isObjectLit()) 
 779               && (parent.getJSType() instanceof EnumType)) { 
 780             ensureTyped(t, n, parent.getJSType()); 
 781           } else { 
 782             ensureTyped(t, n); 
 783           } 
 784         } 
 785         if (n.isObjectLit()) { 
 786           for (Node key : n.children()) { 
 787             visitObjLitKey(t, key, n); 
 788           } 
 789         } 
 790         break; 
 791  
 792       default: 
 793         report(t, n, UNEXPECTED_TOKEN, Token.name(n.getType())); 
 794         ensureTyped(t, n); 
 795         break; 
 796     } 
 797  
 798     // Don't count externs since the user's code may not even use that part. 
 799     typeable = typeable && !inExterns; 
 800  
 801     if (typeable) { 
 802       doPercentTypedAccounting(t, n); 
 803     } 
 804  
 805     checkNoTypeCheckSection(n, false); 
 806   }

 835   /** 
 836    * Visits an assignment <code>lvalue = rvalue</code>. If the 
 837    * <code>lvalue</code> is a prototype modification, we change the schema 
 838    * of the object type it is referring to. 
 839    * @param t the traversal 
 840    * @param assign the assign node 
 841    * (<code>assign.isAssign()</code> is an implicit invariant) 
 842    */ 
 843   private void visitAssign(NodeTraversal t, Node assign) { 
 844     JSDocInfo info = assign.getJSDocInfo(); 
 845     Node lvalue = assign.getFirstChild(); 
 846     Node rvalue = assign.getLastChild(); 
 847  
 848     // Check property sets to 'object.property' when 'object' is known. 
 849     if (lvalue.isGetProp()) { 
 850       Node object = lvalue.getFirstChild(); 
 851       JSType objectJsType = getJSType(object); 
 852       String property = lvalue.getLastChild().getString(); 
 853  
 854       // the first name in this getprop refers to an interface 
 855       // we perform checks in addition to the ones below 
 856       if (object.isGetProp()) { 
 857         JSType jsType = getJSType(object.getFirstChild()); 
 858         if (jsType.isInterface() && 
 859             object.getLastChild().getString().equals("prototype")) { 
 860           visitInterfaceGetprop(t, assign, object, property, lvalue, rvalue); 
 861         } 
 862       } 
 863  
 864       checkEnumAlias(t, info, rvalue); 
 865  
 866       // Prototype assignments are special, because they actually affect 
 867       // the definition of a class. These are mostly validated 
 868       // during TypedScopeCreator, and we only look for the "dumb" cases here. 
 869       // object.prototype = ...; 
 870       if (property.equals("prototype")) { 
 871         if (objectJsType != null && objectJsType.isFunctionType()) { 
 872           FunctionType functionType = objectJsType.toMaybeFunctionType(); 
 873           if (functionType.isConstructor()) { 
 874             JSType rvalueType = rvalue.getJSType(); 
 875             validator.expectObject(t, rvalue, rvalueType, 
 876                 OVERRIDING_PROTOTYPE_WITH_NON_OBJECT); 
 877             return; 
 878           } 
 879         } 
 880       } 
 881  
 882       // The generic checks for 'object.property' when 'object' is known, 
 883       // and 'property' is declared on it. 
 884       // object.property = ...; 
 885       ObjectType type = ObjectType.cast( 
 886           objectJsType.restrictByNotNullOrUndefined()); 
 887       if (type != null) { 
 888         if (type.hasProperty(property) && 
 889             !type.isPropertyTypeInferred(property) && 
 890             !propertyIsImplicitCast(type, property)) { 
 891           JSType expectedType = type.getPropertyType(property); 
 892           if (!expectedType.isUnknownType()) { 
 893             validator.expectCanAssignToPropertyOf( 
 894                 t, assign, getJSType(rvalue), 
 895                 expectedType, object, property); 
 896             checkPropertyInheritanceOnGetpropAssign( 
 897                 t, assign, object, property, info, expectedType); 
 898             return; 
 899           } 
 900         } 
 901       } 
 902  
 903       // If we couldn't get the property type with normal object property 
 904       // lookups, then check inheritance anyway with the unknown type. 
 905       checkPropertyInheritanceOnGetpropAssign( 
 906           t, assign, object, property, info, getNativeType(UNKNOWN_TYPE)); 
 907     } 
 908  
 909     // Check qualified name sets to 'object' and 'object.property'. 
 910     // This can sometimes handle cases when the type of 'object' is not known. 
 911     // e.g., 
 912     // var obj = createUnknownType(); 
 913     // /** @type {number} */ obj.foo = true; 
 914     JSType leftType = getJSType(lvalue); 
 915     if (lvalue.isQualifiedName()) { 
 916       // variable with inferred type case 
 917       JSType rvalueType = getJSType(assign.getLastChild()); 
 918       Var var = t.getScope().getVar(lvalue.getQualifiedName()); 
 919       if (var != null) { 
 920         if (var.isTypeInferred()) { 
 921           return; 
 922         } 
 923  
 924         if (NodeUtil.getRootOfQualifiedName(lvalue).isThis() && 
 925             t.getScope() != var.getScope()) { 
 926           // Don't look at "this.foo" variables from other scopes. 
 927           return; 
 928         } 
 929  
 930         if (var.getType() != null) { 
 931           leftType = var.getType(); 
 932         } 
 933       } 
 934     } 
 935  
 936     // Fall through case for arbitrary LHS and arbitrary RHS. 
 937     Node rightChild = assign.getLastChild(); 
 938     JSType rightType = getJSType(rightChild); 
 939     if (validator.expectCanAssignTo( 
 940             t, assign, rightType, leftType, "assignment")) { 
 941       ensureTyped(t, assign, rightType); 
 942     } else { 
 943       ensureTyped(t, assign); 
 944     } 
 945   }

 947   private void checkPropertyInheritanceOnGetpropAssign( 
 948       NodeTraversal t, Node assign, Node object, String property, 
 949       JSDocInfo info, JSType propertyType) { 
 950     // Inheritance checks for prototype properties. 
 951     // 
 952     // TODO(nicksantos): This isn't the right place to do this check. We 
 953     // really want to do this when we're looking at the constructor. 
 954     // We'd find all its properties and make sure they followed inheritance 
 955     // rules, like we currently do for @implements to make sure 
 956     // all the methods are implemented. 
 957     // 
 958     // As-is, this misses many other ways to override a property. 
 959     // 
 960     // object.prototype.property = ...; 
 961     if (object.isGetProp()) { 
 962       Node object2 = object.getFirstChild(); 
 963       String property2 = NodeUtil.getStringValue(object.getLastChild()); 
 964  
 965       if ("prototype".equals(property2)) { 
 966         JSType jsType = getJSType(object2); 
 967         if (jsType.isFunctionType()) { 
 968           FunctionType functionType = jsType.toMaybeFunctionType(); 
 969           if (functionType.isConstructor() || functionType.isInterface()) { 
 970             checkDeclaredPropertyInheritance( 
 971                 t, assign, functionType, property, info, propertyType); 
 972           } 
 973         } 
 974       } 
 975     } 
 976   }

 978   /** 
 979    * Visits an object literal field definition <code>key : value</code>. 
 980    * 
 981    * If the <code>lvalue</code> is a prototype modification, we change the 
 982    * schema of the object type it is referring to. 
 983    * 
 984    * @param t the traversal 
 985    * @param key the assign node 
 986    */ 
 987   private void visitObjLitKey(NodeTraversal t, Node key, Node objlit) { 
 988     // Do not validate object lit value types in externs. We don't really care, 
 989     // and it makes it easier to generate externs. 
 990     if (objlit.isFromExterns()) { 
 991       ensureTyped(t, key); 
 992       return; 
 993     } 
 994  
 995     // TODO(johnlenz): Validate get and set function declarations are valid 
 996     // as is the functions can have "extraneous" bits. 
 997  
 998     // For getter and setter property definitions the 
 999     // r-value type != the property type. 
1000     Node rvalue = key.getFirstChild(); 
1001     JSType rightType = NodeUtil.getObjectLitKeyTypeFromValueType( 
1002         key, getJSType(rvalue)); 
1003     if (rightType == null) { 
1004       rightType = getNativeType(UNKNOWN_TYPE); 
1005     } 
1006  
1007     Node owner = objlit; 
1008  
1009     // Validate value is assignable to the key type. 
1010  
1011     JSType keyType = getJSType(key); 
1012  
1013     JSType allowedValueType = keyType; 
1014     if (allowedValueType.isEnumElementType()) { 
1015       allowedValueType = 
1016           allowedValueType.toMaybeEnumElementType().getPrimitiveType(); 
1017     } 
1018  
1019     boolean valid = validator.expectCanAssignToPropertyOf(t, key, 
1020         rightType, allowedValueType, 
1021         owner, NodeUtil.getObjectLitKeyName(key)); 
1022     if (valid) { 
1023       ensureTyped(t, key, rightType); 
1024     } else { 
1025       ensureTyped(t, key); 
1026     } 
1027  
1028     // Validate that the key type is assignable to the object property type. 
1029     // This is necessary as the objlit may have been cast to a non-literal 
1030     // object type. 
1031     // TODO(johnlenz): consider introducing a CAST node to the AST (or 
1032     // perhaps a parentheses node). 
1033  
1034     JSType objlitType = getJSType(objlit); 
1035     ObjectType type = ObjectType.cast( 
1036         objlitType.restrictByNotNullOrUndefined()); 
1037     if (type != null) { 
1038       String property = NodeUtil.getObjectLitKeyName(key); 
1039       if (type.hasProperty(property) && 
1040           !type.isPropertyTypeInferred(property) && 
1041           !propertyIsImplicitCast(type, property)) { 
1042         validator.expectCanAssignToPropertyOf( 
1043             t, key, keyType, 
1044             type.getPropertyType(property), owner, property); 
1045       } 
1046       return; 
1047     } 
1048   }

1064   /** 
1065    * Given a constructor type and a property name, check that the property has 
1066    * the JSDoc annotation @override iff the property is declared on a 
1067    * superclass. Several checks regarding inheritance correctness are also 
1068    * performed. 
1069    */ 
1070   private void checkDeclaredPropertyInheritance( 
1071       NodeTraversal t, Node n, FunctionType ctorType, String propertyName, 
1072       JSDocInfo info, JSType propertyType) { 
1073     // If the supertype doesn't resolve correctly, we've warned about this 
1074     // already. 
1075     if (hasUnknownOrEmptySupertype(ctorType)) { 
1076       return; 
1077     } 
1078  
1079     FunctionType superClass = ctorType.getSuperClassConstructor(); 
1080     boolean superClassHasProperty = superClass != null && 
1081         superClass.getInstanceType().hasProperty(propertyName); 
1082     boolean superClassHasDeclaredProperty = superClass != null && 
1083         superClass.getInstanceType().isPropertyTypeDeclared(propertyName); 
1084  
1085     // For interface 
1086     boolean superInterfaceHasProperty = false; 
1087     boolean superInterfaceHasDeclaredProperty = false; 
1088     if (ctorType.isInterface()) { 
1089       for (ObjectType interfaceType : ctorType.getExtendedInterfaces()) { 
1090         superInterfaceHasProperty = 
1091             superInterfaceHasProperty || 
1092             interfaceType.hasProperty(propertyName); 
1093         superInterfaceHasDeclaredProperty = 
1094             superInterfaceHasDeclaredProperty || 
1095             interfaceType.isPropertyTypeDeclared(propertyName); 
1096       } 
1097     } 
1098     boolean declaredOverride = info != null && info.isOverride(); 
1099  
1100     boolean foundInterfaceProperty = false; 
1101     if (ctorType.isConstructor()) { 
1102       for (JSType implementedInterface : 
1103           ctorType.getAllImplementedInterfaces()) { 
1104         if (implementedInterface.isUnknownType() || 
1105             implementedInterface.isEmptyType()) { 
1106           continue; 
1107         } 
1108         FunctionType interfaceType = 
1109             implementedInterface.toObjectType().getConstructor(); 
1110         Preconditions.checkNotNull(interfaceType); 
1111  
1112         boolean interfaceHasProperty = 
1113             interfaceType.getPrototype().hasProperty(propertyName); 
1114         foundInterfaceProperty = foundInterfaceProperty || 
1115             interfaceHasProperty; 
1116         if (reportMissingOverride.isOn() 
1117             && !declaredOverride 
1118             && interfaceHasProperty) { 
1119           // @override not present, but the property does override an interface 
1120           // property 
1121           compiler.report(t.makeError(n, reportMissingOverride, 
1122               HIDDEN_INTERFACE_PROPERTY, propertyName, 
1123               interfaceType.getTopMostDefiningType(propertyName).toString())); 
1124         } 
1125       } 
1126     } 
1127  
1128     if (!declaredOverride 
1129         && !superClassHasProperty 
1130         && !superInterfaceHasProperty) { 
1131       // nothing to do here, it's just a plain new property 
1132       return; 
1133     } 
1134  
1135     ObjectType topInstanceType = superClassHasDeclaredProperty ? 
1136         superClass.getTopMostDefiningType(propertyName) : null; 
1137     boolean declaredLocally = 
1138         ctorType.isConstructor() && 
1139         (ctorType.getPrototype().hasOwnProperty(propertyName) || 
1140          ctorType.getInstanceType().hasOwnProperty(propertyName)); 
1141     if (reportMissingOverride.isOn() 
1142         && !declaredOverride 
1143         && superClassHasDeclaredProperty 
1144         && declaredLocally) { 
1145       // @override not present, but the property does override a superclass 
1146       // property 
1147       compiler.report(t.makeError(n, reportMissingOverride, 
1148           HIDDEN_SUPERCLASS_PROPERTY, propertyName, 
1149           topInstanceType.toString())); 
1150     } 
1151  
1152     // @override is present and we have to check that it is ok 
1153     if (superClassHasDeclaredProperty) { 
1154       // there is a superclass implementation 
1155       JSType superClassPropType = 
1156           superClass.getInstanceType().getPropertyType(propertyName); 
1157       if (!propertyType.canAssignTo(superClassPropType)) { 
1158         compiler.report( 
1159             t.makeError(n, HIDDEN_SUPERCLASS_PROPERTY_MISMATCH, 
1160                 propertyName, topInstanceType.toString(), 
1161                 superClassPropType.toString(), propertyType.toString())); 
1162       } 
1163     } else if (superInterfaceHasDeclaredProperty) { 
1164       // there is an super interface property 
1165       for (ObjectType interfaceType : ctorType.getExtendedInterfaces()) { 
1166         if (interfaceType.hasProperty(propertyName)) { 
1167           JSType superPropertyType = 
1168               interfaceType.getPropertyType(propertyName); 
1169           if (!propertyType.canAssignTo(superPropertyType)) { 
1170             topInstanceType = interfaceType.getConstructor(). 
1171                 getTopMostDefiningType(propertyName); 
1172             compiler.report( 
1173                 t.makeError(n, HIDDEN_SUPERCLASS_PROPERTY_MISMATCH, 
1174                     propertyName, topInstanceType.toString(), 
1175                     superPropertyType.toString(), 
1176                     propertyType.toString())); 
1177           } 
1178         } 
1179       } 
1180     } else if (!foundInterfaceProperty 
1181         && !superClassHasProperty 
1182         && !superInterfaceHasProperty) { 
1183       // there is no superclass nor interface implementation 
1184       compiler.report( 
1185           t.makeError(n, UNKNOWN_OVERRIDE, 
1186               propertyName, ctorType.getInstanceType().toString())); 
1187     } 
1188   }

1190   /** 
1191    * Given a constructor or an interface type, find out whether the unknown 
1192    * type is a supertype of the current type. 
1193    */ 
1194   private static boolean hasUnknownOrEmptySupertype(FunctionType ctor) { 
1195     Preconditions.checkArgument(ctor.isConstructor() || ctor.isInterface()); 
1196     Preconditions.checkArgument(!ctor.isUnknownType()); 
1197  
1198     // The type system should notice inheritance cycles on its own 
1199     // and break the cycle. 
1200     while (true) { 
1201       ObjectType maybeSuperInstanceType = 
1202           ctor.getPrototype().getImplicitPrototype(); 
1203       if (maybeSuperInstanceType == null) { 
1204         return false; 
1205       } 
1206       if (maybeSuperInstanceType.isUnknownType() || 
1207           maybeSuperInstanceType.isEmptyType()) { 
1208         return true; 
1209       } 
1210       ctor = maybeSuperInstanceType.getConstructor(); 
1211       if (ctor == null) { 
1212         return false; 
1213       } 
1214       Preconditions.checkState(ctor.isConstructor() || ctor.isInterface()); 
1215     } 
1216   }

1255   /** 
1256    * Visits a NAME node. 
1257    * 
1258    * @param t The node traversal object that supplies context, such as the 
1259    * scope chain to use in name lookups as well as error reporting. 
1260    * @param n The node being visited. 
1261    * @param parent The parent of the node n. 
1262    * @return whether the node is typeable or not 
1263    */ 
1264   boolean visitName(NodeTraversal t, Node n, Node parent) { 
1265     // At this stage, we need to determine whether this is a leaf 
1266     // node in an expression (which therefore needs to have a type 
1267     // assigned for it) versus some other decorative node that we 
1268     // can safely ignore.  Function names, arguments (children of LP nodes) and 
1269     // variable declarations are ignored. 
1270     // TODO(user): remove this short-circuiting in favor of a 
1271     // pre order traversal of the FUNCTION, CATCH, LP and VAR nodes. 
1272     int parentNodeType = parent.getType(); 
1273     if (parentNodeType == Token.FUNCTION || 
1274         parentNodeType == Token.CATCH || 
1275         parentNodeType == Token.PARAM_LIST || 
1276         parentNodeType == Token.VAR) { 
1277       return false; 
1278     } 
1279  
1280     JSType type = n.getJSType(); 
1281     if (type == null) { 
1282       type = getNativeType(UNKNOWN_TYPE); 
1283       Var var = t.getScope().getVar(n.getString()); 
1284       if (var != null) { 
1285         JSType varType = var.getType(); 
1286         if (varType != null) { 
1287           type = varType; 
1288         } 
1289       } 
1290     } 
1291     ensureTyped(t, n, type); 
1292     return true; 
1293   }

1296    * Visits a GETPROP node. 
1297    * 
1298    * @param t The node traversal object that supplies context, such as the 
1299    * scope chain to use in name lookups as well as error reporting. 
1300    * @param n The node being visited. 
1301    * @param parent The parent of <code>n</code> 
1302    */ 
1303   private void visitGetProp(NodeTraversal t, Node n, Node parent) { 
1304     // obj.prop or obj.method() 
1305     // Lots of types can appear on the left, a call to a void function can 
1306     // never be on the left. getPropertyType will decide what is acceptable 
1307     // and what isn't. 
1308     Node property = n.getLastChild(); 
1309     Node objNode = n.getFirstChild(); 
1310     JSType childType = getJSType(objNode); 
1311  
1312     if (childType.isDict()) { 
1313       report(t, property, TypeValidator.ILLEGAL_PROPERTY_ACCESS, "'.'", "dict"); 
1314     } else if (n.getJSType() != null && parent.isAssign()) { 
1315       return; 
1316     } else if (validator.expectNotNullOrUndefined(t, n, childType, 
1317         "No properties on this expression", getNativeType(OBJECT_TYPE))) { 
1318       checkPropertyAccess(childType, property.getString(), t, n); 
1319     } 
1320     ensureTyped(t, n); 
1321   }

1323   /** 
1324    * Emit a warning if we can prove that a property cannot possibly be 
1325    * defined on an object. Note the difference between JS and a strictly 
1326    * statically typed language: we're checking if the property 
1327    * *cannot be defined*, whereas a java compiler would check if the 
1328    * property *can be undefined*. 
1329    */ 
1330   private void checkPropertyAccess(JSType childType, String propName, 
1331       NodeTraversal t, Node n) { 
1332     // If the property type is unknown, check the object type to see if it 
1333     // can ever be defined. We explicitly exclude CHECKED_UNKNOWN (for 
1334     // properties where we've checked that it exists, or for properties on 
1335     // objects that aren't in this binary). 
1336     JSType propType = getJSType(n); 
1337     if (propType.isEquivalentTo(typeRegistry.getNativeType(UNKNOWN_TYPE))) { 
1338       childType = childType.autobox(); 
1339       ObjectType objectType = ObjectType.cast(childType); 
1340       if (objectType != null) { 
1341         // We special-case object types so that checks on enums can be 
1342         // much stricter, and so that we can use hasProperty (which is much 
1343         // faster in most cases). 
1344         if (!objectType.hasProperty(propName) || 
1345             objectType.isEquivalentTo( 
1346                 typeRegistry.getNativeType(UNKNOWN_TYPE))) { 
1347           if (objectType instanceof EnumType) { 
1348             report(t, n, INEXISTENT_ENUM_ELEMENT, propName); 
1349           } else { 
1350             checkPropertyAccessHelper(objectType, propName, t, n); 
1351           } 
1352         } 
1353  
1354       } else { 
1355         checkPropertyAccessHelper(childType, propName, t, n); 
1356       } 
1357     } 
1358   } 
1359  
1360   private void checkPropertyAccessHelper(JSType objectType, String propName, 
1361       NodeTraversal t, Node n) { 
1362     if (!objectType.isEmptyType() && 
1363         reportMissingProperties && !isPropertyTest(n)) { 
1364       if (!typeRegistry.canPropertyBeDefined(objectType, propName)) { 
1365         report(t, n, INEXISTENT_PROPERTY, propName, 
1366             validator.getReadableJSTypeName(n.getFirstChild(), true)); 
1367       } 
1368     } 
1369   }

1371   /** 
1372    * Determines whether this node is testing for the existence of a property. 
1373    * If true, we will not emit warnings about a missing property. 
1374    * 
1375    * @param getProp The GETPROP being tested. 
1376    */ 
1377   private boolean isPropertyTest(Node getProp) { 
1378     Node parent = getProp.getParent(); 
1379     switch (parent.getType()) { 
1380       case Token.CALL: 
1381         return parent.getFirstChild() != getProp && 
1382             compiler.getCodingConvention().isPropertyTestFunction(parent); 
1383  
1384       case Token.IF: 
1385       case Token.WHILE: 
1386       case Token.DO: 
1387       case Token.FOR: 
1388         return NodeUtil.getConditionExpression(parent) == getProp; 
1389  
1390       case Token.INSTANCEOF: 
1391       case Token.TYPEOF: 
1392         return true; 
1393  
1394       case Token.AND: 
1395       case Token.HOOK: 
1396         return parent.getFirstChild() == getProp; 
1397  
1398       case Token.NOT: 
1399         return parent.getParent().isOr() && 
1400             parent.getParent().getFirstChild() == parent; 
1401     } 
1402     return false; 
1403   }

1428     // merge to fix this.

1476   /** 
1477    * Check whether there's any property conflict for for a particular super 
1478    * interface 
1479    * @param t The node traversal object that supplies context 
1480    * @param n The node being visited 
1481    * @param functionName The function name being checked 
1482    * @param properties The property names in the super interfaces that have 
1483    * been visited 
1484    * @param currentProperties The property names in the super interface 
1485    * that have been visited 
1486    * @param interfaceType The super interface that is being visited 
1487    */ 
1488   private void checkInterfaceConflictProperties(NodeTraversal t, Node n, 
1489       String functionName, HashMap<String, ObjectType> properties, 
1490       HashMap<String, ObjectType> currentProperties, 
1491       ObjectType interfaceType) { 
1492     Set<String> currentPropertyNames = interfaceType.getPropertyNames(); 
1493     for (String name : currentPropertyNames) { 
1494       ObjectType oType = properties.get(name); 
1495       if (oType != null) { 
1496         if (!interfaceType.getPropertyType(name).isEquivalentTo( 
1497             oType.getPropertyType(name))) { 
1498           compiler.report( 
1499               t.makeError(n, INCOMPATIBLE_EXTENDED_PROPERTY_TYPE, 
1500                   functionName, name, oType.toString(), 
1501                   interfaceType.toString())); 
1502         } 
1503       } 
1504       currentProperties.put(name, interfaceType); 
1505     } 
1506     for (ObjectType iType : interfaceType.getCtorExtendedInterfaces()) { 
1507       checkInterfaceConflictProperties(t, n, functionName, properties, 
1508           currentProperties, iType); 
1509     } 
1510   }

1512   /** 
1513    * Visits a {@link Token#FUNCTION} node. 
1514    * 
1515    * @param t The node traversal object that supplies context, such as the 
1516    * scope chain to use in name lookups as well as error reporting. 
1517    * @param n The node being visited. 
1518    */ 
1519   private void visitFunction(NodeTraversal t, Node n) { 
1520     FunctionType functionType = JSType.toMaybeFunctionType(n.getJSType()); 
1521     String functionPrivateName = n.getFirstChild().getString(); 
1522     if (functionType.isConstructor()) { 
1523       FunctionType baseConstructor = functionType.getSuperClassConstructor(); 
1524       if (baseConstructor != getNativeType(OBJECT_FUNCTION_TYPE) && 
1525           baseConstructor != null && 
1526           baseConstructor.isInterface() && functionType.isConstructor()) { 
1527         compiler.report( 
1528             t.makeError(n, CONFLICTING_EXTENDED_TYPE, 
1529                         "constructor", functionPrivateName)); 
1530       } else { 
1531         if (baseConstructor != getNativeType(OBJECT_FUNCTION_TYPE) && 
1532             baseConstructor != null) { 
1533           if (functionType.makesStructs() && !baseConstructor.makesStructs()) { 
1534             compiler.report(t.makeError(n, CONFLICTING_EXTENDED_TYPE, 
1535                                         "struct", functionPrivateName)); 
1536           } else if (functionType.makesDicts() && 
1537                      !baseConstructor.makesDicts()) { 
1538             compiler.report(t.makeError(n, CONFLICTING_EXTENDED_TYPE, 
1539                                         "dict", functionPrivateName)); 
1540           } 
1541         } 
1542         // All interfaces are properly implemented by a class 
1543         for (JSType baseInterface : functionType.getImplementedInterfaces()) { 
1544           boolean badImplementedType = false; 
1545           ObjectType baseInterfaceObj = ObjectType.cast(baseInterface); 
1546           if (baseInterfaceObj != null) { 
1547             FunctionType interfaceConstructor = 
1548               baseInterfaceObj.getConstructor(); 
1549             if (interfaceConstructor != null && 
1550                 !interfaceConstructor.isInterface()) { 
1551               badImplementedType = true; 
1552             } 
1553           } else { 
1554             badImplementedType = true; 
1555           } 
1556           if (badImplementedType) { 
1557             report(t, n, BAD_IMPLEMENTED_TYPE, functionPrivateName); 
1558           } 
1559         } 
1560         // check properties 
1561         validator.expectAllInterfaceProperties(t, n, functionType); 
1562       } 
1563     } else if (functionType.isInterface()) { 
1564       // Interface must extend only interfaces 
1565       for (ObjectType extInterface : functionType.getExtendedInterfaces()) { 
1566         if (extInterface.getConstructor() != null 
1567             && !extInterface.getConstructor().isInterface()) { 
1568           compiler.report( 
1569               t.makeError(n, CONFLICTING_EXTENDED_TYPE, 
1570                           "interface", functionPrivateName)); 
1571         } 
1572       } 
1573       // Interface cannot implement any interfaces 
1574       if (functionType.hasImplementedInterfaces()) { 
1575         compiler.report(t.makeError(n, 
1576             CONFLICTING_IMPLEMENTED_TYPE, functionPrivateName)); 
1577       } 
1578       // Check whether the extended interfaces have any conflicts 
1579       if (functionType.getExtendedInterfacesCount() > 1) { 
1580         // Only check when extending more than one interfaces 
1581         HashMap<String, ObjectType> properties 
1582             = new HashMap<String, ObjectType>(); 
1583         HashMap<String, ObjectType> currentProperties 
1584             = new HashMap<String, ObjectType>(); 
1585         for (ObjectType interfaceType : functionType.getExtendedInterfaces()) { 
1586           currentProperties.clear(); 
1587           checkInterfaceConflictProperties(t, n, functionPrivateName, 
1588               properties, currentProperties, interfaceType); 
1589           properties.putAll(currentProperties); 
1590         } 
1591       } 
1592     } 
1593   }

1595   /** 
1596    * Visits a CALL node. 
1597    * 
1598    * @param t The node traversal object that supplies context, such as the 
1599    * scope chain to use in name lookups as well as error reporting. 
1600    * @param n The node being visited. 
1601    */ 
1602   private void visitCall(NodeTraversal t, Node n) { 
1603     Node child = n.getFirstChild(); 
1604     JSType childType = getJSType(child).restrictByNotNullOrUndefined(); 
1605  
1606     if (!childType.canBeCalled()) { 
1607       report(t, n, NOT_CALLABLE, childType.toString()); 
1608       ensureTyped(t, n); 
1609       return; 
1610     } 
1611  
1612     // A couple of types can be called as if they were functions. 
1613     // If it is a function type, then validate parameters. 
1614     if (childType.isFunctionType()) { 
1615       FunctionType functionType = childType.toMaybeFunctionType(); 
1616  
1617       boolean isExtern = false; 
1618       JSDocInfo functionJSDocInfo = functionType.getJSDocInfo(); 
1619       if( functionJSDocInfo != null  && 
1620           functionJSDocInfo.getAssociatedNode() != null) { 
1621         isExtern = functionJSDocInfo.getAssociatedNode().isFromExterns(); 
1622       } 
1623  
1624       // Non-native constructors should not be called directly 
1625       // unless they specify a return type and are defined 
1626       // in an extern. 
1627       if (functionType.isConstructor() && 
1628           !functionType.isNativeObjectType() && 
1629           (functionType.getReturnType().isUnknownType() || 
1630            functionType.getReturnType().isVoidType() || 
1631            !isExtern)) { 
1632         report(t, n, CONSTRUCTOR_NOT_CALLABLE, childType.toString()); 
1633       } 
1634  
1635       // Functions with explicit 'this' types must be called in a GETPROP 
1636       // or GETELEM. 
1637       if (functionType.isOrdinaryFunction() && 
1638           !functionType.getTypeOfThis().isUnknownType() && 
1639           !functionType.getTypeOfThis().isNativeObjectType() && 
1640           !(child.isGetElem() || 
1641             child.isGetProp())) { 
1642         report(t, n, EXPECTED_THIS_TYPE, functionType.toString()); 
1643       } 
1644  
1645       visitParameterList(t, n, functionType); 
1646       ensureTyped(t, n, functionType.getReturnType()); 
1647     } else { 
1648       ensureTyped(t, n); 
1649     } 
1650  
1651     // TODO: Add something to check for calls of RegExp objects, which is not 
1652     // supported by IE.  Either say something about the return type or warn 
1653     // about the non-portability of the call or both. 
1654   }

1656   /** 
1657    * Visits the parameters of a CALL or a NEW node. 
1658    */ 
1659   private void visitParameterList(NodeTraversal t, Node call, 
1660       FunctionType functionType) { 
1661     Iterator<Node> arguments = call.children().iterator(); 
1662     arguments.next(); // skip the function name 
1663  
1664     Iterator<Node> parameters = functionType.getParameters().iterator(); 
1665     int ordinal = 0; 
1666     Node parameter = null; 
1667     Node argument = null; 
1668     while (arguments.hasNext() && 
1669            (parameters.hasNext() || 
1670             parameter != null && parameter.isVarArgs())) { 
1671       // If there are no parameters left in the list, then the while loop 
1672       // above implies that this must be a var_args function. 
1673       if (parameters.hasNext()) { 
1674         parameter = parameters.next(); 
1675       } 
1676       argument = arguments.next(); 
1677       ordinal++; 
1678  
1679       validator.expectArgumentMatchesParameter(t, argument, 
1680           getJSType(argument), getJSType(parameter), call, ordinal); 
1681     } 
1682  
1683     int numArgs = call.getChildCount() - 1; 
1684     int minArgs = functionType.getMinArguments(); 
1685     int maxArgs = functionType.getMaxArguments(); 
1686     if (minArgs > numArgs || maxArgs < numArgs) { 
1687       report(t, call, WRONG_ARGUMENT_COUNT, 
1688               validator.getReadableJSTypeName(call.getFirstChild(), false), 
1689               String.valueOf(numArgs), String.valueOf(minArgs), 
1690               maxArgs != Integer.MAX_VALUE ? 
1691               " and no more than " + maxArgs + " argument(s)" : ""); 
1692     } 
1693   }

1707  
1708       JSType returnType = functionType.getReturnType();

1728           "inconsistent return type");

1732   /** 
1733    * This function unifies the type checking involved in the core binary 
1734    * operators and the corresponding assignment operators.  The representation 
1735    * used internally is such that common code can handle both kinds of 
1736    * operators easily. 
1737    * 
1738    * @param op The operator. 
1739    * @param t The traversal object, needed to report errors. 
1740    * @param n The node being checked. 
1741    */ 
1742   private void visitBinaryOperator(int op, NodeTraversal t, Node n) { 
1743     Node left = n.getFirstChild(); 
1744     JSType leftType = getJSType(left); 
1745     Node right = n.getLastChild(); 
1746     JSType rightType = getJSType(right); 
1747     switch (op) { 
1748       case Token.ASSIGN_LSH: 
1749       case Token.ASSIGN_RSH: 
1750       case Token.LSH: 
1751       case Token.RSH: 
1752       case Token.ASSIGN_URSH: 
1753       case Token.URSH: 
1754         if (!leftType.matchesInt32Context()) { 
1755           report(t, left, BIT_OPERATION, 
1756                    NodeUtil.opToStr(n.getType()), leftType.toString()); 
1757         } 
1758         if (!rightType.matchesUint32Context()) { 
1759           report(t, right, BIT_OPERATION, 
1760                    NodeUtil.opToStr(n.getType()), rightType.toString()); 
1761         } 
1762         break; 
1763  
1764       case Token.ASSIGN_DIV: 
1765       case Token.ASSIGN_MOD: 
1766       case Token.ASSIGN_MUL: 
1767       case Token.ASSIGN_SUB: 
1768       case Token.DIV: 
1769       case Token.MOD: 
1770       case Token.MUL: 
1771       case Token.SUB: 
1772         validator.expectNumber(t, left, leftType, "left operand"); 
1773         validator.expectNumber(t, right, rightType, "right operand"); 
1774         break; 
1775  
1776       case Token.ASSIGN_BITAND: 
1777       case Token.ASSIGN_BITXOR: 
1778       case Token.ASSIGN_BITOR: 
1779       case Token.BITAND: 
1780       case Token.BITXOR: 
1781       case Token.BITOR: 
1782         validator.expectBitwiseable(t, left, leftType, 
1783             "bad left operand to bitwise operator"); 
1784         validator.expectBitwiseable(t, right, rightType, 
1785             "bad right operand to bitwise operator"); 
1786         break; 
1787  
1788       case Token.ASSIGN_ADD: 
1789       case Token.ADD: 
1790         break; 
1791  
1792       default: 
1793         report(t, n, UNEXPECTED_TOKEN, Token.name(op)); 
1794     } 
1795     ensureTyped(t, n); 
1796   }

1867   /** 
1868    * Enforces type casts, and ensures the node is typed. 
1869    * 
1870    * A cast in the way that we use it in JSDoc annotations never 
1871    * alters the generated code and therefore never can induce any runtime 
1872    * operation. What this means is that a 'cast' is really just a compile 
1873    * time constraint on the underlying value. In the future, we may add 
1874    * support for run-time casts for compiled tests. 
1875    * 
1876    * To ensure some shred of sanity, we enforce the notion that the 
1877    * type you are casting to may only meaningfully be a narrower type 
1878    * than the underlying declared type. We also invalidate optimizations 
1879    * on bad type casts. 
1880    * 
1881    * @param t The traversal object needed to report errors. 
1882    * @param n The node getting a type assigned to it. 
1883    * @param type The type to be assigned. 
1884    */ 
1885   private void ensureTyped(NodeTraversal t, Node n, JSType type) { 
1886     // Make sure FUNCTION nodes always get function type. 
1887     Preconditions.checkState(!n.isFunction() || 
1888             type.isFunctionType() || 
1889             type.isUnknownType()); 
1890     JSDocInfo info = n.getJSDocInfo(); 
1891     if (info != null) { 
1892       if (info.hasType()) { 
1893         JSType infoType = info.getType().evaluate(t.getScope(), typeRegistry); 
1894         validator.expectCanCast(t, n, infoType, type); 
1895         type = infoType; 
1896       } 
1897  
1898       if (info.isImplicitCast() && !inExterns) { 
1899         String propName = n.isGetProp() ? 
1900             n.getLastChild().getString() : "(missing)"; 
1901         compiler.report( 
1902             t.makeError(n, ILLEGAL_IMPLICIT_CAST, propName)); 
1903       } 
1904     } 
1905  
1906     if (n.getJSType() == null) { 
1907       n.setJSType(type); 
1908     }

 422   @Override 
 423   public boolean shouldTraverse( 
 424       NodeTraversal t, Node n, Node parent) { 
 425     checkNoTypeCheckSection(n, true); 
 426     switch (n.getType()) { 
 427       case Token.FUNCTION: 
 428         // normal type checking 
 429         final Scope outerScope = t.getScope(); 
 430         final String functionPrivateName = n.getFirstChild().getString(); 
 431         if (functionPrivateName != null && functionPrivateName.length() > 0 && 
 432             outerScope.isDeclared(functionPrivateName, false) && 
 433             // Ideally, we would want to check whether the type in the scope 
 434             // differs from the type being defined, but then the extern 
 435             // redeclarations of built-in types generates spurious warnings. 
 436             !(outerScope.getVar( 
 437                 functionPrivateName).getType() instanceof FunctionType)) { 
 438           report(t, n, FUNCTION_MASKS_VARIABLE, functionPrivateName); 
 439         } 
 440  
 441         // TODO(user): Only traverse the function's body. The function's 
 442         // name and arguments are traversed by the scope creator, and ideally 
 443         // should not be traversed by the type checker. 
 444         break; 
 445     } 
 446     return true;

 449   /** 
 450    * This is the meat of the type checking.  It is basically one big switch, 
 451    * with each case representing one type of parse tree node.  The individual 
 452    * cases are usually pretty straightforward. 
 453    * 
 454    * @param t The node traversal object that supplies context, such as the 
 455    * scope chain to use in name lookups as well as error reporting. 
 456    * @param n The node being visited. 
 457    * @param parent The parent of the node n. 
 458    */ 
 459   @Override 
 460   public void visit(NodeTraversal t, Node n, Node parent) { 
 461     JSType childType; 
 462     JSType leftType, rightType; 
 463     Node left, right; 
 464     // To be explicitly set to false if the node is not typeable. 
 465     boolean typeable = true; 
 466  
 467     switch (n.getType()) { 
 468       case Token.NAME: 
 469         typeable = visitName(t, n, parent); 
 470         break; 
 471  
 472       case Token.PARAM_LIST: 
 473         typeable = false; 
 474         break; 
 475  
 476       case Token.COMMA: 
 477         ensureTyped(t, n, getJSType(n.getLastChild())); 
 478         break; 
 479  
 480       case Token.TRUE: 
 481       case Token.FALSE: 
 482         ensureTyped(t, n, BOOLEAN_TYPE); 
 483         break; 
 484  
 485       case Token.THIS: 
 486         ensureTyped(t, n, t.getScope().getTypeOfThis()); 
 487         break; 
 488  
 489       case Token.NULL: 
 490         ensureTyped(t, n, NULL_TYPE); 
 491         break; 
 492  
 493       case Token.NUMBER: 
 494         ensureTyped(t, n, NUMBER_TYPE); 
 495         break; 
 496  
 497       case Token.STRING: 
 498         ensureTyped(t, n, STRING_TYPE); 
 499         break; 
 500  
 501       case Token.STRING_KEY: 
 502         typeable = false; 
 503         break; 
 504  
 505       case Token.GETTER_DEF: 
 506       case Token.SETTER_DEF: 
 507         // Object literal keys are handled with OBJECTLIT 
 508         break; 
 509  
 510       case Token.ARRAYLIT: 
 511         ensureTyped(t, n, ARRAY_TYPE); 
 512         break; 
 513  
 514       case Token.REGEXP: 
 515         ensureTyped(t, n, REGEXP_TYPE); 
 516         break; 
 517  
 518       case Token.GETPROP: 
 519         visitGetProp(t, n, parent); 
 520         typeable = !(parent.isAssign() && 
 521                      parent.getFirstChild() == n); 
 522         break; 
 523  
 524       case Token.GETELEM: 
 525         visitGetElem(t, n); 
 526         // The type of GETELEM is always unknown, so no point counting that. 
 527         // If that unknown leaks elsewhere (say by an assignment to another 
 528         // variable), then it will be counted. 
 529         typeable = false; 
 530         break; 
 531  
 532       case Token.VAR: 
 533         visitVar(t, n); 
 534         typeable = false; 
 535         break; 
 536  
 537       case Token.NEW: 
 538         visitNew(t, n); 
 539         break; 
 540  
 541       case Token.CALL: 
 542         visitCall(t, n); 
 543         typeable = !parent.isExprResult(); 
 544         break; 
 545  
 546       case Token.RETURN: 
 547         visitReturn(t, n); 
 548         typeable = false; 
 549         break; 
 550  
 551       case Token.DEC: 
 552       case Token.INC: 
 553         left = n.getFirstChild(); 
 554         validator.expectNumber( 
 555             t, left, getJSType(left), "increment/decrement"); 
 556         ensureTyped(t, n, NUMBER_TYPE); 
 557         break; 
 558  
 559       case Token.NOT: 
 560         ensureTyped(t, n, BOOLEAN_TYPE); 
 561         break; 
 562  
 563       case Token.VOID: 
 564         ensureTyped(t, n, VOID_TYPE); 
 565         break; 
 566  
 567       case Token.TYPEOF: 
 568         ensureTyped(t, n, STRING_TYPE); 
 569         break; 
 570  
 571       case Token.BITNOT: 
 572         childType = getJSType(n.getFirstChild()); 
 573         if (!childType.matchesInt32Context()) { 
 574           report(t, n, BIT_OPERATION, NodeUtil.opToStr(n.getType()), 
 575               childType.toString()); 
 576         } 
 577         ensureTyped(t, n, NUMBER_TYPE); 
 578         break; 
 579  
 580       case Token.POS: 
 581       case Token.NEG: 
 582         left = n.getFirstChild(); 
 583         validator.expectNumber(t, left, getJSType(left), "sign operator"); 
 584         ensureTyped(t, n, NUMBER_TYPE); 
 585         break; 
 586  
 587       case Token.EQ: 
 588       case Token.NE: 
 589       case Token.SHEQ: 
 590       case Token.SHNE: { 
 591         left = n.getFirstChild(); 
 592         right = n.getLastChild(); 
 593  
 594         if (left.isTypeOf()) { 
 595           if (right.isString()) { 
 596             checkTypeofString(t, right, right.getString()); 
 597           } 
 598         } else if (right.isTypeOf() && left.isString()) { 
 599           checkTypeofString(t, left, left.getString()); 
 600         } 
 601  
 602         leftType = getJSType(left); 
 603         rightType = getJSType(right); 
 604  
 605         // We do not want to warn about explicit comparisons to VOID. People 
 606         // often do this if they think their type annotations screwed up. 
 607         // 
 608         // We do want to warn about cases where people compare things like 
 609         // (Array|null) == (Function|null) 
 610         // because it probably means they screwed up. 
 611         // 
 612         // This heuristic here is not perfect, but should catch cases we 
 613         // care about without too many false negatives. 
 614         JSType leftTypeRestricted = leftType.restrictByNotNullOrUndefined(); 
 615         JSType rightTypeRestricted = rightType.restrictByNotNullOrUndefined(); 
 616  
 617         TernaryValue result = TernaryValue.UNKNOWN; 
 618         if (n.getType() == Token.EQ || n.getType() == Token.NE) { 
 619           result = leftTypeRestricted.testForEquality(rightTypeRestricted); 
 620           if (n.isNE()) { 
 621             result = result.not(); 
 622           } 
 623         } else { 
 624           // SHEQ or SHNE 
 625           if (!leftTypeRestricted.canTestForShallowEqualityWith( 
 626                   rightTypeRestricted)) { 
 627             result = n.getType() == Token.SHEQ ? 
 628                 TernaryValue.FALSE : TernaryValue.TRUE; 
 629           } 
 630         } 
 631  
 632         if (result != TernaryValue.UNKNOWN) { 
 633           report(t, n, DETERMINISTIC_TEST, leftType.toString(), 
 634               rightType.toString(), result.toString()); 
 635         } 
 636         ensureTyped(t, n, BOOLEAN_TYPE); 
 637         break; 
 638       } 
 639  
 640       case Token.LT: 
 641       case Token.LE: 
 642       case Token.GT: 
 643       case Token.GE: 
 644         leftType = getJSType(n.getFirstChild()); 
 645         rightType = getJSType(n.getLastChild()); 
 646         if (rightType.isNumber()) { 
 647           validator.expectNumber( 
 648               t, n, leftType, "left side of numeric comparison"); 
 649         } else if (leftType.isNumber()) { 
 650           validator.expectNumber( 
 651               t, n, rightType, "right side of numeric comparison"); 
 652         } else if (leftType.matchesNumberContext() && 
 653                    rightType.matchesNumberContext()) { 
 654           // OK. 
 655         } else { 
 656           // Whether the comparison is numeric will be determined at runtime 
 657           // each time the expression is evaluated. Regardless, both operands 
 658           // should match a string context. 
 659           String message = "left side of comparison"; 
 660           validator.expectString(t, n, leftType, message); 
 661           validator.expectNotNullOrUndefined( 
 662               t, n, leftType, message, getNativeType(STRING_TYPE)); 
 663           message = "right side of comparison"; 
 664           validator.expectString(t, n, rightType, message); 
 665           validator.expectNotNullOrUndefined( 
 666               t, n, rightType, message, getNativeType(STRING_TYPE)); 
 667         } 
 668         ensureTyped(t, n, BOOLEAN_TYPE); 
 669         break; 
 670  
 671       case Token.IN: 
 672         validator.expectObject(t, n, getJSType(n.getLastChild()), 
 673                                "'in' requires an object"); 
 674         left = n.getFirstChild(); 
 675         validator.expectString(t, left, getJSType(left), "left side of 'in'"); 
 676         ensureTyped(t, n, BOOLEAN_TYPE); 
 677         break; 
 678  
 679       case Token.INSTANCEOF: 
 680         left = n.getFirstChild(); 
 681         right = n.getLastChild(); 
 682         rightType = getJSType(right).restrictByNotNullOrUndefined(); 
 683         validator.expectAnyObject( 
 684             t, left, getJSType(left), "deterministic instanceof yields false"); 
 685         validator.expectActualObject( 
 686             t, right, rightType, "instanceof requires an object"); 
 687         ensureTyped(t, n, BOOLEAN_TYPE); 
 688         break; 
 689  
 690       case Token.ASSIGN: 
 691         visitAssign(t, n); 
 692         typeable = false; 
 693         break; 
 694  
 695       case Token.ASSIGN_LSH: 
 696       case Token.ASSIGN_RSH: 
 697       case Token.ASSIGN_URSH: 
 698       case Token.ASSIGN_DIV: 
 699       case Token.ASSIGN_MOD: 
 700       case Token.ASSIGN_BITOR: 
 701       case Token.ASSIGN_BITXOR: 
 702       case Token.ASSIGN_BITAND: 
 703       case Token.ASSIGN_SUB: 
 704       case Token.ASSIGN_ADD: 
 705       case Token.ASSIGN_MUL: 
 706       case Token.LSH: 
 707       case Token.RSH: 
 708       case Token.URSH: 
 709       case Token.DIV: 
 710       case Token.MOD: 
 711       case Token.BITOR: 
 712       case Token.BITXOR: 
 713       case Token.BITAND: 
 714       case Token.SUB: 
 715       case Token.ADD: 
 716       case Token.MUL: 
 717         visitBinaryOperator(n.getType(), t, n); 
 718         break; 
 719  
 720       case Token.DELPROP: 
 721         ensureTyped(t, n, BOOLEAN_TYPE); 
 722         break; 
 723  
 724       case Token.CASE: 
 725         JSType switchType = getJSType(parent.getFirstChild()); 
 726         JSType caseType = getJSType(n.getFirstChild()); 
 727         validator.expectSwitchMatchesCase(t, n, switchType, caseType); 
 728         typeable = false; 
 729         break; 
 730  
 731       case Token.WITH: { 
 732         Node child = n.getFirstChild(); 
 733         childType = getJSType(child); 
 734         validator.expectObject(t, child, childType, "with requires an object"); 
 735         typeable = false; 
 736         break; 
 737       } 
 738  
 739       case Token.FUNCTION: 
 740         visitFunction(t, n); 
 741         break; 
 742  
 743       // These nodes have no interesting type behavior. 
 744       case Token.LABEL: 
 745       case Token.LABEL_NAME: 
 746       case Token.SWITCH: 
 747       case Token.BREAK: 
 748       case Token.CATCH: 
 749       case Token.TRY: 
 750       case Token.SCRIPT: 
 751       case Token.EXPR_RESULT: 
 752       case Token.BLOCK: 
 753       case Token.EMPTY: 
 754       case Token.DEFAULT_CASE: 
 755       case Token.CONTINUE: 
 756       case Token.DEBUGGER: 
 757       case Token.THROW: 
 758         typeable = false; 
 759         break; 
 760  
 761       // These nodes require data flow analysis. 
 762       case Token.DO: 
 763       case Token.FOR: 
 764       case Token.IF: 
 765       case Token.WHILE: 
 766         typeable = false; 
 767         break; 
 768  
 769       // These nodes are typed during the type inference. 
 770       case Token.AND: 
 771       case Token.HOOK: 
 772       case Token.OBJECTLIT: 
 773       case Token.OR: 
 774         if (n.getJSType() != null) { // If we didn't run type inference. 
 775           ensureTyped(t, n); 
 776         } else { 
 777           // If this is an enum, then give that type to the objectlit as well. 
 778           if ((n.isObjectLit()) 
 779               && (parent.getJSType() instanceof EnumType)) { 
 780             ensureTyped(t, n, parent.getJSType()); 
 781           } else { 
 782             ensureTyped(t, n); 
 783           } 
 784         } 
 785         if (n.isObjectLit()) { 
 786           for (Node key : n.children()) { 
 787             visitObjLitKey(t, key, n); 
 788           } 
 789         } 
 790         break; 
 791  
 792       default: 
 793         report(t, n, UNEXPECTED_TOKEN, Token.name(n.getType())); 
 794         ensureTyped(t, n); 
 795         break; 
 796     } 
 797  
 798     // Don't count externs since the user's code may not even use that part. 
 799     typeable = typeable && !inExterns; 
 800  
 801     if (typeable) { 
 802       doPercentTypedAccounting(t, n); 
 803     } 
 804  
 805     checkNoTypeCheckSection(n, false); 
 806   }

 835   /** 
 836    * Visits an assignment <code>lvalue = rvalue</code>. If the 
 837    * <code>lvalue</code> is a prototype modification, we change the schema 
 838    * of the object type it is referring to. 
 839    * @param t the traversal 
 840    * @param assign the assign node 
 841    * (<code>assign.isAssign()</code> is an implicit invariant) 
 842    */ 
 843   private void visitAssign(NodeTraversal t, Node assign) { 
 844     JSDocInfo info = assign.getJSDocInfo(); 
 845     Node lvalue = assign.getFirstChild(); 
 846     Node rvalue = assign.getLastChild(); 
 847  
 848     // Check property sets to 'object.property' when 'object' is known. 
 849     if (lvalue.isGetProp()) { 
 850       Node object = lvalue.getFirstChild(); 
 851       JSType objectJsType = getJSType(object); 
 852       String property = lvalue.getLastChild().getString(); 
 853  
 854       // the first name in this getprop refers to an interface 
 855       // we perform checks in addition to the ones below 
 856       if (object.isGetProp()) { 
 857         JSType jsType = getJSType(object.getFirstChild()); 
 858         if (jsType.isInterface() && 
 859             object.getLastChild().getString().equals("prototype")) { 
 860           visitInterfaceGetprop(t, assign, object, property, lvalue, rvalue); 
 861         } 
 862       } 
 863  
 864       checkEnumAlias(t, info, rvalue); 
 865  
 866       // Prototype assignments are special, because they actually affect 
 867       // the definition of a class. These are mostly validated 
 868       // during TypedScopeCreator, and we only look for the "dumb" cases here. 
 869       // object.prototype = ...; 
 870       if (property.equals("prototype")) { 
 871         if (objectJsType != null && objectJsType.isFunctionType()) { 
 872           FunctionType functionType = objectJsType.toMaybeFunctionType(); 
 873           if (functionType.isConstructor()) { 
 874             JSType rvalueType = rvalue.getJSType(); 
 875             validator.expectObject(t, rvalue, rvalueType, 
 876                 OVERRIDING_PROTOTYPE_WITH_NON_OBJECT); 
 877             return; 
 878           } 
 879         } 
 880       } 
 881  
 882       // The generic checks for 'object.property' when 'object' is known, 
 883       // and 'property' is declared on it. 
 884       // object.property = ...; 
 885       ObjectType type = ObjectType.cast( 
 886           objectJsType.restrictByNotNullOrUndefined()); 
 887       if (type != null) { 
 888         if (type.hasProperty(property) && 
 889             !type.isPropertyTypeInferred(property) && 
 890             !propertyIsImplicitCast(type, property)) { 
 891           JSType expectedType = type.getPropertyType(property); 
 892           if (!expectedType.isUnknownType()) { 
 893             validator.expectCanAssignToPropertyOf( 
 894                 t, assign, getJSType(rvalue), 
 895                 expectedType, object, property); 
 896             checkPropertyInheritanceOnGetpropAssign( 
 897                 t, assign, object, property, info, expectedType); 
 898             return; 
 899           } 
 900         } 
 901       } 
 902  
 903       // If we couldn't get the property type with normal object property 
 904       // lookups, then check inheritance anyway with the unknown type. 
 905       checkPropertyInheritanceOnGetpropAssign( 
 906           t, assign, object, property, info, getNativeType(UNKNOWN_TYPE)); 
 907     } 
 908  
 909     // Check qualified name sets to 'object' and 'object.property'. 
 910     // This can sometimes handle cases when the type of 'object' is not known. 
 911     // e.g., 
 912     // var obj = createUnknownType(); 
 913     // /** @type {number} */ obj.foo = true; 
 914     JSType leftType = getJSType(lvalue); 
 915     if (lvalue.isQualifiedName()) { 
 916       // variable with inferred type case 
 917       JSType rvalueType = getJSType(assign.getLastChild()); 
 918       Var var = t.getScope().getVar(lvalue.getQualifiedName()); 
 919       if (var != null) { 
 920         if (var.isTypeInferred()) { 
 921           return; 
 922         } 
 923  
 924         if (NodeUtil.getRootOfQualifiedName(lvalue).isThis() && 
 925             t.getScope() != var.getScope()) { 
 926           // Don't look at "this.foo" variables from other scopes. 
 927           return; 
 928         } 
 929  
 930         if (var.getType() != null) { 
 931           leftType = var.getType(); 
 932         } 
 933       } 
 934     } 
 935  
 936     // Fall through case for arbitrary LHS and arbitrary RHS. 
 937     Node rightChild = assign.getLastChild(); 
 938     JSType rightType = getJSType(rightChild); 
 939     if (validator.expectCanAssignTo( 
 940             t, assign, rightType, leftType, "assignment")) { 
 941       ensureTyped(t, assign, rightType); 
 942     } else { 
 943       ensureTyped(t, assign); 
 944     } 
 945   }

 947   private void checkPropertyInheritanceOnGetpropAssign( 
 948       NodeTraversal t, Node assign, Node object, String property, 
 949       JSDocInfo info, JSType propertyType) { 
 950     // Inheritance checks for prototype properties. 
 951     // 
 952     // TODO(nicksantos): This isn't the right place to do this check. We 
 953     // really want to do this when we're looking at the constructor. 
 954     // We'd find all its properties and make sure they followed inheritance 
 955     // rules, like we currently do for @implements to make sure 
 956     // all the methods are implemented. 
 957     // 
 958     // As-is, this misses many other ways to override a property. 
 959     // 
 960     // object.prototype.property = ...; 
 961     if (object.isGetProp()) { 
 962       Node object2 = object.getFirstChild(); 
 963       String property2 = NodeUtil.getStringValue(object.getLastChild()); 
 964  
 965       if ("prototype".equals(property2)) { 
 966         JSType jsType = getJSType(object2); 
 967         if (jsType.isFunctionType()) { 
 968           FunctionType functionType = jsType.toMaybeFunctionType(); 
 969           if (functionType.isConstructor() || functionType.isInterface()) { 
 970             checkDeclaredPropertyInheritance( 
 971                 t, assign, functionType, property, info, propertyType); 
 972           } 
 973         } 
 974       } 
 975     } 
 976   }

 978   /** 
 979    * Visits an object literal field definition <code>key : value</code>. 
 980    * 
 981    * If the <code>lvalue</code> is a prototype modification, we change the 
 982    * schema of the object type it is referring to. 
 983    * 
 984    * @param t the traversal 
 985    * @param key the assign node 
 986    */ 
 987   private void visitObjLitKey(NodeTraversal t, Node key, Node objlit) { 
 988     // Do not validate object lit value types in externs. We don't really care, 
 989     // and it makes it easier to generate externs. 
 990     if (objlit.isFromExterns()) { 
 991       ensureTyped(t, key); 
 992       return; 
 993     } 
 994  
 995     // TODO(johnlenz): Validate get and set function declarations are valid 
 996     // as is the functions can have "extraneous" bits. 
 997  
 998     // For getter and setter property definitions the 
 999     // r-value type != the property type. 
1000     Node rvalue = key.getFirstChild(); 
1001     JSType rightType = NodeUtil.getObjectLitKeyTypeFromValueType( 
1002         key, getJSType(rvalue)); 
1003     if (rightType == null) { 
1004       rightType = getNativeType(UNKNOWN_TYPE); 
1005     } 
1006  
1007     Node owner = objlit; 
1008  
1009     // Validate value is assignable to the key type. 
1010  
1011     JSType keyType = getJSType(key); 
1012  
1013     JSType allowedValueType = keyType; 
1014     if (allowedValueType.isEnumElementType()) { 
1015       allowedValueType = 
1016           allowedValueType.toMaybeEnumElementType().getPrimitiveType(); 
1017     } 
1018  
1019     boolean valid = validator.expectCanAssignToPropertyOf(t, key, 
1020         rightType, allowedValueType, 
1021         owner, NodeUtil.getObjectLitKeyName(key)); 
1022     if (valid) { 
1023       ensureTyped(t, key, rightType); 
1024     } else { 
1025       ensureTyped(t, key); 
1026     } 
1027  
1028     // Validate that the key type is assignable to the object property type. 
1029     // This is necessary as the objlit may have been cast to a non-literal 
1030     // object type. 
1031     // TODO(johnlenz): consider introducing a CAST node to the AST (or 
1032     // perhaps a parentheses node). 
1033  
1034     JSType objlitType = getJSType(objlit); 
1035     ObjectType type = ObjectType.cast( 
1036         objlitType.restrictByNotNullOrUndefined()); 
1037     if (type != null) { 
1038       String property = NodeUtil.getObjectLitKeyName(key); 
1039       if (type.hasProperty(property) && 
1040           !type.isPropertyTypeInferred(property) && 
1041           !propertyIsImplicitCast(type, property)) { 
1042         validator.expectCanAssignToPropertyOf( 
1043             t, key, keyType, 
1044             type.getPropertyType(property), owner, property); 
1045       } 
1046       return; 
1047     } 
1048   }

1064   /** 
1065    * Given a constructor type and a property name, check that the property has 
1066    * the JSDoc annotation @override iff the property is declared on a 
1067    * superclass. Several checks regarding inheritance correctness are also 
1068    * performed. 
1069    */ 
1070   private void checkDeclaredPropertyInheritance( 
1071       NodeTraversal t, Node n, FunctionType ctorType, String propertyName, 
1072       JSDocInfo info, JSType propertyType) { 
1073     // If the supertype doesn't resolve correctly, we've warned about this 
1074     // already. 
1075     if (hasUnknownOrEmptySupertype(ctorType)) { 
1076       return; 
1077     } 
1078  
1079     FunctionType superClass = ctorType.getSuperClassConstructor(); 
1080     boolean superClassHasProperty = superClass != null && 
1081         superClass.getInstanceType().hasProperty(propertyName); 
1082     boolean superClassHasDeclaredProperty = superClass != null && 
1083         superClass.getInstanceType().isPropertyTypeDeclared(propertyName); 
1084  
1085     // For interface 
1086     boolean superInterfaceHasProperty = false; 
1087     boolean superInterfaceHasDeclaredProperty = false; 
1088     if (ctorType.isInterface()) { 
1089       for (ObjectType interfaceType : ctorType.getExtendedInterfaces()) { 
1090         superInterfaceHasProperty = 
1091             superInterfaceHasProperty || 
1092             interfaceType.hasProperty(propertyName); 
1093         superInterfaceHasDeclaredProperty = 
1094             superInterfaceHasDeclaredProperty || 
1095             interfaceType.isPropertyTypeDeclared(propertyName); 
1096       } 
1097     } 
1098     boolean declaredOverride = info != null && info.isOverride(); 
1099  
1100     boolean foundInterfaceProperty = false; 
1101     if (ctorType.isConstructor()) { 
1102       for (JSType implementedInterface : 
1103           ctorType.getAllImplementedInterfaces()) { 
1104         if (implementedInterface.isUnknownType() || 
1105             implementedInterface.isEmptyType()) { 
1106           continue; 
1107         } 
1108         FunctionType interfaceType = 
1109             implementedInterface.toObjectType().getConstructor(); 
1110         Preconditions.checkNotNull(interfaceType); 
1111  
1112         boolean interfaceHasProperty = 
1113             interfaceType.getPrototype().hasProperty(propertyName); 
1114         foundInterfaceProperty = foundInterfaceProperty || 
1115             interfaceHasProperty; 
1116         if (reportMissingOverride.isOn() 
1117             && !declaredOverride 
1118             && interfaceHasProperty) { 
1119           // @override not present, but the property does override an interface 
1120           // property 
1121           compiler.report(t.makeError(n, reportMissingOverride, 
1122               HIDDEN_INTERFACE_PROPERTY, propertyName, 
1123               interfaceType.getTopMostDefiningType(propertyName).toString())); 
1124         } 
1125       } 
1126     } 
1127  
1128     if (!declaredOverride 
1129         && !superClassHasProperty 
1130         && !superInterfaceHasProperty) { 
1131       // nothing to do here, it's just a plain new property 
1132       return; 
1133     } 
1134  
1135     ObjectType topInstanceType = superClassHasDeclaredProperty ? 
1136         superClass.getTopMostDefiningType(propertyName) : null; 
1137     boolean declaredLocally = 
1138         ctorType.isConstructor() && 
1139         (ctorType.getPrototype().hasOwnProperty(propertyName) || 
1140          ctorType.getInstanceType().hasOwnProperty(propertyName)); 
1141     if (reportMissingOverride.isOn() 
1142         && !declaredOverride 
1143         && superClassHasDeclaredProperty 
1144         && declaredLocally) { 
1145       // @override not present, but the property does override a superclass 
1146       // property 
1147       compiler.report(t.makeError(n, reportMissingOverride, 
1148           HIDDEN_SUPERCLASS_PROPERTY, propertyName, 
1149           topInstanceType.toString())); 
1150     } 
1151  
1152     // @override is present and we have to check that it is ok 
1153     if (superClassHasDeclaredProperty) { 
1154       // there is a superclass implementation 
1155       JSType superClassPropType = 
1156           superClass.getInstanceType().getPropertyType(propertyName); 
1157       if (!propertyType.canAssignTo(superClassPropType)) { 
1158         compiler.report( 
1159             t.makeError(n, HIDDEN_SUPERCLASS_PROPERTY_MISMATCH, 
1160                 propertyName, topInstanceType.toString(), 
1161                 superClassPropType.toString(), propertyType.toString())); 
1162       } 
1163     } else if (superInterfaceHasDeclaredProperty) { 
1164       // there is an super interface property 
1165       for (ObjectType interfaceType : ctorType.getExtendedInterfaces()) { 
1166         if (interfaceType.hasProperty(propertyName)) { 
1167           JSType superPropertyType = 
1168               interfaceType.getPropertyType(propertyName); 
1169           if (!propertyType.canAssignTo(superPropertyType)) { 
1170             topInstanceType = interfaceType.getConstructor(). 
1171                 getTopMostDefiningType(propertyName); 
1172             compiler.report( 
1173                 t.makeError(n, HIDDEN_SUPERCLASS_PROPERTY_MISMATCH, 
1174                     propertyName, topInstanceType.toString(), 
1175                     superPropertyType.toString(), 
1176                     propertyType.toString())); 
1177           } 
1178         } 
1179       } 
1180     } else if (!foundInterfaceProperty 
1181         && !superClassHasProperty 
1182         && !superInterfaceHasProperty) { 
1183       // there is no superclass nor interface implementation 
1184       compiler.report( 
1185           t.makeError(n, UNKNOWN_OVERRIDE, 
1186               propertyName, ctorType.getInstanceType().toString())); 
1187     } 
1188   }

1190   /** 
1191    * Given a constructor or an interface type, find out whether the unknown 
1192    * type is a supertype of the current type. 
1193    */ 
1194   private static boolean hasUnknownOrEmptySupertype(FunctionType ctor) { 
1195     Preconditions.checkArgument(ctor.isConstructor() || ctor.isInterface()); 
1196     Preconditions.checkArgument(!ctor.isUnknownType()); 
1197  
1198     // The type system should notice inheritance cycles on its own 
1199     // and break the cycle. 
1200     while (true) { 
1201       ObjectType maybeSuperInstanceType = 
1202           ctor.getPrototype().getImplicitPrototype(); 
1203       if (maybeSuperInstanceType == null) { 
1204         return false; 
1205       } 
1206       if (maybeSuperInstanceType.isUnknownType() || 
1207           maybeSuperInstanceType.isEmptyType()) { 
1208         return true; 
1209       } 
1210       ctor = maybeSuperInstanceType.getConstructor(); 
1211       if (ctor == null) { 
1212         return false; 
1213       } 
1214       Preconditions.checkState(ctor.isConstructor() || ctor.isInterface()); 
1215     } 
1216   }

1255   /** 
1256    * Visits a NAME node. 
1257    * 
1258    * @param t The node traversal object that supplies context, such as the 
1259    * scope chain to use in name lookups as well as error reporting. 
1260    * @param n The node being visited. 
1261    * @param parent The parent of the node n. 
1262    * @return whether the node is typeable or not 
1263    */ 
1264   boolean visitName(NodeTraversal t, Node n, Node parent) { 
1265     // At this stage, we need to determine whether this is a leaf 
1266     // node in an expression (which therefore needs to have a type 
1267     // assigned for it) versus some other decorative node that we 
1268     // can safely ignore.  Function names, arguments (children of LP nodes) and 
1269     // variable declarations are ignored. 
1270     // TODO(user): remove this short-circuiting in favor of a 
1271     // pre order traversal of the FUNCTION, CATCH, LP and VAR nodes. 
1272     int parentNodeType = parent.getType(); 
1273     if (parentNodeType == Token.FUNCTION || 
1274         parentNodeType == Token.CATCH || 
1275         parentNodeType == Token.PARAM_LIST || 
1276         parentNodeType == Token.VAR) { 
1277       return false; 
1278     } 
1279  
1280     JSType type = n.getJSType(); 
1281     if (type == null) { 
1282       type = getNativeType(UNKNOWN_TYPE); 
1283       Var var = t.getScope().getVar(n.getString()); 
1284       if (var != null) { 
1285         JSType varType = var.getType();

1301    * @param parent The parent of <code>n</code> 
1302    */ 
1303   private void visitGetProp(NodeTraversal t, Node n, Node parent) { 
1304     // obj.prop or obj.method() 
1305     // Lots of types can appear on the left, a call to a void function can 
1306     // never be on the left. getPropertyType will decide what is acceptable 
1307     // and what isn't. 
1308     Node property = n.getLastChild();

1314     } else if (validator.expectNotNullOrUndefined(t, n, childType, 
1315         "No properties on this expression", getNativeType(OBJECT_TYPE))) { 
1316       checkPropertyAccess(childType, property.getString(), t, n); 
1317     } 
1318     ensureTyped(t, n);

1321   /** 
1322    * Emit a warning if we can prove that a property cannot possibly be 
1323    * defined on an object. Note the difference between JS and a strictly 
1324    * statically typed language: we're checking if the property 
1325    * *cannot be defined*, whereas a java compiler would check if the 
1326    * property *can be undefined*. 
1327    */ 
1328   private void checkPropertyAccess(JSType childType, String propName, 
1329       NodeTraversal t, Node n) { 
1330     // If the property type is unknown, check the object type to see if it 
1331     // can ever be defined. We explicitly exclude CHECKED_UNKNOWN (for 
1332     // properties where we've checked that it exists, or for properties on 
1333     // objects that aren't in this binary). 
1334     JSType propType = getJSType(n); 
1335     if (propType.isEquivalentTo(typeRegistry.getNativeType(UNKNOWN_TYPE))) { 
1336       childType = childType.autobox(); 
1337       ObjectType objectType = ObjectType.cast(childType); 
1338       if (objectType != null) { 
1339         // We special-case object types so that checks on enums can be 
1340         // much stricter, and so that we can use hasProperty (which is much 
1341         // faster in most cases). 
1342         if (!objectType.hasProperty(propName) || 
1343             objectType.isEquivalentTo( 
1344                 typeRegistry.getNativeType(UNKNOWN_TYPE))) { 
1345           if (objectType instanceof EnumType) { 
1346             report(t, n, INEXISTENT_ENUM_ELEMENT, propName); 
1347           } else { 
1348             checkPropertyAccessHelper(objectType, propName, t, n); 
1349           } 
1350         } 
1351  
1352       } else { 
1353         checkPropertyAccessHelper(childType, propName, t, n); 
1354       } 
1355     } 
1356   }

1358   private void checkPropertyAccessHelper(JSType objectType, String propName, 
1359       NodeTraversal t, Node n) { 
1360     if (!objectType.isEmptyType() && 
1361         reportMissingProperties && !isPropertyTest(n)) { 
1362       if (!typeRegistry.canPropertyBeDefined(objectType, propName)) { 
1363         report(t, n, INEXISTENT_PROPERTY, propName, 
1364             validator.getReadableJSTypeName(n.getFirstChild(), true)); 
1365       } 
1366     }

1369   /** 
1370    * Determines whether this node is testing for the existence of a property. 
1371    * If true, we will not emit warnings about a missing property. 
1372    * 
1373    * @param getProp The GETPROP being tested. 
1374    */ 
1375   private boolean isPropertyTest(Node getProp) { 
1376     Node parent = getProp.getParent(); 
1377     switch (parent.getType()) { 
1378       case Token.CALL: 
1379         return parent.getFirstChild() != getProp && 
1380             compiler.getCodingConvention().isPropertyTestFunction(parent); 
1381  
1382       case Token.IF: 
1383       case Token.WHILE: 
1384       case Token.DO: 
1385       case Token.FOR: 
1386         return NodeUtil.getConditionExpression(parent) == getProp; 
1387  
1388       case Token.INSTANCEOF: 
1389       case Token.TYPEOF: 
1390         return true; 
1391  
1392       case Token.AND: 
1393       case Token.HOOK: 
1394         return parent.getFirstChild() == getProp; 
1395  
1396       case Token.NOT: 
1397         return parent.getParent().isOr() && 
1398             parent.getParent().getFirstChild() == parent; 
1399     } 
1400     return false; 
1401   }

1481    * been visited

1483    * that have been visited 
1484    * @param interfaceType The super interface that is being visited

1510   /** 
1511    * Visits a {@link Token#FUNCTION} node. 
1512    * 
1513    * @param t The node traversal object that supplies context, such as the 
1514    * scope chain to use in name lookups as well as error reporting. 
1515    * @param n The node being visited. 
1516    */ 
1517   private void visitFunction(NodeTraversal t, Node n) { 
1518     FunctionType functionType = JSType.toMaybeFunctionType(n.getJSType()); 
1519     String functionPrivateName = n.getFirstChild().getString(); 
1520     if (functionType.isConstructor()) { 
1521       FunctionType baseConstructor = functionType.getSuperClassConstructor(); 
1522       if (baseConstructor != getNativeType(OBJECT_FUNCTION_TYPE) && 
1523           baseConstructor != null && 
1524           baseConstructor.isInterface() && functionType.isConstructor()) { 
1525         compiler.report( 
1526             t.makeError(n, CONFLICTING_EXTENDED_TYPE, 
1527                         "constructor", functionPrivateName)); 
1528       } else { 
1529         if (baseConstructor != getNativeType(OBJECT_FUNCTION_TYPE) && 
1530             baseConstructor != null) { 
1531           if (functionType.makesStructs() && !baseConstructor.makesStructs()) { 
1532             compiler.report(t.makeError(n, CONFLICTING_EXTENDED_TYPE, 
1533                                         "struct", functionPrivateName)); 
1534           } else if (functionType.makesDicts() && 
1535                      !baseConstructor.makesDicts()) { 
1536             compiler.report(t.makeError(n, CONFLICTING_EXTENDED_TYPE, 
1537                                         "dict", functionPrivateName)); 
1538           } 
1539         } 
1540         // All interfaces are properly implemented by a class 
1541         for (JSType baseInterface : functionType.getImplementedInterfaces()) { 
1542           boolean badImplementedType = false; 
1543           ObjectType baseInterfaceObj = ObjectType.cast(baseInterface); 
1544           if (baseInterfaceObj != null) { 
1545             FunctionType interfaceConstructor = 
1546               baseInterfaceObj.getConstructor(); 
1547             if (interfaceConstructor != null && 
1548                 !interfaceConstructor.isInterface()) { 
1549               badImplementedType = true; 
1550             } 
1551           } else { 
1552             badImplementedType = true; 
1553           } 
1554           if (badImplementedType) { 
1555             report(t, n, BAD_IMPLEMENTED_TYPE, functionPrivateName); 
1556           } 
1557         } 
1558         // check properties 
1559         validator.expectAllInterfaceProperties(t, n, functionType); 
1560       } 
1561     } else if (functionType.isInterface()) { 
1562       // Interface must extend only interfaces 
1563       for (ObjectType extInterface : functionType.getExtendedInterfaces()) { 
1564         if (extInterface.getConstructor() != null 
1565             && !extInterface.getConstructor().isInterface()) { 
1566           compiler.report( 
1567               t.makeError(n, CONFLICTING_EXTENDED_TYPE, 
1568                           "interface", functionPrivateName)); 
1569         } 
1570       } 
1571       // Interface cannot implement any interfaces 
1572       if (functionType.hasImplementedInterfaces()) { 
1573         compiler.report(t.makeError(n, 
1574             CONFLICTING_IMPLEMENTED_TYPE, functionPrivateName)); 
1575       } 
1576       // Check whether the extended interfaces have any conflicts 
1577       if (functionType.getExtendedInterfacesCount() > 1) { 
1578         // Only check when extending more than one interfaces 
1579         HashMap<String, ObjectType> properties 
1580             = new HashMap<String, ObjectType>(); 
1581         HashMap<String, ObjectType> currentProperties 
1582             = new HashMap<String, ObjectType>(); 
1583         for (ObjectType interfaceType : functionType.getExtendedInterfaces()) { 
1584           currentProperties.clear(); 
1585           checkInterfaceConflictProperties(t, n, functionPrivateName, 
1586               properties, currentProperties, interfaceType); 
1587           properties.putAll(currentProperties); 
1588         } 
1589       } 
1590     } 
1591   }

1593   /** 
1594    * Visits a CALL node. 
1595    * 
1596    * @param t The node traversal object that supplies context, such as the 
1597    * scope chain to use in name lookups as well as error reporting. 
1598    * @param n The node being visited. 
1599    */ 
1600   private void visitCall(NodeTraversal t, Node n) { 
1601     Node child = n.getFirstChild(); 
1602     JSType childType = getJSType(child).restrictByNotNullOrUndefined(); 
1603  
1604     if (!childType.canBeCalled()) { 
1605       report(t, n, NOT_CALLABLE, childType.toString()); 
1606       ensureTyped(t, n); 
1607       return; 
1608     } 
1609  
1610     // A couple of types can be called as if they were functions. 
1611     // If it is a function type, then validate parameters. 
1612     if (childType.isFunctionType()) { 
1613       FunctionType functionType = childType.toMaybeFunctionType(); 
1614  
1615       boolean isExtern = false; 
1616       JSDocInfo functionJSDocInfo = functionType.getJSDocInfo(); 
1617       if( functionJSDocInfo != null  && 
1618           functionJSDocInfo.getAssociatedNode() != null) { 
1619         isExtern = functionJSDocInfo.getAssociatedNode().isFromExterns(); 
1620       } 
1621  
1622       // Non-native constructors should not be called directly 
1623       // unless they specify a return type and are defined 
1624       // in an extern. 
1625       if (functionType.isConstructor() && 
1626           !functionType.isNativeObjectType() && 
1627           (functionType.getReturnType().isUnknownType() || 
1628            functionType.getReturnType().isVoidType() || 
1629            !isExtern)) { 
1630         report(t, n, CONSTRUCTOR_NOT_CALLABLE, childType.toString()); 
1631       } 
1632  
1633       // Functions with explicit 'this' types must be called in a GETPROP 
1634       // or GETELEM. 
1635       if (functionType.isOrdinaryFunction() && 
1636           !functionType.getTypeOfThis().isUnknownType() && 
1637           !functionType.getTypeOfThis().isNativeObjectType() && 
1638           !(child.isGetElem() || 
1639             child.isGetProp())) { 
1640         report(t, n, EXPECTED_THIS_TYPE, functionType.toString()); 
1641       } 
1642  
1643       visitParameterList(t, n, functionType); 
1644       ensureTyped(t, n, functionType.getReturnType()); 
1645     } else { 
1646       ensureTyped(t, n); 
1647     } 
1648  
1649     // TODO: Add something to check for calls of RegExp objects, which is not 
1650     // supported by IE.  Either say something about the return type or warn 
1651     // about the non-portability of the call or both. 
1652   }

1654   /** 
1655    * Visits the parameters of a CALL or a NEW node. 
1656    */ 
1657   private void visitParameterList(NodeTraversal t, Node call, 
1658       FunctionType functionType) { 
1659     Iterator<Node> arguments = call.children().iterator(); 
1660     arguments.next(); // skip the function name 
1661  
1662     Iterator<Node> parameters = functionType.getParameters().iterator(); 
1663     int ordinal = 0; 
1664     Node parameter = null; 
1665     Node argument = null; 
1666     while (arguments.hasNext() && 
1667            (parameters.hasNext() || 
1668             parameter != null && parameter.isVarArgs())) { 
1669       // If there are no parameters left in the list, then the while loop 
1670       // above implies that this must be a var_args function. 
1671       if (parameters.hasNext()) { 
1672         parameter = parameters.next(); 
1673       } 
1674       argument = arguments.next(); 
1675       ordinal++; 
1676  
1677       validator.expectArgumentMatchesParameter(t, argument, 
1678           getJSType(argument), getJSType(parameter), call, ordinal); 
1679     } 
1680  
1681     int numArgs = call.getChildCount() - 1; 
1682     int minArgs = functionType.getMinArguments(); 
1683     int maxArgs = functionType.getMaxArguments(); 
1684     if (minArgs > numArgs || maxArgs < numArgs) { 
1685       report(t, call, WRONG_ARGUMENT_COUNT, 
1686               validator.getReadableJSTypeName(call.getFirstChild(), false), 
1687               String.valueOf(numArgs), String.valueOf(minArgs), 
1688               maxArgs != Integer.MAX_VALUE ? 
1689               " and no more than " + maxArgs + " argument(s)" : ""); 
1690     } 
1691   }

1730   /** 
1731    * This function unifies the type checking involved in the core binary 
1732    * operators and the corresponding assignment operators.  The representation 
1733    * used internally is such that common code can handle both kinds of 
1734    * operators easily. 
1735    * 
1736    * @param op The operator. 
1737    * @param t The traversal object, needed to report errors. 
1738    * @param n The node being checked. 
1739    */ 
1740   private void visitBinaryOperator(int op, NodeTraversal t, Node n) { 
1741     Node left = n.getFirstChild(); 
1742     JSType leftType = getJSType(left); 
1743     Node right = n.getLastChild(); 
1744     JSType rightType = getJSType(right); 
1745     switch (op) { 
1746       case Token.ASSIGN_LSH: 
1747       case Token.ASSIGN_RSH: 
1748       case Token.LSH: 
1749       case Token.RSH: 
1750       case Token.ASSIGN_URSH: 
1751       case Token.URSH: 
1752         if (!leftType.matchesInt32Context()) { 
1753           report(t, left, BIT_OPERATION, 
1754                    NodeUtil.opToStr(n.getType()), leftType.toString()); 
1755         } 
1756         if (!rightType.matchesUint32Context()) { 
1757           report(t, right, BIT_OPERATION, 
1758                    NodeUtil.opToStr(n.getType()), rightType.toString()); 
1759         } 
1760         break; 
1761  
1762       case Token.ASSIGN_DIV: 
1763       case Token.ASSIGN_MOD: 
1764       case Token.ASSIGN_MUL: 
1765       case Token.ASSIGN_SUB: 
1766       case Token.DIV: 
1767       case Token.MOD: 
1768       case Token.MUL: 
1769       case Token.SUB: 
1770         validator.expectNumber(t, left, leftType, "left operand"); 
1771         validator.expectNumber(t, right, rightType, "right operand"); 
1772         break; 
1773  
1774       case Token.ASSIGN_BITAND: 
1775       case Token.ASSIGN_BITXOR: 
1776       case Token.ASSIGN_BITOR: 
1777       case Token.BITAND: 
1778       case Token.BITXOR: 
1779       case Token.BITOR: 
1780         validator.expectBitwiseable(t, left, leftType, 
1781             "bad left operand to bitwise operator"); 
1782         validator.expectBitwiseable(t, right, rightType, 
1783             "bad right operand to bitwise operator"); 
1784         break; 
1785  
1786       case Token.ASSIGN_ADD: 
1787       case Token.ADD: 
1788         break; 
1789  
1790       default: 
1791         report(t, n, UNEXPECTED_TOKEN, Token.name(op)); 
1792     } 
1793     ensureTyped(t, n); 
1794   }

1865   /** 
1866    * Enforces type casts, and ensures the node is typed. 
1867    * 
1868    * A cast in the way that we use it in JSDoc annotations never 
1869    * alters the generated code and therefore never can induce any runtime 
1870    * operation. What this means is that a 'cast' is really just a compile 
1871    * time constraint on the underlying value. In the future, we may add 
1872    * support for run-time casts for compiled tests. 
1873    * 
1874    * To ensure some shred of sanity, we enforce the notion that the 
1875    * type you are casting to may only meaningfully be a narrower type 
1876    * than the underlying declared type. We also invalidate optimizations 
1877    * on bad type casts. 
1878    * 
1879    * @param t The traversal object needed to report errors. 
1880    * @param n The node getting a type assigned to it. 
1881    * @param type The type to be assigned. 
1882    */ 
1883   private void ensureTyped(NodeTraversal t, Node n, JSType type) { 
1884     // Make sure FUNCTION nodes always get function type. 
1885     Preconditions.checkState(!n.isFunction() || 
1886             type.isFunctionType() || 
1887             type.isUnknownType()); 
1888     JSDocInfo info = n.getJSDocInfo(); 
1889     if (info != null) { 
1890       if (info.hasType()) { 
1891         JSType infoType = info.getType().evaluate(t.getScope(), typeRegistry); 
1892         validator.expectCanCast(t, n, infoType, type); 
1893         type = infoType; 
1894       } 
1895  
1896       if (info.isImplicitCast() && !inExterns) { 
1897         String propName = n.isGetProp() ? 
1898             n.getLastChild().getString() : "(missing)"; 
1899         compiler.report(
