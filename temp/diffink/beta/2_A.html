140     protected final JsonDeserializer<Object> _findDeserializer(DeserializationContext ctxt, 
141             String typeId) throws IOException 
142     { 
143         JsonDeserializer<Object> deser = _deserializers.get(typeId); 
144         if (deser == null) { 
145             /* As per [Databind#305], need to provide contextual info. But for 
146              * backwards compatibility, let's start by only supporting this 
147              * for base class, not via interface. Later on we can add this 
148              * to the interface, assuming deprecation at base class helps. 
149              */ 
150             JavaType type = _idResolver.typeFromId(ctxt, typeId); 
151             if (type == null) { 
152                 // As per [JACKSON-614], use the default impl if no type id available: 
153                 deser = _findDefaultImplDeserializer(ctxt); 
154                 if (deser == null) { 
155                     // 10-May-2016, tatu: We may get some help... 
156                     JavaType actual = _handleUnknownTypeId(ctxt, typeId, _idResolver, _baseType); 
157                     if (actual == null) { // what should this be taken to mean? 
158                         // TODO: try to figure out something better 
159                         return null; 
160                     } 
161                     // ... would this actually work? 
162                     deser = ctxt.findContextualValueDeserializer(actual, _property); 
163                 } 
164             } else { 
165                 /* 16-Dec-2010, tatu: Since nominal type we get here has no (generic) type parameters, 
166                  *   we actually now need to explicitly narrow from base type (which may have parameterization) 
167                  *   using raw type. 
168                  * 
169                  *   One complication, though; can not change 'type class' (simple type to container); otherwise 
170                  *   we may try to narrow a SimpleType (Object.class) into MapType (Map.class), losing actual 
171                  *   type in process (getting SimpleType of Map.class which will not work as expected) 
172                  */ 
173                 if ((_baseType != null) 
174                         && _baseType.getClass() == type.getClass()) { 
175                     /* 09-Aug-2015, tatu: Not sure if the second part of the check makes sense; 
176                      *   but it appears to check that JavaType impl class is the same which is 
177                      *   important for some reason? 
178                      *   Disabling the check will break 2 Enum-related tests. 
179                      */ 
180                     // 19-Jun-2016, tatu: As per [databind#1270] we may actually get full 
181                     //   generic type with custom type resolvers. If so, should try to retain them. 
182                     //  Whether this is sufficient to avoid problems remains to be seen, but for 
183                     //  now it should improve things. 
184                         type = ctxt.getTypeFactory().constructSpecializedType(_baseType, type.getRawClass()); 
185                 } 
186                 deser = ctxt.findContextualValueDeserializer(type, _property); 
187             } 
188             _deserializers.put(typeId, deser); 
189         } 
190         return deser; 
191     }

193     protected final JsonDeserializer<Object> _findDefaultImplDeserializer(DeserializationContext ctxt) throws IOException 
194     { 
195         /* 06-Feb-2013, tatu: As per [databind#148], consider default implementation value of 
196          *   {@link java.lang.Void} to mean "serialize as null"; as well as DeserializationFeature 
197          *   to do swift mapping to null 
198          */ 
199         if (_defaultImpl == null) { 
200             if (!ctxt.isEnabled(DeserializationFeature.FAIL_ON_INVALID_SUBTYPE)) { 
201                 return NullifyingDeserializer.instance; 
202             } 
203             return null; 
204         } 
205         Class<?> raw = _defaultImpl.getRawClass(); 
206         if (ClassUtil.isBogusClass(raw)) { 
207             return NullifyingDeserializer.instance; 
208         } 
209          
210         synchronized (_defaultImpl) { 
211             if (_defaultImplDeserializer == null) { 
212                 _defaultImplDeserializer = ctxt.findContextualValueDeserializer( 
213                         _defaultImpl, _property); 
214             } 
215             return _defaultImplDeserializer; 
216         } 
217     }

237     protected Object _deserializeWithNativeTypeId(JsonParser jp, DeserializationContext ctxt, Object typeId) 
238         throws IOException 
239     { 
240         JsonDeserializer<Object> deser; 
241         if (typeId == null) { 
242             /* 04-May-2014, tatu: Should error be obligatory, or should there be another method 
243              *   for "try to deserialize with native tpye id"? 
244              */ 
245             deser = _findDefaultImplDeserializer(ctxt); 
246             if (deser == null) { 
247                 ctxt.reportMappingException("No (native) type id found when one was expected for polymorphic type handling"); 
248                 return null; 
249             } 
250         } else { 
251             String typeIdStr = (typeId instanceof String) ? (String) typeId : String.valueOf(typeId); 
252             deser = _findDeserializer(ctxt, typeIdStr); 
253         } 
254         return deser.deserialize(jp, ctxt);

258      * Helper method called when given type id can not be resolved into  
259      * concrete deserializer either directly (using given {@link  TypeIdResolver}), 
260      * or using default type. 
261      * Default implementation simply throws a {@link com.fasterxml.jackson.databind.JsonMappingException} to 
262      * indicate the problem; sub-classes may choose 
263      * 
264      * @return If it is possible to resolve type id into a {@link JsonDeserializer} 
265      *   should return that deserializer; otherwise throw an exception to indicate 
266      *   the problem. 
267      * 
268      * @since 2.8 
269      */ 
270     protected JavaType _handleUnknownTypeId(DeserializationContext ctxt, String typeId, 
271             TypeIdResolver idResolver, JavaType baseType) 
272         throws IOException 
273     { 
274         String extraDesc; 
275         if (idResolver instanceof TypeIdResolverBase) { 
276             extraDesc = ((TypeIdResolverBase) idResolver).getDescForKnownTypeIds(); 
277             if (extraDesc == null) { 
278                 extraDesc = "known type ids are not statically known"; 
279             } else { 
280                 extraDesc = "known type ids = " + extraDesc; 
281             } 
282         } else { 
283             extraDesc = null; 
284         } 
285         return ctxt.handleUnknownTypeId(_baseType, typeId, idResolver, extraDesc); 
286     }
