  49 /** 
  50  * Specifies the format of a CSV file and parses input. 
  51  * 
  52  * <h2>Using predefined formats</h2> 
  53  * 
  54  * <p> 
  55  * You can use one of the predefined formats: 
  56  * </p> 
  57  * 
  58  * <ul> 
  59  * <li>{@link #DEFAULT}</li> 
  60  * <li>{@link #EXCEL}</li> 
  61  * <li>{@link #MYSQL}</li> 
  62  * <li>{@link #RFC4180}</li> 
  63  * <li>{@link #TDF}</li> 
  64  * </ul> 
  65  * 
  66  * <p> 
  67  * For example: 
  68  * </p> 
  69  * 
  70  * <pre> 
  71  * CSVParser parser = CSVFormat.EXCEL.parse(reader); 
  72  * </pre> 
  73  * 
  74  * <p> 
  75  * The {@link CSVParser} provides static methods to parse other input types, for example: 
  76  * </p> 
  77  * 
  78  * <pre> 
  79  * CSVParser parser = CSVParser.parse(file, StandardCharsets.US_ASCII, CSVFormat.EXCEL); 
  80  * </pre> 
  81  * 
  82  * <h2>Defining formats</h2> 
  83  * 
  84  * <p> 
  85  * You can extend a format by calling the {@code with} methods. For example: 
  86  * </p> 
  87  * 
  88  * <pre> 
  89  * CSVFormat.EXCEL.withNullString(&quot;N/A&quot;).withIgnoreSurroundingSpaces(true); 
  90  * </pre> 
  91  * 
  92  * <h2>Defining column names</h2> 
  93  * 
  94  * <p> 
  95  * To define the column names you want to use to access records, write: 
  96  * </p> 
  97  * 
  98  * <pre> 
  99  * CSVFormat.EXCEL.withHeader(&quot;Col1&quot;, &quot;Col2&quot;, &quot;Col3&quot;); 
 100  * </pre> 
 101  * 
 102  * <p> 
 103  * Calling {@link #withHeader(String...)} let's you use the given names to address values in a {@link CSVRecord}, and 
 104  * assumes that your CSV source does not contain a first record that also defines column names. 
 105  * 
 106  * If it does, then you are overriding this metadata with your names and you should skip the first record by calling 
 107  * {@link #withSkipHeaderRecord(boolean)} with {@code true}. 
 108  * </p> 
 109  * 
 110  * <h2>Parsing</h2> 
 111  * 
 112  * <p> 
 113  * You can use a format directly to parse a reader. For example, to parse an Excel file with columns header, write: 
 114  * </p> 
 115  * 
 116  * <pre> 
 117  * Reader in = ...; 
 118  * CSVFormat.EXCEL.withHeader(&quot;Col1&quot;, &quot;Col2&quot;, &quot;Col3&quot;).parse(in); 
 119  * </pre> 
 120  * 
 121  * <p> 
 122  * For other input types, like resources, files, and URLs, use the static methods on {@link CSVParser}. 
 123  * </p> 
 124  * 
 125  * <h2>Referencing columns safely</h2> 
 126  * 
 127  * <p> 
 128  * If your source contains a header record, you can simplify your code and safely reference columns, by using 
 129  * {@link #withHeader(String...)} with no arguments: 
 130  * </p> 
 131  * 
 132  * <pre> 
 133  * CSVFormat.EXCEL.withHeader(); 
 134  * </pre> 
 135  * 
 136  * <p> 
 137  * This causes the parser to read the first record and use its values as column names. 
 138  * 
 139  * Then, call one of the {@link CSVRecord} get method that takes a String column name argument: 
 140  * </p> 
 141  * 
 142  * <pre> 
 143  * String value = record.get(&quot;Col1&quot;); 
 144  * </pre> 
 145  * 
 146  * <p> 
 147  * This makes your code impervious to changes in column order in the CSV file. 
 148  * </p> 
 149  * 
 150  * <h2>Notes</h2> 
 151  * 
 152  * <p> 
 153  * This class is immutable. 
 154  * </p> 
 155  */

1090     private void print(final Object object, final CharSequence value, final int offset, final int len, 
1091             final Appendable out, final boolean newRecord) throws IOException { 
1092         if (!newRecord) { 
1093             out.append(getDelimiter()); 
1094         } 
1095         if (object == null) { 
1096             out.append(value); 
1097         } else if (isQuoteCharacterSet()) { 
1098             // the original object is needed so can check for Number 
1099             printAndQuote(object, value, offset, len, out, newRecord); 
1100         } else if (isEscapeCharacterSet()) { 
1101             printAndEscape(value, offset, len, out); 
1102         } else { 
1103             out.append(value, offset, offset + len); 
1104         } 
1105     }

1107     /* 
1108      * Note: must only be called if escaping is enabled, otherwise will generate NPE 
1109      */ 
1110     private void printAndEscape(final CharSequence value, final int offset, final int len, final Appendable out) 
1111             throws IOException { 
1112         int start = offset; 
1113         int pos = offset; 
1114         final int end = offset + len; 
1115  
1116         final char delim = getDelimiter(); 
1117         final char escape = getEscapeCharacter().charValue(); 
1118  
1119         while (pos < end) { 
1120             char c = value.charAt(pos); 
1121             if (c == CR || c == LF || c == delim || c == escape) { 
1122                 // write out segment up until this char 
1123                 if (pos > start) { 
1124                     out.append(value, start, pos); 
1125                 } 
1126                 if (c == LF) { 
1127                     c = 'n'; 
1128                 } else if (c == CR) { 
1129                     c = 'r'; 
1130                 } 
1131  
1132                 out.append(escape); 
1133                 out.append(c); 
1134  
1135                 start = pos + 1; // start on the current char after this one 
1136             } 
1137  
1138             pos++; 
1139         } 
1140  
1141         // write last segment 
1142         if (pos > start) { 
1143             out.append(value, start, pos); 
1144         } 
1145     }

1147     /* 
1148      * Note: must only be called if quoting is enabled, otherwise will generate NPE 
1149      */ 
1150     // the original object is needed so can check for Number 
1151     private void printAndQuote(final Object object, final CharSequence value, final int offset, final int len, 
1152             final Appendable out, final boolean newRecord) throws IOException { 
1153         boolean quote = false; 
1154         int start = offset; 
1155         int pos = offset; 
1156         final int end = offset + len; 
1157  
1158         final char delimChar = getDelimiter(); 
1159         final char quoteChar = getQuoteCharacter().charValue(); 
1160  
1161         QuoteMode quoteModePolicy = getQuoteMode(); 
1162         if (quoteModePolicy == null) { 
1163             quoteModePolicy = QuoteMode.MINIMAL; 
1164         } 
1165         switch (quoteModePolicy) { 
1166         case ALL: 
1167         case ALL_NON_NULL: 
1168             quote = true; 
1169             break; 
1170         case NON_NUMERIC: 
1171             quote = !(object instanceof Number); 
1172             break; 
1173         case NONE: 
1174             // Use the existing escaping code 
1175             printAndEscape(value, offset, len, out); 
1176             return; 
1177         case MINIMAL: 
1178             if (len <= 0) { 
1179                 // always quote an empty token that is the first 
1180                 // on the line, as it may be the only thing on the 
1181                 // line. If it were not quoted in that case, 
1182                 // an empty line has no tokens. 
1183                 if (newRecord) { 
1184                     quote = true; 
1185                 } 
1186             } else { 
1187                 char c = value.charAt(pos); 
1188  
1189                 if (newRecord && (c < 0x20 || c > 0x21 && c < 0x23 || c > 0x2B && c < 0x2D || c > 0x7E)) { 
1190                     quote = true; 
1191                 } else if (c <= COMMENT) { 
1192                     // Some other chars at the start of a value caused the parser to fail, so for now 
1193                     // encapsulate if we start in anything less than '#'. We are being conservative 
1194                     // by including the default comment char too. 
1195                     quote = true; 
1196                 } else { 
1197                     while (pos < end) { 
1198                         c = value.charAt(pos); 
1199                         if (c == LF || c == CR || c == quoteChar || c == delimChar) { 
1200                             quote = true; 
1201                             break; 
1202                         } 
1203                         pos++; 
1204                     } 
1205  
1206                     if (!quote) { 
1207                         pos = end - 1; 
1208                         c = value.charAt(pos); 
1209                         // Some other chars at the end caused the parser to fail, so for now 
1210                         // encapsulate if we end in anything less than ' ' 
1211                         if (c <= SP) { 
1212                             quote = true; 
1213                         } 
1214                     } 
1215                 } 
1216             } 
1217  
1218             if (!quote) { 
1219                 // no encapsulation needed - write out the original value 
1220                 out.append(value, start, end); 
1221                 return; 
1222             } 
1223             break; 
1224         default: 
1225             throw new IllegalStateException("Unexpected Quote value: " + quoteModePolicy); 
1226         } 
1227  
1228         if (!quote) { 
1229             // no encapsulation needed - write out the original value 
1230             out.append(value, start, end); 
1231             return; 
1232         } 
1233  
1234         // we hit something that needed encapsulation 
1235         out.append(quoteChar); 
1236  
1237         // Pick up where we left off: pos should be positioned on the first character that caused 
1238         // the need for encapsulation. 
1239         while (pos < end) { 
1240             final char c = value.charAt(pos); 
1241             if (c == quoteChar) { 
1242                 // write out the chunk up until this point 
1243  
1244                 // add 1 to the length to write out the encapsulator also 
1245                 out.append(value, start, pos + 1); 
1246                 // put the next starting position on the encapsulator so we will 
1247                 // write it out again with the next string (effectively doubling it) 
1248                 start = pos; 
1249             } 
1250             pos++; 
1251         } 
1252  
1253         // write the last segment 
1254         out.append(value, start, pos); 
1255         out.append(quoteChar); 
1256     }

1357     private CharSequence trim(final CharSequence charSequence) { 
1358         if (charSequence instanceof String) { 
1359             return ((String) charSequence).trim(); 
1360         } 
1361         final int count = charSequence.length(); 
1362         int len = count; 
1363         int pos = 0; 
1364  
1365         while (pos < len && charSequence.charAt(pos) <= SP) { 
1366             pos++; 
1367         } 
1368         while (pos < len && charSequence.charAt(len - 1) <= SP) { 
1369             len--; 
1370         } 
1371         return pos > 0 || len < count ? charSequence.subSequence(pos, len) : charSequence; 
1372     }

1374     /** 
1375      * Verifies the consistency of the parameters and throws an IllegalArgumentException if necessary. 
1376      * 
1377      * @throws IllegalArgumentException 
1378      */ 
1379     private void validate() throws IllegalArgumentException { 
1380         if (isLineBreak(delimiter)) { 
1381             throw new IllegalArgumentException("The delimiter cannot be a line break"); 
1382         } 
1383  
1384         if (quoteCharacter != null && delimiter == quoteCharacter.charValue()) { 
1385             throw new IllegalArgumentException( 
1386                     "The quoteChar character and the delimiter cannot be the same ('" + quoteCharacter + "')"); 
1387         } 
1388  
1389         if (escapeCharacter != null && delimiter == escapeCharacter.charValue()) { 
1390             throw new IllegalArgumentException( 
1391                     "The escape character and the delimiter cannot be the same ('" + escapeCharacter + "')"); 
1392         } 
1393  
1394         if (commentMarker != null && delimiter == commentMarker.charValue()) { 
1395             throw new IllegalArgumentException( 
1396                     "The comment start character and the delimiter cannot be the same ('" + commentMarker + "')"); 
1397         } 
1398  
1399         if (quoteCharacter != null && quoteCharacter.equals(commentMarker)) { 
1400             throw new IllegalArgumentException( 
1401                     "The comment start character and the quoteChar cannot be the same ('" + commentMarker + "')"); 
1402         } 
1403  
1404         if (escapeCharacter != null && escapeCharacter.equals(commentMarker)) { 
1405             throw new IllegalArgumentException( 
1406                     "The comment start and the escape character cannot be the same ('" + commentMarker + "')"); 
1407         } 
1408  
1409         if (escapeCharacter == null && quoteMode == QuoteMode.NONE) { 
1410             throw new IllegalArgumentException("No quotes mode set but no escape character is set"); 
1411         } 
1412  
1413         // validate header 
1414         if (header != null) { 
1415             final Set<String> dupCheck = new HashSet<>(); 
1416             for (final String hdr : header) { 
1417                 if (!dupCheck.add(hdr)) { 
1418                     throw new IllegalArgumentException( 
1419                             "The header contains a duplicate entry: '" + hdr + "' in " + Arrays.toString(header)); 
1420                 } 
1421             } 
1422         } 
1423     }

  49 /** 
  50  * Specifies the format of a CSV file and parses input. 
  51  * 
  52  * <h2>Using predefined formats</h2> 
  53  * 
  54  * <p> 
  55  * You can use one of the predefined formats: 
  56  * </p> 
  57  * 
  58  * <ul> 
  59  * <li>{@link #DEFAULT}</li> 
  60  * <li>{@link #EXCEL}</li> 
  61  * <li>{@link #MYSQL}</li> 
  62  * <li>{@link #RFC4180}</li> 
  63  * <li>{@link #TDF}</li> 
  64  * </ul> 
  65  * 
  66  * <p> 
  67  * For example: 
  68  * </p> 
  69  * 
  70  * <pre> 
  71  * CSVParser parser = CSVFormat.EXCEL.parse(reader); 
  72  * </pre> 
  73  * 
  74  * <p> 
  75  * The {@link CSVParser} provides static methods to parse other input types, for example: 
  76  * </p> 
  77  * 
  78  * <pre> 
  79  * CSVParser parser = CSVParser.parse(file, StandardCharsets.US_ASCII, CSVFormat.EXCEL); 
  80  * </pre> 
  81  * 
  82  * <h2>Defining formats</h2> 
  83  * 
  84  * <p> 
  85  * You can extend a format by calling the {@code with} methods. For example: 
  86  * </p> 
  87  * 
  88  * <pre> 
  89  * CSVFormat.EXCEL.withNullString(&quot;N/A&quot;).withIgnoreSurroundingSpaces(true); 
  90  * </pre> 
  91  * 
  92  * <h2>Defining column names</h2> 
  93  * 
  94  * <p> 
  95  * To define the column names you want to use to access records, write: 
  96  * </p> 
  97  * 
  98  * <pre> 
  99  * CSVFormat.EXCEL.withHeader(&quot;Col1&quot;, &quot;Col2&quot;, &quot;Col3&quot;); 
 100  * </pre> 
 101  * 
 102  * <p> 
 103  * Calling {@link #withHeader(String...)} let's you use the given names to address values in a {@link CSVRecord}, and 
 104  * assumes that your CSV source does not contain a first record that also defines column names. 
 105  * 
 106  * If it does, then you are overriding this metadata with your names and you should skip the first record by calling 
 107  * {@link #withSkipHeaderRecord(boolean)} with {@code true}. 
 108  * </p> 
 109  * 
 110  * <h2>Parsing</h2> 
 111  * 
 112  * <p> 
 113  * You can use a format directly to parse a reader. For example, to parse an Excel file with columns header, write: 
 114  * </p> 
 115  * 
 116  * <pre> 
 117  * Reader in = ...; 
 118  * CSVFormat.EXCEL.withHeader(&quot;Col1&quot;, &quot;Col2&quot;, &quot;Col3&quot;).parse(in); 
 119  * </pre> 
 120  * 
 121  * <p> 
 122  * For other input types, like resources, files, and URLs, use the static methods on {@link CSVParser}. 
 123  * </p> 
 124  * 
 125  * <h2>Referencing columns safely</h2> 
 126  * 
 127  * <p> 
 128  * If your source contains a header record, you can simplify your code and safely reference columns, by using 
 129  * {@link #withHeader(String...)} with no arguments: 
 130  * </p> 
 131  * 
 132  * <pre> 
 133  * CSVFormat.EXCEL.withHeader(); 
 134  * </pre> 
 135  * 
 136  * <p> 
 137  * This causes the parser to read the first record and use its values as column names. 
 138  * 
 139  * Then, call one of the {@link CSVRecord} get method that takes a String column name argument: 
 140  * </p> 
 141  * 
 142  * <pre> 
 143  * String value = record.get(&quot;Col1&quot;); 
 144  * </pre> 
 145  * 
 146  * <p> 
 147  * This makes your code impervious to changes in column order in the CSV file. 
 148  * </p> 
 149  * 
 150  * <h2>Notes</h2> 
 151  * 
 152  * <p> 
 153  * This class is immutable. 
 154  * </p> 
 155  */

1090     private void print(final Object object, final CharSequence value, final int offset, final int len, 
1091             final Appendable out, final boolean newRecord) throws IOException { 
1092         if (!newRecord) { 
1093             out.append(getDelimiter()); 
1094         } 
1095         if (object == null) { 
1096             out.append(value); 
1097         } else if (isQuoteCharacterSet()) { 
1098             // the original object is needed so can check for Number 
1099             printAndQuote(object, value, offset, len, out, newRecord); 
1100         } else if (isEscapeCharacterSet()) { 
1101             printAndEscape(value, offset, len, out); 
1102         } else { 
1103             out.append(value, offset, offset + len); 
1104         } 
1105     }

1107     /* 
1108      * Note: must only be called if escaping is enabled, otherwise will generate NPE 
1109      */ 
1110     private void printAndEscape(final CharSequence value, final int offset, final int len, final Appendable out) 
1111             throws IOException { 
1112         int start = offset; 
1113         int pos = offset; 
1114         final int end = offset + len; 
1115  
1116         final char delim = getDelimiter(); 
1117         final char escape = getEscapeCharacter().charValue(); 
1118  
1119         while (pos < end) { 
1120             char c = value.charAt(pos); 
1121             if (c == CR || c == LF || c == delim || c == escape) { 
1122                 // write out segment up until this char 
1123                 if (pos > start) { 
1124                     out.append(value, start, pos); 
1125                 } 
1126                 if (c == LF) { 
1127                     c = 'n'; 
1128                 } else if (c == CR) { 
1129                     c = 'r'; 
1130                 } 
1131  
1132                 out.append(escape); 
1133                 out.append(c); 
1134  
1135                 start = pos + 1; // start on the current char after this one 
1136             } 
1137  
1138             pos++; 
1139         } 
1140  
1141         // write last segment 
1142         if (pos > start) { 
1143             out.append(value, start, pos); 
1144         } 
1145     }

1147     /* 
1148      * Note: must only be called if quoting is enabled, otherwise will generate NPE 
1149      */ 
1150     // the original object is needed so can check for Number 
1151     private void printAndQuote(final Object object, final CharSequence value, final int offset, final int len, 
1152             final Appendable out, final boolean newRecord) throws IOException { 
1153         boolean quote = false; 
1154         int start = offset; 
1155         int pos = offset; 
1156         final int end = offset + len; 
1157  
1158         final char delimChar = getDelimiter(); 
1159         final char quoteChar = getQuoteCharacter().charValue(); 
1160  
1161         QuoteMode quoteModePolicy = getQuoteMode(); 
1162         if (quoteModePolicy == null) { 
1163             quoteModePolicy = QuoteMode.MINIMAL; 
1164         } 
1165         switch (quoteModePolicy) { 
1166         case ALL: 
1167         case ALL_NON_NULL: 
1168             quote = true; 
1169             break; 
1170         case NON_NUMERIC: 
1171             quote = !(object instanceof Number); 
1172             break; 
1173         case NONE: 
1174             // Use the existing escaping code 
1175             printAndEscape(value, offset, len, out); 
1176             return; 
1177         case MINIMAL: 
1178             if (len <= 0) { 
1179                 // always quote an empty token that is the first 
1180                 // on the line, as it may be the only thing on the 
1181                 // line. If it were not quoted in that case, 
1182                 // an empty line has no tokens. 
1183                 if (newRecord) { 
1184                     quote = true; 
1185                 } 
1186             } else { 
1187                 char c = value.charAt(pos); 
1188  
1189                 if (c <= COMMENT) { 
1190                     // Some other chars at the start of a value caused the parser to fail, so for now 
1191                     // encapsulate if we start in anything less than '#'. We are being conservative 
1192                     // by including the default comment char too. 
1193                     quote = true; 
1194                 } else { 
1195                     while (pos < end) { 
1196                         c = value.charAt(pos); 
1197                         if (c == LF || c == CR || c == quoteChar || c == delimChar) { 
1198                             quote = true; 
1199                             break; 
1200                         } 
1201                         pos++; 
1202                     } 
1203  
1204                     if (!quote) { 
1205                         pos = end - 1; 
1206                         c = value.charAt(pos); 
1207                         // Some other chars at the end caused the parser to fail, so for now 
1208                         // encapsulate if we end in anything less than ' ' 
1209                         if (c <= SP) { 
1210                             quote = true; 
1211                         } 
1212                     } 
1213                 } 
1214             } 
1215  
1216             if (!quote) { 
1217                 // no encapsulation needed - write out the original value 
1218                 out.append(value, start, end); 
1219                 return; 
1220             } 
1221             break; 
1222         default: 
1223             throw new IllegalStateException("Unexpected Quote value: " + quoteModePolicy); 
1224         } 
1225  
1226         if (!quote) { 
1227             // no encapsulation needed - write out the original value 
1228             out.append(value, start, end); 
1229             return; 
1230         } 
1231  
1232         // we hit something that needed encapsulation 
1233         out.append(quoteChar); 
1234  
1235         // Pick up where we left off: pos should be positioned on the first character that caused 
1236         // the need for encapsulation. 
1237         while (pos < end) { 
1238             final char c = value.charAt(pos); 
1239             if (c == quoteChar) { 
1240                 // write out the chunk up until this point 
1241  
1242                 // add 1 to the length to write out the encapsulator also 
1243                 out.append(value, start, pos + 1); 
1244                 // put the next starting position on the encapsulator so we will 
1245                 // write it out again with the next string (effectively doubling it) 
1246                 start = pos; 
1247             } 
1248             pos++; 
1249         } 
1250  
1251         // write the last segment 
1252         out.append(value, start, pos); 
1253         out.append(quoteChar); 
1254     }

1355     private CharSequence trim(final CharSequence charSequence) { 
1356         if (charSequence instanceof String) { 
1357             return ((String) charSequence).trim(); 
1358         } 
1359         final int count = charSequence.length(); 
1360         int len = count; 
1361         int pos = 0; 
1362  
1363         while (pos < len && charSequence.charAt(pos) <= SP) { 
1364             pos++; 
1365         } 
1366         while (pos < len && charSequence.charAt(len - 1) <= SP) { 
1367             len--; 
1368         } 
1369         return pos > 0 || len < count ? charSequence.subSequence(pos, len) : charSequence; 
1370     }

1372     /** 
1373      * Verifies the consistency of the parameters and throws an IllegalArgumentException if necessary. 
1374      * 
1375      * @throws IllegalArgumentException 
1376      */ 
1377     private void validate() throws IllegalArgumentException { 
1378         if (isLineBreak(delimiter)) { 
1379             throw new IllegalArgumentException("The delimiter cannot be a line break"); 
1380         } 
1381  
1382         if (quoteCharacter != null && delimiter == quoteCharacter.charValue()) { 
1383             throw new IllegalArgumentException( 
1384                     "The quoteChar character and the delimiter cannot be the same ('" + quoteCharacter + "')"); 
1385         } 
1386  
1387         if (escapeCharacter != null && delimiter == escapeCharacter.charValue()) { 
1388             throw new IllegalArgumentException( 
1389                     "The escape character and the delimiter cannot be the same ('" + escapeCharacter + "')"); 
1390         } 
1391  
1392         if (commentMarker != null && delimiter == commentMarker.charValue()) { 
1393             throw new IllegalArgumentException( 
1394                     "The comment start character and the delimiter cannot be the same ('" + commentMarker + "')"); 
1395         } 
1396  
1397         if (quoteCharacter != null && quoteCharacter.equals(commentMarker)) { 
1398             throw new IllegalArgumentException( 
1399                     "The comment start character and the quoteChar cannot be the same ('" + commentMarker + "')"); 
1400         } 
1401  
1402         if (escapeCharacter != null && escapeCharacter.equals(commentMarker)) { 
1403             throw new IllegalArgumentException( 
1404                     "The comment start and the escape character cannot be the same ('" + commentMarker + "')"); 
1405         } 
1406  
1407         if (escapeCharacter == null && quoteMode == QuoteMode.NONE) { 
1408             throw new IllegalArgumentException("No quotes mode set but no escape character is set"); 
1409         } 
1410  
1411         // validate header 
1412         if (header != null) { 
1413             final Set<String> dupCheck = new HashSet<>(); 
1414             for (final String hdr : header) { 
1415                 if (!dupCheck.add(hdr)) { 
1416                     throw new IllegalArgumentException( 
1417                             "The header contains a duplicate entry: '" + hdr + "' in " + Arrays.toString(header)); 
1418                 } 
1419             } 
1420         } 
1421     }
