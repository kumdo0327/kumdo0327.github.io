112     def get_memory_facts(self): 
113         memory_facts = {} 
114         if not os.access("/proc/meminfo", os.R_OK): 
115             return memory_facts 
116  
117         memstats = {} 
118         for line in get_file_lines("/proc/meminfo"): 
119             data = line.split(":", 1) 
120             key = data[0] 
121             if key in self.ORIGINAL_MEMORY_FACTS: 
122                 val = data[1].strip().split(' ')[0] 
123                 memory_facts["%s_mb" % key.lower()] = int(val) // 1024 
124  
125             if key in self.MEMORY_FACTS: 
126                 val = data[1].strip().split(' ')[0] 
127                 memstats[key.lower()] = int(val) // 1024 
128  
129         if None not in (memstats.get('memtotal'), memstats.get('memfree')): 
130             memstats['real:used'] = memstats['memtotal'] - memstats['memfree'] 
131         if None not in (memstats.get('cached'), memstats.get('memfree'), memstats.get('buffers')): 
132             memstats['nocache:free'] = memstats['cached'] + memstats['memfree'] + memstats['buffers'] 
133         if None not in (memstats.get('memtotal'), memstats.get('nocache:free')): 
134             memstats['nocache:used'] = memstats['memtotal'] - memstats['nocache:free'] 
135         if None not in (memstats.get('swaptotal'), memstats.get('swapfree')): 
136             memstats['swap:used'] = memstats['swaptotal'] - memstats['swapfree'] 
137  
138         memory_facts['memory_mb'] = { 
139             'real': { 
140                 'total': memstats.get('memtotal'), 
141                 'used': memstats.get('real:used'), 
142                 'free': memstats.get('memfree'), 
143             }, 
144             'nocache': { 
145                 'free': memstats.get('nocache:free'), 
146                 'used': memstats.get('nocache:used'), 
147             }, 
148             'swap': { 
149                 'total': memstats.get('swaptotal'), 
150                 'free': memstats.get('swapfree'), 
151                 'used': memstats.get('swap:used'), 
152                 'cached': memstats.get('swapcached'), 
153             }, 
154         } 
155  
156         return memory_facts

158     def get_cpu_facts(self, collected_facts=None): 
159         cpu_facts = {} 
160         collected_facts = collected_facts or {} 
161  
162         i = 0 
163         vendor_id_occurrence = 0 
164         model_name_occurrence = 0 
165         processor_occurence = 0 
166         physid = 0 
167         coreid = 0 
168         sockets = {} 
169         cores = {} 
170  
171         xen = False 
172         xen_paravirt = False 
173         try: 
174             if os.path.exists('/proc/xen'): 
175                 xen = True 
176             else: 
177                 for line in get_file_lines('/sys/hypervisor/type'): 
178                     if line.strip() == 'xen': 
179                         xen = True 
180                     # Only interested in the first line 
181                     break 
182         except IOError: 
183             pass 
184  
185         if not os.access("/proc/cpuinfo", os.R_OK): 
186             return cpu_facts 
187  
188         cpu_facts['processor'] = [] 
189         for line in get_file_lines('/proc/cpuinfo'): 
190             data = line.split(":", 1) 
191             key = data[0].strip() 
192  
193             try: 
194                 val = data[1].strip() 
195             except IndexError: 
196                 val = "" 
197  
198             if xen: 
199                 if key == 'flags': 
200                     # Check for vme cpu flag, Xen paravirt does not expose this. 
201                     #   Need to detect Xen paravirt because it exposes cpuinfo 
202                     #   differently than Xen HVM or KVM and causes reporting of 
203                     #   only a single cpu core. 
204                     if 'vme' not in val: 
205                         xen_paravirt = True 
206  
207             # model name is for Intel arch, Processor (mind the uppercase P) 
208             # works for some ARM devices, like the Sheevaplug. 
209             # 'ncpus active' is SPARC attribute 
210             if key in ['model name', 'Processor', 'vendor_id', 'cpu', 'Vendor', 'processor']: 
211                 if 'processor' not in cpu_facts: 
212                     cpu_facts['processor'] = [] 
213                 cpu_facts['processor'].append(val) 
214                 if key == 'vendor_id': 
215                     vendor_id_occurrence += 1 
216                 if key == 'model name': 
217                     model_name_occurrence += 1 
218                 if key == 'processor': 
219                     processor_occurence += 1 
220                 i += 1 
221             elif key == 'physical id': 
222                 physid = val 
223                 if physid not in sockets: 
224                     sockets[physid] = 1 
225             elif key == 'core id': 
226                 coreid = val 
227                 if coreid not in sockets: 
228                     cores[coreid] = 1 
229             elif key == 'cpu cores': 
230                 sockets[physid] = int(val) 
231             elif key == 'siblings': 
232                 cores[coreid] = int(val) 
233             elif key == '# processors': 
234                 cpu_facts['processor_cores'] = int(val) 
235             elif key == 'ncpus active': 
236                 i = int(val) 
237  
238         # Skip for platforms without vendor_id/model_name in cpuinfo (e.g ppc64le) 
239         if vendor_id_occurrence > 0: 
240             if vendor_id_occurrence == model_name_occurrence: 
241                 i = vendor_id_occurrence 
242  
243         # The fields for ARM CPUs do not always include 'vendor_id' or 'model name', 
244         # and sometimes includes both 'processor' and 'Processor'. 
245         # Always use 'processor' count for ARM systems 
246         if collected_facts.get('ansible_architecture', '').startswith(('armv', 'aarch')): 
247             i = processor_occurence 
248  
249         # FIXME 
250         if collected_facts.get('ansible_architecture') != 's390x': 
251             if xen_paravirt: 
252                 cpu_facts['processor_count'] = i 
253                 cpu_facts['processor_cores'] = i 
254                 cpu_facts['processor_threads_per_core'] = 1 
255                 cpu_facts['processor_vcpus'] = i 
256             else: 
257                 if sockets: 
258                     cpu_facts['processor_count'] = len(sockets) 
259                 else: 
260                     cpu_facts['processor_count'] = i 
261  
262                 socket_values = list(sockets.values()) 
263                 if socket_values and socket_values[0]: 
264                     cpu_facts['processor_cores'] = socket_values[0] 
265                 else: 
266                     cpu_facts['processor_cores'] = 1 
267  
268                 core_values = list(cores.values()) 
269                 if core_values: 
270                     cpu_facts['processor_threads_per_core'] = core_values[0] // cpu_facts['processor_cores'] 
271                 else: 
272                     cpu_facts['processor_threads_per_core'] = 1 // cpu_facts['processor_cores'] 
273  
274                 cpu_facts['processor_vcpus'] = (cpu_facts['processor_threads_per_core'] * 
275                                                 cpu_facts['processor_count'] * cpu_facts['processor_cores']) 
276  
277         return cpu_facts

368     def _lsblk_uuid(self): 
369         uuids = {} 
370         lsblk_path = self.module.get_bin_path("lsblk") 
371         if not lsblk_path: 
372             return uuids 
373  
374         rc, out, err = self._run_lsblk(lsblk_path) 
375         if rc != 0: 
376             return uuids 
377  
378         # each line will be in format: 
379         # <devicename><some whitespace><uuid> 
380         # /dev/sda1  32caaec3-ef40-4691-a3b6-438c3f9bc1c0 
381         for lsblk_line in out.splitlines(): 
382             if not lsblk_line: 
383                 continue 
384  
385             line = lsblk_line.strip() 
386             fields = line.rsplit(None, 1) 
387  
388             if len(fields) < 2: 
389                 continue 
390  
391             device_name, uuid = fields[0].strip(), fields[1].strip() 
392             if device_name in uuids: 
393                 continue 
394             uuids[device_name] = uuid 
395  
396         return uuids

398     def _udevadm_uuid(self, device): 
399         # fallback for versions of lsblk <= 2.23 that don't have --paths, see _run_lsblk() above 
400         uuid = 'N/A' 
401  
402         udevadm_path = self.module.get_bin_path('udevadm') 
403         if not udevadm_path: 
404             return uuid 
405  
406         cmd = [udevadm_path, 'info', '--query', 'property', '--name', device] 
407         rc, out, err = self.module.run_command(cmd) 
408         if rc != 0: 
409             return uuid 
410  
411         # a snippet of the output of the udevadm command below will be: 
412         # ... 
413         # ID_FS_TYPE=ext4 
414         # ID_FS_USAGE=filesystem 
415         # ID_FS_UUID=57b1a3e7-9019-4747-9809-7ec52bba9179 
416         # ... 
417         m = re.search('ID_FS_UUID=(.*)\n', out) 
418         if m: 
419             uuid = m.group(1) 
420  
421         return uuid

429     def _find_bind_mounts(self): 
430         bind_mounts = set() 
431         findmnt_path = self.module.get_bin_path("findmnt") 
432         if not findmnt_path: 
433             return bind_mounts 
434  
435         rc, out, err = self._run_findmnt(findmnt_path) 
436         if rc != 0: 
437             return bind_mounts 
438  
439         # find bind mounts, in case /etc/mtab is a symlink to /proc/mounts 
440         for line in out.splitlines(): 
441             fields = line.split() 
442             # fields[0] is the TARGET, fields[1] is the SOURCE 
443             if len(fields) < 2: 
444                 continue 
445  
446             # bind mounts will have a [/directory_name] in the SOURCE column 
447             if self.BIND_MOUNT_RE.match(fields[1]): 
448                 bind_mounts.add(fields[0]) 
449  
450         return bind_mounts

485     def get_mount_facts(self): 
486  
487         mounts = [] 
488  
489         # gather system lists 
490         bind_mounts = self._find_bind_mounts() 
491         uuids = self._lsblk_uuid() 
492         mtab_entries = self._mtab_entries() 
493  
494         # start threads to query each mount 
495         results = {} 
496         pool = ThreadPool(processes=min(len(mtab_entries), cpu_count())) 
497         maxtime = globals().get('GATHER_TIMEOUT') or timeout.DEFAULT_GATHER_TIMEOUT 
498         for fields in mtab_entries: 
499             # Transform octal escape sequences 
500             fields = [self._replace_octal_escapes(field) for field in fields] 
501  
502             device, mount, fstype, options = fields[0], fields[1], fields[2], fields[3] 
503  
504             if not device.startswith('/') and ':/' not in device or fstype == 'none': 
505                 continue 
506  
507             mount_info = {'mount': mount, 
508                           'device': device, 
509                           'fstype': fstype, 
510                           'options': options} 
511  
512             if mount in bind_mounts: 
513                 # only add if not already there, we might have a plain /etc/mtab 
514                 if not self.MTAB_BIND_MOUNT_RE.match(options): 
515                     mount_info['options'] += ",bind" 
516  
517             results[mount] = {'info': mount_info, 
518                               'extra': pool.apply_async(self.get_mount_info, (mount, device, uuids)), 
519                               'timelimit': time.time() + maxtime} 
520  
521         pool.close()  # done with new workers, start gc 
522  
523         # wait for workers and get results 
524         while results: 
525             for mount in results: 
526                 res = results[mount]['extra'] 
527                 if res.ready(): 
528                     if res.successful(): 
529                         mount_size, uuid = res.get() 
530                         if mount_size: 
531                             results[mount]['info'].update(mount_size) 
532                         results[mount]['info']['uuid'] = uuid or 'N/A' 
533                     else: 
534                         # give incomplete data 
535                         errmsg = to_text(res.get()) 
536                         self.module.warn("Error prevented getting extra info for mount %s: %s." % (mount, errmsg)) 
537                         results[mount]['info']['note'] = 'Could not get extra information: %s.' % (errmsg) 
538  
539                     mounts.append(results[mount]['info']) 
540                     del results[mount] 
541                     break 
542                 elif time.time() > results[mount]['timelimit']: 
543                     results[mount]['info']['note'] = 'Timed out while attempting to get extra information.' 
544                     mounts.append(results[mount]['info']) 
545                     del results[mount] 
546                     break 
547             else: 
548                 # avoid cpu churn 
549                 time.sleep(0.1) 
550  
551         return {'mounts': mounts}

117         memstats = {}

158     def get_cpu_facts(self, collected_facts=None): 
159         cpu_facts = {} 
160         collected_facts = collected_facts or {} 
161  
162         i = 0 
163         vendor_id_occurrence = 0 
164         model_name_occurrence = 0 
165         processor_occurence = 0 
166         physid = 0 
167         coreid = 0 
168         sockets = {} 
169         cores = {} 
170  
171         xen = False 
172         xen_paravirt = False 
173         try: 
174             if os.path.exists('/proc/xen'): 
175                 xen = True 
176             else: 
177                 for line in get_file_lines('/sys/hypervisor/type'): 
178                     if line.strip() == 'xen': 
179                         xen = True 
180                     # Only interested in the first line 
181                     break 
182         except IOError: 
183             pass 
184  
185         if not os.access("/proc/cpuinfo", os.R_OK): 
186             return cpu_facts 
187  
188         cpu_facts['processor'] = [] 
189         for line in get_file_lines('/proc/cpuinfo'): 
190             data = line.split(":", 1) 
191             key = data[0].strip() 
192  
193             try: 
194                 val = data[1].strip() 
195             except IndexError: 
196                 val = "" 
197  
198             if xen: 
199                 if key == 'flags': 
200                     # Check for vme cpu flag, Xen paravirt does not expose this. 
201                     #   Need to detect Xen paravirt because it exposes cpuinfo 
202                     #   differently than Xen HVM or KVM and causes reporting of 
203                     #   only a single cpu core. 
204                     if 'vme' not in val: 
205                         xen_paravirt = True 
206  
207             # model name is for Intel arch, Processor (mind the uppercase P) 
208             # works for some ARM devices, like the Sheevaplug. 
209             # 'ncpus active' is SPARC attribute 
210             if key in ['model name', 'Processor', 'vendor_id', 'cpu', 'Vendor', 'processor']: 
211                 if 'processor' not in cpu_facts: 
212                     cpu_facts['processor'] = [] 
213                 cpu_facts['processor'].append(val) 
214                 if key == 'vendor_id': 
215                     vendor_id_occurrence += 1 
216                 if key == 'model name': 
217                     model_name_occurrence += 1 
218                 if key == 'processor': 
219                     processor_occurence += 1 
220                 i += 1 
221             elif key == 'physical id': 
222                 physid = val 
223                 if physid not in sockets: 
224                     sockets[physid] = 1 
225             elif key == 'core id': 
226                 coreid = val 
227                 if coreid not in sockets: 
228                     cores[coreid] = 1 
229             elif key == 'cpu cores': 
230                 sockets[physid] = int(val) 
231             elif key == 'siblings': 
232                 cores[coreid] = int(val) 
233             elif key == '# processors': 
234                 cpu_facts['processor_cores'] = int(val) 
235             elif key == 'ncpus active': 
236                 i = int(val) 
237  
238         # Skip for platforms without vendor_id/model_name in cpuinfo (e.g ppc64le) 
239         if vendor_id_occurrence > 0: 
240             if vendor_id_occurrence == model_name_occurrence: 
241                 i = vendor_id_occurrence 
242  
243         # The fields for ARM CPUs do not always include 'vendor_id' or 'model name', 
244         # and sometimes includes both 'processor' and 'Processor'. 
245         # The fields for Power CPUs include 'processor' and 'cpu'. 
246         # Always use 'processor' count for ARM and Power systems 
247         if collected_facts.get('ansible_architecture', '').startswith(('armv', 'aarch', 'ppc')): 
248             i = processor_occurence 
249  
250         # FIXME 
251         if collected_facts.get('ansible_architecture') != 's390x': 
252             if xen_paravirt: 
253                 cpu_facts['processor_count'] = i 
254                 cpu_facts['processor_cores'] = i 
255                 cpu_facts['processor_threads_per_core'] = 1 
256                 cpu_facts['processor_vcpus'] = i 
257             else: 
258                 if sockets: 
259                     cpu_facts['processor_count'] = len(sockets) 
260                 else: 
261                     cpu_facts['processor_count'] = i 
262  
263                 socket_values = list(sockets.values()) 
264                 if socket_values and socket_values[0]: 
265                     cpu_facts['processor_cores'] = socket_values[0] 
266                 else: 
267                     cpu_facts['processor_cores'] = 1 
268  
269                 core_values = list(cores.values()) 
270                 if core_values: 
271                     cpu_facts['processor_threads_per_core'] = core_values[0] // cpu_facts['processor_cores'] 
272                 else: 
273                     cpu_facts['processor_threads_per_core'] = 1 // cpu_facts['processor_cores'] 
274  
275                 cpu_facts['processor_vcpus'] = (cpu_facts['processor_threads_per_core'] * 
276                                                 cpu_facts['processor_count'] * cpu_facts['processor_cores']) 
277  
278         return cpu_facts

400         # fallback for versions of lsblk <= 2.23 that don't have --paths, see _run_lsblk() above 
401         uuid = 'N/A' 
402  
403         udevadm_path = self.module.get_bin_path('udevadm') 
404         if not udevadm_path: 
405             return uuid 
406  
407         cmd = [udevadm_path, 'info', '--query', 'property', '--name', device] 
408         rc, out, err = self.module.run_command(cmd) 
409         if rc != 0: 
410             return uuid 
411  
412         # a snippet of the output of the udevadm command below will be: 
413         # ... 
414         # ID_FS_TYPE=ext4 
415         # ID_FS_USAGE=filesystem 
416         # ID_FS_UUID=57b1a3e7-9019-4747-9809-7ec52bba9179 
417         # ... 
418         m = re.search('ID_FS_UUID=(.*)\n', out) 
419         if m: 
420             uuid = m.group(1) 
421  
422         return uuid

430     def _find_bind_mounts(self): 
431         bind_mounts = set() 
432         findmnt_path = self.module.get_bin_path("findmnt") 
433         if not findmnt_path: 
434             return bind_mounts 
435  
436         rc, out, err = self._run_findmnt(findmnt_path) 
437         if rc != 0: 
438             return bind_mounts 
439  
440         # find bind mounts, in case /etc/mtab is a symlink to /proc/mounts 
441         for line in out.splitlines(): 
442             fields = line.split() 
443             # fields[0] is the TARGET, fields[1] is the SOURCE 
444             if len(fields) < 2: 
445                 continue 
446  
447             # bind mounts will have a [/directory_name] in the SOURCE column 
448             if self.BIND_MOUNT_RE.match(fields[1]): 
449                 bind_mounts.add(fields[0]) 
450  
451         return bind_mounts

475     def get_mount_info(self, mount, device, uuids): 
476  
477         mount_size = get_mount_size(mount) 
478  
479         # _udevadm_uuid is a fallback for versions of lsblk <= 2.23 that don't have --paths 
480         # see _run_lsblk() above 
481         # https://github.com/ansible/ansible/issues/36077 
482         uuid = uuids.get(device, self._udevadm_uuid(device)) 
483  
484         return mount_size, uuid

486     def get_mount_facts(self): 
487  
488         mounts = [] 
489  
490         # gather system lists 
491         bind_mounts = self._find_bind_mounts() 
492         uuids = self._lsblk_uuid() 
493         mtab_entries = self._mtab_entries() 
494  
495         # start threads to query each mount 
496         results = {} 
497         pool = ThreadPool(processes=min(len(mtab_entries), cpu_count())) 
498         maxtime = globals().get('GATHER_TIMEOUT') or timeout.DEFAULT_GATHER_TIMEOUT 
499         for fields in mtab_entries: 
500             # Transform octal escape sequences 
501             fields = [self._replace_octal_escapes(field) for field in fields] 
502  
503             device, mount, fstype, options = fields[0], fields[1], fields[2], fields[3] 
504  
505             if not device.startswith('/') and ':/' not in device or fstype == 'none': 
506                 continue 
507  
508             mount_info = {'mount': mount, 
509                           'device': device, 
510                           'fstype': fstype, 
511                           'options': options} 
512  
513             if mount in bind_mounts: 
514                 # only add if not already there, we might have a plain /etc/mtab 
515                 if not self.MTAB_BIND_MOUNT_RE.match(options): 
516                     mount_info['options'] += ",bind" 
517  
518             results[mount] = {'info': mount_info, 
519                               'extra': pool.apply_async(self.get_mount_info, (mount, device, uuids)), 
520                               'timelimit': time.time() + maxtime} 
521  
522         pool.close()  # done with new workers, start gc 
523  
524         # wait for workers and get results 
525         while results: 
526             for mount in results: 
527                 res = results[mount]['extra'] 
528                 if res.ready(): 
529                     if res.successful(): 
530                         mount_size, uuid = res.get() 
531                         if mount_size: 
532                             results[mount]['info'].update(mount_size) 
533                         results[mount]['info']['uuid'] = uuid or 'N/A' 
534                     else: 
535                         # give incomplete data 
536                         errmsg = to_text(res.get()) 
537                         self.module.warn("Error prevented getting extra info for mount %s: %s." % (mount, errmsg)) 
538                         results[mount]['info']['note'] = 'Could not get extra information: %s.' % (errmsg) 
539  
540                     mounts.append(results[mount]['info']) 
541                     del results[mount] 
542                     break 
543                 elif time.time() > results[mount]['timelimit']: 
544                     results[mount]['info']['note'] = 'Timed out while attempting to get extra information.' 
545                     mounts.append(results[mount]['info']) 
546                     del results[mount] 
547                     break 
548             else: 
549                 # avoid cpu churn 
550                 time.sleep(0.1) 
551  
552         return {'mounts': mounts}

601     def get_device_facts(self): 
602         device_facts = {} 
603  
604         device_facts['devices'] = {} 
605         lspci = self.module.get_bin_path('lspci') 
606         if lspci: 
607             rc, pcidata, err = self.module.run_command([lspci, '-D'], errors='surrogate_then_replace') 
608         else: 
609             pcidata = None 
610  
611         try: 
612             block_devs = os.listdir("/sys/block") 
613         except OSError: 
614             return device_facts 
615  
616         devs_wwn = {} 
617         try: 
618             devs_by_id = os.listdir("/dev/disk/by-id") 
619         except OSError: 
620             pass 
621         else: 
622             for link_name in devs_by_id: 
623                 if link_name.startswith("wwn-"): 
624                     try: 
625                         wwn_link = os.readlink(os.path.join("/dev/disk/by-id", link_name)) 
626                     except OSError: 
627                         continue 
628                     devs_wwn[os.path.basename(wwn_link)] = link_name[4:] 
629  
630         links = self.get_all_device_links() 
631         device_facts['device_links'] = links 
632  
633         for block in block_devs: 
634             virtual = 1 
635             sysfs_no_links = 0 
636             try: 
637                 path = os.readlink(os.path.join("/sys/block/", block)) 
638             except OSError: 
639                 e = sys.exc_info()[1] 
640                 if e.errno == errno.EINVAL: 
641                     path = block 
642                     sysfs_no_links = 1 
643                 else: 
644                     continue 
645             sysdir = os.path.join("/sys/block", path) 
646             if sysfs_no_links == 1: 
647                 for folder in os.listdir(sysdir): 
648                     if "device" in folder: 
649                         virtual = 0 
650                         break 
651             d = {} 
652             d['virtual'] = virtual 
653             d['links'] = {} 
654             for (link_type, link_values) in iteritems(links): 
655                 d['links'][link_type] = link_values.get(block, []) 
656             diskname = os.path.basename(sysdir) 
657             for key in ['vendor', 'model', 'sas_address', 'sas_device_handle']: 
658                 d[key] = get_file_content(sysdir + "/device/" + key) 
659  
660             sg_inq = self.module.get_bin_path('sg_inq') 
661  
662             if sg_inq: 
663                 device = "/dev/%s" % (block) 
664                 rc, drivedata, err = self.module.run_command([sg_inq, device]) 
665                 if rc == 0: 
666                     serial = re.search(r"Unit serial number:\s+(\w+)", drivedata) 
667                     if serial: 
668                         d['serial'] = serial.group(1) 
669  
670             for key, test in [('removable', '/removable'), 
671                               ('support_discard', '/queue/discard_granularity'), 
672                               ]: 
673                 d[key] = get_file_content(sysdir + test) 
674  
675             if diskname in devs_wwn: 
676                 d['wwn'] = devs_wwn[diskname] 
677  
678             d['partitions'] = {} 
679             for folder in os.listdir(sysdir): 
680                 m = re.search("(" + diskname + r"[p]?\d+)", folder) 
681                 if m: 
682                     part = {} 
683                     partname = m.group(1) 
684                     part_sysdir = sysdir + "/" + partname 
685  
686                     part['links'] = {} 
687                     for (link_type, link_values) in iteritems(links): 
688                         part['links'][link_type] = link_values.get(partname, []) 
689  
690                     part['start'] = get_file_content(part_sysdir + "/start", 0) 
691                     part['sectors'] = get_file_content(part_sysdir + "/size", 0) 
692  
693                     part['sectorsize'] = get_file_content(part_sysdir + "/queue/logical_block_size") 
694                     if not part['sectorsize']: 
695                         part['sectorsize'] = get_file_content(part_sysdir + "/queue/hw_sector_size", 512) 
696                     part['size'] = bytes_to_human((float(part['sectors']) * 512.0)) 
697                     part['uuid'] = get_partition_uuid(partname) 
698                     self.get_holders(part, part_sysdir) 
699  
700                     d['partitions'][partname] = part 
701  
702             d['rotational'] = get_file_content(sysdir + "/queue/rotational") 
703             d['scheduler_mode'] = "" 
704             scheduler = get_file_content(sysdir + "/queue/scheduler") 
705             if scheduler is not None: 
706                 m = re.match(r".*?(\[(.*)\])", scheduler) 
707                 if m: 
708                     d['scheduler_mode'] = m.group(2) 
709  
710             d['sectors'] = get_file_content(sysdir + "/size") 
711             if not d['sectors']: 
712                 d['sectors'] = 0 
713             d['sectorsize'] = get_file_content(sysdir + "/queue/logical_block_size") 
714             if not d['sectorsize']: 
715                 d['sectorsize'] = get_file_content(sysdir + "/queue/hw_sector_size", 512) 
716             d['size'] = bytes_to_human(float(d['sectors']) * 512.0) 
717  
718             d['host'] = "" 
719  
720             # domains are numbered (0 to ffff), bus (0 to ff), slot (0 to 1f), and function (0 to 7). 
721             m = re.match(r".+/([a-f0-9]{4}:[a-f0-9]{2}:[0|1][a-f0-9]\.[0-7])/", sysdir) 
722             if m and pcidata: 
723                 pciid = m.group(1) 
724                 did = re.escape(pciid) 
725                 m = re.search("^" + did + r"\s(.*)$", pcidata, re.MULTILINE) 
726                 if m: 
727                     d['host'] = m.group(1) 
728  
729             self.get_holders(d, sysdir) 
730  
731             device_facts['devices'][diskname] = d 
732  
733         return device_facts

755     def get_lvm_facts(self): 
756         """ Get LVM Facts if running as root and lvm utils are available """ 
757  
758         lvm_facts = {} 
759  
760         if os.getuid() == 0 and self.module.get_bin_path('vgs'): 
761             lvm_util_options = '--noheadings --nosuffix --units g --separator ,' 
762  
763             vgs_path = self.module.get_bin_path('vgs') 
764             # vgs fields: VG #PV #LV #SN Attr VSize VFree 
765             vgs = {} 
766             if vgs_path: 
767                 rc, vg_lines, err = self.module.run_command('%s %s' % (vgs_path, lvm_util_options)) 
768                 for vg_line in vg_lines.splitlines(): 
769                     items = vg_line.strip().split(',') 
770                     vgs[items[0]] = {'size_g': items[-2], 
771                                      'free_g': items[-1], 
772                                      'num_lvs': items[2], 
773                                      'num_pvs': items[1]} 
774  
775             lvs_path = self.module.get_bin_path('lvs') 
776             # lvs fields: 
777             # LV VG Attr LSize Pool Origin Data% Move Log Copy% Convert 
778             lvs = {} 
779             if lvs_path: 
780                 rc, lv_lines, err = self.module.run_command('%s %s' % (lvs_path, lvm_util_options)) 
781                 for lv_line in lv_lines.splitlines(): 
782                     items = lv_line.strip().split(',') 
783                     lvs[items[0]] = {'size_g': items[3], 'vg': items[1]} 
784  
785             pvs_path = self.module.get_bin_path('pvs') 
786             # pvs fields: PV VG #Fmt #Attr PSize PFree 
787             pvs = {} 
788             if pvs_path: 
789                 rc, pv_lines, err = self.module.run_command('%s %s' % (pvs_path, lvm_util_options)) 
790                 for pv_line in pv_lines.splitlines(): 
791                     items = pv_line.strip().split(',') 
792                     pvs[self._find_mapper_device_name(items[0])] = { 
793                         'size_g': items[4], 
794                         'free_g': items[5], 
795                         'vg': items[1]} 
796  
797             lvm_facts['lvm'] = {'lvs': lvs, 'vgs': vgs, 'pvs': pvs} 
798  
799         return lvm_facts
