  45 /** 
  46  * <p>An implementation of the active Covariance Matrix Adaptation Evolution Strategy (CMA-ES) 
  47  * for non-linear, non-convex, non-smooth, global function minimization. 
  48  * The CMA-Evolution Strategy (CMA-ES) is a reliable stochastic optimization method 
  49  * which should be applied if derivative-based methods, e.g. quasi-Newton BFGS or 
  50  * conjugate gradient, fail due to a rugged search landscape (e.g. noise, local 
  51  * optima, outlier, etc.) of the objective function. Like a 
  52  * quasi-Newton method, the CMA-ES learns and applies a variable metric 
  53  * on the underlying search space. Unlike a quasi-Newton method, the 
  54  * CMA-ES neither estimates nor uses gradients, making it considerably more 
  55  * reliable in terms of finding a good, or even close to optimal, solution.</p> 
  56  * 
  57  * <p>In general, on smooth objective functions the CMA-ES is roughly ten times 
  58  * slower than BFGS (counting objective function evaluations, no gradients provided). 
  59  * For up to <math>N=10</math> variables also the derivative-free simplex 
  60  * direct search method (Nelder and Mead) can be faster, but it is 
  61  * far less reliable than CMA-ES.</p> 
  62  * 
  63  * <p>The CMA-ES is particularly well suited for non-separable 
  64  * and/or badly conditioned problems. To observe the advantage of CMA compared 
  65  * to a conventional evolution strategy, it will usually take about 
  66  * <math>30 N</math> function evaluations. On difficult problems the complete 
  67  * optimization (a single run) is expected to take <em>roughly</em> between 
  68  * <math>30 N</math> and <math>300 N<sup>2</sup></math> 
  69  * function evaluations.</p> 
  70  * 
  71  * <p>This implementation is translated and adapted from the Matlab version 
  72  * of the CMA-ES algorithm as implemented in module {@code cmaes.m} version 3.51.</p> 
  73  * 
  74  * For more information, please refer to the following links: 
  75  * <ul> 
  76  *  <li><a href="http://www.lri.fr/~hansen/cmaes.m">Matlab code</a></li> 
  77  *  <li><a href="http://www.lri.fr/~hansen/cmaesintro.html">Introduction to CMA-ES</a></li> 
  78  *  <li><a href="http://en.wikipedia.org/wiki/CMA-ES">Wikipedia</a></li> 
  79  * </ul> 
  80  * 
  81  * When simple constraints (boundaries) are used, care must be taken that the 
  82  * difference between the upper and lower bounds does not overflow; should it 
  83  * be the case, a {@link NumberIsTooLargeException} will be thrown by the 
  84  * {@link BaseAbstractMultivariateSimpleBoundsOptimizer#optimize(int, 
  85  * MultivariateFunction,GoalType,double[],double[],double[]) optimize} method. 
  86  * 
  87  * @version $Id$ 
  88  * @since 3.0 
  89  */

 353     /** {@inheritDoc} */ 
 354     @Override 
 355     protected PointValuePair doOptimize() { 
 356         checkParameters(); 
 357          // -------------------- Initialization -------------------------------- 
 358         isMinimize = getGoalType().equals(GoalType.MINIMIZE); 
 359         final FitnessFunction fitfun = new FitnessFunction(); 
 360         final double[] guess = fitfun.encode(getStartPoint()); 
 361         // number of objective variables/problem dimension 
 362         dimension = guess.length; 
 363         initializeCMA(guess); 
 364         iterations = 0; 
 365         double bestValue = fitfun.value(guess); 
 366         push(fitnessHistory, bestValue); 
 367         PointValuePair optimum = new PointValuePair(getStartPoint(), 
 368                 isMinimize ? bestValue : -bestValue); 
 369         PointValuePair lastResult = null; 
 370  
 371         // -------------------- Generation Loop -------------------------------- 
 372  
 373         generationLoop: 
 374             for (iterations = 1; iterations <= maxIterations; iterations++) { 
 375                 // Generate and evaluate lambda offspring 
 376                 RealMatrix arz = randn1(dimension, lambda); 
 377                 RealMatrix arx = zeros(dimension, lambda); 
 378                 double[] fitness = new double[lambda]; 
 379                 // generate random offspring 
 380                 for (int k = 0; k < lambda; k++) { 
 381                     RealMatrix arxk = null; 
 382                     for (int i = 0; i < checkFeasableCount+1; i++) { 
 383                         if (diagonalOnly <= 0) { 
 384                             arxk = xmean.add(BD.multiply(arz.getColumnMatrix(k)) 
 385                                     .scalarMultiply(sigma)); // m + sig * Normal(0,C) 
 386                         } else { 
 387                             arxk = xmean.add(times(diagD,arz.getColumnMatrix(k)) 
 388                                     .scalarMultiply(sigma)); 
 389                         } 
 390                         if (i >= checkFeasableCount || fitfun.isFeasible(arxk.getColumn(0))) { 
 391                             break; 
 392                         } 
 393                         // regenerate random arguments for row 
 394                         arz.setColumn(k, randn(dimension)); 
 395                     } 
 396                     copyColumn(arxk, 0, arx, k); 
 397                     try { 
 398                         fitness[k] = fitfun.value(arx.getColumn(k)); // compute fitness 
 399                     } catch (TooManyEvaluationsException e) { 
 400                         break generationLoop; 
 401                     } 
 402                 } 
 403                 // Sort by fitness and compute weighted mean into xmean 
 404                 int[] arindex = sortedIndices(fitness); 
 405                 // Calculate new xmean, this is selection and recombination 
 406                 RealMatrix xold = xmean; // for speed up of Eq. (2) and (3) 
 407                 RealMatrix bestArx = selectColumns(arx, MathArrays.copyOf(arindex, mu)); 
 408                 xmean = bestArx.multiply(weights); 
 409                 RealMatrix bestArz = selectColumns(arz, MathArrays.copyOf(arindex, mu)); 
 410                 RealMatrix zmean = bestArz.multiply(weights); 
 411                 boolean hsig = updateEvolutionPaths(zmean, xold); 
 412                 if (diagonalOnly <= 0) { 
 413                     updateCovariance(hsig, bestArx, arz, arindex, xold); 
 414                 } else { 
 415                     updateCovarianceDiagonalOnly(hsig, bestArz, xold); 
 416                 } 
 417                 // Adapt step size sigma - Eq. (5) 
 418                 sigma *= Math.exp(Math.min(1.0,(normps/chiN - 1.)*cs/damps)); 
 419                 double bestFitness = fitness[arindex[0]]; 
 420                 double worstFitness = fitness[arindex[arindex.length-1]]; 
 421                 if (bestValue > bestFitness) { 
 422                     bestValue = bestFitness; 
 423                     lastResult = optimum; 
 424                     optimum = new PointValuePair( 
 425                             fitfun.repairAndDecode(bestArx.getColumn(0)), 
 426                             isMinimize ? bestFitness : -bestFitness); 
 427                     if (getConvergenceChecker() != null && lastResult != null) { 
 428                         if (getConvergenceChecker().converged(iterations, optimum, lastResult)) { 
 429                             break generationLoop; 
 430                         } 
 431                     } 
 432                 } 
 433                 // handle termination criteria 
 434                 // Break, if fitness is good enough 
 435                 if (stopFitness != 0) { // only if stopFitness is defined 
 436                     if (bestFitness < (isMinimize ? stopFitness : -stopFitness)) { 
 437                         break generationLoop; 
 438                     } 
 439                 } 
 440                 double[] sqrtDiagC = sqrt(diagC).getColumn(0); 
 441                 double[] pcCol = pc.getColumn(0); 
 442                 for (int i = 0; i < dimension; i++) { 
 443                     if (sigma*(Math.max(Math.abs(pcCol[i]), sqrtDiagC[i])) > stopTolX) { 
 444                         break; 
 445                     } 
 446                     if (i >= dimension-1) { 
 447                         break generationLoop; 
 448                     } 
 449                 } 
 450                 for (int i = 0; i < dimension; i++) { 
 451                     if (sigma*sqrtDiagC[i] > stopTolUpX) { 
 452                         break generationLoop; 
 453                     } 
 454                 } 
 455                 double historyBest = min(fitnessHistory); 
 456                 double historyWorst = max(fitnessHistory); 
 457                 if (iterations > 2 && Math.max(historyWorst, worstFitness) - 
 458                         Math.min(historyBest, bestFitness) < stopTolFun) { 
 459                     break generationLoop; 
 460                 } 
 461                 if (iterations > fitnessHistory.length && 
 462                         historyWorst-historyBest < stopTolHistFun) { 
 463                     break generationLoop; 
 464                 } 
 465                 // condition number of the covariance matrix exceeds 1e14 
 466                 if (max(diagD)/min(diagD) > 1e7) { 
 467                     break generationLoop; 
 468                 } 
 469                 // user defined termination 
 470                 if (getConvergenceChecker() != null) { 
 471                     PointValuePair current = 
 472                         new PointValuePair(bestArx.getColumn(0), 
 473                                 isMinimize ? bestFitness : -bestFitness); 
 474                     if (lastResult != null && 
 475                         getConvergenceChecker().converged(iterations, current, lastResult)) { 
 476                         break generationLoop; 
 477                     } 
 478                     lastResult = current; 
 479                 } 
 480                 // Adjust step size in case of equal function values (flat fitness) 
 481                 if (bestValue == fitness[arindex[(int)(0.1+lambda/4.)]]) { 
 482                     sigma = sigma * Math.exp(0.2+cs/damps); 
 483                 } 
 484                 if (iterations > 2 && Math.max(historyWorst, bestFitness) - 
 485                         Math.min(historyBest, bestFitness) == 0) { 
 486                     sigma = sigma * Math.exp(0.2+cs/damps); 
 487                 } 
 488                 // store best in history 
 489                 push(fitnessHistory,bestFitness); 
 490                 fitfun.setValueRange(worstFitness-bestFitness); 
 491                 if (generateStatistics) { 
 492                     statisticsSigmaHistory.add(sigma); 
 493                     statisticsFitnessHistory.add(bestFitness); 
 494                     statisticsMeanHistory.add(xmean.transpose()); 
 495                     statisticsDHistory.add(diagD.transpose().scalarMultiply(1E5)); 
 496                 } 
 497             } 
 498         return optimum; 
 499     }

 501     /** 
 502      * Checks dimensions and values of boundaries and inputSigma if defined. 
 503      */ 
 504     private void checkParameters() { 
 505         final double[] init = getStartPoint(); 
 506         final double[] lB = getLowerBound(); 
 507         final double[] uB = getUpperBound(); 
 508  
 509         // Checks whether there is at least one finite bound value. 
 510         boolean hasFiniteBounds = false; 
 511         for (int i = 0; i < lB.length; i++) { 
 512             if (!Double.isInfinite(lB[i]) || 
 513                 !Double.isInfinite(uB[i])) { 
 514                 hasFiniteBounds = true; 
 515                 break; 
 516             } 
 517         } 
 518         // Checks whether there is at least one infinite bound value. 
 519         boolean hasInfiniteBounds = false; 
 520         if (hasFiniteBounds) { 
 521             for (int i = 0; i < lB.length; i++) { 
 522                 if (Double.isInfinite(lB[i]) || 
 523                     Double.isInfinite(uB[i])) { 
 524                     hasInfiniteBounds = true; 
 525                     break; 
 526                 } 
 527             } 
 528  
 529             if (hasInfiniteBounds) { 
 530                 // If there is at least one finite bound, none can be infinite, 
 531                 // because mixed cases are not supported by the current code. 
 532                 throw new MathUnsupportedOperationException(); 
 533             } else { 
 534                 // Convert API to internal handling of boundaries. 
 535                 boundaries = new double[2][]; 
 536                 boundaries[0] = lB; 
 537                 boundaries[1] = uB; 
 538  
 539                 // Abort early if the normalization will overflow (cf. "encode" method). 
 540                 for (int i = 0; i < lB.length; i++) { 
 541                     if (Double.isInfinite(boundaries[1][i] - boundaries[0][i])) { 
 542                         final double max = Double.MAX_VALUE + boundaries[0][i]; 
 543                         final NumberIsTooLargeException e 
 544                             = new NumberIsTooLargeException(boundaries[1][i], 
 545                                                             max, 
 546                                                             true); 
 547                         e.getContext().addMessage(LocalizedFormats.OVERFLOW); 
 548                         e.getContext().addMessage(LocalizedFormats.INDEX, i); 
 549  
 550                         throw e; 
 551                     } 
 552                 } 
 553             } 
 554         } else { 
 555             // Convert API to internal handling of boundaries. 
 556             boundaries = null; 
 557         } 
 558  
 559         if (inputSigma != null) { 
 560             if (inputSigma.length != init.length) { 
 561                 throw new DimensionMismatchException(inputSigma.length, init.length); 
 562             } 
 563             for (int i = 0; i < init.length; i++) { 
 564                 if (inputSigma[i] < 0) { 
 565                     throw new NotPositiveException(inputSigma[i]); 
 566                 } 
 567                 if (boundaries != null) { 
 568                     if (inputSigma[i] > boundaries[1][i] - boundaries[0][i]) { 
 569                         throw new OutOfRangeException(inputSigma[i], 0, boundaries[1][i] - boundaries[0][i]); 
 570                     } 
 571                 } 
 572             } 
 573         } 
 574     }

 576     /** 
 577      * Initialization of the dynamic search parameters 
 578      * 
 579      * @param guess Initial guess for the arguments of the fitness function. 
 580      */ 
 581     private void initializeCMA(double[] guess) { 
 582         if (lambda <= 0) { 
 583             lambda = 4 + (int) (3. * Math.log(dimension)); 
 584         } 
 585         // initialize sigma 
 586         double[][] sigmaArray = new double[guess.length][1]; 
 587         for (int i = 0; i < guess.length; i++) { 
 588             final double range = (boundaries == null) ? 1.0 : boundaries[1][i] - boundaries[0][i]; 
 589             sigmaArray[i][0] = ((inputSigma == null) ? 0.3 : inputSigma[i]) / range; 
 590         } 
 591         RealMatrix insigma = new Array2DRowRealMatrix(sigmaArray, false); 
 592         sigma = max(insigma); // overall standard deviation 
 593  
 594         // initialize termination criteria 
 595         stopTolUpX = 1e3 * max(insigma); 
 596         stopTolX = 1e-11 * max(insigma); 
 597         stopTolFun = 1e-12; 
 598         stopTolHistFun = 1e-13; 
 599  
 600         // initialize selection strategy parameters 
 601         mu = lambda / 2; // number of parents/points for recombination 
 602         logMu2 = Math.log(mu + 0.5); 
 603         weights = log(sequence(1, mu, 1)).scalarMultiply(-1.).scalarAdd(logMu2); 
 604         double sumw = 0; 
 605         double sumwq = 0; 
 606         for (int i = 0; i < mu; i++) { 
 607             double w = weights.getEntry(i, 0); 
 608             sumw += w; 
 609             sumwq += w * w; 
 610         } 
 611         weights = weights.scalarMultiply(1. / sumw); 
 612         mueff = sumw * sumw / sumwq; // variance-effectiveness of sum w_i x_i 
 613  
 614         // initialize dynamic strategy parameters and constants 
 615         cc = (4. + mueff / dimension) / 
 616                 (dimension + 4. + 2. * mueff / dimension); 
 617         cs = (mueff + 2.) / (dimension + mueff + 3.); 
 618         damps = (1. + 2. * Math.max(0, Math.sqrt((mueff - 1.) / 
 619                 (dimension + 1.)) - 1.)) * 
 620                 Math.max(0.3, 1. - dimension / 
 621                         (1e-6 + Math.min(maxIterations, getMaxEvaluations() / 
 622                                 lambda))) + cs; // minor increment 
 623         ccov1 = 2. / ((dimension + 1.3) * (dimension + 1.3) + mueff); 
 624         ccovmu = Math.min(1 - ccov1, 2. * (mueff - 2. + 1. / mueff) / 
 625                 ((dimension + 2.) * (dimension + 2.) + mueff)); 
 626         ccov1Sep = Math.min(1, ccov1 * (dimension + 1.5) / 3.); 
 627         ccovmuSep = Math.min(1 - ccov1, ccovmu * (dimension + 1.5) / 3.); 
 628         chiN = Math.sqrt(dimension) * 
 629                 (1. - 1. / (4. * dimension) + 1 / (21. * dimension * dimension)); 
 630         // intialize CMA internal values - updated each generation 
 631         xmean = MatrixUtils.createColumnRealMatrix(guess); // objective 
 632                                                            // variables 
 633         diagD = insigma.scalarMultiply(1. / sigma); 
 634         diagC = square(diagD); 
 635         pc = zeros(dimension, 1); // evolution paths for C and sigma 
 636         ps = zeros(dimension, 1); // B defines the coordinate system 
 637         normps = ps.getFrobeniusNorm(); 
 638  
 639         B = eye(dimension, dimension); 
 640         D = ones(dimension, 1); // diagonal D defines the scaling 
 641         BD = times(B, repmat(diagD.transpose(), dimension, 1)); 
 642         C = B.multiply(diag(square(D)).multiply(B.transpose())); // covariance 
 643         historySize = 10 + (int) (3. * 10. * dimension / lambda); 
 644         fitnessHistory = new double[historySize]; // history of fitness values 
 645         for (int i = 0; i < historySize; i++) { 
 646             fitnessHistory[i] = Double.MAX_VALUE; 
 647         } 
 648     }

 650     /** 
 651      * Update of the evolution paths ps and pc. 
 652      * 
 653      * @param zmean Weighted row matrix of the gaussian random numbers generating 
 654      * the current offspring. 
 655      * @param xold xmean matrix of the previous generation. 
 656      * @return hsig flag indicating a small correction. 
 657      */ 
 658     private boolean updateEvolutionPaths(RealMatrix zmean, RealMatrix xold) { 
 659         ps = ps.scalarMultiply(1. - cs).add( 
 660                 B.multiply(zmean).scalarMultiply( 
 661                         Math.sqrt(cs * (2. - cs) * mueff))); 
 662         normps = ps.getFrobeniusNorm(); 
 663         boolean hsig = normps / 
 664             Math.sqrt(1. - Math.pow(1. - cs, 2. * iterations)) / 
 665                 chiN < 1.4 + 2. / (dimension + 1.); 
 666         pc = pc.scalarMultiply(1. - cc); 
 667         if (hsig) { 
 668             pc = pc.add(xmean.subtract(xold).scalarMultiply( 
 669                     Math.sqrt(cc * (2. - cc) * mueff) / sigma)); 
 670         } 
 671         return hsig; 
 672     }

 675      * Update of the covariance matrix C for diagonalOnly > 0 
 676      * 
 677      * @param hsig Flag indicating a small correction. 
 678      * @param bestArz Fitness-sorted matrix of the gaussian random values of the 
 679      * current offspring. 
 680      * @param xold xmean matrix of the previous generation. 
 681      */ 
 682     private void updateCovarianceDiagonalOnly(boolean hsig, 
 683                                               final RealMatrix bestArz, 
 684                                               final RealMatrix xold) { 
 685         // minor correction if hsig==false 
 686         double oldFac = hsig ? 0 : ccov1Sep * cc * (2. - cc); 
 687         oldFac += 1. - ccov1Sep - ccovmuSep; 
 688         diagC = diagC.scalarMultiply(oldFac) // regard old matrix 
 689                 // plus rank one update 
 690                 .add(square(pc).scalarMultiply(ccov1Sep)) 
 691                 // plus rank mu update 
 692                 .add((times(diagC, square(bestArz).multiply(weights))) 
 693                         .scalarMultiply(ccovmuSep)); 
 694         diagD = sqrt(diagC); // replaces eig(C) 
 695         if (diagonalOnly > 1 && iterations > diagonalOnly) { 
 696             // full covariance matrix from now on 
 697             diagonalOnly = 0; 
 698             B = eye(dimension, dimension); 
 699             BD = diag(diagD); 
 700             C = diag(diagC); 
 701         } 
 702     }

 704     /** 
 705      * Update of the covariance matrix C. 
 706      * 
 707      * @param hsig Flag indicating a small correction. 
 708      * @param bestArx Fitness-sorted matrix of the argument vectors producing the 
 709      * current offspring. 
 710      * @param arz Unsorted matrix containing the gaussian random values of the 
 711      * current offspring. 
 712      * @param arindex Indices indicating the fitness-order of the current offspring. 
 713      * @param xold xmean matrix of the previous generation. 
 714      */ 
 715     private void updateCovariance(boolean hsig, final RealMatrix bestArx, 
 716             final RealMatrix arz, final int[] arindex, final RealMatrix xold) { 
 717         double negccov = 0; 
 718         if (ccov1 + ccovmu > 0) { 
 719             RealMatrix arpos = bestArx.subtract(repmat(xold, 1, mu)) 
 720                     .scalarMultiply(1. / sigma); // mu difference vectors 
 721             RealMatrix roneu = pc.multiply(pc.transpose()) 
 722                     .scalarMultiply(ccov1); // rank one update 
 723             // minor correction if hsig==false 
 724             double oldFac = hsig ? 0 : ccov1 * cc * (2. - cc); 
 725             oldFac += 1. - ccov1 - ccovmu; 
 726             if (isActiveCMA) { 
 727                 // Adapt covariance matrix C active CMA 
 728                 negccov = (1. - ccovmu) * 0.25 * mueff / 
 729                 (Math.pow(dimension + 2., 1.5) + 2. * mueff); 
 730                 double negminresidualvariance = 0.66; 
 731                 // keep at least 0.66 in all directions, small popsize are most 
 732                 // critical 
 733                 double negalphaold = 0.5; // where to make up for the variance 
 734                                           // loss, 
 735                 // prepare vectors, compute negative updating matrix Cneg 
 736                 int[] arReverseIndex = reverse(arindex); 
 737                 RealMatrix arzneg 
 738                     = selectColumns(arz, MathArrays.copyOf(arReverseIndex, mu)); 
 739                 RealMatrix arnorms = sqrt(sumRows(square(arzneg))); 
 740                 int[] idxnorms = sortedIndices(arnorms.getRow(0)); 
 741                 RealMatrix arnormsSorted = selectColumns(arnorms, idxnorms); 
 742                 int[] idxReverse = reverse(idxnorms); 
 743                 RealMatrix arnormsReverse = selectColumns(arnorms, idxReverse); 
 744                 arnorms = divide(arnormsReverse, arnormsSorted); 
 745                 int[] idxInv = inverse(idxnorms); 
 746                 RealMatrix arnormsInv = selectColumns(arnorms, idxInv); 
 747                 // check and set learning rate negccov 
 748                 double negcovMax = (1. - negminresidualvariance) / 
 749                         square(arnormsInv).multiply(weights).getEntry(0, 0); 
 750                 if (negccov > negcovMax) { 
 751                     negccov = negcovMax; 
 752                 } 
 753                 arzneg = times(arzneg, repmat(arnormsInv, dimension, 1)); 
 754                 RealMatrix artmp = BD.multiply(arzneg); 
 755                 RealMatrix Cneg = artmp.multiply(diag(weights)).multiply( 
 756                         artmp.transpose()); 
 757                 oldFac += negalphaold * negccov; 
 758                 C = C.scalarMultiply(oldFac) 
 759                         // regard old matrix 
 760                         .add(roneu) 
 761                         // plus rank one update 
 762                         .add(arpos.scalarMultiply( 
 763                                 // plus rank mu update 
 764                                 ccovmu + (1. - negalphaold) * negccov) 
 765                                 .multiply( 
 766                                         times(repmat(weights, 1, dimension), 
 767                                                 arpos.transpose()))) 
 768                         .subtract(Cneg.scalarMultiply(negccov)); 
 769             } else { 
 770                 // Adapt covariance matrix C - nonactive 
 771                 C = C.scalarMultiply(oldFac) // regard old matrix 
 772                         .add(roneu) 
 773                         // plus rank one update 
 774                         .add(arpos.scalarMultiply(ccovmu) // plus rank mu update 
 775                                 .multiply( 
 776                                         times(repmat(weights, 1, dimension), 
 777                                                 arpos.transpose()))); 
 778             } 
 779         } 
 780         updateBD(negccov); 
 781     }

 783     /** 
 784      * Update B and D from C. 
 785      * 
 786      * @param negccov Negative covariance factor. 
 787      */ 
 788     private void updateBD(double negccov) { 
 789         if (ccov1 + ccovmu + negccov > 0 && 
 790                 (iterations % 1. / (ccov1 + ccovmu + negccov) / dimension / 10.) < 1.) { 
 791             // to achieve O(N^2) 
 792             C = triu(C, 0).add(triu(C, 1).transpose()); 
 793             // enforce symmetry to prevent complex numbers 
 794             EigenDecomposition eig = new EigenDecomposition(C); 
 795             B = eig.getV(); // eigen decomposition, B==normalized eigenvectors 
 796             D = eig.getD(); 
 797             diagD = diag(D); 
 798             if (min(diagD) <= 0) { 
 799                 for (int i = 0; i < dimension; i++) { 
 800                     if (diagD.getEntry(i, 0) < 0) { 
 801                         diagD.setEntry(i, 0, 0.); 
 802                     } 
 803                 } 
 804                 double tfac = max(diagD) / 1e14; 
 805                 C = C.add(eye(dimension, dimension).scalarMultiply(tfac)); 
 806                 diagD = diagD.add(ones(dimension, 1).scalarMultiply(tfac)); 
 807             } 
 808             if (max(diagD) > 1e14 * min(diagD)) { 
 809                 double tfac = max(diagD) / 1e14 - min(diagD); 
 810                 C = C.add(eye(dimension, dimension).scalarMultiply(tfac)); 
 811                 diagD = diagD.add(ones(dimension, 1).scalarMultiply(tfac)); 
 812             } 
 813             diagC = diag(C); 
 814             diagD = sqrt(diagD); // D contains standard deviations now 
 815             BD = times(B, repmat(diagD.transpose(), dimension, 1)); // O(n^2) 
 816         } 
 817     }

 900     /** 
 901      * Normalizes fitness values to the range [0,1]. Adds a penalty to the 
 902      * fitness value if out of range. The penalty is adjusted by calling 
 903      * setValueRange(). 
 904      */ 
 905     private class FitnessFunction { 
 906         /** Determines the penalty for boundary violations */ 
 907         private double valueRange; 
 908         /** 
 909          * Flag indicating whether the objective variables are forced into their 
 910          * bounds if defined 
 911          */ 
 912         private boolean isRepairMode; 
 913  
 914         /** Simple constructor. 
 915          */ 
 916         public FitnessFunction() { 
 917             valueRange = 1.0; 
 918             isRepairMode = true; 
 919         } 
 920  
 921         /** 
 922          * @param x Original objective variables. 
 923          * @return the normalized objective variables. 
 924          */ 
 925         public double[] encode(final double[] x) { 
 926             if (boundaries == null) { 
 927                 return x; 
 928             } 
 929             double[] res = new double[x.length]; 
 930             for (int i = 0; i < x.length; i++) { 
 931                 double diff = boundaries[1][i] - boundaries[0][i]; 
 932                 res[i] = (x[i] - boundaries[0][i]) / diff; 
 933             } 
 934             return res; 
 935         } 
 936  
 937         /** 
 938          * @param x Normalized objective variables. 
 939          * @return the original objective variables, possibly repaired. 
 940          */ 
 941         public double[] repairAndDecode(final double[] x) { 
 942             return boundaries != null && isRepairMode ? 
 943                 decode(repair(x)) : 
 944                 decode(x); 
 945         } 
 946  
 947         /** 
 948          * @param x Normalized objective variables. 
 949          * @return the original objective variables. 
 950          */ 
 951         public double[] decode(final double[] x) { 
 952             if (boundaries == null) { 
 953                 return x; 
 954             } 
 955             double[] res = new double[x.length]; 
 956             for (int i = 0; i < x.length; i++) { 
 957                 double diff = boundaries[1][i] - boundaries[0][i]; 
 958                 res[i] = diff * x[i] + boundaries[0][i]; 
 959             } 
 960             return res; 
 961         } 
 962  
 963         /** 
 964          * @param point Normalized objective variables. 
 965          * @return the objective value + penalty for violated bounds. 
 966          */ 
 967         public double value(final double[] point) { 
 968             double value; 
 969             if (boundaries != null && isRepairMode) { 
 970                 double[] repaired = repair(point); 
 971                 value = CMAESOptimizer.this 
 972                         .computeObjectiveValue(decode(repaired)) + 
 973                         penalty(point, repaired); 
 974             } else { 
 975                 value = CMAESOptimizer.this 
 976                         .computeObjectiveValue(decode(point)); 
 977             } 
 978             return isMinimize ? value : -value; 
 979         } 
 980  
 981         /** 
 982          * @param x Normalized objective variables. 
 983          * @return {@code true} if in bounds. 
 984          */ 
 985         public boolean isFeasible(final double[] x) { 
 986             if (boundaries == null) { 
 987                 return true; 
 988             } 
 989  
 990  
 991             for (int i = 0; i < x.length; i++) { 
 992                 if (x[i] < 0) { 
 993                     return false; 
 994                 } 
 995                 if (x[i] > 1.0) { 
 996                     return false; 
 997                 } 
 998             } 
 999             return true; 
1000         }

1002         /** 
1003          * @param valueRange Adjusts the penalty computation. 
1004          */ 
1005         public void setValueRange(double valueRange) { 
1006             this.valueRange = valueRange; 
1007         } 
1008  
1009         /** 
1010          * @param x Normalized objective variables. 
1011          * @return the repaired objective variables - all in bounds. 
1012          */ 
1013         private double[] repair(final double[] x) { 
1014             double[] repaired = new double[x.length]; 
1015             for (int i = 0; i < x.length; i++) { 
1016                 if (x[i] < 0) { 
1017                     repaired[i] = 0; 
1018                 } else if (x[i] > 1.0) { 
1019                     repaired[i] = 1.0; 
1020                 } else { 
1021                     repaired[i] = x[i]; 
1022                 } 
1023             } 
1024             return repaired; 
1025         }

1027         /** 
1028          * @param x Normalized objective variables. 
1029          * @param repaired Repaired objective variables. 
1030          * @return Penalty value according to the violation of the bounds. 
1031          */ 
1032         private double penalty(final double[] x, final double[] repaired) { 
1033             double penalty = 0; 
1034             for (int i = 0; i < x.length; i++) { 
1035                 double diff = Math.abs(x[i] - repaired[i]); 
1036                 penalty += diff * valueRange; 
1037             } 
1038             return isMinimize ? penalty : -penalty; 
1039         }

  45 /** 
  46  * <p>An implementation of the active Covariance Matrix Adaptation Evolution Strategy (CMA-ES) 
  47  * for non-linear, non-convex, non-smooth, global function minimization. 
  48  * The CMA-Evolution Strategy (CMA-ES) is a reliable stochastic optimization method 
  49  * which should be applied if derivative-based methods, e.g. quasi-Newton BFGS or 
  50  * conjugate gradient, fail due to a rugged search landscape (e.g. noise, local 
  51  * optima, outlier, etc.) of the objective function. Like a 
  52  * quasi-Newton method, the CMA-ES learns and applies a variable metric 
  53  * on the underlying search space. Unlike a quasi-Newton method, the 
  54  * CMA-ES neither estimates nor uses gradients, making it considerably more 
  55  * reliable in terms of finding a good, or even close to optimal, solution.</p> 
  56  * 
  57  * <p>In general, on smooth objective functions the CMA-ES is roughly ten times 
  58  * slower than BFGS (counting objective function evaluations, no gradients provided). 
  59  * For up to <math>N=10</math> variables also the derivative-free simplex 
  60  * direct search method (Nelder and Mead) can be faster, but it is 
  61  * far less reliable than CMA-ES.</p> 
  62  * 
  63  * <p>The CMA-ES is particularly well suited for non-separable 
  64  * and/or badly conditioned problems. To observe the advantage of CMA compared 
  65  * to a conventional evolution strategy, it will usually take about 
  66  * <math>30 N</math> function evaluations. On difficult problems the complete 
  67  * optimization (a single run) is expected to take <em>roughly</em> between 
  68  * <math>30 N</math> and <math>300 N<sup>2</sup></math> 
  69  * function evaluations.</p> 
  70  * 
  71  * <p>This implementation is translated and adapted from the Matlab version 
  72  * of the CMA-ES algorithm as implemented in module {@code cmaes.m} version 3.51.</p> 
  73  * 
  74  * For more information, please refer to the following links: 
  75  * <ul> 
  76  *  <li><a href="http://www.lri.fr/~hansen/cmaes.m">Matlab code</a></li> 
  77  *  <li><a href="http://www.lri.fr/~hansen/cmaesintro.html">Introduction to CMA-ES</a></li> 
  78  *  <li><a href="http://en.wikipedia.org/wiki/CMA-ES">Wikipedia</a></li> 
  79  * </ul>

 353     /** {@inheritDoc} */ 
 354     @Override 
 355     protected PointValuePair doOptimize() { 
 356         checkParameters(); 
 357          // -------------------- Initialization -------------------------------- 
 358         isMinimize = getGoalType().equals(GoalType.MINIMIZE); 
 359         final FitnessFunction fitfun = new FitnessFunction(); 
 360         final double[] guess = fitfun.encode(getStartPoint()); 
 361         // number of objective variables/problem dimension 
 362         dimension = guess.length; 
 363         initializeCMA(guess); 
 364         iterations = 0; 
 365         double bestValue = fitfun.value(guess); 
 366         push(fitnessHistory, bestValue); 
 367         PointValuePair optimum = new PointValuePair(getStartPoint(), 
 368                 isMinimize ? bestValue : -bestValue); 
 369         PointValuePair lastResult = null; 
 370  
 371         // -------------------- Generation Loop -------------------------------- 
 372  
 373         generationLoop: 
 374             for (iterations = 1; iterations <= maxIterations; iterations++) { 
 375                 // Generate and evaluate lambda offspring 
 376                 RealMatrix arz = randn1(dimension, lambda); 
 377                 RealMatrix arx = zeros(dimension, lambda); 
 378                 double[] fitness = new double[lambda]; 
 379                 // generate random offspring 
 380                 for (int k = 0; k < lambda; k++) { 
 381                     RealMatrix arxk = null; 
 382                     for (int i = 0; i < checkFeasableCount+1; i++) { 
 383                         if (diagonalOnly <= 0) { 
 384                             arxk = xmean.add(BD.multiply(arz.getColumnMatrix(k)) 
 385                                     .scalarMultiply(sigma)); // m + sig * Normal(0,C) 
 386                         } else { 
 387                             arxk = xmean.add(times(diagD,arz.getColumnMatrix(k)) 
 388                                     .scalarMultiply(sigma)); 
 389                         } 
 390                         if (i >= checkFeasableCount || fitfun.isFeasible(arxk.getColumn(0))) { 
 391                             break; 
 392                         } 
 393                         // regenerate random arguments for row 
 394                         arz.setColumn(k, randn(dimension)); 
 395                     } 
 396                     copyColumn(arxk, 0, arx, k); 
 397                     try { 
 398                         fitness[k] = fitfun.value(arx.getColumn(k)); // compute fitness 
 399                     } catch (TooManyEvaluationsException e) { 
 400                         break generationLoop; 
 401                     } 
 402                 } 
 403                 // Sort by fitness and compute weighted mean into xmean 
 404                 int[] arindex = sortedIndices(fitness); 
 405                 // Calculate new xmean, this is selection and recombination 
 406                 RealMatrix xold = xmean; // for speed up of Eq. (2) and (3) 
 407                 RealMatrix bestArx = selectColumns(arx, MathArrays.copyOf(arindex, mu)); 
 408                 xmean = bestArx.multiply(weights); 
 409                 RealMatrix bestArz = selectColumns(arz, MathArrays.copyOf(arindex, mu)); 
 410                 RealMatrix zmean = bestArz.multiply(weights); 
 411                 boolean hsig = updateEvolutionPaths(zmean, xold); 
 412                 if (diagonalOnly <= 0) { 
 413                     updateCovariance(hsig, bestArx, arz, arindex, xold); 
 414                 } else { 
 415                     updateCovarianceDiagonalOnly(hsig, bestArz, xold); 
 416                 } 
 417                 // Adapt step size sigma - Eq. (5) 
 418                 sigma *= Math.exp(Math.min(1.0,(normps/chiN - 1.)*cs/damps)); 
 419                 double bestFitness = fitness[arindex[0]]; 
 420                 double worstFitness = fitness[arindex[arindex.length-1]]; 
 421                 if (bestValue > bestFitness) { 
 422                     bestValue = bestFitness; 
 423                     lastResult = optimum; 
 424                     optimum = new PointValuePair( 
 425                             fitfun.repairAndDecode(bestArx.getColumn(0)), 
 426                             isMinimize ? bestFitness : -bestFitness); 
 427                     if (getConvergenceChecker() != null && lastResult != null) { 
 428                         if (getConvergenceChecker().converged(iterations, optimum, lastResult)) { 
 429                             break generationLoop; 
 430                         } 
 431                     } 
 432                 } 
 433                 // handle termination criteria 
 434                 // Break, if fitness is good enough 
 435                 if (stopFitness != 0) { // only if stopFitness is defined 
 436                     if (bestFitness < (isMinimize ? stopFitness : -stopFitness)) { 
 437                         break generationLoop; 
 438                     } 
 439                 } 
 440                 double[] sqrtDiagC = sqrt(diagC).getColumn(0); 
 441                 double[] pcCol = pc.getColumn(0); 
 442                 for (int i = 0; i < dimension; i++) { 
 443                     if (sigma*(Math.max(Math.abs(pcCol[i]), sqrtDiagC[i])) > stopTolX) { 
 444                         break; 
 445                     } 
 446                     if (i >= dimension-1) { 
 447                         break generationLoop; 
 448                     } 
 449                 } 
 450                 for (int i = 0; i < dimension; i++) { 
 451                     if (sigma*sqrtDiagC[i] > stopTolUpX) { 
 452                         break generationLoop; 
 453                     } 
 454                 } 
 455                 double historyBest = min(fitnessHistory); 
 456                 double historyWorst = max(fitnessHistory); 
 457                 if (iterations > 2 && Math.max(historyWorst, worstFitness) - 
 458                         Math.min(historyBest, bestFitness) < stopTolFun) { 
 459                     break generationLoop; 
 460                 } 
 461                 if (iterations > fitnessHistory.length && 
 462                         historyWorst-historyBest < stopTolHistFun) { 
 463                     break generationLoop; 
 464                 } 
 465                 // condition number of the covariance matrix exceeds 1e14 
 466                 if (max(diagD)/min(diagD) > 1e7) { 
 467                     break generationLoop; 
 468                 } 
 469                 // user defined termination 
 470                 if (getConvergenceChecker() != null) { 
 471                     PointValuePair current = 
 472                         new PointValuePair(bestArx.getColumn(0), 
 473                                 isMinimize ? bestFitness : -bestFitness); 
 474                     if (lastResult != null && 
 475                         getConvergenceChecker().converged(iterations, current, lastResult)) { 
 476                         break generationLoop; 
 477                     } 
 478                     lastResult = current; 
 479                 } 
 480                 // Adjust step size in case of equal function values (flat fitness) 
 481                 if (bestValue == fitness[arindex[(int)(0.1+lambda/4.)]]) { 
 482                     sigma = sigma * Math.exp(0.2+cs/damps); 
 483                 } 
 484                 if (iterations > 2 && Math.max(historyWorst, bestFitness) - 
 485                         Math.min(historyBest, bestFitness) == 0) { 
 486                     sigma = sigma * Math.exp(0.2+cs/damps); 
 487                 } 
 488                 // store best in history 
 489                 push(fitnessHistory,bestFitness); 
 490                 fitfun.setValueRange(worstFitness-bestFitness); 
 491                 if (generateStatistics) { 
 492                     statisticsSigmaHistory.add(sigma); 
 493                     statisticsFitnessHistory.add(bestFitness); 
 494                     statisticsMeanHistory.add(xmean.transpose()); 
 495                     statisticsDHistory.add(diagD.transpose().scalarMultiply(1E5)); 
 496                 } 
 497             } 
 498         return optimum; 
 499     }

 501     /** 
 502      * Checks dimensions and values of boundaries and inputSigma if defined. 
 503      */ 
 504     private void checkParameters() { 
 505         final double[] init = getStartPoint(); 
 506         final double[] lB = getLowerBound(); 
 507         final double[] uB = getUpperBound(); 
 508  
 509         // Checks whether there is at least one finite bound value. 
 510         boolean hasFiniteBounds = false; 
 511         for (int i = 0; i < lB.length; i++) { 
 512             if (!Double.isInfinite(lB[i]) || 
 513                 !Double.isInfinite(uB[i])) { 
 514                 hasFiniteBounds = true; 
 515                 break; 
 516             } 
 517         } 
 518         // Checks whether there is at least one infinite bound value. 
 519         boolean hasInfiniteBounds = false; 
 520         if (hasFiniteBounds) { 
 521             for (int i = 0; i < lB.length; i++) { 
 522                 if (Double.isInfinite(lB[i]) || 
 523                     Double.isInfinite(uB[i])) { 
 524                     hasInfiniteBounds = true; 
 525                     break; 
 526                 } 
 527             } 
 528  
 529             if (hasInfiniteBounds) { 
 530                 // If there is at least one finite bound, none can be infinite, 
 531                 // because mixed cases are not supported by the current code. 
 532                 throw new MathUnsupportedOperationException(); 
 533             } else { 
 534                 // Convert API to internal handling of boundaries. 
 535                 boundaries = new double[2][]; 
 536                 boundaries[0] = lB; 
 537                 boundaries[1] = uB; 
 538  
 539                 // Abort early if the normalization will overflow (cf. "encode" method). 
 540                 for (int i = 0; i < lB.length; i++) { 
 541                     if (Double.isInfinite(boundaries[1][i] - boundaries[0][i])) { 
 542                         final double max = Double.MAX_VALUE + boundaries[0][i]; 
 543                         final NumberIsTooLargeException e 
 544                             = new NumberIsTooLargeException(boundaries[1][i], 
 545                                                             max, 
 546                                                             true); 
 547                         e.getContext().addMessage(LocalizedFormats.OVERFLOW); 
 548                         e.getContext().addMessage(LocalizedFormats.INDEX, i); 
 549  
 550                         throw e; 
 551                     } 
 552                 } 
 553             } 
 554         } else { 
 555             // Convert API to internal handling of boundaries. 
 556             boundaries = null; 
 557         } 
 558  
 559         if (inputSigma != null) { 
 560             if (inputSigma.length != init.length) { 
 561                 throw new DimensionMismatchException(inputSigma.length, init.length); 
 562             } 
 563             for (int i = 0; i < init.length; i++) { 
 564                 if (inputSigma[i] < 0) { 
 565                     throw new NotPositiveException(inputSigma[i]); 
 566                 } 
 567                 if (boundaries != null) { 
 568                     if (inputSigma[i] > boundaries[1][i] - boundaries[0][i]) { 
 569                         throw new OutOfRangeException(inputSigma[i], 0, boundaries[1][i] - boundaries[0][i]); 
 570                     } 
 571                 } 
 572             } 
 573         } 
 574     }

 576     /** 
 577      * Initialization of the dynamic search parameters 
 578      * 
 579      * @param guess Initial guess for the arguments of the fitness function. 
 580      */ 
 581     private void initializeCMA(double[] guess) { 
 582         if (lambda <= 0) { 
 583             lambda = 4 + (int) (3. * Math.log(dimension)); 
 584         } 
 585         // initialize sigma 
 586         double[][] sigmaArray = new double[guess.length][1]; 
 587         for (int i = 0; i < guess.length; i++) { 
 588             final double range = (boundaries == null) ? 1.0 : boundaries[1][i] - boundaries[0][i]; 
 589             sigmaArray[i][0] = ((inputSigma == null) ? 0.3 : inputSigma[i]) / range; 
 590         } 
 591         RealMatrix insigma = new Array2DRowRealMatrix(sigmaArray, false); 
 592         sigma = max(insigma); // overall standard deviation 
 593  
 594         // initialize termination criteria 
 595         stopTolUpX = 1e3 * max(insigma); 
 596         stopTolX = 1e-11 * max(insigma); 
 597         stopTolFun = 1e-12; 
 598         stopTolHistFun = 1e-13; 
 599  
 600         // initialize selection strategy parameters 
 601         mu = lambda / 2; // number of parents/points for recombination 
 602         logMu2 = Math.log(mu + 0.5); 
 603         weights = log(sequence(1, mu, 1)).scalarMultiply(-1.).scalarAdd(logMu2); 
 604         double sumw = 0; 
 605         double sumwq = 0; 
 606         for (int i = 0; i < mu; i++) { 
 607             double w = weights.getEntry(i, 0); 
 608             sumw += w; 
 609             sumwq += w * w; 
 610         } 
 611         weights = weights.scalarMultiply(1. / sumw); 
 612         mueff = sumw * sumw / sumwq; // variance-effectiveness of sum w_i x_i 
 613  
 614         // initialize dynamic strategy parameters and constants 
 615         cc = (4. + mueff / dimension) / 
 616                 (dimension + 4. + 2. * mueff / dimension); 
 617         cs = (mueff + 2.) / (dimension + mueff + 3.); 
 618         damps = (1. + 2. * Math.max(0, Math.sqrt((mueff - 1.) / 
 619                 (dimension + 1.)) - 1.)) * 
 620                 Math.max(0.3, 1. - dimension / 
 621                         (1e-6 + Math.min(maxIterations, getMaxEvaluations() / 
 622                                 lambda))) + cs; // minor increment 
 623         ccov1 = 2. / ((dimension + 1.3) * (dimension + 1.3) + mueff); 
 624         ccovmu = Math.min(1 - ccov1, 2. * (mueff - 2. + 1. / mueff) / 
 625                 ((dimension + 2.) * (dimension + 2.) + mueff)); 
 626         ccov1Sep = Math.min(1, ccov1 * (dimension + 1.5) / 3.); 
 627         ccovmuSep = Math.min(1 - ccov1, ccovmu * (dimension + 1.5) / 3.); 
 628         chiN = Math.sqrt(dimension) * 
 629                 (1. - 1. / (4. * dimension) + 1 / (21. * dimension * dimension)); 
 630         // intialize CMA internal values - updated each generation 
 631         xmean = MatrixUtils.createColumnRealMatrix(guess); // objective 
 632                                                            // variables 
 633         diagD = insigma.scalarMultiply(1. / sigma); 
 634         diagC = square(diagD); 
 635         pc = zeros(dimension, 1); // evolution paths for C and sigma 
 636         ps = zeros(dimension, 1); // B defines the coordinate system 
 637         normps = ps.getFrobeniusNorm(); 
 638  
 639         B = eye(dimension, dimension); 
 640         D = ones(dimension, 1); // diagonal D defines the scaling 
 641         BD = times(B, repmat(diagD.transpose(), dimension, 1)); 
 642         C = B.multiply(diag(square(D)).multiply(B.transpose())); // covariance 
 643         historySize = 10 + (int) (3. * 10. * dimension / lambda); 
 644         fitnessHistory = new double[historySize]; // history of fitness values 
 645         for (int i = 0; i < historySize; i++) { 
 646             fitnessHistory[i] = Double.MAX_VALUE; 
 647         } 
 648     }

 650     /** 
 651      * Update of the evolution paths ps and pc. 
 652      * 
 653      * @param zmean Weighted row matrix of the gaussian random numbers generating 
 654      * the current offspring. 
 655      * @param xold xmean matrix of the previous generation. 
 656      * @return hsig flag indicating a small correction. 
 657      */ 
 658     private boolean updateEvolutionPaths(RealMatrix zmean, RealMatrix xold) { 
 659         ps = ps.scalarMultiply(1. - cs).add( 
 660                 B.multiply(zmean).scalarMultiply( 
 661                         Math.sqrt(cs * (2. - cs) * mueff))); 
 662         normps = ps.getFrobeniusNorm(); 
 663         boolean hsig = normps / 
 664             Math.sqrt(1. - Math.pow(1. - cs, 2. * iterations)) / 
 665                 chiN < 1.4 + 2. / (dimension + 1.); 
 666         pc = pc.scalarMultiply(1. - cc); 
 667         if (hsig) { 
 668             pc = pc.add(xmean.subtract(xold).scalarMultiply( 
 669                     Math.sqrt(cc * (2. - cc) * mueff) / sigma)); 
 670         } 
 671         return hsig; 
 672     }

 674     /** 
 675      * Update of the covariance matrix C for diagonalOnly > 0 
 676      * 
 677      * @param hsig Flag indicating a small correction. 
 678      * @param bestArz Fitness-sorted matrix of the gaussian random values of the 
 679      * current offspring. 
 680      * @param xold xmean matrix of the previous generation. 
 681      */ 
 682     private void updateCovarianceDiagonalOnly(boolean hsig, 
 683                                               final RealMatrix bestArz, 
 684                                               final RealMatrix xold) { 
 685         // minor correction if hsig==false 
 686         double oldFac = hsig ? 0 : ccov1Sep * cc * (2. - cc); 
 687         oldFac += 1. - ccov1Sep - ccovmuSep; 
 688         diagC = diagC.scalarMultiply(oldFac) // regard old matrix 
 689                 // plus rank one update 
 690                 .add(square(pc).scalarMultiply(ccov1Sep)) 
 691                 // plus rank mu update 
 692                 .add((times(diagC, square(bestArz).multiply(weights))) 
 693                         .scalarMultiply(ccovmuSep)); 
 694         diagD = sqrt(diagC); // replaces eig(C) 
 695         if (diagonalOnly > 1 && iterations > diagonalOnly) { 
 696             // full covariance matrix from now on 
 697             diagonalOnly = 0; 
 698             B = eye(dimension, dimension); 
 699             BD = diag(diagD); 
 700             C = diag(diagC); 
 701         } 
 702     }

 704     /** 
 705      * Update of the covariance matrix C. 
 706      * 
 707      * @param hsig Flag indicating a small correction. 
 708      * @param bestArx Fitness-sorted matrix of the argument vectors producing the 
 709      * current offspring. 
 710      * @param arz Unsorted matrix containing the gaussian random values of the 
 711      * current offspring. 
 712      * @param arindex Indices indicating the fitness-order of the current offspring. 
 713      * @param xold xmean matrix of the previous generation. 
 714      */ 
 715     private void updateCovariance(boolean hsig, final RealMatrix bestArx, 
 716             final RealMatrix arz, final int[] arindex, final RealMatrix xold) { 
 717         double negccov = 0; 
 718         if (ccov1 + ccovmu > 0) { 
 719             RealMatrix arpos = bestArx.subtract(repmat(xold, 1, mu)) 
 720                     .scalarMultiply(1. / sigma); // mu difference vectors 
 721             RealMatrix roneu = pc.multiply(pc.transpose()) 
 722                     .scalarMultiply(ccov1); // rank one update 
 723             // minor correction if hsig==false 
 724             double oldFac = hsig ? 0 : ccov1 * cc * (2. - cc); 
 725             oldFac += 1. - ccov1 - ccovmu; 
 726             if (isActiveCMA) { 
 727                 // Adapt covariance matrix C active CMA 
 728                 negccov = (1. - ccovmu) * 0.25 * mueff / 
 729                 (Math.pow(dimension + 2., 1.5) + 2. * mueff); 
 730                 double negminresidualvariance = 0.66; 
 731                 // keep at least 0.66 in all directions, small popsize are most 
 732                 // critical 
 733                 double negalphaold = 0.5; // where to make up for the variance 
 734                                           // loss, 
 735                 // prepare vectors, compute negative updating matrix Cneg 
 736                 int[] arReverseIndex = reverse(arindex); 
 737                 RealMatrix arzneg 
 738                     = selectColumns(arz, MathArrays.copyOf(arReverseIndex, mu)); 
 739                 RealMatrix arnorms = sqrt(sumRows(square(arzneg))); 
 740                 int[] idxnorms = sortedIndices(arnorms.getRow(0)); 
 741                 RealMatrix arnormsSorted = selectColumns(arnorms, idxnorms); 
 742                 int[] idxReverse = reverse(idxnorms); 
 743                 RealMatrix arnormsReverse = selectColumns(arnorms, idxReverse); 
 744                 arnorms = divide(arnormsReverse, arnormsSorted); 
 745                 int[] idxInv = inverse(idxnorms); 
 746                 RealMatrix arnormsInv = selectColumns(arnorms, idxInv); 
 747                 // check and set learning rate negccov 
 748                 double negcovMax = (1. - negminresidualvariance) / 
 749                         square(arnormsInv).multiply(weights).getEntry(0, 0); 
 750                 if (negccov > negcovMax) { 
 751                     negccov = negcovMax; 
 752                 } 
 753                 arzneg = times(arzneg, repmat(arnormsInv, dimension, 1)); 
 754                 RealMatrix artmp = BD.multiply(arzneg); 
 755                 RealMatrix Cneg = artmp.multiply(diag(weights)).multiply( 
 756                         artmp.transpose()); 
 757                 oldFac += negalphaold * negccov; 
 758                 C = C.scalarMultiply(oldFac) 
 759                         // regard old matrix 
 760                         .add(roneu) 
 761                         // plus rank one update 
 762                         .add(arpos.scalarMultiply( 
 763                                 // plus rank mu update 
 764                                 ccovmu + (1. - negalphaold) * negccov) 
 765                                 .multiply( 
 766                                         times(repmat(weights, 1, dimension), 
 767                                                 arpos.transpose()))) 
 768                         .subtract(Cneg.scalarMultiply(negccov)); 
 769             } else { 
 770                 // Adapt covariance matrix C - nonactive 
 771                 C = C.scalarMultiply(oldFac) // regard old matrix 
 772                         .add(roneu) 
 773                         // plus rank one update 
 774                         .add(arpos.scalarMultiply(ccovmu) // plus rank mu update 
 775                                 .multiply( 
 776                                         times(repmat(weights, 1, dimension), 
 777                                                 arpos.transpose()))); 
 778             } 
 779         } 
 780         updateBD(negccov); 
 781     }

 783     /** 
 784      * Update B and D from C. 
 785      * 
 786      * @param negccov Negative covariance factor. 
 787      */ 
 788     private void updateBD(double negccov) { 
 789         if (ccov1 + ccovmu + negccov > 0 && 
 790                 (iterations % 1. / (ccov1 + ccovmu + negccov) / dimension / 10.) < 1.) { 
 791             // to achieve O(N^2) 
 792             C = triu(C, 0).add(triu(C, 1).transpose()); 
 793             // enforce symmetry to prevent complex numbers 
 794             EigenDecomposition eig = new EigenDecomposition(C); 
 795             B = eig.getV(); // eigen decomposition, B==normalized eigenvectors 
 796             D = eig.getD(); 
 797             diagD = diag(D); 
 798             if (min(diagD) <= 0) { 
 799                 for (int i = 0; i < dimension; i++) { 
 800                     if (diagD.getEntry(i, 0) < 0) { 
 801                         diagD.setEntry(i, 0, 0.); 
 802                     } 
 803                 } 
 804                 double tfac = max(diagD) / 1e14; 
 805                 C = C.add(eye(dimension, dimension).scalarMultiply(tfac)); 
 806                 diagD = diagD.add(ones(dimension, 1).scalarMultiply(tfac)); 
 807             } 
 808             if (max(diagD) > 1e14 * min(diagD)) { 
 809                 double tfac = max(diagD) / 1e14 - min(diagD); 
 810                 C = C.add(eye(dimension, dimension).scalarMultiply(tfac)); 
 811                 diagD = diagD.add(ones(dimension, 1).scalarMultiply(tfac)); 
 812             } 
 813             diagC = diag(C); 
 814             diagD = sqrt(diagD); // D contains standard deviations now 
 815             BD = times(B, repmat(diagD.transpose(), dimension, 1)); // O(n^2) 
 816         } 
 817     }

 900     /** 
 901      * Normalizes fitness values to the range [0,1]. Adds a penalty to the 
 902      * fitness value if out of range. The penalty is adjusted by calling 
 903      * setValueRange(). 
 904      */ 
 905     private class FitnessFunction { 
 906         /** Determines the penalty for boundary violations */ 
 907         private double valueRange; 
 908         /** 
 909          * Flag indicating whether the objective variables are forced into their 
 910          * bounds if defined 
 911          */ 
 912         private boolean isRepairMode; 
 913  
 914         /** Simple constructor. 
 915          */ 
 916         public FitnessFunction() { 
 917             valueRange = 1.0; 
 918             isRepairMode = true; 
 919         } 
 920  
 921         /** 
 922          * @param x Original objective variables. 
 923          * @return the normalized objective variables. 
 924          */ 
 925         public double[] encode(final double[] x) { 
 926             if (boundaries == null) { 
 927                 return x; 
 928             } 
 929             double[] res = new double[x.length]; 
 930             for (int i = 0; i < x.length; i++) { 
 931                 double diff = boundaries[1][i] - boundaries[0][i]; 
 932                 res[i] = x[i] / diff; 
 933             } 
 934             return res; 
 935         } 
 936  
 937         /** 
 938          * @param x Normalized objective variables. 
 939          * @return the original objective variables, possibly repaired. 
 940          */ 
 941         public double[] repairAndDecode(final double[] x) { 
 942             return boundaries != null && isRepairMode ? 
 943                 decode(repair(x)) : 
 944                 decode(x); 
 945         } 
 946  
 947         /** 
 948          * @param x Normalized objective variables. 
 949          * @return the original objective variables. 
 950          */ 
 951         public double[] decode(final double[] x) { 
 952             if (boundaries == null) { 
 953                 return x; 
 954             } 
 955             double[] res = new double[x.length]; 
 956             for (int i = 0; i < x.length; i++) { 
 957                 double diff = boundaries[1][i] - boundaries[0][i]; 
 958                 res[i] = diff * x[i]; 
 959             } 
 960             return res; 
 961         } 
 962  
 963         /** 
 964          * @param point Normalized objective variables. 
 965          * @return the objective value + penalty for violated bounds. 
 966          */ 
 967         public double value(final double[] point) { 
 968             double value; 
 969             if (boundaries != null && isRepairMode) { 
 970                 double[] repaired = repair(point); 
 971                 value = CMAESOptimizer.this 
 972                         .computeObjectiveValue(decode(repaired)) + 
 973                         penalty(point, repaired); 
 974             } else { 
 975                 value = CMAESOptimizer.this 
 976                         .computeObjectiveValue(decode(point)); 
 977             } 
 978             return isMinimize ? value : -value; 
 979         } 
 980  
 981         /** 
 982          * @param x Normalized objective variables. 
 983          * @return {@code true} if in bounds. 
 984          */ 
 985         public boolean isFeasible(final double[] x) { 
 986             if (boundaries == null) { 
 987                 return true; 
 988             } 
 989  
 990             final double[] bLoEnc = encode(boundaries[0]); 
 991             final double[] bHiEnc = encode(boundaries[1]); 
 992  
 993             for (int i = 0; i < x.length; i++) { 
 994                 if (x[i] < bLoEnc[i]) { 
 995                     return false; 
 996                 } 
 997                 if (x[i] > bHiEnc[i]) { 
 998                     return false; 
 999                 } 
1000             } 
1001             return true; 
1002         }

1004         /** 
1005          * @param valueRange Adjusts the penalty computation. 
1006          */ 
1007         public void setValueRange(double valueRange) { 
1008             this.valueRange = valueRange; 
1009         }

1011         /** 
1012          * @param x Normalized objective variables. 
1013          * @return the repaired objective variables - all in bounds. 
1014          */ 
1015         private double[] repair(final double[] x) { 
1016             double[] repaired = new double[x.length]; 
1017             for (int i = 0; i < x.length; i++) { 
1018                 if (x[i] < 0) { 
1019                     repaired[i] = 0; 
1020                 } else if (x[i] > 1.0) { 
1021                     repaired[i] = 1.0; 
1022                 } else { 
1023                     repaired[i] = x[i]; 
1024                 } 
1025             } 
1026             return repaired; 
1027         }

1029         /** 
1030          * @param x Normalized objective variables. 
1031          * @param repaired Repaired objective variables. 
1032          * @return Penalty value according to the violation of the bounds. 
1033          */ 
1034         private double penalty(final double[] x, final double[] repaired) { 
1035             double penalty = 0; 
1036             for (int i = 0; i < x.length; i++) { 
1037                 double diff = Math.abs(x[i] - repaired[i]); 
1038                 penalty += diff * valueRange; 
1039             } 
1040             return isMinimize ? penalty : -penalty; 
1041         }
