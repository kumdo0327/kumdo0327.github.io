171         callback_model.stop_training = False 
172         # Construct epoch logs. 
173         epoch_logs = {} 
174         while epoch < epochs: 
175             for m in model.stateful_metric_functions: 
176                 m.reset_states() 
177             callbacks.on_epoch_begin(epoch) 
178             steps_done = 0 
179             batch_index = 0 
180             while steps_done < steps_per_epoch: 
181                 generator_output = next(output_generator) 
182  
183                 if not hasattr(generator_output, '__len__'): 
184                     raise ValueError('Output of generator should be ' 
185                                      'a tuple `(x, y, sample_weight)` ' 
186                                      'or `(x, y)`. Found: ' + 
187                                      str(generator_output)) 
188  
189                 if len(generator_output) == 2: 
190                     x, y = generator_output 
191                     sample_weight = None 
192                 elif len(generator_output) == 3: 
193                     x, y, sample_weight = generator_output 
194                 else: 
195                     raise ValueError('Output of generator should be ' 
196                                      'a tuple `(x, y, sample_weight)` ' 
197                                      'or `(x, y)`. Found: ' + 
198                                      str(generator_output)) 
199                 # build batch logs 
200                 batch_logs = {} 
201                 if x is None or len(x) == 0: 
202                     # Handle data tensors support when no input given 
203                     # step-size = 1 for data tensors 
204                     batch_size = 1 
205                 elif isinstance(x, list): 
206                     batch_size = x[0].shape[0] 
207                 elif isinstance(x, dict): 
208                     batch_size = list(x.values())[0].shape[0] 
209                 else: 
210                     batch_size = x.shape[0] 
211                 batch_logs['batch'] = batch_index 
212                 batch_logs['size'] = batch_size 
213                 callbacks.on_batch_begin(batch_index, batch_logs) 
214  
215                 outs = model.train_on_batch(x, y, 
216                                             sample_weight=sample_weight, 
217                                             class_weight=class_weight) 
218  
219                 outs = to_list(outs) 
220                 for l, o in zip(out_labels, outs): 
221                     batch_logs[l] = o 
222  
223                 callbacks.on_batch_end(batch_index, batch_logs) 
224  
225                 batch_index += 1 
226                 steps_done += 1 
227  
228                 # Epoch finished. 
229                 if steps_done >= steps_per_epoch and do_validation: 
230                     if val_gen: 
231                         val_outs = model.evaluate_generator( 
232                             val_enqueuer_gen, 
233                             validation_steps, 
234                             workers=0) 
235                     else: 
236                         # No need for try/except because 
237                         # data has already been validated. 
238                         val_outs = model.evaluate( 
239                             val_x, val_y, 
240                             batch_size=batch_size, 
241                             sample_weight=val_sample_weights, 
242                             verbose=0) 
243                     val_outs = to_list(val_outs) 
244                     # Same labels assumed. 
245                     for l, o in zip(out_labels, val_outs): 
246                         epoch_logs['val_' + l] = o 
247  
248                 if callback_model.stop_training: 
249                     break 
250  
251             callbacks.on_epoch_end(epoch, epoch_logs) 
252             epoch += 1 
253             if callback_model.stop_training: 
254                 break 
255  
256     finally: 
257         try: 
258             if enqueuer is not None: 
259                 enqueuer.stop() 
260         finally: 
261             if val_enqueuer is not None: 
262                 val_enqueuer.stop() 
263  
264     callbacks.on_train_end() 
265     return model.history

268 def evaluate_generator(model, generator, 
269                        steps=None, 
270                        max_queue_size=10, 
271                        workers=1, 
272                        use_multiprocessing=False, 
273                        verbose=0): 
274     """See docstring for `Model.evaluate_generator`.""" 
275     model._make_test_function() 
276  
277     if hasattr(model, 'metrics'): 
278         for m in model.stateful_metric_functions: 
279             m.reset_states() 
280         stateful_metric_indices = [ 
281             i for i, name in enumerate(model.metrics_names) 
282             if str(name) in model.stateful_metric_names] 
283     else: 
284         stateful_metric_indices = [] 
285  
286     steps_done = 0 
287     outs_per_batch = [] 
288     batch_sizes = [] 
289     use_sequence_api = is_sequence(generator) 
290     if not use_sequence_api and use_multiprocessing and workers > 1: 
291         warnings.warn( 
292             UserWarning('Using a generator with `use_multiprocessing=True`' 
293                         ' and multiple workers may duplicate your data.' 
294                         ' Please consider using the`keras.utils.Sequence' 
295                         ' class.')) 
296     if steps is None: 
297         if use_sequence_api: 
298             steps = len(generator) 
299         else: 
300             raise ValueError('`steps=None` is only valid for a generator' 
301                              ' based on the `keras.utils.Sequence` class.' 
302                              ' Please specify `steps` or use the' 
303                              ' `keras.utils.Sequence` class.') 
304     enqueuer = None 
305  
306     try: 
307         if workers > 0: 
308             if use_sequence_api: 
309                 enqueuer = OrderedEnqueuer( 
310                     generator, 
311                     use_multiprocessing=use_multiprocessing) 
312             else: 
313                 enqueuer = GeneratorEnqueuer( 
314                     generator, 
315                     use_multiprocessing=use_multiprocessing) 
316             enqueuer.start(workers=workers, max_queue_size=max_queue_size) 
317             output_generator = enqueuer.get() 
318         else: 
319             if use_sequence_api: 
320                 output_generator = iter_sequence_infinite(generator) 
321             else: 
322                 output_generator = generator 
323  
324         if verbose == 1: 
325             progbar = Progbar(target=steps) 
326  
327         while steps_done < steps: 
328             generator_output = next(output_generator) 
329             if not hasattr(generator_output, '__len__'): 
330                 raise ValueError('Output of generator should be a tuple ' 
331                                  '(x, y, sample_weight) ' 
332                                  'or (x, y). Found: ' + 
333                                  str(generator_output)) 
334             if len(generator_output) == 2: 
335                 x, y = generator_output 
336                 sample_weight = None 
337             elif len(generator_output) == 3: 
338                 x, y, sample_weight = generator_output 
339             else: 
340                 raise ValueError('Output of generator should be a tuple ' 
341                                  '(x, y, sample_weight) ' 
342                                  'or (x, y). Found: ' + 
343                                  str(generator_output)) 
344             outs = model.test_on_batch(x, y, sample_weight=sample_weight) 
345             outs = to_list(outs) 
346             outs_per_batch.append(outs) 
347  
348             if x is None or len(x) == 0: 
349                 # Handle data tensors support when no input given 
350                 # step-size = 1 for data tensors 
351                 batch_size = 1 
352             elif isinstance(x, list): 
353                 batch_size = x[0].shape[0] 
354             elif isinstance(x, dict): 
355                 batch_size = list(x.values())[0].shape[0] 
356             else: 
357                 batch_size = x.shape[0] 
358             if batch_size == 0: 
359                 raise ValueError('Received an empty batch. ' 
360                                  'Batches should contain ' 
361                                  'at least one item.') 
362             steps_done += 1 
363             batch_sizes.append(batch_size) 
364             if verbose == 1: 
365                 progbar.update(steps_done) 
366  
367     finally: 
368         if enqueuer is not None: 
369             enqueuer.stop() 
370  
371     averages = [] 
372     for i in range(len(outs)): 
373         if i not in stateful_metric_indices: 
374             averages.append(np.average([out[i] for out in outs_per_batch], 
375                                        weights=batch_sizes)) 
376         else: 
377             averages.append(np.float64(outs_per_batch[-1][i])) 
378     return unpack_singleton(averages)

381 def predict_generator(model, generator, 
382                       steps=None, 
383                       max_queue_size=10, 
384                       workers=1, 
385                       use_multiprocessing=False, 
386                       verbose=0): 
387     """See docstring for `Model.predict_generator`.""" 
388     model._make_predict_function() 
389  
390     steps_done = 0 
391     all_outs = [] 
392     use_sequence_api = is_sequence(generator) 
393     if not use_sequence_api and use_multiprocessing and workers > 1: 
394         warnings.warn( 
395             UserWarning('Using a generator with `use_multiprocessing=True`' 
396                         ' and multiple workers may duplicate your data.' 
397                         ' Please consider using the`keras.utils.Sequence' 
398                         ' class.')) 
399     if steps is None: 
400         if use_sequence_api: 
401             steps = len(generator) 
402         else: 
403             raise ValueError('`steps=None` is only valid for a generator' 
404                              ' based on the `keras.utils.Sequence` class.' 
405                              ' Please specify `steps` or use the' 
406                              ' `keras.utils.Sequence` class.') 
407     enqueuer = None 
408  
409     try: 
410         if workers > 0: 
411             if use_sequence_api: 
412                 enqueuer = OrderedEnqueuer( 
413                     generator, 
414                     use_multiprocessing=use_multiprocessing) 
415             else: 
416                 enqueuer = GeneratorEnqueuer( 
417                     generator, 
418                     use_multiprocessing=use_multiprocessing) 
419             enqueuer.start(workers=workers, max_queue_size=max_queue_size) 
420             output_generator = enqueuer.get() 
421         else: 
422             if use_sequence_api: 
423                 output_generator = iter_sequence_infinite(generator) 
424             else: 
425                 output_generator = generator 
426  
427         if verbose == 1: 
428             progbar = Progbar(target=steps) 
429  
430         while steps_done < steps: 
431             generator_output = next(output_generator) 
432             if isinstance(generator_output, tuple): 
433                 # Compatibility with the generators 
434                 # used for training. 
435                 if len(generator_output) == 2: 
436                     x, _ = generator_output 
437                 elif len(generator_output) == 3: 
438                     x, _, _ = generator_output 
439                 else: 
440                     raise ValueError('Output of generator should be ' 
441                                      'a tuple `(x, y, sample_weight)` ' 
442                                      'or `(x, y)`. Found: ' + 
443                                      str(generator_output)) 
444             else: 
445                 # Assumes a generator that only 
446                 # yields inputs (not targets and sample weights). 
447                 x = generator_output 
448  
449             outs = model.predict_on_batch(x) 
450             outs = to_list(outs) 
451  
452             if not all_outs: 
453                 for out in outs: 
454                     all_outs.append([]) 
455  
456             for i, out in enumerate(outs): 
457                 all_outs[i].append(out) 
458             steps_done += 1 
459             if verbose == 1: 
460                 progbar.update(steps_done) 
461  
462     finally: 
463         if enqueuer is not None: 
464             enqueuer.stop() 
465  
466     if len(all_outs) == 1: 
467         if steps_done == 1: 
468             return all_outs[0][0] 
469         else: 
470             return np.concatenate(all_outs[0]) 
471     if steps_done == 1: 
472         return [out[0] for out in all_outs] 
473     else: 
474         return [np.concatenate(out) for out in all_outs]
