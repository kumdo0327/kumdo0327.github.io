 209 // Standard permits specialization of std::numeric_limits. This specialization 
 210 // is used to resolve ambiguity between isinf and std::isinf in glibc: 
 211 // https://gcc.gnu.org/bugzilla/show_bug.cgi?id=48891 
 212 // and the same for isnan and signbit. 
 213 template <> 
 214 class numeric_limits<fmt::internal::dummy_int> : 
 215     public std::numeric_limits<int> { 
 216  public: 
 217   // Portable version of isinf. 
 218   template <typename T> 
 219   static bool isinfinity(T x) { 
 220     using namespace fmt::internal; 
 221     // The resolution "priority" is: 
 222     // isinf macro > std::isinf > ::isinf > fmt::internal::isinf 
 223     if (const_check(sizeof(isinf(x)) == sizeof(bool) || 
 224                     sizeof(isinf(x)) == sizeof(int))) { 
 225       return isinf(x) != 0; 
 226     } 
 227     return !_finite(static_cast<double>(x)); 
 228   } 
 229  
 230   // Portable version of isnan. 
 231   template <typename T> 
 232   static bool isnotanumber(T x) { 
 233     using namespace fmt::internal; 
 234     if (const_check(sizeof(isnan(x)) == sizeof(bool) || 
 235                     sizeof(isnan(x)) == sizeof(int))) { 
 236       return isnan(x) != 0; 
 237     } 
 238     return _isnan(static_cast<double>(x)) != 0; 
 239   } 
 240  
 241   // Portable version of signbit. 
 242   static bool isnegative(double x) { 
 243     using namespace fmt::internal; 
 244     if (const_check(sizeof(signbit(x)) == sizeof(int))) 
 245       return signbit(x) != 0; 
 246     if (x < 0) return true; 
 247     if (!isnotanumber(x)) return false; 
 248     int dec = 0, sign = 0; 
 249     char buffer[2];  // The buffer size must be >= 2 or _ecvt_s will fail. 
 250     _ecvt_s(buffer, sizeof(buffer), x, 0, &dec, &sign); 
 251     return sign != 0; 
 252   } 
 253 };

 319   /** Sets the buffer data and capacity. */

 321     ptr_ = data; 
 322     capacity_ = capacity;

 325   /**

 328     \endrst 
 329    */

 395 /** 
 396   \rst 
 397   A dynamically growing memory buffer for trivially copyable/constructible types 
 398   with the first SIZE elements stored in the object itself. 
 399  
 400   You can use one of the following typedefs for common character types: 
 401  
 402   +----------------+------------------------------+ 
 403   | Type           | Definition                   | 
 404   +================+==============================+ 
 405   | memory_buffer  | basic_memory_buffer<char>    | 
 406   +----------------+------------------------------+ 
 407   | wmemory_buffer | basic_memory_buffer<wchar_t> | 
 408   +----------------+------------------------------+ 
 409  
 410   **Example**:: 
 411  
 412      memory_buffer out; 
 413      format_to(out, "The answer is {}.", 42); 
 414  
 415   This will write the following output to the ``out`` object: 
 416  
 417   .. code-block:: none 
 418  
 419      The answer is 42. 
 420  
 421   The output can be converted to an ``std::string`` with ``to_string(out)``. 
 422   \endrst 
 423  */ 
 424 template <typename T, std::size_t SIZE = internal::INLINE_BUFFER_SIZE, 
 425           typename Allocator = std::allocator<T> > 
 426 class basic_memory_buffer : private Allocator, public basic_buffer<T> { 
 427  private: 
 428   T store_[SIZE]; 
 429  
 430   // Deallocate memory allocated by the buffer. 
 431   void deallocate() { 
 432     T* data = this->data(); 
 433     if (data != store_) Allocator::deallocate(data, this->capacity()); 
 434   } 
 435  
 436  protected: 
 437   void grow(std::size_t size); 
 438  
 439  public: 
 440   explicit basic_memory_buffer(const Allocator &alloc = Allocator()) 
 441       : Allocator(alloc) { 
 442     this->set(store_, SIZE); 
 443   } 
 444   ~basic_memory_buffer() { deallocate(); } 
 445  
 446  private: 
 447   // Move data from other to this buffer. 
 448   void move(basic_memory_buffer &other) { 
 449     Allocator &this_alloc = *this, &other_alloc = other; 
 450     this_alloc = std::move(other_alloc); 
 451     T* data = other.data(); 
 452     std::size_t size = other.size(), capacity = other.capacity(); 
 453     if (data == other.store_) { 
 454       this->set(store_, capacity); 
 455       std::uninitialized_copy(other.store_, other.store_ + size, 
 456                               internal::make_ptr(store_, capacity)); 
 457     } else { 
 458       this->set(data, capacity); 
 459       // Set pointer to the inline array so that delete is not called 
 460       // when deallocating. 
 461       other.set(other.store_, 0); 
 462     } 
 463     this->resize(size); 
 464   } 
 465  
 466  public: 
 467   /** 
 468     \rst 
 469     Constructs a :class:`fmt::basic_memory_buffer` object moving the content 
 470     of the other object to it. 
 471     \endrst 
 472    */ 
 473   basic_memory_buffer(basic_memory_buffer &&other) { 
 474     move(other); 
 475   } 
 476  
 477   /** 
 478     \rst 
 479     Moves the content of the other ``basic_memory_buffer`` object to this one. 
 480     \endrst 
 481    */ 
 482   basic_memory_buffer &operator=(basic_memory_buffer &&other) { 
 483     assert(this != &other); 
 484     deallocate(); 
 485     move(other); 
 486     return *this; 
 487   } 
 488  
 489   // Returns a copy of the allocator associated with this buffer. 
 490   Allocator get_allocator() const { return *this; } 
 491 };

 493 template <typename T, std::size_t SIZE, typename Allocator> 
 494 void basic_memory_buffer<T, SIZE, Allocator>::grow(std::size_t size) { 
 495   std::size_t old_capacity = this->capacity(); 
 496   std::size_t new_capacity = old_capacity + old_capacity / 2; 
 497   if (size > new_capacity) 
 498       new_capacity = size; 
 499   T *old_data = this->data(); 
 500   T *new_data = this->allocate(new_capacity); 
 501   // The following code doesn't throw, so the raw pointer above doesn't leak. 
 502   std::uninitialized_copy(old_data, old_data + this->size(), 
 503                           internal::make_ptr(new_data, new_capacity)); 
 504   this->set(new_data, new_capacity); 
 505   // deallocate must not throw according to the standard, but even if it does, 
 506   // the buffer already uses the new storage and will deallocate it in 
 507   // destructor. 
 508   if (old_data != store_) 
 509     Allocator::deallocate(old_data, old_capacity); 
 510 }

 596 // and allows comparing the performance of processing a null-terminated string 
 597 // vs string_view. 
 598 template <typename Char> 
 599 class null_terminating_iterator { 
 600  public: 
 601   using difference_type = std::ptrdiff_t; 
 602   using value_type = Char; 
 603   using pointer = const Char*; 
 604   using reference = const Char&; 
 605   using iterator_category = std::random_access_iterator_tag; 
 606  
 607   null_terminating_iterator() : ptr_(0), end_(0) {} 
 608  
 609   constexpr null_terminating_iterator(const Char *ptr, const Char *end) 
 610     : ptr_(ptr), end_(end) {} 
 611  
 612   template <typename Range> 
 613   constexpr explicit null_terminating_iterator(const Range &r) 
 614     : ptr_(r.begin()), end_(r.end()) {} 
 615  
 616   null_terminating_iterator &operator=(const Char *ptr) { 
 617     assert(ptr <= end_); 
 618     ptr_ = ptr; 
 619     return *this; 
 620   } 
 621

 623     return ptr_ != end_ ? *ptr_ : 0; 
 624   }

 627     ++ptr_; 
 628     return *this; 
 629   }

 632     null_terminating_iterator result(*this); 
 633     ++ptr_; 
 634     return result; 
 635   }

 638     --ptr_; 
 639     return *this; 
 640   }

 643     return null_terminating_iterator(ptr_ + n, end_); 
 644   }

 647     return null_terminating_iterator(ptr_ - n, end_);

 651     ptr_ += n;

 725 // Returns the number of decimal digits in n. Leading zeros are not counted 
 726 // except for n == 0 in which case count_digits returns 1. 
 727 inline unsigned count_digits(uint64_t n) { 
 728   // Based on http://graphics.stanford.edu/~seander/bithacks.html#IntegerLog10 
 729   // and the benchmark https://github.com/localvoid/cxx-benchmark-count-digits. 
 730   int t = (64 - FMT_BUILTIN_CLZLL(n | 1)) * 1233 >> 12; 
 731   return to_unsigned(t) - (n < data::POWERS_OF_10_64[t]) + 1; 
 732 } 
 733 #else 
 734 // Fallback version of count_digits used when __builtin_clz is not available. 
 735 inline unsigned count_digits(uint64_t n) { 
 736   unsigned count = 1; 
 737   for (;;) { 
 738     // Integer division is slow so do it for a group of four digits instead 
 739     // of for every digit. The idea comes from the talk by Alexandrescu 
 740     // "Three Optimization Tips for C++". See speed-test for a comparison. 
 741     if (n < 10) return count; 
 742     if (n < 100) return count + 1; 
 743     if (n < 1000) return count + 2; 
 744     if (n < 10000) return count + 3; 
 745     n /= 10000u; 
 746     count += 4; 
 747   } 
 748 } 
 749 #endif

 751 #ifdef FMT_BUILTIN_CLZ 
 752 // Optional version of count_digits for better performance on 32-bit platforms. 
 753 inline unsigned count_digits(uint32_t n) { 
 754   int t = (32 - FMT_BUILTIN_CLZ(n | 1)) * 1233 >> 12; 
 755   return to_unsigned(t) - (n < data::POWERS_OF_10_32[t]) + 1; 
 756 }

 768  private: 
 769   basic_string_view<Char> sep_;

 771   // Index of a decimal digit with the least significant digit having index 0.

 790 // Formats a decimal unsigned integer value writing into buffer. 
 791 // thousands_sep is a functor that is called after writing each char to 
 792 // add a thousands separator if necessary. 
 793 template <typename UInt, typename Char, typename ThousandsSep> 
 794 inline void format_decimal(Char *buffer, UInt value, unsigned num_digits, 
 795                            ThousandsSep thousands_sep) { 
 796   buffer += num_digits; 
 797   while (value >= 100) { 
 798     // Integer division is slow so do it for a group of two digits instead 
 799     // of for every digit. The idea comes from the talk by Alexandrescu 
 800     // "Three Optimization Tips for C++". See speed-test for a comparison. 
 801     unsigned index = static_cast<unsigned>((value % 100) * 2); 
 802     value /= 100; 
 803     *--buffer = data::DIGITS[index + 1]; 
 804     thousands_sep(buffer); 
 805     *--buffer = data::DIGITS[index]; 
 806     thousands_sep(buffer); 
 807   } 
 808   if (value < 10) { 
 809     *--buffer = static_cast<char>('0' + value); 
 810     return; 
 811   } 
 812   unsigned index = static_cast<unsigned>(value * 2); 
 813   *--buffer = data::DIGITS[index + 1]; 
 814   thousands_sep(buffer); 
 815   *--buffer = data::DIGITS[index]; 
 816 }

 876 /** 
 877   \rst 
 878   Visits an argument dispatching to the appropriate visit method based on 
 879   the argument type. For example, if the argument type is ``double`` then 
 880   ``vis(value)`` will be called with the value of type ``double``. 
 881   \endrst 
 882  */ 
 883 template <typename Visitor, typename Context> 
 884 constexpr typename std::result_of<Visitor(int)>::type 
 885     visit(Visitor &&vis, basic_arg<Context> arg) { 
 886   using char_type = typename Context::char_type; 
 887   switch (arg.type_) { 
 888   case internal::NONE: 
 889     return vis(monostate()); 
 890   case internal::NAMED_ARG: 
 891     FMT_ASSERT(false, "invalid argument type"); 
 892     break; 
 893   case internal::INT: 
 894     return vis(arg.value_.int_value); 
 895   case internal::UINT: 
 896     return vis(arg.value_.uint_value); 
 897   case internal::LONG_LONG: 
 898     return vis(arg.value_.long_long_value); 
 899   case internal::ULONG_LONG: 
 900     return vis(arg.value_.ulong_long_value); 
 901   case internal::BOOL: 
 902     return vis(arg.value_.int_value != 0); 
 903   case internal::CHAR: 
 904     return vis(static_cast<char_type>(arg.value_.int_value)); 
 905   case internal::DOUBLE: 
 906     return vis(arg.value_.double_value); 
 907   case internal::LONG_DOUBLE: 
 908     return vis(arg.value_.long_double_value); 
 909   case internal::CSTRING: 
 910     return vis(arg.value_.string.value); 
 911   case internal::STRING: 
 912     return vis(basic_string_view<char_type>( 
 913                  arg.value_.string.value, arg.value_.string.size)); 
 914   case internal::POINTER: 
 915     return vis(arg.value_.pointer); 
 916   case internal::CUSTOM: 
 917     return vis(typename basic_arg<Context>::handle(arg.value_.custom)); 
 918   } 
 919   return typename std::result_of<Visitor(int)>::type(); 
 920 }

1336 // Parses the input as an unsigned integer. This function assumes that the 
1337 // first character is a digit and presence of a non-digit character at the end. 
1338 // it: an iterator pointing to the beginning of the input range. 
1339 template <typename Iterator, typename ErrorHandler> 
1340 constexpr unsigned parse_nonnegative_int(Iterator &it, ErrorHandler &&eh) { 
1341   assert('0' <= *it && *it <= '9'); 
1342   unsigned value = 0; 
1343   // Convert to unsigned to prevent a warning. 
1344   unsigned max_int = (std::numeric_limits<int>::max)(); 
1345   unsigned big = max_int / 10; 
1346   do { 
1347     // Check for overflow. 
1348     if (value > big) { 
1349       value = max_int + 1; 
1350       break; 
1351     } 
1352     value = value * 10 + (*it - '0'); 
1353     // Workaround for MSVC "setup_exception stack overflow" error: 
1354     auto next = it; 
1355     ++next; 
1356     it = next; 
1357   } while ('0' <= *it && *it <= '9'); 
1358   if (value > max_int) 
1359     eh.on_error("number is too big"); 
1360   return value; 
1361 }

1664 template <typename Iterator, typename IDHandler> 
1665 constexpr Iterator parse_arg_id(Iterator it, IDHandler &&handler) { 
1666   using char_type = typename std::iterator_traits<Iterator>::value_type; 
1667   char_type c = *it; 
1668   if (c == '}' || c == ':') { 
1669     handler(); 
1670     return it; 
1671   } 
1672   if (c >= '0' && c <= '9') { 
1673     unsigned index = parse_nonnegative_int(it, handler); 
1674     if (*it != '}' && *it != ':') { 
1675       handler.on_error("invalid format string"); 
1676       return it; 
1677     } 
1678     handler(index); 
1679     return it; 
1680   } 
1681   if (!is_name_start(c)) { 
1682     handler.on_error("invalid format string"); 
1683     return it; 
1684   } 
1685   auto start = it; 
1686   do { 
1687     c = *++it; 
1688   } while (is_name_start(c) || ('0' <= c && c <= '9')); 
1689   handler(basic_string_view<char_type>(pointer_from(start), it - start)); 
1690   return it; 
1691 }

1725 // Parses standard format specifiers and sends notifications about parsed 
1726 // components to handler. 
1727 // it: an iterator pointing to the beginning of a null-terminated range of 
1728 //     characters, possibly emulated via null_terminating_iterator, representing 
1729 //     format specifiers. 
1730 template <typename Iterator, typename SpecHandler> 
1731 constexpr Iterator parse_format_specs(Iterator it, SpecHandler &&handler) { 
1732   using char_type = typename std::iterator_traits<Iterator>::value_type; 
1733   // Parse fill and alignment. 
1734   if (char_type c = *it) { 
1735     alignment align = ALIGN_DEFAULT; 
1736     int i = 1; 
1737     do { 
1738       auto p = it + i; 
1739       switch (*p) { 
1740         case '<': 
1741           align = ALIGN_LEFT; 
1742           break; 
1743         case '>': 
1744           align = ALIGN_RIGHT; 
1745           break; 
1746         case '=': 
1747           align = ALIGN_NUMERIC; 
1748           break; 
1749         case '^': 
1750           align = ALIGN_CENTER; 
1751           break; 
1752       } 
1753       if (align != ALIGN_DEFAULT) { 
1754         handler.on_align(align); 
1755         if (p != it) { 
1756           if (c == '}') break; 
1757           if (c == '{') { 
1758             handler.on_error("invalid fill character '{'"); 
1759             return it; 
1760           } 
1761           it += 2; 
1762           handler.on_fill(c); 
1763         } else ++it; 
1764         break; 
1765       } 
1766     } while (--i >= 0); 
1767   } 
1768  
1769   // Parse sign. 
1770   switch (*it) { 
1771     case '+': 
1772       handler.on_plus(); 
1773       ++it; 
1774       break; 
1775     case '-': 
1776       handler.on_minus(); 
1777       ++it; 
1778       break; 
1779     case ' ': 
1780       handler.on_space(); 
1781       ++it; 
1782       break; 
1783   } 
1784  
1785   if (*it == '#') { 
1786     handler.on_hash(); 
1787     ++it; 
1788   } 
1789  
1790   // Parse zero flag. 
1791   if (*it == '0') { 
1792     handler.on_zero(); 
1793     ++it; 
1794   } 
1795  
1796   // Parse width. 
1797   if ('0' <= *it && *it <= '9') { 
1798     handler.on_width(parse_nonnegative_int(it, handler)); 
1799   } else if (*it == '{') { 
1800     it = parse_arg_id(it + 1, width_adapter<SpecHandler, char_type>(handler)); 
1801     if (*it++ != '}') { 
1802       handler.on_error("invalid format string"); 
1803       return it; 
1804     } 
1805   } 
1806  
1807   // Parse precision. 
1808   if (*it == '.') { 
1809     ++it; 
1810     if ('0' <= *it && *it <= '9') { 
1811       handler.on_precision(parse_nonnegative_int(it, handler)); 
1812     } else if (*it == '{') { 
1813       it = parse_arg_id( 
1814             it + 1, precision_adapter<SpecHandler, char_type>(handler)); 
1815       if (*it++ != '}') { 
1816         handler.on_error("invalid format string"); 
1817         return it; 
1818       } 
1819     } else { 
1820       handler.on_error("missing precision specifier"); 
1821       return it; 
1822     } 
1823     handler.end_precision(); 
1824   } 
1825  
1826   // Parse type. 
1827   if (*it != '}' && *it) 
1828     handler.on_type(*it++); 
1829   return it; 
1830 }

1849 template <typename Iterator, typename Handler> 
1850 constexpr void parse_format_string(Iterator it, Handler &&handler) { 
1851   using char_type = typename std::iterator_traits<Iterator>::value_type; 
1852   auto start = it; 
1853   while (*it) { 
1854     char_type ch = *it++; 
1855     if (ch != '{' && ch != '}') continue; 
1856     if (*it == ch) { 
1857       handler.on_text(start, it); 
1858       start = ++it; 
1859       continue; 
1860     } 
1861     if (ch == '}') { 
1862       handler.on_error("unmatched '}' in format string"); 
1863       return; 
1864     } 
1865     handler.on_text(start, it - 1); 
1866  
1867     it = parse_arg_id(it, id_adapter<Handler, char_type>(handler)); 
1868     if (*it == '}') { 
1869       handler.on_replacement_field(it); 
1870     } else if (*it == ':') { 
1871       ++it; 
1872       it = handler.on_format_specs(it); 
1873       if (*it != '}') { 
1874         handler.on_error("unknown format specifier"); 
1875         return; 
1876       } 
1877     } else { 
1878       handler.on_error("missing '}' in format string"); 
1879       return; 
1880     } 
1881  
1882     start = ++it; 
1883   } 
1884   handler.on_text(start, it); 
1885 }

2344 template <typename Char> 
2345 typename basic_writer<Char>::pointer_type basic_writer<Char>::fill_padding( 
2346     pointer_type buffer, unsigned total_size, 
2347     std::size_t content_size, wchar_t fill) { 
2348   std::size_t padding = total_size - content_size; 
2349   std::size_t left_padding = padding / 2; 
2350   Char fill_char = internal::char_traits<Char>::cast(fill); 
2351   std::uninitialized_fill_n(buffer, left_padding, fill_char); 
2352   buffer += left_padding; 
2353   pointer_type content = buffer; 
2354   std::uninitialized_fill_n(buffer + content_size, 
2355                             padding - left_padding, fill_char); 
2356   return content;

2359 template <typename Char> 
2360 template <typename Spec> 
2361 typename basic_writer<Char>::pointer_type 
2362   basic_writer<Char>::prepare_int_buffer( 
2363     unsigned num_digits, const Spec &spec, 
2364     const char *prefix, unsigned prefix_size) { 
2365   unsigned width = spec.width(); 
2366   alignment align = spec.align(); 
2367   Char fill = internal::char_traits<Char>::cast(spec.fill()); 
2368   if (spec.precision() > static_cast<int>(num_digits)) { 
2369     // Octal prefix '0' is counted as a digit, so ignore it if precision 
2370     // is specified. 
2371     if (prefix_size > 0 && prefix[prefix_size - 1] == '0') 
2372       --prefix_size; 
2373     unsigned number_size = 
2374         prefix_size + internal::to_unsigned(spec.precision()); 
2375     align_spec subspec(number_size, '0', ALIGN_NUMERIC); 
2376     if (number_size >= width) 
2377       return prepare_int_buffer(num_digits, subspec, prefix, prefix_size); 
2378     buffer_.reserve(width); 
2379     unsigned fill_size = width - number_size; 
2380     if (align != ALIGN_LEFT) { 
2381       pointer_type p = grow_buffer(fill_size); 
2382       std::uninitialized_fill(p, p + fill_size, fill); 
2383     } 
2384     pointer_type result = prepare_int_buffer( 
2385         num_digits, subspec, prefix, prefix_size); 
2386     if (align == ALIGN_LEFT) { 
2387       pointer_type p = grow_buffer(fill_size); 
2388       std::uninitialized_fill(p, p + fill_size, fill); 
2389     } 
2390     return result; 
2391   } 
2392   unsigned size = prefix_size + num_digits; 
2393   if (width <= size) { 
2394     pointer_type p = grow_buffer(size); 
2395     std::uninitialized_copy(prefix, prefix + prefix_size, p); 
2396     return p + size - 1; 
2397   } 
2398   pointer_type p = grow_buffer(width); 
2399   pointer_type end = p + width; 
2400   if (align == ALIGN_LEFT) { 
2401     std::uninitialized_copy(prefix, prefix + prefix_size, p); 
2402     p += size; 
2403     std::uninitialized_fill(p, end, fill); 
2404   } else if (align == ALIGN_CENTER) { 
2405     p = fill_padding(p, width, size, fill); 
2406     std::uninitialized_copy(prefix, prefix + prefix_size, p); 
2407     p += size; 
2408   } else { 
2409     if (align == ALIGN_NUMERIC) { 
2410       if (prefix_size != 0) { 
2411         p = std::uninitialized_copy(prefix, prefix + prefix_size, p); 
2412         size -= prefix_size; 
2413       } 
2414     } else { 
2415       std::uninitialized_copy(prefix, prefix + prefix_size, end - size); 
2416     } 
2417     std::uninitialized_fill(p, end - size, fill); 
2418     p = end; 
2419   } 
2420   return p - 1; 
2421 }

2424 template <typename T, typename Spec> 
2425 void basic_writer<Char>::write_int(T value, const Spec& spec) { 
2426   using unsigned_type = typename internal::int_traits<T>::main_type; 
2427   struct spec_handler { 
2428     basic_writer<Char> &writer; 
2429     const Spec& spec; 
2430     unsigned prefix_size = 0; 
2431     unsigned_type abs_value; 
2432     char prefix[4] = ""; 
2433  
2434     spec_handler(basic_writer<Char> &w, T val, const Spec& s) 
2435       : writer(w), spec(s), abs_value(static_cast<unsigned_type>(val)) { 
2436       if (internal::is_negative(val)) { 
2437         prefix[0] = '-'; 
2438         ++prefix_size; 
2439         abs_value = 0 - abs_value; 
2440       } else if (spec.flag(SIGN_FLAG)) { 
2441         prefix[0] = spec.flag(PLUS_FLAG) ? '+' : ' '; 
2442         ++prefix_size; 
2443       } 
2444     } 
2445  
2446     void on_dec() { 
2447       unsigned num_digits = internal::count_digits(abs_value); 
2448       pointer_type p = 
2449           writer.prepare_int_buffer(num_digits, spec, prefix, prefix_size) + 1; 
2450       internal::format_decimal(get(p), abs_value, 0); 
2451     } 
2452  
2453     void on_hex() { 
2454       unsigned_type n = abs_value; 
2455       if (spec.flag(HASH_FLAG)) { 
2456         prefix[prefix_size++] = '0'; 
2457         prefix[prefix_size++] = spec.type(); 
2458       } 
2459       unsigned num_digits = 0; 
2460       do { 
2461         ++num_digits; 
2462       } while ((n >>= 4) != 0); 
2463       Char *p = 
2464           get(writer.prepare_int_buffer(num_digits, spec, prefix, prefix_size)); 
2465       n = abs_value; 
2466       const char *digits = spec.type() == 'x' ? 
2467           "0123456789abcdef" : "0123456789ABCDEF"; 
2468       do { 
2469         *p-- = digits[n & 0xf]; 
2470       } while ((n >>= 4) != 0); 
2471     }

2473     void on_bin() { 
2474       unsigned_type n = abs_value; 
2475       if (spec.flag(HASH_FLAG)) { 
2476         prefix[prefix_size++] = '0'; 
2477         prefix[prefix_size++] = spec.type(); 
2478       } 
2479       unsigned num_digits = 0; 
2480       do { 
2481         ++num_digits; 
2482       } while ((n >>= 1) != 0); 
2483       Char *p = 
2484           get(writer.prepare_int_buffer(num_digits, spec, prefix, prefix_size)); 
2485       n = abs_value; 
2486       do { 
2487         *p-- = static_cast<Char>('0' + (n & 1)); 
2488       } while ((n >>= 1) != 0); 
2489     } 
2490

2492       unsigned_type n = abs_value; 
2493       if (spec.flag(HASH_FLAG)) 
2494         prefix[prefix_size++] = '0'; 
2495       unsigned num_digits = 0; 
2496       do { 
2497         ++num_digits; 
2498       } while ((n >>= 3) != 0); 
2499       Char *p = 
2500           get(writer.prepare_int_buffer(num_digits, spec, prefix, prefix_size)); 
2501       n = abs_value; 
2502       do { 
2503         *p-- = static_cast<Char>('0' + (n & 7)); 
2504       } while ((n >>= 3) != 0); 
2505     }

2507     void on_num() { 
2508       unsigned num_digits = internal::count_digits(abs_value); 
2509       Char thousands_sep = internal::thousands_sep(writer.buffer_); 
2510       fmt::basic_string_view<Char> sep(&thousands_sep, 1); 
2511       unsigned size = static_cast<unsigned>( 
2512             num_digits + sep.size() * ((num_digits - 1) / 3)); 
2513       pointer_type p = 
2514           writer.prepare_int_buffer(size, spec, prefix, prefix_size) + 1; 
2515       internal::format_decimal(get(p), abs_value, 0, 
2516                                internal::add_thousands_sep<Char>(sep)); 
2517     }

2521     }

2527 template <typename T> 
2528 void basic_writer<Char>::write_double(T value, const format_specs &spec) { 
2529   // Check type. 
2530   struct spec_handler { 
2531     char type; 
2532     bool upper = false; 
2533  
2534     explicit spec_handler(char t) : type(t) {} 
2535  
2536     void on_general() { 
2537       if (type == 'G') 
2538         upper = true; 
2539       else 
2540         type = 'g'; 
2541     } 
2542  
2543     void on_exp() { 
2544       if (type == 'E') 
2545         upper = true; 
2546     } 
2547  
2548     void on_fixed() { 
2549       if (type == 'F') { 
2550         upper = true; 
2551 #if FMT_MSC_VER 
2552         // MSVC's printf doesn't support 'F'. 
2553         type = 'f'; 
2554 #endif 
2555       } 
2556     } 
2557  
2558     void on_hex() { 
2559       if (type == 'A') 
2560         upper = true; 
2561     }

2563     void on_error() { 
2564       FMT_THROW(format_error("invalid type specifier")); 
2565     } 
2566   }; 
2567   spec_handler handler(spec.type()); 
2568   internal::handle_float_type_spec(spec.type(), handler); 
2569  
2570   char sign = 0; 
2571   // Use isnegative instead of value < 0 because the latter is always 
2572   // false for NaN. 
2573   if (internal::fputil::isnegative(static_cast<double>(value))) { 
2574     sign = '-'; 
2575     value = -value; 
2576   } else if (spec.flag(SIGN_FLAG)) { 
2577     sign = spec.flag(PLUS_FLAG) ? '+' : ' '; 
2578   } 
2579  
2580   if (internal::fputil::isnotanumber(value)) { 
2581     // Format NaN ourselves because sprintf's output is not consistent 
2582     // across platforms. 
2583     std::size_t nan_size = 4; 
2584     const char *nan = handler.upper ? " NAN" : " nan"; 
2585     if (!sign) { 
2586       --nan_size; 
2587       ++nan; 
2588     } 
2589     pointer_type out = write_str(nan, nan_size, spec); 
2590     if (sign) 
2591       *out = sign; 
2592     return; 
2593   } 
2594  
2595   if (internal::fputil::isinfinity(value)) { 
2596     // Format infinity ourselves because sprintf's output is not consistent 
2597     // across platforms. 
2598     std::size_t inf_size = 4; 
2599     const char *inf = handler.upper ? " INF" : " inf"; 
2600     if (!sign) { 
2601       --inf_size; 
2602       ++inf; 
2603     } 
2604     pointer_type out = write_str(inf, inf_size, spec); 
2605     if (sign) 
2606       *out = sign; 
2607     return; 
2608   } 
2609  
2610   std::size_t offset = buffer_.size(); 
2611   unsigned width = spec.width(); 
2612   if (sign) { 
2613     buffer_.reserve(buffer_.size() + (width > 1u ? width : 1u)); 
2614     if (width > 0) 
2615       --width; 
2616     ++offset; 
2617   } 
2618  
2619   // Build format string. 
2620   enum { MAX_FORMAT_SIZE = 10}; // longest format: %#-*.*Lg 
2621   Char format[MAX_FORMAT_SIZE]; 
2622   Char *format_ptr = format; 
2623   *format_ptr++ = '%'; 
2624   unsigned width_for_sprintf = width; 
2625   if (spec.flag(HASH_FLAG)) 
2626     *format_ptr++ = '#'; 
2627   if (spec.align() == ALIGN_CENTER) { 
2628     width_for_sprintf = 0; 
2629   } else {

2637     *format_ptr++ = '*';

2641   *format_ptr++ = handler.type; 
2642   *format_ptr = '\0'; 
2643  
2644   // Format using snprintf. 
2645   Char fill = internal::char_traits<Char>::cast(spec.fill()); 
2646   unsigned n = 0; 
2647   Char *start = 0; 
2648   for (;;) { 
2649     std::size_t buffer_size = buffer_.capacity() - offset; 
2650 #if FMT_MSC_VER 
2651     // MSVC's vsnprintf_s doesn't work with zero size, so reserve 
2652     // space for at least one extra character to make the size non-zero. 
2653     // Note that the buffer's capacity will increase by more than 1. 
2654     if (buffer_size == 0) { 
2655       buffer_.reserve(offset + 1); 
2656       buffer_size = buffer_.capacity() - offset; 
2657     } 
2658 #endif 
2659     start = &buffer_[offset];

2661         start, buffer_size, format, width_for_sprintf, spec.precision(), value); 
2662     if (result >= 0) { 
2663       n = internal::to_unsigned(result); 
2664       if (offset + n < buffer_.capacity()) 
2665         break;  // The buffer is large enough - continue with formatting. 
2666       buffer_.reserve(offset + n + 1); 
2667     } else { 
2668       // If result is negative we ask to increase the capacity by at least 1, 
2669       // but as std::vector, the buffer grows exponentially. 
2670       buffer_.reserve(buffer_.capacity() + 1); 
2671     } 
2672   } 
2673   if (sign) { 
2674     if ((spec.align() != ALIGN_RIGHT && spec.align() != ALIGN_DEFAULT) || 
2675         *start != ' ') { 
2676       *(start - 1) = sign; 
2677       sign = 0; 
2678     } else { 
2679       *(start - 1) = fill; 
2680     } 
2681     ++n; 
2682   } 
2683   if (spec.align() == ALIGN_CENTER && spec.width() > n) { 
2684     width = spec.width(); 
2685     pointer_type p = grow_buffer(width); 
2686     std::memmove(get(p) + (width - n) / 2, get(p), n * sizeof(Char)); 
2687     fill_padding(p, spec.width(), n, fill); 
2688     return; 
2689   } 
2690   if (spec.fill() != ' ' || sign) { 
2691     while (*start == ' ') 
2692       *start++ = fill; 
2693     if (sign) 
2694       *(start - 1) = sign; 
2695   } 
2696   grow_buffer(n); 
2697 }

2756 class FormatInt { 
2757  private: 
2758   // Buffer should be large enough to hold all digits (digits10 + 1), 
2759   // a sign and a null character. 
2760   enum {BUFFER_SIZE = std::numeric_limits<unsigned long long>::digits10 + 3}; 
2761   mutable char buffer_[BUFFER_SIZE]; 
2762   char *str_; 
2763  
2764   // Formats value in reverse and returns the number of digits. 
2765   char *format_decimal(unsigned long long value) { 
2766     char *buffer_end = buffer_ + BUFFER_SIZE - 1; 
2767     while (value >= 100) { 
2768       // Integer division is slow so do it for a group of two digits instead 
2769       // of for every digit. The idea comes from the talk by Alexandrescu 
2770       // "Three Optimization Tips for C++". See speed-test for a comparison. 
2771       unsigned index = static_cast<unsigned>((value % 100) * 2); 
2772       value /= 100; 
2773       *--buffer_end = internal::data::DIGITS[index + 1]; 
2774       *--buffer_end = internal::data::DIGITS[index]; 
2775     } 
2776     if (value < 10) { 
2777       *--buffer_end = static_cast<char>('0' + value); 
2778       return buffer_end; 
2779     } 
2780     unsigned index = static_cast<unsigned>(value * 2); 
2781     *--buffer_end = internal::data::DIGITS[index + 1]; 
2782     *--buffer_end = internal::data::DIGITS[index]; 
2783     return buffer_end; 
2784   } 
2785  
2786   void FormatSigned(long long value) { 
2787     unsigned long long abs_value = static_cast<unsigned long long>(value); 
2788     bool negative = value < 0; 
2789     if (negative) 
2790       abs_value = 0 - abs_value; 
2791     str_ = format_decimal(abs_value); 
2792     if (negative) 
2793       *--str_ = '-'; 
2794   } 
2795  
2796  public: 
2797   explicit FormatInt(int value) { FormatSigned(value); } 
2798   explicit FormatInt(long value) { FormatSigned(value); } 
2799   explicit FormatInt(long long value) { FormatSigned(value); } 
2800   explicit FormatInt(unsigned value) : str_(format_decimal(value)) {} 
2801   explicit FormatInt(unsigned long value) : str_(format_decimal(value)) {} 
2802   explicit FormatInt(unsigned long long value) : str_(format_decimal(value)) {}

2805   std::size_t size() const { 
2806     return internal::to_unsigned(buffer_ - str_ + BUFFER_SIZE - 1); 
2807   } 
2808  
2809   /** 
2810     Returns a pointer to the output buffer content. No terminating null 
2811     character is appended. 
2812    */ 
2813   const char *data() const { return str_; } 
2814  
2815   /** 
2816     Returns a pointer to the output buffer content with terminating null 
2817     character appended. 
2818    */ 
2819   const char *c_str() const { 
2820     buffer_[BUFFER_SIZE - 1] = '\0'; 
2821     return str_; 
2822   }

2824   /** 
2825     \rst 
2826     Returns the content of the output buffer as an ``std::string``. 
2827     \endrst 
2828    */

2832 // Formats a decimal integer value writing into buffer and returns 
2833 // a pointer to the end of the formatted string. This function doesn't 
2834 // write a terminating null character. 
2835 template <typename T> 
2836 inline void format_decimal(char *&buffer, T value) { 
2837   typedef typename internal::int_traits<T>::main_type main_type; 
2838   main_type abs_value = static_cast<main_type>(value); 
2839   if (internal::is_negative(value)) { 
2840     *buffer++ = '-'; 
2841     abs_value = 0 - abs_value; 
2842   } 
2843   if (abs_value < 100) { 
2844     if (abs_value < 10) { 
2845       *buffer++ = static_cast<char>('0' + abs_value); 
2846       return; 
2847     } 
2848     unsigned index = static_cast<unsigned>(abs_value * 2); 
2849     *buffer++ = internal::data::DIGITS[index]; 
2850     *buffer++ = internal::data::DIGITS[index + 1]; 
2851     return; 
2852   } 
2853   unsigned num_digits = internal::count_digits(abs_value); 
2854   internal::format_decimal(buffer, abs_value, num_digits); 
2855   buffer += num_digits; 
2856 }

 209 // Standard permits specialization of std::numeric_limits. This specialization 
 210 // is used to resolve ambiguity between isinf and std::isinf in glibc: 
 211 // https://gcc.gnu.org/bugzilla/show_bug.cgi?id=48891 
 212 // and the same for isnan and signbit. 
 213 template <> 
 214 class numeric_limits<fmt::internal::dummy_int> : 
 215     public std::numeric_limits<int> { 
 216  public: 
 217   // Portable version of isinf. 
 218   template <typename T> 
 219   static bool isinfinity(T x) { 
 220     using namespace fmt::internal; 
 221     // The resolution "priority" is: 
 222     // isinf macro > std::isinf > ::isinf > fmt::internal::isinf 
 223     if (const_check(sizeof(isinf(x)) == sizeof(bool) || 
 224                     sizeof(isinf(x)) == sizeof(int))) { 
 225       return isinf(x) != 0; 
 226     } 
 227     return !_finite(static_cast<double>(x)); 
 228   } 
 229  
 230   // Portable version of isnan. 
 231   template <typename T> 
 232   static bool isnotanumber(T x) { 
 233     using namespace fmt::internal; 
 234     if (const_check(sizeof(isnan(x)) == sizeof(bool) || 
 235                     sizeof(isnan(x)) == sizeof(int))) { 
 236       return isnan(x) != 0; 
 237     } 
 238     return _isnan(static_cast<double>(x)) != 0; 
 239   } 
 240  
 241   // Portable version of signbit. 
 242   static bool isnegative(double x) { 
 243     using namespace fmt::internal; 
 244     if (const_check(sizeof(signbit(x)) == sizeof(int))) 
 245       return signbit(x) != 0; 
 246     if (x < 0) return true; 
 247     if (!isnotanumber(x)) return false; 
 248     int dec = 0, sign = 0; 
 249     char buffer[2];  // The buffer size must be >= 2 or _ecvt_s will fail. 
 250     _ecvt_s(buffer, sizeof(buffer), x, 0, &dec, &sign); 
 251     return sign != 0; 
 252   } 
 253 };

 307 template <typename T> 
 308 class basic_buffer { 
 309  private: 
 310   FMT_DISALLOW_COPY_AND_ASSIGN(basic_buffer); 
 311  
 312   T *ptr_; 
 313   std::size_t size_; 
 314   std::size_t capacity_; 
 315  
 316  protected: 
 317   basic_buffer() FMT_NOEXCEPT : ptr_(0), size_(0), capacity_(0) {} 
 318  
 319   /** Sets the buffer data and capacity. */ 
 320   void set(T* data, std::size_t capacity) FMT_NOEXCEPT { 
 321     ptr_ = data; 
 322     capacity_ = capacity; 
 323   } 
 324  
 325   /** 
 326     \rst 
 327     Increases the buffer capacity to hold at least *capacity* elements. 
 328     \endrst 
 329    */ 
 330   virtual void grow(std::size_t capacity) = 0; 
 331  
 332  public: 
 333   virtual ~basic_buffer() {} 
 334  
 335   /** Returns the size of this buffer. */ 
 336   std::size_t size() const FMT_NOEXCEPT { return size_; } 
 337  
 338   /** Returns the capacity of this buffer. */ 
 339   std::size_t capacity() const FMT_NOEXCEPT { return capacity_; } 
 340  
 341   /** Returns a pointer to the buffer data. */ 
 342   T *data() FMT_NOEXCEPT { return ptr_; } 
 343  
 344   /** Returns a pointer to the buffer data. */ 
 345   const T *data() const FMT_NOEXCEPT { return ptr_; } 
 346  
 347   /** 
 348     Resizes the buffer. If T is a POD type new elements may not be initialized. 
 349    */ 
 350   void resize(std::size_t new_size) { 
 351     reserve(new_size); 
 352     size_ = new_size; 
 353   } 
 354  
 355   /** 
 356     \rst 
 357     Reserves space to store at least *capacity* elements. 
 358     \endrst 
 359    */ 
 360   void reserve(std::size_t capacity) { 
 361     if (capacity > capacity_) 
 362       grow(capacity); 
 363   } 
 364  
 365   void push_back(const T &value) { 
 366     reserve(size_ + 1); 
 367     ptr_[size_++] = value; 
 368   }

 370   /** Appends data to the end of the buffer. */ 
 371   template <typename U> 
 372   void append(const U *begin, const U *end); 
 373  
 374   T &operator[](std::size_t index) { return ptr_[index]; } 
 375   const T &operator[](std::size_t index) const { return ptr_[index]; } 
 376  
 377   virtual fmt::locale locale() const; 
 378 };

 384   reserve(new_size);

 395 /** 
 396   \rst 
 397   A dynamically growing memory buffer for trivially copyable/constructible types 
 398   with the first SIZE elements stored in the object itself. 
 399  
 400   You can use one of the following typedefs for common character types: 
 401  
 402   +----------------+------------------------------+ 
 403   | Type           | Definition                   | 
 404   +================+==============================+ 
 405   | memory_buffer  | basic_memory_buffer<char>    | 
 406   +----------------+------------------------------+ 
 407   | wmemory_buffer | basic_memory_buffer<wchar_t> | 
 408   +----------------+------------------------------+ 
 409  
 410   **Example**:: 
 411  
 412      memory_buffer out; 
 413      format_to(out, "The answer is {}.", 42); 
 414  
 415   This will write the following output to the ``out`` object: 
 416  
 417   .. code-block:: none 
 418  
 419      The answer is 42. 
 420  
 421   The output can be converted to an ``std::string`` with ``to_string(out)``. 
 422   \endrst 
 423  */ 
 424 template <typename T, std::size_t SIZE = internal::INLINE_BUFFER_SIZE, 
 425           typename Allocator = std::allocator<T> > 
 426 class basic_memory_buffer : private Allocator, public basic_buffer<T> { 
 427  private: 
 428   T store_[SIZE]; 
 429  
 430   // Deallocate memory allocated by the buffer. 
 431   void deallocate() { 
 432     T* data = this->data(); 
 433     if (data != store_) Allocator::deallocate(data, this->capacity()); 
 434   } 
 435  
 436  protected: 
 437   void grow(std::size_t size); 
 438  
 439  public: 
 440   explicit basic_memory_buffer(const Allocator &alloc = Allocator()) 
 441       : Allocator(alloc) { 
 442     this->set(store_, SIZE); 
 443   } 
 444   ~basic_memory_buffer() { deallocate(); } 
 445  
 446  private: 
 447   // Move data from other to this buffer. 
 448   void move(basic_memory_buffer &other) { 
 449     Allocator &this_alloc = *this, &other_alloc = other; 
 450     this_alloc = std::move(other_alloc); 
 451     T* data = other.data(); 
 452     std::size_t size = other.size(), capacity = other.capacity(); 
 453     if (data == other.store_) { 
 454       this->set(store_, capacity); 
 455       std::uninitialized_copy(other.store_, other.store_ + size, 
 456                               internal::make_ptr(store_, capacity)); 
 457     } else { 
 458       this->set(data, capacity); 
 459       // Set pointer to the inline array so that delete is not called 
 460       // when deallocating. 
 461       other.set(other.store_, 0); 
 462     } 
 463     this->resize(size); 
 464   } 
 465  
 466  public: 
 467   /** 
 468     \rst 
 469     Constructs a :class:`fmt::basic_memory_buffer` object moving the content 
 470     of the other object to it. 
 471     \endrst 
 472    */ 
 473   basic_memory_buffer(basic_memory_buffer &&other) { 
 474     move(other); 
 475   } 
 476  
 477   /** 
 478     \rst 
 479     Moves the content of the other ``basic_memory_buffer`` object to this one. 
 480     \endrst 
 481    */ 
 482   basic_memory_buffer &operator=(basic_memory_buffer &&other) { 
 483     assert(this != &other); 
 484     deallocate(); 
 485     move(other); 
 486     return *this; 
 487   } 
 488  
 489   // Returns a copy of the allocator associated with this buffer. 
 490   Allocator get_allocator() const { return *this; } 
 491 };

 493 template <typename T, std::size_t SIZE, typename Allocator> 
 494 void basic_memory_buffer<T, SIZE, Allocator>::grow(std::size_t size) { 
 495   std::size_t old_capacity = this->capacity(); 
 496   std::size_t new_capacity = old_capacity + old_capacity / 2; 
 497   if (size > new_capacity) 
 498       new_capacity = size; 
 499   T *old_data = this->data(); 
 500   T *new_data = this->allocate(new_capacity); 
 501   // The following code doesn't throw, so the raw pointer above doesn't leak. 
 502   std::uninitialized_copy(old_data, old_data + this->size(), 
 503                           internal::make_ptr(new_data, new_capacity)); 
 504   this->set(new_data, new_capacity); 
 505   // deallocate must not throw according to the standard, but even if it does, 
 506   // the buffer already uses the new storage and will deallocate it in 
 507   // destructor. 
 508   if (old_data != store_) 
 509     Allocator::deallocate(old_data, old_capacity); 
 510 }

 614     : ptr_(r.begin()), end_(r.end()) {}

 617     assert(ptr <= end_);

 639     return *this;

 725 // Returns the number of decimal digits in n. Leading zeros are not counted 
 726 // except for n == 0 in which case count_digits returns 1. 
 727 inline unsigned count_digits(uint64_t n) { 
 728   // Based on http://graphics.stanford.edu/~seander/bithacks.html#IntegerLog10 
 729   // and the benchmark https://github.com/localvoid/cxx-benchmark-count-digits. 
 730   int t = (64 - FMT_BUILTIN_CLZLL(n | 1)) * 1233 >> 12; 
 731   return to_unsigned(t) - (n < data::POWERS_OF_10_64[t]) + 1; 
 732 } 
 733 #else 
 734 // Fallback version of count_digits used when __builtin_clz is not available. 
 735 inline unsigned count_digits(uint64_t n) { 
 736   unsigned count = 1; 
 737   for (;;) { 
 738     // Integer division is slow so do it for a group of four digits instead 
 739     // of for every digit. The idea comes from the talk by Alexandrescu 
 740     // "Three Optimization Tips for C++". See speed-test for a comparison. 
 741     if (n < 10) return count; 
 742     if (n < 100) return count + 1; 
 743     if (n < 1000) return count + 2; 
 744     if (n < 10000) return count + 3; 
 745     n /= 10000u; 
 746     count += 4; 
 747   } 
 748 } 
 749 #endif

 753 inline unsigned count_digits(uint32_t n) { 
 754   int t = (32 - FMT_BUILTIN_CLZ(n | 1)) * 1233 >> 12; 
 755   return to_unsigned(t) - (n < data::POWERS_OF_10_32[t]) + 1;

 769   basic_string_view<Char> sep_;

 790 // Formats a decimal unsigned integer value writing into buffer. 
 791 // thousands_sep is a functor that is called after writing each char to 
 792 // add a thousands separator if necessary. 
 793 template <typename UInt, typename Char, typename ThousandsSep> 
 794 inline void format_decimal(Char *buffer, UInt value, unsigned num_digits, 
 795                            ThousandsSep thousands_sep) { 
 796   buffer += num_digits; 
 797   while (value >= 100) { 
 798     // Integer division is slow so do it for a group of two digits instead 
 799     // of for every digit. The idea comes from the talk by Alexandrescu 
 800     // "Three Optimization Tips for C++". See speed-test for a comparison. 
 801     unsigned index = static_cast<unsigned>((value % 100) * 2); 
 802     value /= 100; 
 803     *--buffer = data::DIGITS[index + 1]; 
 804     thousands_sep(buffer); 
 805     *--buffer = data::DIGITS[index]; 
 806     thousands_sep(buffer); 
 807   } 
 808   if (value < 10) { 
 809     *--buffer = static_cast<char>('0' + value); 
 810     return; 
 811   } 
 812   unsigned index = static_cast<unsigned>(value * 2); 
 813   *--buffer = data::DIGITS[index + 1]; 
 814   thousands_sep(buffer); 
 815   *--buffer = data::DIGITS[index]; 
 816 }

 876 /** 
 877   \rst 
 878   Visits an argument dispatching to the appropriate visit method based on 
 879   the argument type. For example, if the argument type is ``double`` then 
 880   ``vis(value)`` will be called with the value of type ``double``. 
 881   \endrst 
 882  */ 
 883 template <typename Visitor, typename Context> 
 884 constexpr typename std::result_of<Visitor(int)>::type 
 885     visit(Visitor &&vis, basic_arg<Context> arg) { 
 886   using char_type = typename Context::char_type; 
 887   switch (arg.type_) { 
 888   case internal::NONE: 
 889     return vis(monostate()); 
 890   case internal::NAMED_ARG: 
 891     FMT_ASSERT(false, "invalid argument type"); 
 892     break; 
 893   case internal::INT: 
 894     return vis(arg.value_.int_value); 
 895   case internal::UINT: 
 896     return vis(arg.value_.uint_value); 
 897   case internal::LONG_LONG: 
 898     return vis(arg.value_.long_long_value); 
 899   case internal::ULONG_LONG: 
 900     return vis(arg.value_.ulong_long_value); 
 901   case internal::BOOL: 
 902     return vis(arg.value_.int_value != 0); 
 903   case internal::CHAR: 
 904     return vis(static_cast<char_type>(arg.value_.int_value)); 
 905   case internal::DOUBLE: 
 906     return vis(arg.value_.double_value); 
 907   case internal::LONG_DOUBLE: 
 908     return vis(arg.value_.long_double_value); 
 909   case internal::CSTRING: 
 910     return vis(arg.value_.string.value); 
 911   case internal::STRING: 
 912     return vis(basic_string_view<char_type>( 
 913                  arg.value_.string.value, arg.value_.string.size)); 
 914   case internal::POINTER: 
 915     return vis(arg.value_.pointer); 
 916   case internal::CUSTOM: 
 917     return vis(typename basic_arg<Context>::handle(arg.value_.custom)); 
 918   } 
 919   return typename std::result_of<Visitor(int)>::type(); 
 920 }

1336 // Parses the input as an unsigned integer. This function assumes that the 
1337 // first character is a digit and presence of a non-digit character at the end. 
1338 // it: an iterator pointing to the beginning of the input range. 
1339 template <typename Iterator, typename ErrorHandler> 
1340 constexpr unsigned parse_nonnegative_int(Iterator &it, ErrorHandler &&eh) { 
1341   assert('0' <= *it && *it <= '9'); 
1342   unsigned value = 0; 
1343   // Convert to unsigned to prevent a warning. 
1344   unsigned max_int = (std::numeric_limits<int>::max)(); 
1345   unsigned big = max_int / 10; 
1346   do { 
1347     // Check for overflow. 
1348     if (value > big) { 
1349       value = max_int + 1; 
1350       break; 
1351     } 
1352     value = value * 10 + (*it - '0'); 
1353     // Workaround for MSVC "setup_exception stack overflow" error: 
1354     auto next = it; 
1355     ++next; 
1356     it = next; 
1357   } while ('0' <= *it && *it <= '9'); 
1358   if (value > max_int) 
1359     eh.on_error("number is too big"); 
1360   return value; 
1361 }

1664 template <typename Iterator, typename IDHandler> 
1665 constexpr Iterator parse_arg_id(Iterator it, IDHandler &&handler) { 
1666   using char_type = typename std::iterator_traits<Iterator>::value_type; 
1667   char_type c = *it; 
1668   if (c == '}' || c == ':') { 
1669     handler(); 
1670     return it; 
1671   } 
1672   if (c >= '0' && c <= '9') { 
1673     unsigned index = parse_nonnegative_int(it, handler); 
1674     if (*it != '}' && *it != ':') { 
1675       handler.on_error("invalid format string"); 
1676       return it; 
1677     } 
1678     handler(index); 
1679     return it; 
1680   } 
1681   if (!is_name_start(c)) { 
1682     handler.on_error("invalid format string"); 
1683     return it; 
1684   } 
1685   auto start = it; 
1686   do { 
1687     c = *++it; 
1688   } while (is_name_start(c) || ('0' <= c && c <= '9')); 
1689   handler(basic_string_view<char_type>(pointer_from(start), it - start)); 
1690   return it; 
1691 }

1725 // Parses standard format specifiers and sends notifications about parsed 
1726 // components to handler. 
1727 // it: an iterator pointing to the beginning of a null-terminated range of 
1728 //     characters, possibly emulated via null_terminating_iterator, representing 
1729 //     format specifiers. 
1730 template <typename Iterator, typename SpecHandler> 
1731 constexpr Iterator parse_format_specs(Iterator it, SpecHandler &&handler) { 
1732   using char_type = typename std::iterator_traits<Iterator>::value_type; 
1733   // Parse fill and alignment. 
1734   if (char_type c = *it) { 
1735     alignment align = ALIGN_DEFAULT; 
1736     int i = 1; 
1737     do { 
1738       auto p = it + i; 
1739       switch (*p) { 
1740         case '<': 
1741           align = ALIGN_LEFT; 
1742           break; 
1743         case '>': 
1744           align = ALIGN_RIGHT; 
1745           break; 
1746         case '=': 
1747           align = ALIGN_NUMERIC; 
1748           break; 
1749         case '^': 
1750           align = ALIGN_CENTER; 
1751           break; 
1752       } 
1753       if (align != ALIGN_DEFAULT) { 
1754         handler.on_align(align); 
1755         if (p != it) { 
1756           if (c == '}') break; 
1757           if (c == '{') { 
1758             handler.on_error("invalid fill character '{'"); 
1759             return it; 
1760           } 
1761           it += 2; 
1762           handler.on_fill(c); 
1763         } else ++it; 
1764         break; 
1765       } 
1766     } while (--i >= 0); 
1767   } 
1768  
1769   // Parse sign. 
1770   switch (*it) { 
1771     case '+': 
1772       handler.on_plus(); 
1773       ++it; 
1774       break; 
1775     case '-': 
1776       handler.on_minus(); 
1777       ++it; 
1778       break; 
1779     case ' ': 
1780       handler.on_space(); 
1781       ++it; 
1782       break; 
1783   } 
1784  
1785   if (*it == '#') { 
1786     handler.on_hash(); 
1787     ++it; 
1788   } 
1789  
1790   // Parse zero flag. 
1791   if (*it == '0') { 
1792     handler.on_zero(); 
1793     ++it; 
1794   } 
1795  
1796   // Parse width. 
1797   if ('0' <= *it && *it <= '9') { 
1798     handler.on_width(parse_nonnegative_int(it, handler)); 
1799   } else if (*it == '{') { 
1800     it = parse_arg_id(it + 1, width_adapter<SpecHandler, char_type>(handler)); 
1801     if (*it++ != '}') { 
1802       handler.on_error("invalid format string"); 
1803       return it; 
1804     } 
1805   } 
1806  
1807   // Parse precision. 
1808   if (*it == '.') { 
1809     ++it; 
1810     if ('0' <= *it && *it <= '9') { 
1811       handler.on_precision(parse_nonnegative_int(it, handler)); 
1812     } else if (*it == '{') { 
1813       it = parse_arg_id( 
1814             it + 1, precision_adapter<SpecHandler, char_type>(handler)); 
1815       if (*it++ != '}') { 
1816         handler.on_error("invalid format string"); 
1817         return it; 
1818       } 
1819     } else { 
1820       handler.on_error("missing precision specifier"); 
1821       return it; 
1822     } 
1823     handler.end_precision(); 
1824   } 
1825  
1826   // Parse type. 
1827   if (*it != '}' && *it) 
1828     handler.on_type(*it++); 
1829   return it; 
1830 }

1849 template <typename Iterator, typename Handler> 
1850 constexpr void parse_format_string(Iterator it, Handler &&handler) { 
1851   using char_type = typename std::iterator_traits<Iterator>::value_type; 
1852   auto start = it; 
1853   while (*it) { 
1854     char_type ch = *it++; 
1855     if (ch != '{' && ch != '}') continue; 
1856     if (*it == ch) { 
1857       handler.on_text(start, it); 
1858       start = ++it; 
1859       continue; 
1860     } 
1861     if (ch == '}') { 
1862       handler.on_error("unmatched '}' in format string"); 
1863       return; 
1864     } 
1865     handler.on_text(start, it - 1); 
1866  
1867     it = parse_arg_id(it, id_adapter<Handler, char_type>(handler)); 
1868     if (*it == '}') { 
1869       handler.on_replacement_field(it); 
1870     } else if (*it == ':') { 
1871       ++it; 
1872       it = handler.on_format_specs(it); 
1873       if (*it != '}') { 
1874         handler.on_error("unknown format specifier"); 
1875         return; 
1876       } 
1877     } else { 
1878       handler.on_error("missing '}' in format string"); 
1879       return; 
1880     } 
1881  
1882     start = ++it; 
1883   } 
1884   handler.on_text(start, it); 
1885 }

2359 template <typename Char> 
2360 template <typename Spec> 
2361 typename basic_writer<Char>::pointer_type 
2362   basic_writer<Char>::prepare_int_buffer( 
2363     unsigned num_digits, const Spec &spec, 
2364     const char *prefix, unsigned prefix_size) { 
2365   unsigned width = spec.width(); 
2366   alignment align = spec.align(); 
2367   Char fill = internal::char_traits<Char>::cast(spec.fill()); 
2368   if (spec.precision() > static_cast<int>(num_digits)) { 
2369     // Octal prefix '0' is counted as a digit, so ignore it if precision 
2370     // is specified. 
2371     if (prefix_size > 0 && prefix[prefix_size - 1] == '0') 
2372       --prefix_size; 
2373     unsigned number_size = 
2374         prefix_size + internal::to_unsigned(spec.precision()); 
2375     align_spec subspec(number_size, '0', ALIGN_NUMERIC); 
2376     if (number_size >= width) 
2377       return prepare_int_buffer(num_digits, subspec, prefix, prefix_size); 
2378     buffer_.reserve(width); 
2379     unsigned fill_size = width - number_size; 
2380     if (align != ALIGN_LEFT) { 
2381       pointer_type p = grow_buffer(fill_size); 
2382       std::uninitialized_fill(p, p + fill_size, fill); 
2383     } 
2384     pointer_type result = prepare_int_buffer( 
2385         num_digits, subspec, prefix, prefix_size); 
2386     if (align == ALIGN_LEFT) { 
2387       pointer_type p = grow_buffer(fill_size); 
2388       std::uninitialized_fill(p, p + fill_size, fill); 
2389     } 
2390     return result; 
2391   } 
2392   unsigned size = prefix_size + num_digits; 
2393   if (width <= size) { 
2394     pointer_type p = grow_buffer(size); 
2395     std::uninitialized_copy(prefix, prefix + prefix_size, p); 
2396     return p + size - 1; 
2397   } 
2398   pointer_type p = grow_buffer(width); 
2399   pointer_type end = p + width; 
2400   if (align == ALIGN_LEFT) { 
2401     std::uninitialized_copy(prefix, prefix + prefix_size, p); 
2402     p += size; 
2403     std::uninitialized_fill(p, end, fill); 
2404   } else if (align == ALIGN_CENTER) { 
2405     p = fill_padding(p, width, size, fill); 
2406     std::uninitialized_copy(prefix, prefix + prefix_size, p); 
2407     p += size; 
2408   } else { 
2409     if (align == ALIGN_NUMERIC) { 
2410       if (prefix_size != 0) { 
2411         p = std::uninitialized_copy(prefix, prefix + prefix_size, p); 
2412         size -= prefix_size; 
2413       } 
2414     } else { 
2415       std::uninitialized_copy(prefix, prefix + prefix_size, end - size); 
2416     } 
2417     std::uninitialized_fill(p, end - size, fill); 
2418     p = end; 
2419   } 
2420   return p - 1; 
2421 }

2426   using unsigned_type = typename internal::int_traits<T>::main_type; 
2427   struct spec_handler { 
2428     basic_writer<Char> &writer; 
2429     const Spec& spec; 
2430     unsigned prefix_size = 0; 
2431     unsigned_type abs_value; 
2432     char prefix[4] = ""; 
2433  
2434     spec_handler(basic_writer<Char> &w, T val, const Spec& s) 
2435       : writer(w), spec(s), abs_value(static_cast<unsigned_type>(val)) { 
2436       if (internal::is_negative(val)) { 
2437         prefix[0] = '-'; 
2438         ++prefix_size; 
2439         abs_value = 0 - abs_value; 
2440       } else if (spec.flag(SIGN_FLAG)) { 
2441         prefix[0] = spec.flag(PLUS_FLAG) ? '+' : ' '; 
2442         ++prefix_size; 
2443       } 
2444     } 
2445  
2446     void on_dec() { 
2447       unsigned num_digits = internal::count_digits(abs_value); 
2448       pointer_type p = 
2449           writer.prepare_int_buffer(num_digits, spec, prefix, prefix_size) + 1; 
2450       internal::format_decimal(get(p), abs_value, 0); 
2451     } 
2452  
2453     void on_hex() { 
2454       unsigned_type n = abs_value; 
2455       if (spec.flag(HASH_FLAG)) { 
2456         prefix[prefix_size++] = '0'; 
2457         prefix[prefix_size++] = spec.type(); 
2458       } 
2459       unsigned num_digits = 0; 
2460       do { 
2461         ++num_digits; 
2462       } while ((n >>= 4) != 0); 
2463       Char *p = 
2464           get(writer.prepare_int_buffer(num_digits, spec, prefix, prefix_size)); 
2465       n = abs_value; 
2466       const char *digits = spec.type() == 'x' ? 
2467           "0123456789abcdef" : "0123456789ABCDEF"; 
2468       do { 
2469         *p-- = digits[n & 0xf];

2475       if (spec.flag(HASH_FLAG)) {

2480       do { 
2481         ++num_digits; 
2482       } while ((n >>= 1) != 0);

2484           get(writer.prepare_int_buffer(num_digits, spec, prefix, prefix_size));

2529   // Check type. 
2530   struct spec_handler { 
2531     char type; 
2532     bool upper = false; 
2533  
2534     explicit spec_handler(char t) : type(t) {} 
2535  
2536     void on_general() {

2538         upper = true;

2541     } 
2542  
2543     void on_exp() { 
2544       if (type == 'E') 
2545         upper = true; 
2546     } 
2547  
2548     void on_fixed() { 
2549       if (type == 'F') { 
2550         upper = true; 
2551 #if FMT_MSC_VER 
2552         // MSVC's printf doesn't support 'F'. 
2553         type = 'f'; 
2554 #endif 
2555       } 
2556     } 
2557  
2558     void on_hex() { 
2559       if (type == 'A') 
2560         upper = true; 
2561     } 
2562  
2563     void on_error() { 
2564       FMT_THROW(format_error("invalid type specifier")); 
2565     } 
2566   }; 
2567   spec_handler handler(spec.type()); 
2568   internal::handle_float_type_spec(spec.type(), handler); 
2569  
2570   char sign = 0; 
2571   // Use isnegative instead of value < 0 because the latter is always 
2572   // false for NaN. 
2573   if (internal::fputil::isnegative(static_cast<double>(value))) { 
2574     sign = '-'; 
2575     value = -value; 
2576   } else if (spec.flag(SIGN_FLAG)) { 
2577     sign = spec.flag(PLUS_FLAG) ? '+' : ' '; 
2578   } 
2579  
2580   if (internal::fputil::isnotanumber(value)) { 
2581     // Format NaN ourselves because sprintf's output is not consistent 
2582     // across platforms. 
2583     std::size_t nan_size = 4; 
2584     const char *nan = handler.upper ? " NAN" : " nan"; 
2585     if (!sign) { 
2586       --nan_size; 
2587       ++nan; 
2588     } 
2589     pointer_type out = write_str(nan, nan_size, spec); 
2590     if (sign) 
2591       *out = sign; 
2592     return; 
2593   } 
2594  
2595   if (internal::fputil::isinfinity(value)) { 
2596     // Format infinity ourselves because sprintf's output is not consistent 
2597     // across platforms. 
2598     std::size_t inf_size = 4; 
2599     const char *inf = handler.upper ? " INF" : " inf"; 
2600     if (!sign) { 
2601       --inf_size; 
2602       ++inf; 
2603     } 
2604     pointer_type out = write_str(inf, inf_size, spec); 
2605     if (sign) 
2606       *out = sign; 
2607     return; 
2608   } 
2609  
2610   std::size_t offset = buffer_.size(); 
2611   unsigned width = spec.width(); 
2612   if (sign) { 
2613     buffer_.reserve(buffer_.size() + (width > 1u ? width : 1u)); 
2614     if (width > 0) 
2615       --width; 
2616     ++offset; 
2617   } 
2618  
2619   // Build format string. 
2620   enum { MAX_FORMAT_SIZE = 10}; // longest format: %#-*.*Lg 
2621   Char format[MAX_FORMAT_SIZE]; 
2622   Char *format_ptr = format; 
2623   *format_ptr++ = '%'; 
2624   unsigned width_for_sprintf = width; 
2625   if (spec.flag(HASH_FLAG)) 
2626     *format_ptr++ = '#'; 
2627   if (spec.align() == ALIGN_CENTER) { 
2628     width_for_sprintf = 0; 
2629   } else { 
2630     if (spec.align() == ALIGN_LEFT) 
2631       *format_ptr++ = '-'; 
2632     if (width != 0) 
2633       *format_ptr++ = '*'; 
2634   } 
2635   if (spec.precision() >= 0) { 
2636     *format_ptr++ = '.'; 
2637     *format_ptr++ = '*'; 
2638   } 
2639  
2640   append_float_length(format_ptr, value); 
2641   *format_ptr++ = handler.type; 
2642   *format_ptr = '\0'; 
2643  
2644   // Format using snprintf. 
2645   Char fill = internal::char_traits<Char>::cast(spec.fill()); 
2646   unsigned n = 0; 
2647   Char *start = 0; 
2648   for (;;) { 
2649     std::size_t buffer_size = buffer_.capacity() - offset; 
2650 #if FMT_MSC_VER 
2651     // MSVC's vsnprintf_s doesn't work with zero size, so reserve 
2652     // space for at least one extra character to make the size non-zero. 
2653     // Note that the buffer's capacity will increase by more than 1. 
2654     if (buffer_size == 0) { 
2655       buffer_.reserve(offset + 1); 
2656       buffer_size = buffer_.capacity() - offset; 
2657     } 
2658 #endif 
2659     start = &buffer_[offset]; 
2660     int result = internal::char_traits<Char>::format_float( 
2661         start, buffer_size, format, width_for_sprintf, spec.precision(), value); 
2662     if (result >= 0) { 
2663       n = internal::to_unsigned(result); 
2664       if (offset + n < buffer_.capacity()) 
2665         break;  // The buffer is large enough - continue with formatting. 
2666       buffer_.reserve(offset + n + 1); 
2667     } else { 
2668       // If result is negative we ask to increase the capacity by at least 1, 
2669       // but as std::vector, the buffer grows exponentially. 
2670       buffer_.reserve(buffer_.capacity() + 1); 
2671     } 
2672   } 
2673   if (sign) { 
2674     if ((spec.align() != ALIGN_RIGHT && spec.align() != ALIGN_DEFAULT) || 
2675         *start != ' ') { 
2676       *(start - 1) = sign; 
2677       sign = 0; 
2678     } else { 
2679       *(start - 1) = fill; 
2680     } 
2681     ++n; 
2682   } 
2683   if (spec.align() == ALIGN_CENTER && spec.width() > n) { 
2684     width = spec.width(); 
2685     pointer_type p = grow_buffer(width); 
2686     std::memmove(get(p) + (width - n) / 2, get(p), n * sizeof(Char)); 
2687     fill_padding(p, spec.width(), n, fill); 
2688     return; 
2689   } 
2690   if (spec.fill() != ' ' || sign) { 
2691     while (*start == ' ') 
2692       *start++ = fill; 
2693     if (sign) 
2694       *(start - 1) = sign; 
2695   } 
2696   grow_buffer(n); 
2697 }

2753 /** 
2754   Fast integer formatter. 
2755  */ 
2756 class FormatInt { 
2757  private: 
2758   // Buffer should be large enough to hold all digits (digits10 + 1), 
2759   // a sign and a null character. 
2760   enum {BUFFER_SIZE = std::numeric_limits<unsigned long long>::digits10 + 3}; 
2761   mutable char buffer_[BUFFER_SIZE]; 
2762   char *str_; 
2763  
2764   // Formats value in reverse and returns the number of digits. 
2765   char *format_decimal(unsigned long long value) { 
2766     char *buffer_end = buffer_ + BUFFER_SIZE - 1; 
2767     while (value >= 100) { 
2768       // Integer division is slow so do it for a group of two digits instead 
2769       // of for every digit. The idea comes from the talk by Alexandrescu 
2770       // "Three Optimization Tips for C++". See speed-test for a comparison. 
2771       unsigned index = static_cast<unsigned>((value % 100) * 2); 
2772       value /= 100; 
2773       *--buffer_end = internal::data::DIGITS[index + 1]; 
2774       *--buffer_end = internal::data::DIGITS[index]; 
2775     } 
2776     if (value < 10) { 
2777       *--buffer_end = static_cast<char>('0' + value); 
2778       return buffer_end; 
2779     } 
2780     unsigned index = static_cast<unsigned>(value * 2); 
2781     *--buffer_end = internal::data::DIGITS[index + 1]; 
2782     *--buffer_end = internal::data::DIGITS[index]; 
2783     return buffer_end; 
2784   } 
2785  
2786   void FormatSigned(long long value) { 
2787     unsigned long long abs_value = static_cast<unsigned long long>(value); 
2788     bool negative = value < 0; 
2789     if (negative) 
2790       abs_value = 0 - abs_value; 
2791     str_ = format_decimal(abs_value); 
2792     if (negative) 
2793       *--str_ = '-'; 
2794   } 
2795  
2796  public: 
2797   explicit FormatInt(int value) { FormatSigned(value); } 
2798   explicit FormatInt(long value) { FormatSigned(value); } 
2799   explicit FormatInt(long long value) { FormatSigned(value); } 
2800   explicit FormatInt(unsigned value) : str_(format_decimal(value)) {} 
2801   explicit FormatInt(unsigned long value) : str_(format_decimal(value)) {} 
2802   explicit FormatInt(unsigned long long value) : str_(format_decimal(value)) {} 
2803  
2804   /** Returns the number of characters written to the output buffer. */ 
2805   std::size_t size() const { 
2806     return internal::to_unsigned(buffer_ - str_ + BUFFER_SIZE - 1); 
2807   } 
2808  
2809   /** 
2810     Returns a pointer to the output buffer content. No terminating null 
2811     character is appended. 
2812    */ 
2813   const char *data() const { return str_; } 
2814  
2815   /** 
2816     Returns a pointer to the output buffer content with terminating null 
2817     character appended. 
2818    */ 
2819   const char *c_str() const { 
2820     buffer_[BUFFER_SIZE - 1] = '\0'; 
2821     return str_; 
2822   } 
2823  
2824   /** 
2825     \rst 
2826     Returns the content of the output buffer as an ``std::string``. 
2827     \endrst 
2828    */ 
2829   std::string str() const { return std::string(str_, size()); } 
2830 };

2832 // Formats a decimal integer value writing into buffer and returns 
2833 // a pointer to the end of the formatted string. This function doesn't 
2834 // write a terminating null character. 
2835 template <typename T> 
2836 inline void format_decimal(char *&buffer, T value) { 
2837   typedef typename internal::int_traits<T>::main_type main_type; 
2838   main_type abs_value = static_cast<main_type>(value); 
2839   if (internal::is_negative(value)) { 
2840     *buffer++ = '-'; 
2841     abs_value = 0 - abs_value; 
2842   } 
2843   if (abs_value < 100) { 
2844     if (abs_value < 10) { 
2845       *buffer++ = static_cast<char>('0' + abs_value); 
2846       return; 
2847     } 
2848     unsigned index = static_cast<unsigned>(abs_value * 2); 
2849     *buffer++ = internal::data::DIGITS[index]; 
2850     *buffer++ = internal::data::DIGITS[index + 1]; 
2851     return; 
2852   } 
2853   unsigned num_digits = internal::count_digits(abs_value); 
2854   internal::format_decimal(buffer, abs_value, num_digits); 
2855   buffer += num_digits; 
2856 }

2980       checker.on_minus(); 
2981     } else if (specs_.flag(HASH_FLAG)) { 
2982       checker.on_hash(); 
2983     } 
2984     if (specs_.precision_ != -1) 
2985       checker.end_precision();
