 18 package org.apache.commons.csv; 
 19  
 20 import java.io.BufferedReader; 
 21 import java.io.IOException; 
 22 import java.io.Reader; 
 23  
 24 /** 
 25  * ExtendedBufferedReader 
 26  * 
 27  * A special reader decorator which supports more 
 28  * sophisticated access to the underlying reader object. 
 29  * 
 30  * In particular the reader supports a look-ahead option, 
 31  * which allows you to see the next char returned by 
 32  * next(). 
 33  */ 
 34 class ExtendedBufferedReader extends BufferedReader { 
 35  
 36     /** The end of stream symbol */ 
 37     static final int END_OF_STREAM = -1; 
 38  
 39     /** Undefined state for the lookahead char */ 
 40     static final int UNDEFINED = -2; 
 41  
 42     /** The last char returned */ 
 43     private int lastChar = UNDEFINED; 
 44  
 45     /** The line counter */ 
 46     private int lineCounter = 0; 
 47  
 48     /** 
 49      * Created extended buffered reader using default buffer-size 
 50      */ 
 51     ExtendedBufferedReader(Reader r) { 
 52         super(r); 
 53     } 
 54  
 55     @Override 
 56     public int read() throws IOException { 
 57         int current = super.read(); 
 58         if (current == '\n') { 
 59             lineCounter++; 
 60         } 
 61         lastChar = current; 
 62         return lastChar; 
 63     } 
 64  
 65     /** 
 66      * Returns the last character that was read as an integer (0 to 65535). This 
 67      * will be the last character returned by any of the read methods. This will 
 68      * not include a character read using the {@link #peek()} method. If no 
 69      * character has been read then this will return {@link #UNDEFINED}. If the 
 70      * end of the stream was reached on the last read then this will return 
 71      * {@link #END_OF_STREAM}. 
 72      *  
 73      * @return the last character that was read 
 74      */ 
 75     int readAgain() { 
 76         return lastChar; 
 77     } 
 78  
 79     @Override 
 80     public int read(char[] buf, int offset, int length) throws IOException { 
 81         if (length == 0) { 
 82             return 0; 
 83         } 
 84          
 85         int len = super.read(buf, offset, length); 
 86          
 87         if (len > 0) { 
 88  
 89             for (int i = offset; i < offset + len; i++) { 
 90                 char ch = buf[i]; 
 91                 if (ch == '\n') { 
 92                     if ('\r' != (i > 0 ? buf[i-1]: lastChar)) { 
 93                         lineCounter++;                         
 94                     } 
 95                 } else if (ch == '\r') { 
 96                     lineCounter++; 
 97                 } 
 98             } 
 99  
100             lastChar = buf[offset + len - 1]; 
101  
102         } else if (len == -1) { 
103             lastChar = END_OF_STREAM; 
104         } 
105          
106         return len; 
107     } 
108  
109     /** 
110      * Calls {@link BufferedReader#readLine()} which drops the line terminator(s). 
111      * This method should only be called when processing a comment, otherwise 
112      * information can be lost. 
113      * <p> 
114      * Increments  {@link #lineCounter} 
115      * <p> 
116      * Sets {@link #lastChar} to {@link #END_OF_STREAM} at EOF,  
117      * otherwise to last character on the line (won't be CR or LF)  
118      *  
119      * @return the line that was read, or null if reached EOF. 
120      */ 
121     @Override 
122     public String readLine() throws IOException { 
123         String line = super.readLine(); 
124  
125         if (line != null) { 
126             if (line.length() > 0) { 
127                 lastChar = line.charAt(line.length() - 1); 
128             } 
129             lineCounter++; 
130         } else { 
131             lastChar = END_OF_STREAM; 
132         } 
133  
134         return line; 
135     } 
136  
137     /** 
138      * Returns the next character in the current reader without consuming it. So 
139      * the next call to {@link #read()} will still return this value. 
140      *  
141      * @return the next character 
142      *  
143      * @throws IOException if there is an error in reading 
144      */ 
145     int lookAhead() throws IOException { 
146         super.mark(1); 
147         int c = super.read(); 
148         super.reset(); 
149  
150         return c; 
151     } 
152  
153     /** 
154      * Returns the nof line read 
155      * 
156      * @return the current-line-number (or -1) 
157      */ 
158     int getLineNumber() { 
159         return lineCounter; 
160     } 
161 }
