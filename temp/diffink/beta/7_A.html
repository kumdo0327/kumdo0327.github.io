 89     /** 
 90      * Adds a block to the arrangement manager at the specified edge. 
 91      * 
 92      * @param block  the block (<code>null</code> permitted). 
 93      * @param key  the edge (an instance of {@link RectangleEdge}) or 
 94      *             <code>null</code> for the center block. 
 95      */ 
 96     public void add(Block block, Object key) { 
 97  
 98         if (key == null) { 
 99             this.centerBlock = block; 
100         } 
101         else { 
102             RectangleEdge edge = (RectangleEdge) key; 
103             if (edge == RectangleEdge.TOP) { 
104                 this.topBlock = block; 
105             } 
106             else if (edge == RectangleEdge.BOTTOM) { 
107                 this.bottomBlock = block; 
108             } 
109             else if (edge == RectangleEdge.LEFT) { 
110                 this.leftBlock = block; 
111             } 
112             else if (edge == RectangleEdge.RIGHT) { 
113                 this.rightBlock = block; 
114             } 
115         } 
116     }

118     /** 
119      * Arranges the items in the specified container, subject to the given 
120      * constraint. 
121      * 
122      * @param container  the container. 
123      * @param g2  the graphics device. 
124      * @param constraint  the constraint. 
125      * 
126      * @return The block size. 
127      */ 
128     public Size2D arrange(BlockContainer container, 
129                           Graphics2D g2, 
130                           RectangleConstraint constraint) { 
131         RectangleConstraint contentConstraint 
132                 = container.toContentConstraint(constraint); 
133         Size2D contentSize = null; 
134         LengthConstraintType w = contentConstraint.getWidthConstraintType(); 
135         LengthConstraintType h = contentConstraint.getHeightConstraintType(); 
136         if (w == LengthConstraintType.NONE) { 
137             if (h == LengthConstraintType.NONE) { 
138                 contentSize = arrangeNN(container, g2); 
139             } 
140             else if (h == LengthConstraintType.FIXED) { 
141                 throw new RuntimeException("Not implemented."); 
142             } 
143             else if (h == LengthConstraintType.RANGE) { 
144                 throw new RuntimeException("Not implemented."); 
145             } 
146         } 
147         else if (w == LengthConstraintType.FIXED) { 
148             if (h == LengthConstraintType.NONE) { 
149                 contentSize = arrangeFN(container, g2, constraint.getWidth()); 
150             } 
151             else if (h == LengthConstraintType.FIXED) { 
152                 contentSize = arrangeFF(container, g2, constraint); 
153             } 
154             else if (h == LengthConstraintType.RANGE) { 
155                 contentSize = arrangeFR(container, g2, constraint); 
156             } 
157         } 
158         else if (w == LengthConstraintType.RANGE) { 
159             if (h == LengthConstraintType.NONE) { 
160                 throw new RuntimeException("Not implemented."); 
161             } 
162             else if (h == LengthConstraintType.FIXED) { 
163                 throw new RuntimeException("Not implemented."); 
164             } 
165             else if (h == LengthConstraintType.RANGE) { 
166                 contentSize = arrangeRR(container, constraint.getWidthRange(), 
167                         constraint.getHeightRange(), g2); 
168             } 
169         } 
170         return new Size2D(container.calculateTotalWidth(contentSize.getWidth()), 
171                 container.calculateTotalHeight(contentSize.getHeight())); 
172     }

174     /** 
175      * Performs an arrangement without constraints. 
176      * 
177      * @param container  the container. 
178      * @param g2  the graphics device. 
179      * 
180      * @return The container size after the arrangement. 
181      */ 
182     protected Size2D arrangeNN(BlockContainer container, Graphics2D g2) { 
183         double[] w = new double[5]; 
184         double[] h = new double[5]; 
185         if (this.topBlock != null) { 
186             Size2D size = this.topBlock.arrange(g2, RectangleConstraint.NONE); 
187             w[0] = size.width; 
188             h[0] = size.height; 
189         } 
190         if (this.bottomBlock != null) { 
191             Size2D size = this.bottomBlock.arrange(g2, 
192                     RectangleConstraint.NONE); 
193             w[1] = size.width; 
194             h[1] = size.height; 
195         } 
196         if (this.leftBlock != null) { 
197             Size2D size = this.leftBlock.arrange(g2, RectangleConstraint.NONE); 
198             w[2] = size.width; 
199             h[2] = size.height; 
200        } 
201         if (this.rightBlock != null) { 
202             Size2D size = this.rightBlock.arrange(g2, RectangleConstraint.NONE); 
203             w[3] = size.width; 
204             h[3] = size.height; 
205         } 
206  
207         h[2] = Math.max(h[2], h[3]); 
208         h[3] = h[2]; 
209  
210         if (this.centerBlock != null) { 
211             Size2D size = this.centerBlock.arrange(g2, 
212                     RectangleConstraint.NONE); 
213             w[4] = size.width; 
214             h[4] = size.height; 
215         } 
216         double width = Math.max(w[0], Math.max(w[1], w[2] + w[4] + w[3])); 
217         double centerHeight = Math.max(h[2], Math.max(h[3], h[4])); 
218         double height = h[0] + h[1] + centerHeight; 
219         if (this.topBlock != null) { 
220             this.topBlock.setBounds(new Rectangle2D.Double(0.0, 0.0, width, 
221                     h[0])); 
222         } 
223         if (this.bottomBlock != null) { 
224             this.bottomBlock.setBounds(new Rectangle2D.Double(0.0, 
225                     height - h[1], width, h[1])); 
226         } 
227         if (this.leftBlock != null) { 
228             this.leftBlock.setBounds(new Rectangle2D.Double(0.0, h[0], w[2], 
229                     centerHeight)); 
230         } 
231         if (this.rightBlock != null) { 
232             this.rightBlock.setBounds(new Rectangle2D.Double(width - w[3], 
233                     h[0], w[3], centerHeight)); 
234         } 
235  
236         if (this.centerBlock != null) { 
237             this.centerBlock.setBounds(new Rectangle2D.Double(w[2], h[0], 
238                     width - w[2] - w[3], centerHeight)); 
239         } 
240         return new Size2D(width, height); 
241     }

243     /** 
244      * Performs an arrangement with a fixed width and a range for the height. 
245      * 
246      * @param container  the container. 
247      * @param g2  the graphics device. 
248      * @param constraint  the constraint. 
249      * 
250      * @return The container size after the arrangement. 
251      */ 
252     protected Size2D arrangeFR(BlockContainer container, Graphics2D g2, 
253                                RectangleConstraint constraint) { 
254         Size2D size1 = arrangeFN(container, g2, constraint.getWidth()); 
255         if (constraint.getHeightRange().contains(size1.getHeight())) { 
256             return size1; 
257         } 
258         else { 
259             double h = constraint.getHeightRange().constrain(size1.getHeight()); 
260             RectangleConstraint c2 = constraint.toFixedHeight(h); 
261             return arrange(container, g2, c2); 
262         } 
263     }

265     /** 
266      * Arranges the container width a fixed width and no constraint on the 
267      * height. 
268      * 
269      * @param container  the container. 
270      * @param g2  the graphics device. 
271      * @param width  the fixed width. 
272      * 
273      * @return The container size after arranging the contents. 
274      */ 
275     protected Size2D arrangeFN(BlockContainer container, Graphics2D g2, 
276                                double width) { 
277         double[] w = new double[5]; 
278         double[] h = new double[5]; 
279         RectangleConstraint c1 = new RectangleConstraint(width, null, 
280                 LengthConstraintType.FIXED, 0.0, null, 
281                 LengthConstraintType.NONE); 
282         if (this.topBlock != null) { 
283             Size2D size = this.topBlock.arrange(g2, c1); 
284             w[0] = size.width; 
285             h[0] = size.height; 
286         } 
287         if (this.bottomBlock != null) { 
288             Size2D size = this.bottomBlock.arrange(g2, c1); 
289             w[1] = size.width; 
290             h[1] = size.height; 
291         } 
292         RectangleConstraint c2 = new RectangleConstraint(0.0, 
293                 new Range(0.0, width), LengthConstraintType.RANGE, 
294                 0.0, null, LengthConstraintType.NONE); 
295         if (this.leftBlock != null) { 
296             Size2D size = this.leftBlock.arrange(g2, c2); 
297             w[2] = size.width; 
298             h[2] = size.height; 
299         } 
300         if (this.rightBlock != null) { 
301             double maxW = Math.max(width - w[2], 0.0); 
302             RectangleConstraint c3 = new RectangleConstraint(0.0, 
303                     new Range(Math.min(w[2], maxW), maxW), 
304                     LengthConstraintType.RANGE, 0.0, null, 
305                     LengthConstraintType.NONE); 
306             Size2D size = this.rightBlock.arrange(g2, c3); 
307             w[3] = size.width; 
308             h[3] = size.height; 
309         } 
310  
311         h[2] = Math.max(h[2], h[3]); 
312         h[3] = h[2]; 
313  
314         if (this.centerBlock != null) { 
315             RectangleConstraint c4 = new RectangleConstraint(width - w[2] 
316                     - w[3], null, LengthConstraintType.FIXED, 0.0, null, 
317                     LengthConstraintType.NONE); 
318             Size2D size = this.centerBlock.arrange(g2, c4); 
319             w[4] = size.width; 
320             h[4] = size.height; 
321         } 
322         double height = h[0] + h[1] + Math.max(h[2], Math.max(h[3], h[4])); 
323         return arrange(container, g2, new RectangleConstraint(width, height)); 
324     }

326     /** 
327      * Performs an arrangement with range constraints on both the vertical 
328      * and horizontal sides. 
329      * 
330      * @param container  the container. 
331      * @param widthRange  the allowable range for the container width. 
332      * @param heightRange  the allowable range for the container height. 
333      * @param g2  the graphics device. 
334      * 
335      * @return The container size. 
336      */ 
337     protected Size2D arrangeRR(BlockContainer container, 
338                                Range widthRange, Range heightRange, 
339                                Graphics2D g2) { 
340         double[] w = new double[5]; 
341         double[] h = new double[5]; 
342         if (this.topBlock != null) { 
343             RectangleConstraint c1 = new RectangleConstraint(widthRange, 
344                     heightRange); 
345             Size2D size = this.topBlock.arrange(g2, c1); 
346             w[0] = size.width; 
347             h[0] = size.height; 
348         } 
349         if (this.bottomBlock != null) { 
350             Range heightRange2 = Range.shift(heightRange, -h[0], false); 
351             RectangleConstraint c2 = new RectangleConstraint(widthRange, 
352                     heightRange2); 
353             Size2D size = this.bottomBlock.arrange(g2, c2); 
354             w[1] = size.width; 
355             h[1] = size.height; 
356         } 
357         Range heightRange3 = Range.shift(heightRange, -(h[0] + h[1])); 
358         if (this.leftBlock != null) { 
359             RectangleConstraint c3 = new RectangleConstraint(widthRange, 
360                     heightRange3); 
361             Size2D size = this.leftBlock.arrange(g2, c3); 
362             w[2] = size.width; 
363             h[2] = size.height; 
364         } 
365         Range widthRange2 = Range.shift(widthRange, -w[2], false); 
366         if (this.rightBlock != null) { 
367             RectangleConstraint c4 = new RectangleConstraint(widthRange2, 
368                     heightRange3); 
369             Size2D size = this.rightBlock.arrange(g2, c4); 
370             w[3] = size.width; 
371             h[3] = size.height; 
372         } 
373  
374         h[2] = Math.max(h[2], h[3]); 
375         h[3] = h[2]; 
376         Range widthRange3 = Range.shift(widthRange, -(w[2] + w[3]), false); 
377         if (this.centerBlock != null) { 
378             RectangleConstraint c5 = new RectangleConstraint(widthRange3, 
379                     heightRange3); 
380             // TODO:  the width and height ranges should be reduced by the 
381             // height required for the top and bottom, and the width required 
382             // by the left and right 
383             Size2D size = this.centerBlock.arrange(g2, c5); 
384             w[4] = size.width; 
385             h[4] = size.height; 
386         } 
387         double width = Math.max(w[0], Math.max(w[1], w[2] + w[4] + w[3])); 
388         double height = h[0] + h[1] + Math.max(h[2], Math.max(h[3], h[4])); 
389         if (this.topBlock != null) { 
390             this.topBlock.setBounds(new Rectangle2D.Double(0.0, 0.0, width, 
391                     h[0])); 
392         } 
393         if (this.bottomBlock != null) { 
394             this.bottomBlock.setBounds(new Rectangle2D.Double(0.0, 
395                     height - h[1], width, h[1])); 
396         } 
397         if (this.leftBlock != null) { 
398             this.leftBlock.setBounds(new Rectangle2D.Double(0.0, h[0], w[2], 
399                     h[2])); 
400         } 
401         if (this.rightBlock != null) { 
402             this.rightBlock.setBounds(new Rectangle2D.Double(width - w[3], 
403                     h[0], w[3], h[3])); 
404         } 
405  
406         if (this.centerBlock != null) { 
407             this.centerBlock.setBounds(new Rectangle2D.Double(w[2], h[0], 
408                     width - w[2] - w[3], height - h[0] - h[1])); 
409         } 
410         return new Size2D(width, height); 
411     }

413     /** 
414      * Arranges the items within a container. 
415      * 
416      * @param container  the container. 
417      * @param constraint  the constraint. 
418      * @param g2  the graphics device. 
419      * 
420      * @return The container size after the arrangement. 
421      */ 
422     protected Size2D arrangeFF(BlockContainer container, Graphics2D g2, 
423                                RectangleConstraint constraint) { 
424         double[] w = new double[5]; 
425         double[] h = new double[5]; 
426         w[0] = constraint.getWidth(); 
427         if (this.topBlock != null) { 
428             RectangleConstraint c1 = new RectangleConstraint(w[0], null, 
429                     LengthConstraintType.FIXED, 0.0, 
430                     new Range(0.0, constraint.getHeight()), 
431                     LengthConstraintType.RANGE); 
432             Size2D size = this.topBlock.arrange(g2, c1); 
433             h[0] = size.height; 
434         } 
435         w[1] = w[0]; 
436         if (this.bottomBlock != null) { 
437             RectangleConstraint c2 = new RectangleConstraint(w[0], null, 
438                     LengthConstraintType.FIXED, 0.0, new Range(0.0, 
439                     constraint.getHeight() - h[0]), LengthConstraintType.RANGE); 
440             Size2D size = this.bottomBlock.arrange(g2, c2); 
441             h[1] = size.height; 
442         } 
443         h[2] = constraint.getHeight() - h[1] - h[0]; 
444         if (this.leftBlock != null) { 
445             RectangleConstraint c3 = new RectangleConstraint(0.0, 
446                     new Range(0.0, constraint.getWidth()), 
447                     LengthConstraintType.RANGE, h[2], null, 
448                     LengthConstraintType.FIXED); 
449             Size2D size = this.leftBlock.arrange(g2, c3); 
450             w[2] = size.width; 
451         } 
452         h[3] = h[2]; 
453         if (this.rightBlock != null) { 
454             RectangleConstraint c4 = new RectangleConstraint(0.0, 
455                     new Range(0.0, constraint.getWidth() - w[2]), 
456                     LengthConstraintType.RANGE, h[2], null, 
457                     LengthConstraintType.FIXED); 
458             Size2D size = this.rightBlock.arrange(g2, c4); 
459             w[3] = size.width; 
460         } 
461         h[4] = h[2]; 
462         w[4] = constraint.getWidth() - w[3] - w[2]; 
463         RectangleConstraint c5 = new RectangleConstraint(w[4], h[4]); 
464         if (this.centerBlock != null) { 
465             this.centerBlock.arrange(g2, c5); 
466         } 
467  
468         if (this.topBlock != null) { 
469             this.topBlock.setBounds(new Rectangle2D.Double(0.0, 0.0, w[0], 
470                     h[0])); 
471         } 
472         if (this.bottomBlock != null) { 
473             this.bottomBlock.setBounds(new Rectangle2D.Double(0.0, h[0] + h[2], 
474                     w[1], h[1])); 
475         } 
476         if (this.leftBlock != null) { 
477             this.leftBlock.setBounds(new Rectangle2D.Double(0.0, h[0], w[2], 
478                     h[2])); 
479         } 
480         if (this.rightBlock != null) { 
481             this.rightBlock.setBounds(new Rectangle2D.Double(w[2] + w[4], h[0], 
482                     w[3], h[3])); 
483         } 
484         if (this.centerBlock != null) { 
485             this.centerBlock.setBounds(new Rectangle2D.Double(w[2], h[0], w[4], 
486                     h[4])); 
487         } 
488         return new Size2D(constraint.getWidth(), constraint.getHeight()); 
489     }

 89     /** 
 90      * Adds a block to the arrangement manager at the specified edge. 
 91      * 
 92      * @param block  the block (<code>null</code> permitted). 
 93      * @param key  the edge (an instance of {@link RectangleEdge}) or 
 94      *             <code>null</code> for the center block. 
 95      */ 
 96     public void add(Block block, Object key) { 
 97  
 98         if (key == null) { 
 99             this.centerBlock = block; 
100         } 
101         else { 
102             RectangleEdge edge = (RectangleEdge) key; 
103             if (edge == RectangleEdge.TOP) { 
104                 this.topBlock = block; 
105             } 
106             else if (edge == RectangleEdge.BOTTOM) { 
107                 this.bottomBlock = block; 
108             } 
109             else if (edge == RectangleEdge.LEFT) { 
110                 this.leftBlock = block; 
111             } 
112             else if (edge == RectangleEdge.RIGHT) { 
113                 this.rightBlock = block; 
114             } 
115         } 
116     }

118     /** 
119      * Arranges the items in the specified container, subject to the given 
120      * constraint. 
121      * 
122      * @param container  the container. 
123      * @param g2  the graphics device. 
124      * @param constraint  the constraint. 
125      * 
126      * @return The block size. 
127      */ 
128     public Size2D arrange(BlockContainer container, 
129                           Graphics2D g2, 
130                           RectangleConstraint constraint) { 
131         RectangleConstraint contentConstraint 
132                 = container.toContentConstraint(constraint); 
133         Size2D contentSize = null; 
134         LengthConstraintType w = contentConstraint.getWidthConstraintType(); 
135         LengthConstraintType h = contentConstraint.getHeightConstraintType(); 
136         if (w == LengthConstraintType.NONE) { 
137             if (h == LengthConstraintType.NONE) { 
138                 contentSize = arrangeNN(container, g2); 
139             } 
140             else if (h == LengthConstraintType.FIXED) { 
141                 throw new RuntimeException("Not implemented."); 
142             } 
143             else if (h == LengthConstraintType.RANGE) { 
144                 throw new RuntimeException("Not implemented."); 
145             } 
146         } 
147         else if (w == LengthConstraintType.FIXED) { 
148             if (h == LengthConstraintType.NONE) { 
149                 contentSize = arrangeFN(container, g2, constraint.getWidth()); 
150             } 
151             else if (h == LengthConstraintType.FIXED) { 
152                 contentSize = arrangeFF(container, g2, constraint); 
153             } 
154             else if (h == LengthConstraintType.RANGE) { 
155                 contentSize = arrangeFR(container, g2, constraint); 
156             } 
157         } 
158         else if (w == LengthConstraintType.RANGE) { 
159             if (h == LengthConstraintType.NONE) { 
160                 throw new RuntimeException("Not implemented."); 
161             } 
162             else if (h == LengthConstraintType.FIXED) { 
163                 throw new RuntimeException("Not implemented."); 
164             } 
165             else if (h == LengthConstraintType.RANGE) { 
166                 contentSize = arrangeRR(container, constraint.getWidthRange(), 
167                         constraint.getHeightRange(), g2); 
168             } 
169         } 
170         return new Size2D(container.calculateTotalWidth(contentSize.getWidth()), 
171                 container.calculateTotalHeight(contentSize.getHeight())); 
172     }

174     /** 
175      * Performs an arrangement without constraints. 
176      * 
177      * @param container  the container. 
178      * @param g2  the graphics device. 
179      * 
180      * @return The container size after the arrangement. 
181      */ 
182     protected Size2D arrangeNN(BlockContainer container, Graphics2D g2) { 
183         double[] w = new double[5]; 
184         double[] h = new double[5]; 
185         if (this.topBlock != null) { 
186             Size2D size = this.topBlock.arrange(g2, RectangleConstraint.NONE); 
187             w[0] = size.width; 
188             h[0] = size.height; 
189         } 
190         if (this.bottomBlock != null) { 
191             Size2D size = this.bottomBlock.arrange(g2, 
192                     RectangleConstraint.NONE); 
193             w[1] = size.width; 
194             h[1] = size.height; 
195         } 
196         if (this.leftBlock != null) { 
197             Size2D size = this.leftBlock.arrange(g2, RectangleConstraint.NONE); 
198             w[2] = size.width; 
199             h[2] = size.height; 
200        } 
201         if (this.rightBlock != null) { 
202             Size2D size = this.rightBlock.arrange(g2, RectangleConstraint.NONE); 
203             w[3] = size.width; 
204             h[3] = size.height; 
205         } 
206  
207         h[2] = Math.max(h[2], h[3]); 
208         h[3] = h[2]; 
209  
210         if (this.centerBlock != null) { 
211             Size2D size = this.centerBlock.arrange(g2, 
212                     RectangleConstraint.NONE); 
213             w[4] = size.width; 
214             h[4] = size.height; 
215         } 
216         double width = Math.max(w[0], Math.max(w[1], w[2] + w[4] + w[3])); 
217         double centerHeight = Math.max(h[2], Math.max(h[3], h[4])); 
218         double height = h[0] + h[1] + centerHeight; 
219         if (this.topBlock != null) { 
220             this.topBlock.setBounds(new Rectangle2D.Double(0.0, 0.0, width, 
221                     h[0])); 
222         } 
223         if (this.bottomBlock != null) { 
224             this.bottomBlock.setBounds(new Rectangle2D.Double(0.0, 
225                     height - h[1], width, h[1])); 
226         } 
227         if (this.leftBlock != null) { 
228             this.leftBlock.setBounds(new Rectangle2D.Double(0.0, h[0], w[2], 
229                     centerHeight)); 
230         } 
231         if (this.rightBlock != null) { 
232             this.rightBlock.setBounds(new Rectangle2D.Double(width - w[3], 
233                     h[0], w[3], centerHeight)); 
234         } 
235  
236         if (this.centerBlock != null) { 
237             this.centerBlock.setBounds(new Rectangle2D.Double(w[2], h[0], 
238                     width - w[2] - w[3], centerHeight)); 
239         } 
240         return new Size2D(width, height); 
241     }

243     /** 
244      * Performs an arrangement with a fixed width and a range for the height. 
245      * 
246      * @param container  the container. 
247      * @param g2  the graphics device. 
248      * @param constraint  the constraint. 
249      * 
250      * @return The container size after the arrangement. 
251      */ 
252     protected Size2D arrangeFR(BlockContainer container, Graphics2D g2, 
253                                RectangleConstraint constraint) { 
254         Size2D size1 = arrangeFN(container, g2, constraint.getWidth()); 
255         if (constraint.getHeightRange().contains(size1.getHeight())) { 
256             return size1; 
257         } 
258         else { 
259             double h = constraint.getHeightRange().constrain(size1.getHeight()); 
260             RectangleConstraint c2 = constraint.toFixedHeight(h); 
261             return arrange(container, g2, c2); 
262         } 
263     }

265     /** 
266      * Arranges the container width a fixed width and no constraint on the 
267      * height. 
268      * 
269      * @param container  the container. 
270      * @param g2  the graphics device. 
271      * @param width  the fixed width. 
272      * 
273      * @return The container size after arranging the contents. 
274      */ 
275     protected Size2D arrangeFN(BlockContainer container, Graphics2D g2, 
276                                double width) { 
277         double[] w = new double[5]; 
278         double[] h = new double[5]; 
279         RectangleConstraint c1 = new RectangleConstraint(width, null, 
280                 LengthConstraintType.FIXED, 0.0, null, 
281                 LengthConstraintType.NONE); 
282         if (this.topBlock != null) { 
283             Size2D size = this.topBlock.arrange(g2, c1); 
284             w[0] = size.width; 
285             h[0] = size.height; 
286         } 
287         if (this.bottomBlock != null) { 
288             Size2D size = this.bottomBlock.arrange(g2, c1); 
289             w[1] = size.width; 
290             h[1] = size.height; 
291         } 
292         RectangleConstraint c2 = new RectangleConstraint(0.0, 
293                 new Range(0.0, width), LengthConstraintType.RANGE, 
294                 0.0, null, LengthConstraintType.NONE); 
295         if (this.leftBlock != null) { 
296             Size2D size = this.leftBlock.arrange(g2, c2); 
297             w[2] = size.width; 
298             h[2] = size.height; 
299         } 
300         if (this.rightBlock != null) { 
301             double maxW = Math.max(width - w[2], 0.0); 
302             RectangleConstraint c3 = new RectangleConstraint(0.0, 
303                     new Range(Math.min(w[2], maxW), maxW), 
304                     LengthConstraintType.RANGE, 0.0, null, 
305                     LengthConstraintType.NONE); 
306             Size2D size = this.rightBlock.arrange(g2, c3); 
307             w[3] = size.width; 
308             h[3] = size.height; 
309         } 
310  
311         h[2] = Math.max(h[2], h[3]); 
312         h[3] = h[2]; 
313  
314         if (this.centerBlock != null) { 
315             RectangleConstraint c4 = new RectangleConstraint(width - w[2] 
316                     - w[3], null, LengthConstraintType.FIXED, 0.0, null, 
317                     LengthConstraintType.NONE); 
318             Size2D size = this.centerBlock.arrange(g2, c4); 
319             w[4] = size.width; 
320             h[4] = size.height; 
321         } 
322         double height = h[0] + h[1] + Math.max(h[2], Math.max(h[3], h[4])); 
323         return arrange(container, g2, new RectangleConstraint(width, height)); 
324     }

326     /** 
327      * Performs an arrangement with range constraints on both the vertical 
328      * and horizontal sides. 
329      * 
330      * @param container  the container. 
331      * @param widthRange  the allowable range for the container width. 
332      * @param heightRange  the allowable range for the container height. 
333      * @param g2  the graphics device. 
334      * 
335      * @return The container size. 
336      */ 
337     protected Size2D arrangeRR(BlockContainer container, 
338                                Range widthRange, Range heightRange, 
339                                Graphics2D g2) { 
340         double[] w = new double[5]; 
341         double[] h = new double[5]; 
342         if (this.topBlock != null) { 
343             RectangleConstraint c1 = new RectangleConstraint(widthRange, 
344                     heightRange); 
345             Size2D size = this.topBlock.arrange(g2, c1); 
346             w[0] = size.width; 
347             h[0] = size.height; 
348         } 
349         if (this.bottomBlock != null) { 
350             Range heightRange2 = Range.shift(heightRange, -h[0], false); 
351             RectangleConstraint c2 = new RectangleConstraint(widthRange, 
352                     heightRange2); 
353             Size2D size = this.bottomBlock.arrange(g2, c2); 
354             w[1] = size.width; 
355             h[1] = size.height; 
356         } 
357         Range heightRange3 = Range.shift(heightRange, -(h[0] + h[1])); 
358         if (this.leftBlock != null) { 
359             RectangleConstraint c3 = new RectangleConstraint(widthRange, 
360                     heightRange3); 
361             Size2D size = this.leftBlock.arrange(g2, c3); 
362             w[2] = size.width; 
363             h[2] = size.height; 
364         } 
365         Range widthRange2 = Range.shift(widthRange, -w[2], false); 
366         if (this.rightBlock != null) { 
367             RectangleConstraint c4 = new RectangleConstraint(widthRange2, 
368                     heightRange3); 
369             Size2D size = this.rightBlock.arrange(g2, c4); 
370             w[3] = size.width; 
371             h[3] = size.height; 
372         } 
373  
374         h[2] = Math.max(h[2], h[3]); 
375         h[3] = h[2]; 
376         Range widthRange3 = Range.shift(widthRange, -(w[2] + w[3]), false); 
377         if (this.centerBlock != null) { 
378             RectangleConstraint c5 = new RectangleConstraint(widthRange3, 
379                     heightRange3); 
380             // TODO:  the width and height ranges should be reduced by the 
381             // height required for the top and bottom, and the width required 
382             // by the left and right 
383             Size2D size = this.centerBlock.arrange(g2, c5); 
384             w[4] = size.width; 
385             h[4] = size.height; 
386         } 
387         double width = Math.max(w[0], Math.max(w[1], w[2] + w[4] + w[3])); 
388         double height = h[0] + h[1] + Math.max(h[2], Math.max(h[3], h[4])); 
389         if (this.topBlock != null) { 
390             this.topBlock.setBounds(new Rectangle2D.Double(0.0, 0.0, width, 
391                     h[0])); 
392         } 
393         if (this.bottomBlock != null) { 
394             this.bottomBlock.setBounds(new Rectangle2D.Double(0.0, 
395                     height - h[1], width, h[1])); 
396         } 
397         if (this.leftBlock != null) { 
398             this.leftBlock.setBounds(new Rectangle2D.Double(0.0, h[0], w[2], 
399                     h[2])); 
400         } 
401         if (this.rightBlock != null) { 
402             this.rightBlock.setBounds(new Rectangle2D.Double(width - w[3], 
403                     h[0], w[3], h[3])); 
404         } 
405  
406         if (this.centerBlock != null) { 
407             this.centerBlock.setBounds(new Rectangle2D.Double(w[2], h[0], 
408                     width - w[2] - w[3], height - h[0] - h[1])); 
409         } 
410         return new Size2D(width, height); 
411     }

413     /** 
414      * Arranges the items within a container. 
415      * 
416      * @param container  the container. 
417      * @param constraint  the constraint. 
418      * @param g2  the graphics device. 
419      * 
420      * @return The container size after the arrangement. 
421      */ 
422     protected Size2D arrangeFF(BlockContainer container, Graphics2D g2, 
423                                RectangleConstraint constraint) { 
424         double[] w = new double[5]; 
425         double[] h = new double[5]; 
426         w[0] = constraint.getWidth(); 
427         if (this.topBlock != null) { 
428             RectangleConstraint c1 = new RectangleConstraint(w[0], null, 
429                     LengthConstraintType.FIXED, 0.0, 
430                     new Range(0.0, constraint.getHeight()), 
431                     LengthConstraintType.RANGE); 
432             Size2D size = this.topBlock.arrange(g2, c1); 
433             h[0] = size.height; 
434         } 
435         w[1] = w[0]; 
436         if (this.bottomBlock != null) { 
437             RectangleConstraint c2 = new RectangleConstraint(w[0], null, 
438                     LengthConstraintType.FIXED, 0.0, new Range(0.0, 
439                     constraint.getHeight() - h[0]), LengthConstraintType.RANGE); 
440             Size2D size = this.bottomBlock.arrange(g2, c2); 
441             h[1] = size.height; 
442         } 
443         h[2] = constraint.getHeight() - h[1] - h[0]; 
444         if (this.leftBlock != null) { 
445             RectangleConstraint c3 = new RectangleConstraint(0.0, 
446                     new Range(0.0, constraint.getWidth()), 
447                     LengthConstraintType.RANGE, h[2], null, 
448                     LengthConstraintType.FIXED); 
449             Size2D size = this.leftBlock.arrange(g2, c3); 
450             w[2] = size.width; 
451         } 
452         h[3] = h[2]; 
453         if (this.rightBlock != null) { 
454             RectangleConstraint c4 = new RectangleConstraint(0.0, 
455                     new Range(0.0, Math.max(constraint.getWidth() - w[2], 0.0)), 
456                     LengthConstraintType.RANGE, h[2], null, 
457                     LengthConstraintType.FIXED); 
458             Size2D size = this.rightBlock.arrange(g2, c4); 
459             w[3] = size.width; 
460         } 
461         h[4] = h[2]; 
462         w[4] = constraint.getWidth() - w[3] - w[2]; 
463         RectangleConstraint c5 = new RectangleConstraint(w[4], h[4]); 
464         if (this.centerBlock != null) { 
465             this.centerBlock.arrange(g2, c5); 
466         } 
467  
468         if (this.topBlock != null) { 
469             this.topBlock.setBounds(new Rectangle2D.Double(0.0, 0.0, w[0], 
470                     h[0])); 
471         } 
472         if (this.bottomBlock != null) { 
473             this.bottomBlock.setBounds(new Rectangle2D.Double(0.0, h[0] + h[2], 
474                     w[1], h[1])); 
475         } 
476         if (this.leftBlock != null) { 
477             this.leftBlock.setBounds(new Rectangle2D.Double(0.0, h[0], w[2], 
478                     h[2])); 
479         } 
480         if (this.rightBlock != null) { 
481             this.rightBlock.setBounds(new Rectangle2D.Double(w[2] + w[4], h[0], 
482                     w[3], h[3])); 
483         } 
484         if (this.centerBlock != null) { 
485             this.centerBlock.setBounds(new Rectangle2D.Double(w[2], h[0], w[4], 
486                     h[4])); 
487         } 
488         return new Size2D(constraint.getWidth(), constraint.getHeight()); 
489     }

491     /** 
492      * Clears the layout. 
493      */ 
494     public void clear() { 
495         this.centerBlock = null; 
496         this.topBlock = null; 
497         this.bottomBlock = null; 
498         this.leftBlock = null; 
499         this.rightBlock = null;

502     /** 
503      * Tests this arrangement for equality with an arbitrary object. 
504      * 
505      * @param obj  the object (<code>null</code> permitted). 
506      * 
507      * @return A boolean. 
508      */ 
509     public boolean equals(Object obj) { 
510         if (obj == this) { 
511             return true; 
512         } 
513         if (!(obj instanceof BorderArrangement)) { 
514             return false; 
515         } 
516         BorderArrangement that = (BorderArrangement) obj; 
517         if (!ObjectUtilities.equal(this.topBlock, that.topBlock)) { 
518             return false; 
519         } 
520         if (!ObjectUtilities.equal(this.bottomBlock, that.bottomBlock)) { 
521             return false; 
522         } 
523         if (!ObjectUtilities.equal(this.leftBlock, that.leftBlock)) { 
524             return false; 
525         } 
526         if (!ObjectUtilities.equal(this.rightBlock, that.rightBlock)) { 
527             return false; 
528         } 
529         if (!ObjectUtilities.equal(this.centerBlock, that.centerBlock)) { 
530             return false; 
531         } 
532         return true; 
533     }
