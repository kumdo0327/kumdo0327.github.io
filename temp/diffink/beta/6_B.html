  47 /** 
  48  * Specifies the format of a CSV file and parses input. 
  49  * 
  50  * <h2>Using predefined formats</h2> 
  51  * 
  52  * <p> 
  53  * You can use one of the predefined formats: 
  54  * </p> 
  55  * 
  56  * <ul> 
  57  * <li>{@link #DEFAULT}</li> 
  58  * <li>{@link #EXCEL}</li> 
  59  * <li>{@link #MYSQL}</li> 
  60  * <li>{@link #RFC4180}</li> 
  61  * <li>{@link #TDF}</li> 
  62  * </ul> 
  63  * 
  64  * <p> 
  65  * For example: 
  66  * </p> 
  67  * 
  68  * <pre> 
  69  * CSVParser parser = CSVFormat.EXCEL.parse(reader); 
  70  * </pre> 
  71  * 
  72  * <p> 
  73  * The {@link CSVParser} provides static methods to parse other input types, for example: 
  74  * </p> 
  75  * 
  76  * <pre> 
  77  * CSVParser parser = CSVParser.parse(file, StandardCharsets.US_ASCII, CSVFormat.EXCEL); 
  78  * </pre> 
  79  * 
  80  * <h2>Defining formats</h2> 
  81  * 
  82  * <p> 
  83  * You can extend a format by calling the {@code with} methods. For example: 
  84  * </p> 
  85  * 
  86  * <pre> 
  87  * CSVFormat.EXCEL.withNullString(&quot;N/A&quot;).withIgnoreSurroundingSpaces(true); 
  88  * </pre> 
  89  * 
  90  * <h2>Defining column names</h2> 
  91  * 
  92  * <p> 
  93  * To define the column names you want to use to access records, write: 
  94  * </p> 
  95  * 
  96  * <pre> 
  97  * CSVFormat.EXCEL.withHeader(&quot;Col1&quot;, &quot;Col2&quot;, &quot;Col3&quot;); 
  98  * </pre> 
  99  * 
 100  * <p> 
 101  * Calling {@link #withHeader(String...)} let's you use the given names to address values in a {@link CSVRecord}, and 
 102  * assumes that your CSV source does not contain a first record that also defines column names. 
 103  * 
 104  * If it does, then you are overriding this metadata with your names and you should skip the first record by calling 
 105  * {@link #withSkipHeaderRecord(boolean)} with {@code true}. 
 106  * </p> 
 107  * 
 108  * <h2>Parsing</h2> 
 109  * 
 110  * <p> 
 111  * You can use a format directly to parse a reader. For example, to parse an Excel file with columns header, write: 
 112  * </p> 
 113  * 
 114  * <pre> 
 115  * Reader in = ...; 
 116  * CSVFormat.EXCEL.withHeader(&quot;Col1&quot;, &quot;Col2&quot;, &quot;Col3&quot;).parse(in); 
 117  * </pre> 
 118  * 
 119  * <p> 
 120  * For other input types, like resources, files, and URLs, use the static methods on {@link CSVParser}. 
 121  * </p> 
 122  * 
 123  * <h2>Referencing columns safely</h2> 
 124  * 
 125  * <p> 
 126  * If your source contains a header record, you can simplify your code and safely reference columns, by using 
 127  * {@link #withHeader(String...)} with no arguments: 
 128  * </p> 
 129  * 
 130  * <pre> 
 131  * CSVFormat.EXCEL.withHeader(); 
 132  * </pre> 
 133  * 
 134  * <p> 
 135  * This causes the parser to read the first record and use its values as column names. 
 136  * 
 137  * Then, call one of the {@link CSVRecord} get method that takes a String column name argument: 
 138  * </p> 
 139  * 
 140  * <pre> 
 141  * String value = record.get(&quot;Col1&quot;); 
 142  * </pre> 
 143  * 
 144  * <p> 
 145  * This makes your code impervious to changes in column order in the CSV file. 
 146  * </p> 
 147  * 
 148  * <h2>Notes</h2> 
 149  * 
 150  * <p> 
 151  * This class is immutable. 
 152  * </p> 
 153  * 
 154  * @version $Id$ 
 155  */

 539     @Override 
 540     public boolean equals(final Object obj) { 
 541         if (this == obj) { 
 542             return true; 
 543         } 
 544         if (obj == null) { 
 545             return false; 
 546         } 
 547         if (getClass() != obj.getClass()) { 
 548             return false; 
 549         } 
 550  
 551         final CSVFormat other = (CSVFormat) obj; 
 552         if (delimiter != other.delimiter) { 
 553             return false; 
 554         } 
 555         if (quoteMode != other.quoteMode) { 
 556             return false; 
 557         } 
 558         if (quoteCharacter == null) { 
 559             if (other.quoteCharacter != null) { 
 560                 return false; 
 561             } 
 562         } else if (!quoteCharacter.equals(other.quoteCharacter)) { 
 563             return false; 
 564         } 
 565         if (commentMarker == null) { 
 566             if (other.commentMarker != null) { 
 567                 return false; 
 568             } 
 569         } else if (!commentMarker.equals(other.commentMarker)) { 
 570             return false; 
 571         } 
 572         if (escapeCharacter == null) { 
 573             if (other.escapeCharacter != null) { 
 574                 return false; 
 575             } 
 576         } else if (!escapeCharacter.equals(other.escapeCharacter)) { 
 577             return false; 
 578         } 
 579         if (nullString == null) { 
 580             if (other.nullString != null) { 
 581                 return false; 
 582             } 
 583         } else if (!nullString.equals(other.nullString)) { 
 584             return false; 
 585         } 
 586         if (!Arrays.equals(header, other.header)) { 
 587             return false; 
 588         } 
 589         if (ignoreSurroundingSpaces != other.ignoreSurroundingSpaces) { 
 590             return false; 
 591         } 
 592         if (ignoreEmptyLines != other.ignoreEmptyLines) { 
 593             return false; 
 594         } 
 595         if (skipHeaderRecord != other.skipHeaderRecord) { 
 596             return false; 
 597         } 
 598         if (recordSeparator == null) { 
 599             if (other.recordSeparator != null) { 
 600                 return false; 
 601             } 
 602         } else if (!recordSeparator.equals(other.recordSeparator)) { 
 603             return false; 
 604         } 
 605         return true; 
 606     }

 608     /** 
 609      * Formats the specified values. 
 610      * 
 611      * @param values 
 612      *            the values to format 
 613      * @return the formatted values 
 614      */ 
 615     public String format(final Object... values) { 
 616         final StringWriter out = new StringWriter(); 
 617         try (final CSVPrinter csvPrinter = new CSVPrinter(out, this)) { 
 618             csvPrinter.printRecord(values); 
 619             return out.toString().trim(); 
 620         } catch (final IOException e) { 
 621             // should not happen because a StringWriter does not do IO. 
 622             throw new IllegalStateException(e); 
 623         } 
 624     }

 940     private void print(final Object object, final CharSequence value, final int offset, final int len, 
 941             final Appendable out, final boolean newRecord) throws IOException { 
 942         if (!newRecord) { 
 943             out.append(getDelimiter()); 
 944         } 
 945         if (object == null) { 
 946             out.append(value); 
 947         } else if (isQuoteCharacterSet()) { 
 948             // the original object is needed so can check for Number 
 949             printAndQuote(object, value, offset, len, out, newRecord); 
 950         } else if (isEscapeCharacterSet()) { 
 951             printAndEscape(value, offset, len, out); 
 952         } else { 
 953             out.append(value, offset, offset + len); 
 954         } 
 955     }

 957     /* 
 958      * Note: must only be called if escaping is enabled, otherwise will generate NPE 
 959      */ 
 960     private void printAndEscape(final CharSequence value, final int offset, final int len, final Appendable out) 
 961             throws IOException { 
 962         int start = offset; 
 963         int pos = offset; 
 964         final int end = offset + len; 
 965  
 966         final char delim = getDelimiter(); 
 967         final char escape = getEscapeCharacter().charValue(); 
 968  
 969         while (pos < end) { 
 970             char c = value.charAt(pos); 
 971             if (c == CR || c == LF || c == delim || c == escape) { 
 972                 // write out segment up until this char 
 973                 if (pos > start) { 
 974                     out.append(value, start, pos); 
 975                 } 
 976                 if (c == LF) { 
 977                     c = 'n'; 
 978                 } else if (c == CR) { 
 979                     c = 'r'; 
 980                 } 
 981  
 982                 out.append(escape); 
 983                 out.append(c); 
 984  
 985                 start = pos + 1; // start on the current char after this one 
 986             } 
 987  
 988             pos++; 
 989         } 
 990  
 991         // write last segment 
 992         if (pos > start) { 
 993             out.append(value, start, pos); 
 994         } 
 995     }

 997     /* 
 998      * Note: must only be called if quoting is enabled, otherwise will generate NPE 
 999      */ 
1000     // the original object is needed so can check for Number 
1001     private void printAndQuote(final Object object, final CharSequence value, final int offset, final int len, 
1002             final Appendable out, final boolean newRecord) throws IOException { 
1003         boolean quote = false; 
1004         int start = offset; 
1005         int pos = offset; 
1006         final int end = offset + len; 
1007  
1008         final char delimChar = getDelimiter(); 
1009         final char quoteChar = getQuoteCharacter().charValue(); 
1010  
1011         QuoteMode quoteModePolicy = getQuoteMode(); 
1012         if (quoteModePolicy == null) { 
1013             quoteModePolicy = QuoteMode.MINIMAL; 
1014         } 
1015         switch (quoteModePolicy) { 
1016         case ALL: 
1017             quote = true; 
1018             break; 
1019         case NON_NUMERIC: 
1020             quote = !(object instanceof Number); 
1021             break; 
1022         case NONE: 
1023             // Use the existing escaping code 
1024             printAndEscape(value, offset, len, out); 
1025             return; 
1026         case MINIMAL: 
1027             if (len <= 0) { 
1028                 // always quote an empty token that is the first 
1029                 // on the line, as it may be the only thing on the 
1030                 // line. If it were not quoted in that case, 
1031                 // an empty line has no tokens. 
1032                 if (newRecord) { 
1033                     quote = true; 
1034                 } 
1035             } else { 
1036                 char c = value.charAt(pos); 
1037  
1038                 // RFC4180 (https://tools.ietf.org/html/rfc4180) TEXTDATA =  %x20-21 / %x23-2B / %x2D-7E 
1039                 if (newRecord && (c < '0' || c > '9' && c < 'A' || c > 'Z' && c < 'a' || c > 'z')) { 
1040                     quote = true; 
1041                 } else if (c <= COMMENT) { 
1042                     // Some other chars at the start of a value caused the parser to fail, so for now 
1043                     // encapsulate if we start in anything less than '#'. We are being conservative 
1044                     // by including the default comment char too. 
1045                     quote = true; 
1046                 } else { 
1047                     while (pos < end) { 
1048                         c = value.charAt(pos); 
1049                         if (c == LF || c == CR || c == quoteChar || c == delimChar) { 
1050                             quote = true; 
1051                             break; 
1052                         } 
1053                         pos++; 
1054                     } 
1055  
1056                     if (!quote) { 
1057                         pos = end - 1; 
1058                         c = value.charAt(pos); 
1059                         // Some other chars at the end caused the parser to fail, so for now 
1060                         // encapsulate if we end in anything less than ' ' 
1061                         if (c <= SP) { 
1062                             quote = true; 
1063                         } 
1064                     } 
1065                 } 
1066             } 
1067  
1068             if (!quote) { 
1069                 // no encapsulation needed - write out the original value 
1070                 out.append(value, start, end); 
1071                 return; 
1072             } 
1073             break; 
1074         default: 
1075             throw new IllegalStateException("Unexpected Quote value: " + quoteModePolicy); 
1076         } 
1077  
1078         if (!quote) { 
1079             // no encapsulation needed - write out the original value 
1080             out.append(value, start, end); 
1081             return; 
1082         } 
1083  
1084         // we hit something that needed encapsulation 
1085         out.append(quoteChar); 
1086  
1087         // Pick up where we left off: pos should be positioned on the first character that caused 
1088         // the need for encapsulation. 
1089         while (pos < end) { 
1090             final char c = value.charAt(pos); 
1091             if (c == quoteChar) { 
1092                 // write out the chunk up until this point 
1093  
1094                 // add 1 to the length to write out the encapsulator also 
1095                 out.append(value, start, pos + 1); 
1096                 // put the next starting position on the encapsulator so we will 
1097                 // write it out again with the next string (effectively doubling it) 
1098                 start = pos; 
1099             } 
1100             pos++; 
1101         } 
1102  
1103         // write the last segment 
1104         out.append(value, start, pos); 
1105         out.append(quoteChar); 
1106     }

1207     private CharSequence trim(final CharSequence charSequence) { 
1208         if (charSequence instanceof String) { 
1209             return ((String) charSequence).trim(); 
1210         } 
1211         final int count = charSequence.length(); 
1212         int len = count; 
1213         int pos = 0; 
1214  
1215         while (pos < len && charSequence.charAt(pos) <= SP) { 
1216             pos++; 
1217         } 
1218         while (pos < len && charSequence.charAt(len - 1) <= SP) { 
1219             len--; 
1220         } 
1221         return pos > 0 || len < count ? charSequence.subSequence(pos, len) : charSequence; 
1222     }

1225      * Verifies the consistency of the parameters and throws an IllegalArgumentException if necessary. 
1226      * 
1227      * @throws IllegalArgumentException 
1228      */ 
1229     private void validate() throws IllegalArgumentException { 
1230         if (isLineBreak(delimiter)) { 
1231             throw new IllegalArgumentException("The delimiter cannot be a line break"); 
1232         } 
1233  
1234         if (quoteCharacter != null && delimiter == quoteCharacter.charValue()) { 
1235             throw new IllegalArgumentException( 
1236                     "The quoteChar character and the delimiter cannot be the same ('" + quoteCharacter + "')"); 
1237         } 
1238  
1239         if (escapeCharacter != null && delimiter == escapeCharacter.charValue()) { 
1240             throw new IllegalArgumentException( 
1241                     "The escape character and the delimiter cannot be the same ('" + escapeCharacter + "')"); 
1242         } 
1243  
1244         if (commentMarker != null && delimiter == commentMarker.charValue()) { 
1245             throw new IllegalArgumentException( 
1246                     "The comment start character and the delimiter cannot be the same ('" + commentMarker + "')"); 
1247         } 
1248  
1249         if (quoteCharacter != null && quoteCharacter.equals(commentMarker)) { 
1250             throw new IllegalArgumentException( 
1251                     "The comment start character and the quoteChar cannot be the same ('" + commentMarker + "')"); 
1252         } 
1253  
1254         if (escapeCharacter != null && escapeCharacter.equals(commentMarker)) { 
1255             throw new IllegalArgumentException( 
1256                     "The comment start and the escape character cannot be the same ('" + commentMarker + "')"); 
1257         } 
1258  
1259         if (escapeCharacter == null && quoteMode == QuoteMode.NONE) { 
1260             throw new IllegalArgumentException("No quotes mode set but no escape character is set"); 
1261         } 
1262  
1263         // validate header 
1264         if (header != null) { 
1265             final Set<String> dupCheck = new HashSet<>(); 
1266             for (final String hdr : header) { 
1267                 if (!dupCheck.add(hdr)) { 
1268                     throw new IllegalArgumentException( 
1269                             "The header contains a duplicate entry: '" + hdr + "' in " + Arrays.toString(header)); 
1270                 } 
1271             } 
1272         }

  47 /** 
  48  * Specifies the format of a CSV file and parses input. 
  49  * 
  50  * <h2>Using predefined formats</h2> 
  51  * 
  52  * <p> 
  53  * You can use one of the predefined formats: 
  54  * </p> 
  55  * 
  56  * <ul> 
  57  * <li>{@link #DEFAULT}</li> 
  58  * <li>{@link #EXCEL}</li> 
  59  * <li>{@link #MYSQL}</li> 
  60  * <li>{@link #RFC4180}</li> 
  61  * <li>{@link #TDF}</li> 
  62  * </ul> 
  63  * 
  64  * <p> 
  65  * For example: 
  66  * </p> 
  67  * 
  68  * <pre> 
  69  * CSVParser parser = CSVFormat.EXCEL.parse(reader); 
  70  * </pre> 
  71  * 
  72  * <p> 
  73  * The {@link CSVParser} provides static methods to parse other input types, for example: 
  74  * </p> 
  75  * 
  76  * <pre> 
  77  * CSVParser parser = CSVParser.parse(file, StandardCharsets.US_ASCII, CSVFormat.EXCEL); 
  78  * </pre> 
  79  * 
  80  * <h2>Defining formats</h2> 
  81  * 
  82  * <p> 
  83  * You can extend a format by calling the {@code with} methods. For example: 
  84  * </p> 
  85  * 
  86  * <pre> 
  87  * CSVFormat.EXCEL.withNullString(&quot;N/A&quot;).withIgnoreSurroundingSpaces(true); 
  88  * </pre> 
  89  * 
  90  * <h2>Defining column names</h2> 
  91  * 
  92  * <p> 
  93  * To define the column names you want to use to access records, write: 
  94  * </p> 
  95  * 
  96  * <pre> 
  97  * CSVFormat.EXCEL.withHeader(&quot;Col1&quot;, &quot;Col2&quot;, &quot;Col3&quot;); 
  98  * </pre> 
  99  * 
 100  * <p> 
 101  * Calling {@link #withHeader(String...)} let's you use the given names to address values in a {@link CSVRecord}, and 
 102  * assumes that your CSV source does not contain a first record that also defines column names. 
 103  * 
 104  * If it does, then you are overriding this metadata with your names and you should skip the first record by calling 
 105  * {@link #withSkipHeaderRecord(boolean)} with {@code true}. 
 106  * </p> 
 107  * 
 108  * <h2>Parsing</h2> 
 109  * 
 110  * <p> 
 111  * You can use a format directly to parse a reader. For example, to parse an Excel file with columns header, write: 
 112  * </p> 
 113  * 
 114  * <pre> 
 115  * Reader in = ...; 
 116  * CSVFormat.EXCEL.withHeader(&quot;Col1&quot;, &quot;Col2&quot;, &quot;Col3&quot;).parse(in); 
 117  * </pre> 
 118  * 
 119  * <p> 
 120  * For other input types, like resources, files, and URLs, use the static methods on {@link CSVParser}. 
 121  * </p> 
 122  * 
 123  * <h2>Referencing columns safely</h2> 
 124  * 
 125  * <p> 
 126  * If your source contains a header record, you can simplify your code and safely reference columns, by using 
 127  * {@link #withHeader(String...)} with no arguments: 
 128  * </p> 
 129  * 
 130  * <pre> 
 131  * CSVFormat.EXCEL.withHeader(); 
 132  * </pre> 
 133  * 
 134  * <p> 
 135  * This causes the parser to read the first record and use its values as column names. 
 136  * 
 137  * Then, call one of the {@link CSVRecord} get method that takes a String column name argument: 
 138  * </p> 
 139  * 
 140  * <pre> 
 141  * String value = record.get(&quot;Col1&quot;); 
 142  * </pre> 
 143  * 
 144  * <p> 
 145  * This makes your code impervious to changes in column order in the CSV file. 
 146  * </p> 
 147  * 
 148  * <h2>Notes</h2> 
 149  * 
 150  * <p> 
 151  * This class is immutable.

 548             return false;

 553             return false; 
 554         }

 556             return false; 
 557         } 
 558         if (quoteCharacter == null) {

 560                 return false; 
 561             } 
 562         } else if (!quoteCharacter.equals(other.quoteCharacter)) { 
 563             return false; 
 564         } 
 565         if (commentMarker == null) { 
 566             if (other.commentMarker != null) { 
 567                 return false; 
 568             } 
 569         } else if (!commentMarker.equals(other.commentMarker)) { 
 570             return false; 
 571         } 
 572         if (escapeCharacter == null) { 
 573             if (other.escapeCharacter != null) { 
 574                 return false; 
 575             } 
 576         } else if (!escapeCharacter.equals(other.escapeCharacter)) { 
 577             return false; 
 578         }

 580             if (other.nullString != null) { 
 581                 return false; 
 582             } 
 583         } else if (!nullString.equals(other.nullString)) { 
 584             return false; 
 585         } 
 586         if (!Arrays.equals(header, other.header)) {

 588         }

 590             return false; 
 591         } 
 592         if (ignoreEmptyLines != other.ignoreEmptyLines) { 
 593             return false; 
 594         }

 597         }

 600                 return false; 
 601             } 
 602         } else if (!recordSeparator.equals(other.recordSeparator)) { 
 603             return false; 
 604         } 
 605         return true;

 619             return out.toString().trim();

 622             throw new IllegalStateException(e); 
 623         }

 789         result = prime * result + ((nullString == null) ? 0 : nullString.hashCode()); 
 790         result = prime * result + (ignoreSurroundingSpaces ? 1231 : 1237); 
 791         result = prime * result + (ignoreHeaderCase ? 1231 : 1237); 
 792         result = prime * result + (ignoreEmptyLines ? 1231 : 1237); 
 793         result = prime * result + (skipHeaderRecord ? 1231 : 1237); 
 794         result = prime * result + ((recordSeparator == null) ? 0 : recordSeparator.hashCode()); 
 795         result = prime * result + Arrays.hashCode(header); 
 796         return result; 
 797     }

 940     private void print(final Object object, final CharSequence value, final int offset, final int len, 
 941             final Appendable out, final boolean newRecord) throws IOException { 
 942         if (!newRecord) { 
 943             out.append(getDelimiter()); 
 944         } 
 945         if (object == null) { 
 946             out.append(value); 
 947         } else if (isQuoteCharacterSet()) { 
 948             // the original object is needed so can check for Number 
 949             printAndQuote(object, value, offset, len, out, newRecord); 
 950         } else if (isEscapeCharacterSet()) { 
 951             printAndEscape(value, offset, len, out); 
 952         } else { 
 953             out.append(value, offset, offset + len); 
 954         } 
 955     }

 957     /* 
 958      * Note: must only be called if escaping is enabled, otherwise will generate NPE 
 959      */ 
 960     private void printAndEscape(final CharSequence value, final int offset, final int len, final Appendable out) 
 961             throws IOException { 
 962         int start = offset; 
 963         int pos = offset; 
 964         final int end = offset + len; 
 965  
 966         final char delim = getDelimiter(); 
 967         final char escape = getEscapeCharacter().charValue(); 
 968  
 969         while (pos < end) { 
 970             char c = value.charAt(pos); 
 971             if (c == CR || c == LF || c == delim || c == escape) { 
 972                 // write out segment up until this char 
 973                 if (pos > start) { 
 974                     out.append(value, start, pos); 
 975                 } 
 976                 if (c == LF) { 
 977                     c = 'n'; 
 978                 } else if (c == CR) { 
 979                     c = 'r'; 
 980                 } 
 981  
 982                 out.append(escape); 
 983                 out.append(c); 
 984  
 985                 start = pos + 1; // start on the current char after this one 
 986             } 
 987  
 988             pos++; 
 989         } 
 990  
 991         // write last segment 
 992         if (pos > start) { 
 993             out.append(value, start, pos); 
 994         } 
 995     }

 997     /* 
 998      * Note: must only be called if quoting is enabled, otherwise will generate NPE 
 999      */ 
1000     // the original object is needed so can check for Number 
1001     private void printAndQuote(final Object object, final CharSequence value, final int offset, final int len, 
1002             final Appendable out, final boolean newRecord) throws IOException { 
1003         boolean quote = false; 
1004         int start = offset; 
1005         int pos = offset; 
1006         final int end = offset + len; 
1007  
1008         final char delimChar = getDelimiter(); 
1009         final char quoteChar = getQuoteCharacter().charValue(); 
1010  
1011         QuoteMode quoteModePolicy = getQuoteMode(); 
1012         if (quoteModePolicy == null) { 
1013             quoteModePolicy = QuoteMode.MINIMAL; 
1014         } 
1015         switch (quoteModePolicy) { 
1016         case ALL: 
1017             quote = true; 
1018             break; 
1019         case NON_NUMERIC: 
1020             quote = !(object instanceof Number); 
1021             break; 
1022         case NONE: 
1023             // Use the existing escaping code 
1024             printAndEscape(value, offset, len, out); 
1025             return; 
1026         case MINIMAL: 
1027             if (len <= 0) { 
1028                 // always quote an empty token that is the first 
1029                 // on the line, as it may be the only thing on the 
1030                 // line. If it were not quoted in that case, 
1031                 // an empty line has no tokens. 
1032                 if (newRecord) { 
1033                     quote = true; 
1034                 } 
1035             } else { 
1036                 char c = value.charAt(pos); 
1037  
1038                 // RFC4180 (https://tools.ietf.org/html/rfc4180) TEXTDATA =  %x20-21 / %x23-2B / %x2D-7E 
1039                 if (newRecord && (c < 0x20 || c > 0x21 && c < 0x23 || c > 0x2B && c < 0x2D || c > 0x7E)) { 
1040                     quote = true; 
1041                 } else if (c <= COMMENT) { 
1042                     // Some other chars at the start of a value caused the parser to fail, so for now 
1043                     // encapsulate if we start in anything less than '#'. We are being conservative 
1044                     // by including the default comment char too. 
1045                     quote = true; 
1046                 } else { 
1047                     while (pos < end) { 
1048                         c = value.charAt(pos); 
1049                         if (c == LF || c == CR || c == quoteChar || c == delimChar) { 
1050                             quote = true; 
1051                             break; 
1052                         } 
1053                         pos++; 
1054                     } 
1055  
1056                     if (!quote) { 
1057                         pos = end - 1; 
1058                         c = value.charAt(pos); 
1059                         // Some other chars at the end caused the parser to fail, so for now 
1060                         // encapsulate if we end in anything less than ' ' 
1061                         if (c <= SP) { 
1062                             quote = true; 
1063                         } 
1064                     } 
1065                 } 
1066             } 
1067  
1068             if (!quote) { 
1069                 // no encapsulation needed - write out the original value 
1070                 out.append(value, start, end); 
1071                 return; 
1072             } 
1073             break; 
1074         default: 
1075             throw new IllegalStateException("Unexpected Quote value: " + quoteModePolicy); 
1076         } 
1077  
1078         if (!quote) { 
1079             // no encapsulation needed - write out the original value 
1080             out.append(value, start, end); 
1081             return; 
1082         } 
1083  
1084         // we hit something that needed encapsulation 
1085         out.append(quoteChar); 
1086  
1087         // Pick up where we left off: pos should be positioned on the first character that caused 
1088         // the need for encapsulation. 
1089         while (pos < end) { 
1090             final char c = value.charAt(pos); 
1091             if (c == quoteChar) { 
1092                 // write out the chunk up until this point 
1093  
1094                 // add 1 to the length to write out the encapsulator also 
1095                 out.append(value, start, pos + 1); 
1096                 // put the next starting position on the encapsulator so we will 
1097                 // write it out again with the next string (effectively doubling it) 
1098                 start = pos; 
1099             } 
1100             pos++; 
1101         } 
1102  
1103         // write the last segment 
1104         out.append(value, start, pos); 
1105         out.append(quoteChar); 
1106     }

1207     private CharSequence trim(final CharSequence charSequence) { 
1208         if (charSequence instanceof String) { 
1209             return ((String) charSequence).trim(); 
1210         } 
1211         final int count = charSequence.length(); 
1212         int len = count; 
1213         int pos = 0; 
1214  
1215         while (pos < len && charSequence.charAt(pos) <= SP) { 
1216             pos++; 
1217         } 
1218         while (pos < len && charSequence.charAt(len - 1) <= SP) { 
1219             len--; 
1220         } 
1221         return pos > 0 || len < count ? charSequence.subSequence(pos, len) : charSequence; 
1222     }

1224     /** 
1225      * Verifies the consistency of the parameters and throws an IllegalArgumentException if necessary. 
1226      * 
1227      * @throws IllegalArgumentException 
1228      */ 
1229     private void validate() throws IllegalArgumentException { 
1230         if (isLineBreak(delimiter)) { 
1231             throw new IllegalArgumentException("The delimiter cannot be a line break"); 
1232         } 
1233  
1234         if (quoteCharacter != null && delimiter == quoteCharacter.charValue()) { 
1235             throw new IllegalArgumentException( 
1236                     "The quoteChar character and the delimiter cannot be the same ('" + quoteCharacter + "')"); 
1237         } 
1238  
1239         if (escapeCharacter != null && delimiter == escapeCharacter.charValue()) { 
1240             throw new IllegalArgumentException( 
1241                     "The escape character and the delimiter cannot be the same ('" + escapeCharacter + "')"); 
1242         } 
1243  
1244         if (commentMarker != null && delimiter == commentMarker.charValue()) { 
1245             throw new IllegalArgumentException( 
1246                     "The comment start character and the delimiter cannot be the same ('" + commentMarker + "')"); 
1247         } 
1248  
1249         if (quoteCharacter != null && quoteCharacter.equals(commentMarker)) { 
1250             throw new IllegalArgumentException( 
1251                     "The comment start character and the quoteChar cannot be the same ('" + commentMarker + "')"); 
1252         } 
1253  
1254         if (escapeCharacter != null && escapeCharacter.equals(commentMarker)) { 
1255             throw new IllegalArgumentException( 
1256                     "The comment start and the escape character cannot be the same ('" + commentMarker + "')"); 
1257         } 
1258  
1259         if (escapeCharacter == null && quoteMode == QuoteMode.NONE) { 
1260             throw new IllegalArgumentException("No quotes mode set but no escape character is set"); 
1261         } 
1262  
1263         // validate header 
1264         if (header != null) { 
1265             final Set<String> dupCheck = new HashSet<>(); 
1266             for (final String hdr : header) { 
1267                 if (!dupCheck.add(hdr)) { 
1268                     throw new IllegalArgumentException( 
1269                             "The header contains a duplicate entry: '" + hdr + "' in " + Arrays.toString(header)); 
1270                 } 
1271             } 
1272         } 
1273     }
