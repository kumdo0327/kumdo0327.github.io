 43 static const struct lys_module * 
 44 ly_schema_resolve_prefix(const struct ly_ctx *UNUSED(ctx), const char *prefix, size_t prefix_len, const void *prefix_data) 
 45 { 
 46     const struct lysp_module *prefix_mod = prefix_data; 
 47     struct lys_module *m = NULL; 
 48     LY_ARRAY_COUNT_TYPE u; 
 49     const char *local_prefix; 
 50  
 51     local_prefix = prefix_mod->is_submod ? ((struct lysp_submodule *)prefix_mod)->prefix : prefix_mod->mod->prefix; 
 52     if (!prefix_len || !ly_strncmp(local_prefix, prefix, prefix_len)) { 
 53         /* it is the prefix of the module itself */ 
 54         m = prefix_mod->mod; 
 55     } 
 56  
 57     /* search in imports */ 
 58     if (!m) { 
 59         LY_ARRAY_FOR(prefix_mod->imports, u) { 
 60             if (!ly_strncmp(prefix_mod->imports[u].prefix, prefix, prefix_len)) { 
 61                 m = prefix_mod->imports[u].module; 
 62                 break; 
 63             } 
 64         } 
 65     } 
 66  
 67     return m; 
 68 }

196         } 
197     } 
198

369 LIBYANG_API_DEF LY_ERR 
370 lyplg_type_parse_dec64(uint8_t fraction_digits, const char *value, size_t value_len, int64_t *ret, struct ly_err_item **err) 
371 { 
372     LY_ERR ret_val; 
373     char *valcopy = NULL; 
374     size_t fraction = 0, size, len = 0, trailing_zeros; 
375     int64_t d; 
376  
377     *err = NULL; 
378  
379     /* consume leading whitespaces */ 
380     for ( ; value_len && isspace(*value); ++value, --value_len) {} 
381  
382     /* parse value */ 
383     if (!value_len) { 
384         return ly_err_new(err, LY_EVALID, LYVE_DATA, NULL, NULL, "Invalid empty decimal64 value."); 
385     } else if (!isdigit(value[len]) && (value[len] != '-') && (value[len] != '+')) { 
386         return ly_err_new(err, LY_EVALID, LYVE_DATA, NULL, NULL, "Invalid %zu. character of decimal64 value \"%.*s\".", 
387                 len + 1, (int)value_len, value); 
388     } 
389  
390     if ((value[len] == '-') || (value[len] == '+')) { 
391         ++len; 
392     } 
393  
394     while (len < value_len && isdigit(value[len])) { 
395         ++len; 
396     } 
397  
398     trailing_zeros = 0; 
399     if ((len < value_len) && ((value[len] != '.') || !isdigit(value[len + 1]))) { 
400         goto decimal; 
401     } 
402     fraction = len; 
403     ++len; 
404     while (len < value_len && isdigit(value[len])) { 
405         if (value[len] == '0') { 
406             ++trailing_zeros; 
407         } else { 
408             trailing_zeros = 0; 
409         } 
410         ++len; 
411     } 
412     len = len - trailing_zeros; 
413  
414 decimal: 
415     if (fraction && (len - 1 - fraction > fraction_digits)) { 
416         return ly_err_new(err, LY_EVALID, LYVE_DATA, NULL, NULL, 
417                 "Value \"%.*s\" of decimal64 type exceeds defined number (%u) of fraction digits.", 
418                 (int)len, value, fraction_digits); 
419     } 
420     if (fraction) { 
421         size = len + (fraction_digits - (len - 1 - fraction)); 
422     } else { 
423         size = len + fraction_digits + 1; 
424     } 
425  
426     if (len + trailing_zeros < value_len) { 
427         /* consume trailing whitespaces to check that there is nothing after it */ 
428         uint64_t u; 
429         for (u = len + trailing_zeros; u < value_len && isspace(value[u]); ++u) {} 
430         if (u != value_len) { 
431             return ly_err_new(err, LY_EVALID, LYVE_DATA, NULL, NULL, 
432                     "Invalid %" PRIu64 ". character of decimal64 value \"%.*s\".", u + 1, (int)value_len, value); 
433         } 
434     } 
435  
436     /* prepare value string without decimal point to easily parse using standard functions */ 
437     valcopy = malloc(size * sizeof *valcopy); 
438     if (!valcopy) { 
439         return ly_err_new(err, LY_EMEM, 0, NULL, NULL, LY_EMEM_MSG); 
440     } 
441  
442     valcopy[size - 1] = '\0'; 
443     if (fraction) { 
444         memcpy(&valcopy[0], &value[0], fraction); 
445         memcpy(&valcopy[fraction], &value[fraction + 1], len - 1 - (fraction)); 
446         /* add trailing zero characters */ 
447         memset(&valcopy[len - 1], '0', fraction_digits - (len - 1 - fraction)); 
448     } else { 
449         memcpy(&valcopy[0], &value[0], len); 
450         /* add trailing zero characters */ 
451         memset(&valcopy[len], '0', fraction_digits); 
452     } 
453  
454     ret_val = lyplg_type_parse_int("decimal64", LY_BASE_DEC, INT64_C(-9223372036854775807) - INT64_C(1), 
455             INT64_C(9223372036854775807), valcopy, size - 1, &d, err); 
456     if (!ret_val && ret) { 
457         *ret = d; 
458     } 
459     free(valcopy); 
460  
461     return ret_val; 
462 }

465 lyplg_type_validate_patterns(struct lysc_pattern **patterns, const char *str, size_t str_len, struct ly_err_item **err) 
466 { 
467     int rc, match_opts; 
468     LY_ARRAY_COUNT_TYPE u; 
469     pcre2_match_data *match_data = NULL; 
470  
471     LY_CHECK_ARG_RET(NULL, str, err, LY_EINVAL); 
472  
473     *err = NULL; 
474  
475     LY_ARRAY_FOR(patterns, u) { 
476         /* match_data needs to be allocated each time because of possible multi-threaded evaluation */ 
477         match_data = pcre2_match_data_create_from_pattern(patterns[u]->code, NULL); 
478         if (!match_data) { 
479             return ly_err_new(err, LY_EMEM, 0, NULL, NULL, LY_EMEM_MSG); 
480         } 
481  
482         match_opts = PCRE2_ANCHORED; 
483 #ifdef PCRE2_ENDANCHORED 
484         /* PCRE2_ENDANCHORED was added in PCRE2 version 10.30 */ 
485         match_opts |= PCRE2_ENDANCHORED; 
486 #endif 
487         rc = pcre2_match(patterns[u]->code, (PCRE2_SPTR)str, str_len, 0, match_opts, match_data, NULL); 
488         pcre2_match_data_free(match_data); 
489  
490         if ((rc != PCRE2_ERROR_NOMATCH) && (rc < 0)) { 
491             PCRE2_UCHAR pcre2_errmsg[LY_PCRE2_MSG_LIMIT] = {0}; 
492             pcre2_get_error_message(rc, pcre2_errmsg, LY_PCRE2_MSG_LIMIT); 
493  
494             return ly_err_new(err, LY_ESYS, 0, NULL, NULL, "%s", (const char *)pcre2_errmsg); 
495         } else if (((rc == PCRE2_ERROR_NOMATCH) && !patterns[u]->inverted) || 
496                 ((rc != PCRE2_ERROR_NOMATCH) && patterns[u]->inverted)) { 
497             char *eapptag = patterns[u]->eapptag ? strdup(patterns[u]->eapptag) : NULL; 
498  
499             if (patterns[u]->emsg) { 
500                 return ly_err_new(err, LY_EVALID, LYVE_DATA, NULL, eapptag, "%s", patterns[u]->emsg); 
501             } else { 
502                 const char *inverted = patterns[u]->inverted ? "inverted " : ""; 
503                 return ly_err_new(err, LY_EVALID, LYVE_DATA, NULL, eapptag, 
504                         LY_ERRMSG_NOPATTERN, (int)str_len, str, inverted, patterns[u]->expr); 
505             } 
506         } 
507     } 
508     return LY_SUCCESS; 
509 }

511 LIBYANG_API_DEF LY_ERR 
512 lyplg_type_validate_range(LY_DATA_TYPE basetype, struct lysc_range *range, int64_t value, const char *strval, 
513         size_t strval_len, struct ly_err_item **err) 
514 { 
515     LY_ARRAY_COUNT_TYPE u; 
516     ly_bool is_length; /* length or range */ 
517  
518     *err = NULL; 
519     is_length = (basetype == LY_TYPE_BINARY || basetype == LY_TYPE_STRING) ? 1 : 0; 
520  
521     LY_ARRAY_FOR(range->parts, u) { 
522         if (basetype < LY_TYPE_DEC64) { 
523             /* unsigned */ 
524             if ((uint64_t)value < range->parts[u].min_u64) { 
525                 char *eapptag = range->eapptag ? strdup(range->eapptag) : NULL; 
526                 if (range->emsg) { 
527                     return ly_err_new(err, LY_EVALID, LYVE_DATA, NULL, eapptag, "%s", range->emsg); 
528                 } else { 
529                     return ly_err_new(err, LY_EVALID, LYVE_DATA, NULL, eapptag, 
530                             is_length ? LY_ERRMSG_NOLENGTH : LY_ERRMSG_NORANGE, (int)strval_len, strval); 
531                 } 
532             } else if ((uint64_t)value <= range->parts[u].max_u64) { 
533                 /* inside the range */ 
534                 return LY_SUCCESS; 
535             } else if (u == LY_ARRAY_COUNT(range->parts) - 1) { 
536                 /* we have the last range part, so the value is out of bounds */ 
537                 char *eapptag = range->eapptag ? strdup(range->eapptag) : NULL; 
538                 if (range->emsg) { 
539                     return ly_err_new(err, LY_EVALID, LYVE_DATA, NULL, eapptag, "%s", range->emsg); 
540                 } else { 
541                     return ly_err_new(err, LY_EVALID, LYVE_DATA, NULL, eapptag, 
542                             is_length ? LY_ERRMSG_NOLENGTH : LY_ERRMSG_NORANGE, (int)strval_len, strval); 
543                 } 
544             } 
545         } else { 
546             /* signed */ 
547             if (value < range->parts[u].min_64) { 
548                 char *eapptag = range->eapptag ? strdup(range->eapptag) : NULL; 
549                 if (range->emsg) { 
550                     return ly_err_new(err, LY_EVALID, LYVE_DATA, NULL, eapptag, "%s", range->emsg); 
551                 } else { 
552                     return ly_err_new(err, LY_EVALID, LYVE_DATA, NULL, eapptag, LY_ERRMSG_NORANGE, (int)strval_len, strval); 
553                 } 
554             } else if (value <= range->parts[u].max_64) { 
555                 /* inside the range */ 
556                 return LY_SUCCESS; 
557             } else if (u == LY_ARRAY_COUNT(range->parts) - 1) { 
558                 /* we have the last range part, so the value is out of bounds */ 
559                 char *eapptag = range->eapptag ? strdup(range->eapptag) : NULL; 
560                 if (range->emsg) { 
561                     return ly_err_new(err, LY_EVALID, LYVE_DATA, NULL, eapptag, "%s", range->emsg); 
562                 } else { 
563                     return ly_err_new(err, LY_EVALID, LYVE_DATA, NULL, eapptag, LY_ERRMSG_NORANGE, (int)strval_len, strval); 
564                 } 
565             } 
566         } 
567     } 
568  
569     return LY_SUCCESS; 
570 }

650                     lys_datatype2str(type), (int)value_len, value);

652         *base = type_get_hints_base(hints);

654     case LY_TYPE_STRING: 
655     case LY_TYPE_DEC64:

659     case LY_TYPE_IDENT:

675                     lys_datatype2str(type), (int)value_len, value);

679     case LY_TYPE_LEAFREF: 
680     case LY_TYPE_UNION:

687 LIBYANG_API_DEF LY_ERR 
688 lyplg_type_check_status(const struct lysc_node *ctx_node, uint16_t val_flags, LY_VALUE_FORMAT format, void *prefix_data, 
689         const char *val_name, struct ly_err_item **err) 
690 { 
691     LY_ERR ret; 
692     const struct lys_module *mod2; 
693     uint16_t flg1, flg2; 
694  
695     if (format != LY_VALUE_SCHEMA) { 
696         /* nothing/unable to check */ 
697         return LY_SUCCESS; 
698     } 
699  
700     mod2 = ((struct lysp_module *)prefix_data)->mod; 
701  
702     if (mod2 == ctx_node->module) { 
703         /* use flags of the context node since the definition is local */ 
704         flg1 = (ctx_node->flags & LYS_STATUS_MASK) ? (ctx_node->flags & LYS_STATUS_MASK) : LYS_STATUS_CURR; 
705     } else { 
706         /* definition is foreign (deviation, refine), always current */ 
707         flg1 = LYS_STATUS_CURR; 
708     } 
709     flg2 = (val_flags & LYS_STATUS_MASK) ? (val_flags & LYS_STATUS_MASK) : LYS_STATUS_CURR; 
710  
711     if ((flg1 < flg2) && (ctx_node->module == mod2)) { 
712         ret = ly_err_new(err, LY_EVALID, LYVE_REFERENCE, NULL, NULL, 
713                 "A %s definition \"%s\" is not allowed to reference %s value \"%s\".", 
714                 flg1 == LYS_STATUS_CURR ? "current" : "deprecated", ctx_node->name, 
715                 flg2 == LYS_STATUS_OBSLT ? "obsolete" : "deprecated", val_name); 
716         return ret; 
717     } 
718  
719     return LY_SUCCESS; 
720 }

762 LIBYANG_API_DEF LY_ERR 
763 lyplg_type_lypath_new(const struct ly_ctx *ctx, const char *value, size_t value_len, uint32_t options, 
764         LY_VALUE_FORMAT format, void *prefix_data, const struct lysc_node *ctx_node, struct lys_glob_unres *unres, 
765         struct ly_path **path, struct ly_err_item **err) 
766 { 
767     LY_ERR ret = LY_SUCCESS; 
768     struct lyxp_expr *exp = NULL; 
769     uint32_t prefix_opt = 0; 
770  
771     LY_CHECK_ARG_RET(ctx, ctx, value, ctx_node, path, err, LY_EINVAL); 
772  
773     *path = NULL; 
774     *err = NULL; 
775  
776     switch (format) { 
777     case LY_VALUE_SCHEMA: 
778     case LY_VALUE_SCHEMA_RESOLVED: 
779     case LY_VALUE_XML: 
780         prefix_opt = LY_PATH_PREFIX_MANDATORY; 
781         break; 
782     case LY_VALUE_CANON: 
783     case LY_VALUE_LYB: 
784     case LY_VALUE_JSON: 
785         prefix_opt = LY_PATH_PREFIX_STRICT_INHERIT; 
786         break; 
787     } 
788  
789     /* parse the value */ 
790     ret = ly_path_parse(ctx, ctx_node, value, value_len, 0, LY_PATH_BEGIN_ABSOLUTE, prefix_opt, LY_PATH_PRED_SIMPLE, &exp); 
791     if (ret) { 
792         ret = ly_err_new(err, LY_EVALID, LYVE_DATA, NULL, NULL, 
793                 "Invalid instance-identifier \"%.*s\" value - syntax error.", (int)value_len, value); 
794         goto cleanup; 
795     } 
796  
797     if (options & LYPLG_TYPE_STORE_IMPLEMENT) { 
798         /* implement all prefixes */ 
799         LY_CHECK_GOTO(ret = lys_compile_expr_implement(ctx, exp, format, prefix_data, 1, unres, NULL), cleanup); 
800     } 
801  
802     /* resolve it on schema tree */ 
803     ret = ly_path_compile(ctx, NULL, ctx_node, NULL, exp, (ctx_node->flags & LYS_IS_OUTPUT) ? 
804             LY_PATH_OPER_OUTPUT : LY_PATH_OPER_INPUT, LY_PATH_TARGET_SINGLE, 1, format, prefix_data, path); 
805     if (ret) { 
806         ret = ly_err_new(err, ret, LYVE_DATA, NULL, NULL, 
807                 "Invalid instance-identifier \"%.*s\" value - semantic error.", (int)value_len, value); 
808         goto cleanup; 
809     } 
810  
811 cleanup: 
812     lyxp_expr_free(ctx, exp); 
813     if (ret) { 
814         ly_path_free(ctx, *path); 
815         *path = NULL; 
816     } 
817  
818     return ret; 
819 }

857 lyplg_type_resolve_leafref(const struct lysc_type_leafref *lref, const struct lyd_node *node, struct lyd_value *value, 
858         const struct lyd_node *tree, struct lyd_node **target, char **errmsg) 
859 { 
860     LY_ERR ret; 
861     struct lyxp_set set = {0}; 
862     const char *val_str; 
863     uint32_t i; 
864     int rc; 
865  
866     LY_CHECK_ARG_RET(NULL, lref, node, value, errmsg, LY_EINVAL); 
867  
868     /* find all target data instances */ 
869     ret = lyxp_eval(LYD_CTX(node), lref->path, node->schema->module, LY_VALUE_SCHEMA_RESOLVED, lref->prefixes, 
870             node, tree, NULL, &set, LYXP_IGNORE_WHEN); 
871     if (ret) { 
872         ret = LY_ENOTFOUND; 
873         val_str = lref->plugin->print(LYD_CTX(node), value, LY_VALUE_CANON, NULL, NULL, NULL); 
874         if (asprintf(errmsg, "Invalid leafref value \"%s\" - XPath evaluation error.", val_str) == -1) { 
875             *errmsg = NULL; 
876             ret = LY_EMEM; 
877         } 
878         goto error; 
879     } 
880  
881     /* check whether any matches */ 
882     for (i = 0; i < set.used; ++i) { 
883         if (set.val.nodes[i].type != LYXP_NODE_ELEM) { 
884             continue; 
885         } 
886  
887         if (!lref->plugin->compare(&((struct lyd_node_term *)set.val.nodes[i].node)->value, value)) { 
888             break; 
889         } 
890     } 
891     if (i == set.used) { 
892         ret = LY_ENOTFOUND; 
893         val_str = lref->plugin->print(LYD_CTX(node), value, LY_VALUE_CANON, NULL, NULL, NULL); 
894         if (set.used) { 
895             rc = asprintf(errmsg, LY_ERRMSG_NOLREF_VAL, val_str, lref->path->expr); 
896         } else { 
897             rc = asprintf(errmsg, LY_ERRMSG_NOLREF_INST, val_str, lref->path->expr); 
898         } 
899         if (rc == -1) { 
900             *errmsg = NULL; 
901             ret = LY_EMEM; 
902         } 
903         goto error; 
904     } 
905  
906     if (target) { 
907         *target = set.val.nodes[i].node; 
908     } 
909  
910     lyxp_set_free_content(&set); 
911     return LY_SUCCESS; 
912  
913 error: 
914     lyxp_set_free_content(&set); 
915     return ret; 
916 }

 44 ly_schema_resolve_prefix(const struct ly_ctx *UNUSED(ctx), const char *prefix, size_t prefix_len, const void *prefix_data) 
 45 { 
 46     const struct lysp_module *prefix_mod = prefix_data; 
 47     struct lys_module *m = NULL; 
 48     LY_ARRAY_COUNT_TYPE u; 
 49     const char *local_prefix; 
 50  
 51     local_prefix = prefix_mod->is_submod ? ((struct lysp_submodule *)prefix_mod)->prefix : prefix_mod->mod->prefix; 
 52     if (!prefix_len || !ly_strncmp(local_prefix, prefix, prefix_len)) { 
 53         /* it is the prefix of the module itself */ 
 54         m = prefix_mod->mod; 
 55     } 
 56  
 57     /* search in imports */ 
 58     if (!m) { 
 59         LY_ARRAY_FOR(prefix_mod->imports, u) { 
 60             if (!ly_strncmp(prefix_mod->imports[u].prefix, prefix, prefix_len)) { 
 61                 m = prefix_mod->imports[u].module; 
 62                 break; 
 63             } 
 64         } 
 65     } 
 66  
 67     return m; 
 68 }

371 LIBYANG_API_DEF LY_ERR 
372 lyplg_type_parse_dec64(uint8_t fraction_digits, const char *value, size_t value_len, int64_t *ret, struct ly_err_item **err) 
373 { 
374     LY_ERR ret_val; 
375     char *valcopy = NULL; 
376     size_t fraction = 0, size, len = 0, trailing_zeros; 
377     int64_t d; 
378  
379     *err = NULL; 
380  
381     /* consume leading whitespaces */ 
382     for ( ; value_len && isspace(*value); ++value, --value_len) {} 
383  
384     /* parse value */ 
385     if (!value_len) { 
386         return ly_err_new(err, LY_EVALID, LYVE_DATA, NULL, NULL, "Invalid empty decimal64 value."); 
387     } else if (!isdigit(value[len]) && (value[len] != '-') && (value[len] != '+')) { 
388         return ly_err_new(err, LY_EVALID, LYVE_DATA, NULL, NULL, "Invalid %zu. character of decimal64 value \"%.*s\".", 
389                 len + 1, (int)value_len, value); 
390     } 
391  
392     if ((value[len] == '-') || (value[len] == '+')) { 
393         ++len; 
394     } 
395  
396     while (len < value_len && isdigit(value[len])) { 
397         ++len; 
398     } 
399  
400     trailing_zeros = 0; 
401     if ((len < value_len) && ((value[len] != '.') || !isdigit(value[len + 1]))) { 
402         goto decimal; 
403     } 
404     fraction = len; 
405     ++len; 
406     while (len < value_len && isdigit(value[len])) { 
407         if (value[len] == '0') { 
408             ++trailing_zeros; 
409         } else { 
410             trailing_zeros = 0; 
411         } 
412         ++len; 
413     } 
414     len = len - trailing_zeros; 
415  
416 decimal: 
417     if (fraction && (len - 1 - fraction > fraction_digits)) { 
418         return ly_err_new(err, LY_EVALID, LYVE_DATA, NULL, NULL, 
419                 "Value \"%.*s\" of decimal64 type exceeds defined number (%u) of fraction digits.", 
420                 (int)len, value, fraction_digits); 
421     } 
422     if (fraction) { 
423         size = len + (fraction_digits - (len - 1 - fraction)); 
424     } else { 
425         size = len + fraction_digits + 1; 
426     } 
427  
428     if (len + trailing_zeros < value_len) { 
429         /* consume trailing whitespaces to check that there is nothing after it */ 
430         uint64_t u; 
431         for (u = len + trailing_zeros; u < value_len && isspace(value[u]); ++u) {} 
432         if (u != value_len) { 
433             return ly_err_new(err, LY_EVALID, LYVE_DATA, NULL, NULL, 
434                     "Invalid %" PRIu64 ". character of decimal64 value \"%.*s\".", u + 1, (int)value_len, value); 
435         } 
436     } 
437  
438     /* prepare value string without decimal point to easily parse using standard functions */ 
439     valcopy = malloc(size * sizeof *valcopy); 
440     if (!valcopy) { 
441         return ly_err_new(err, LY_EMEM, 0, NULL, NULL, LY_EMEM_MSG); 
442     } 
443  
444     valcopy[size - 1] = '\0'; 
445     if (fraction) { 
446         memcpy(&valcopy[0], &value[0], fraction); 
447         memcpy(&valcopy[fraction], &value[fraction + 1], len - 1 - (fraction)); 
448         /* add trailing zero characters */ 
449         memset(&valcopy[len - 1], '0', fraction_digits - (len - 1 - fraction)); 
450     } else { 
451         memcpy(&valcopy[0], &value[0], len); 
452         /* add trailing zero characters */ 
453         memset(&valcopy[len], '0', fraction_digits); 
454     } 
455  
456     ret_val = lyplg_type_parse_int("decimal64", LY_BASE_DEC, INT64_C(-9223372036854775807) - INT64_C(1), 
457             INT64_C(9223372036854775807), valcopy, size - 1, &d, err); 
458     if (!ret_val && ret) { 
459         *ret = d; 
460     } 
461     free(valcopy); 
462  
463     return ret_val; 
464 }

466 LIBYANG_API_DEF LY_ERR 
467 lyplg_type_validate_patterns(struct lysc_pattern **patterns, const char *str, size_t str_len, struct ly_err_item **err) 
468 { 
469     int rc, match_opts; 
470     LY_ARRAY_COUNT_TYPE u; 
471     pcre2_match_data *match_data = NULL; 
472  
473     LY_CHECK_ARG_RET(NULL, str, err, LY_EINVAL); 
474  
475     *err = NULL; 
476  
477     LY_ARRAY_FOR(patterns, u) { 
478         /* match_data needs to be allocated each time because of possible multi-threaded evaluation */ 
479         match_data = pcre2_match_data_create_from_pattern(patterns[u]->code, NULL); 
480         if (!match_data) { 
481             return ly_err_new(err, LY_EMEM, 0, NULL, NULL, LY_EMEM_MSG); 
482         } 
483  
484         match_opts = PCRE2_ANCHORED; 
485 #ifdef PCRE2_ENDANCHORED 
486         /* PCRE2_ENDANCHORED was added in PCRE2 version 10.30 */ 
487         match_opts |= PCRE2_ENDANCHORED; 
488 #endif 
489         rc = pcre2_match(patterns[u]->code, (PCRE2_SPTR)str, str_len, 0, match_opts, match_data, NULL); 
490         pcre2_match_data_free(match_data); 
491  
492         if ((rc != PCRE2_ERROR_NOMATCH) && (rc < 0)) { 
493             PCRE2_UCHAR pcre2_errmsg[LY_PCRE2_MSG_LIMIT] = {0}; 
494             pcre2_get_error_message(rc, pcre2_errmsg, LY_PCRE2_MSG_LIMIT); 
495  
496             return ly_err_new(err, LY_ESYS, 0, NULL, NULL, "%s", (const char *)pcre2_errmsg); 
497         } else if (((rc == PCRE2_ERROR_NOMATCH) && !patterns[u]->inverted) || 
498                 ((rc != PCRE2_ERROR_NOMATCH) && patterns[u]->inverted)) { 
499             char *eapptag = patterns[u]->eapptag ? strdup(patterns[u]->eapptag) : NULL; 
500  
501             if (patterns[u]->emsg) { 
502                 return ly_err_new(err, LY_EVALID, LYVE_DATA, NULL, eapptag, "%s", patterns[u]->emsg); 
503             } else { 
504                 const char *inverted = patterns[u]->inverted ? "inverted " : ""; 
505                 return ly_err_new(err, LY_EVALID, LYVE_DATA, NULL, eapptag, 
506                         LY_ERRMSG_NOPATTERN, (int)str_len, str, inverted, patterns[u]->expr); 
507             } 
508         } 
509     } 
510     return LY_SUCCESS; 
511 }

513 LIBYANG_API_DEF LY_ERR 
514 lyplg_type_validate_range(LY_DATA_TYPE basetype, struct lysc_range *range, int64_t value, const char *strval, 
515         size_t strval_len, struct ly_err_item **err) 
516 { 
517     LY_ARRAY_COUNT_TYPE u; 
518     ly_bool is_length; /* length or range */ 
519  
520     *err = NULL; 
521     is_length = (basetype == LY_TYPE_BINARY || basetype == LY_TYPE_STRING) ? 1 : 0; 
522  
523     LY_ARRAY_FOR(range->parts, u) { 
524         if (basetype < LY_TYPE_DEC64) { 
525             /* unsigned */ 
526             if ((uint64_t)value < range->parts[u].min_u64) { 
527                 char *eapptag = range->eapptag ? strdup(range->eapptag) : NULL; 
528                 if (range->emsg) { 
529                     return ly_err_new(err, LY_EVALID, LYVE_DATA, NULL, eapptag, "%s", range->emsg); 
530                 } else { 
531                     return ly_err_new(err, LY_EVALID, LYVE_DATA, NULL, eapptag, 
532                             is_length ? LY_ERRMSG_NOLENGTH : LY_ERRMSG_NORANGE, (int)strval_len, strval); 
533                 } 
534             } else if ((uint64_t)value <= range->parts[u].max_u64) { 
535                 /* inside the range */ 
536                 return LY_SUCCESS; 
537             } else if (u == LY_ARRAY_COUNT(range->parts) - 1) { 
538                 /* we have the last range part, so the value is out of bounds */ 
539                 char *eapptag = range->eapptag ? strdup(range->eapptag) : NULL; 
540                 if (range->emsg) { 
541                     return ly_err_new(err, LY_EVALID, LYVE_DATA, NULL, eapptag, "%s", range->emsg); 
542                 } else { 
543                     return ly_err_new(err, LY_EVALID, LYVE_DATA, NULL, eapptag, 
544                             is_length ? LY_ERRMSG_NOLENGTH : LY_ERRMSG_NORANGE, (int)strval_len, strval); 
545                 } 
546             } 
547         } else { 
548             /* signed */ 
549             if (value < range->parts[u].min_64) { 
550                 char *eapptag = range->eapptag ? strdup(range->eapptag) : NULL; 
551                 if (range->emsg) { 
552                     return ly_err_new(err, LY_EVALID, LYVE_DATA, NULL, eapptag, "%s", range->emsg); 
553                 } else { 
554                     return ly_err_new(err, LY_EVALID, LYVE_DATA, NULL, eapptag, LY_ERRMSG_NORANGE, (int)strval_len, strval); 
555                 } 
556             } else if (value <= range->parts[u].max_64) { 
557                 /* inside the range */ 
558                 return LY_SUCCESS; 
559             } else if (u == LY_ARRAY_COUNT(range->parts) - 1) { 
560                 /* we have the last range part, so the value is out of bounds */ 
561                 char *eapptag = range->eapptag ? strdup(range->eapptag) : NULL; 
562                 if (range->emsg) { 
563                     return ly_err_new(err, LY_EVALID, LYVE_DATA, NULL, eapptag, "%s", range->emsg); 
564                 } else { 
565                     return ly_err_new(err, LY_EVALID, LYVE_DATA, NULL, eapptag, LY_ERRMSG_NORANGE, (int)strval_len, strval); 
566                 } 
567             } 
568         } 
569     } 
570  
571     return LY_SUCCESS; 
572 }

700     } 
701  
702     mod2 = ((struct lysp_module *)prefix_data)->mod;

707     } else { 
708         /* definition is foreign (deviation, refine), always current */

714         ret = ly_err_new(err, LY_EVALID, LYVE_REFERENCE, NULL, NULL, 
715                 "A %s definition \"%s\" is not allowed to reference %s value \"%s\".", 
716                 flg1 == LYS_STATUS_CURR ? "current" : "deprecated", ctx_node->name, 
717                 flg2 == LYS_STATUS_OBSLT ? "obsolete" : "deprecated", val_name); 
718         return ret; 
719     } 
720  
721     return LY_SUCCESS; 
722 }

811     } 
812

814     lyxp_expr_free(ctx, exp); 
815     if (ret) { 
816         ly_path_free(ctx, *path); 
817         *path = NULL; 
818     } 
819

858 LIBYANG_API_DEF LY_ERR 
859 lyplg_type_resolve_leafref(const struct lysc_type_leafref *lref, const struct lyd_node *node, struct lyd_value *value, 
860         const struct lyd_node *tree, struct lyd_node **target, char **errmsg) 
861 { 
862     LY_ERR ret; 
863     struct lyxp_set set = {0}; 
864     const char *val_str; 
865     uint32_t i; 
866     int rc; 
867  
868     LY_CHECK_ARG_RET(NULL, lref, node, value, errmsg, LY_EINVAL); 
869  
870     /* find all target data instances */ 
871     ret = lyxp_eval(LYD_CTX(node), lref->path, node->schema->module, LY_VALUE_SCHEMA_RESOLVED, lref->prefixes, 
872             node, tree, NULL, &set, LYXP_IGNORE_WHEN); 
873     if (ret) { 
874         ret = LY_ENOTFOUND; 
875         val_str = lref->plugin->print(LYD_CTX(node), value, LY_VALUE_CANON, NULL, NULL, NULL); 
876         if (asprintf(errmsg, "Invalid leafref value \"%s\" - XPath evaluation error.", val_str) == -1) { 
877             *errmsg = NULL; 
878             ret = LY_EMEM; 
879         } 
880         goto error; 
881     } 
882  
883     /* check whether any matches */ 
884     for (i = 0; i < set.used; ++i) { 
885         if (set.val.nodes[i].type != LYXP_NODE_ELEM) { 
886             continue; 
887         } 
888  
889         if (!lref->plugin->compare(&((struct lyd_node_term *)set.val.nodes[i].node)->value, value)) { 
890             break; 
891         } 
892     } 
893     if (i == set.used) { 
894         ret = LY_ENOTFOUND; 
895         val_str = lref->plugin->print(LYD_CTX(node), value, LY_VALUE_CANON, NULL, NULL, NULL); 
896         if (set.used) { 
897             rc = asprintf(errmsg, LY_ERRMSG_NOLREF_VAL, val_str, lref->path->expr); 
898         } else { 
899             rc = asprintf(errmsg, LY_ERRMSG_NOLREF_INST, val_str, lref->path->expr); 
900         } 
901         if (rc == -1) { 
902             *errmsg = NULL; 
903             ret = LY_EMEM; 
904         } 
905         goto error; 
906     } 
907  
908     if (target) { 
909         *target = set.val.nodes[i].node; 
910     } 
911  
912     lyxp_set_free_content(&set); 
913     return LY_SUCCESS; 
914  
915 error: 
916     lyxp_set_free_content(&set); 
917     return ret; 
918 }
