 35     /** 
 36      * Parse an octal string from a buffer. 
 37      * 
 38      * <p>Leading spaces are ignored. 
 39      * The buffer must contain a trailing space or NUL, 
 40      * and may contain an additional trailing space or NUL.</p> 
 41      * 
 42      * <p>The input buffer is allowed to contain all NULs, 
 43      * in which case the method returns 0L 
 44      * (this allows for missing fields).</p> 
 45      * 
 46      * <p>To work-around some tar implementations that insert a 
 47      * leading NUL this method returns 0 if it detects a leading NUL 
 48      * since Commons Compress 1.4.</p> 
 49      * 
 50      * @param buffer The buffer from which to parse. 
 51      * @param offset The offset into the buffer from which to parse. 
 52      * @param length The maximum number of bytes to parse - must be at least 2 bytes. 
 53      * @return The long value of the octal string. 
 54      * @throws IllegalArgumentException if the trailing space/NUL is missing or if a invalid byte is detected. 
 55      */ 
 56     public static long parseOctal(final byte[] buffer, final int offset, final int length) { 
 57         long    result = 0; 
 58         int     end = offset + length; 
 59         int     start = offset; 
 60  
 61         if (length < 2){ 
 62             throw new IllegalArgumentException("Length "+length+" must be at least 2"); 
 63         } 
 64  
 65         boolean allNUL = true; 
 66         for (int i = start; i < end; i++){ 
 67             if (buffer[i] != 0){ 
 68                 allNUL = false; 
 69                 break; 
 70             } 
 71         } 
 72         if (allNUL) { 
 73             return 0L; 
 74         } 
 75  
 76         // Skip leading spaces 
 77         while (start < end){ 
 78             if (buffer[start] == ' '){ 
 79                 start++; 
 80             } else { 
 81                 break; 
 82             } 
 83         } 
 84  
 85         // Must have trailing NUL or space 
 86         byte trailer; 
 87         trailer = buffer[end-1]; 
 88         if (trailer == 0 || trailer == ' '){ 
 89             end--; 
 90         } else { 
 91             throw new IllegalArgumentException( 
 92                     exceptionMessage(buffer, offset, length, end-1, trailer)); 
 93         } 
 94         // May have additional NUL or space 
 95         trailer = buffer[end-1]; 
 96         if (trailer == 0 || trailer == ' '){ 
 97             end--; 
 98         } 
 99  
100         for ( ;start < end; start++) { 
101             final byte currentByte = buffer[start]; 
102             // CheckStyle:MagicNumber OFF 
103             if (currentByte < '0' || currentByte > '7'){ 
104                 throw new IllegalArgumentException( 
105                         exceptionMessage(buffer, offset, length, start, currentByte)); 
106             } 
107             result = (result << 3) + (currentByte - '0'); // convert from ASCII 
108             // CheckStyle:MagicNumber ON 
109         } 
110  
111         return result; 
112     }

114     /**  
115      * Compute the value contained in a byte buffer.  If the most 
116      * significant bit of the first byte in the buffer is set, this 
117      * bit is ignored and the rest of the buffer is interpreted as a 
118      * binary number.  Otherwise, the buffer is interpreted as an 
119      * octal number as per the parseOctal function above. 
120      * 
121      * @param buffer The buffer from which to parse. 
122      * @param offset The offset into the buffer from which to parse. 
123      * @param length The maximum number of bytes to parse. 
124      * @return The long value of the octal or binary string. 
125      * @throws IllegalArgumentException if the trailing space/NUL is 
126      * missing or an invalid byte is detected in an octal number, or 
127      * if a binary number would exceed the size of a signed long 
128      * 64-bit integer. 
129      * @since Apache Commons Compress 1.4 
130      */ 
131     public static long parseOctalOrBinary(final byte[] buffer, final int offset, 
132                                           final int length) { 
133  
134         if ((buffer[offset] & 0x80) == 0) { 
135             return parseOctal(buffer, offset, length); 
136         } 
137  
138         long val = buffer[offset] & 0x7f; 
139         for (int i = 1; i < length; i++) { 
140             if (val >= (1L << (63 - 8))) { 
141                 throw new IllegalArgumentException( 
142                     "At offset " + offset + ", " + length + " byte " + 
143                     "binary number exceeds maximum signed long value"); 
144             } 
145             val = (val << 8) + (buffer[offset + i] & 0xff); 
146         } 
147         return val; 
148     }

229     /** 
230      * Fill buffer with unsigned octal number, padded with leading zeroes. 
231      *  
232      * @param value number to convert to octal - treated as unsigned 
233      * @param buffer destination buffer 
234      * @param offset starting offset in buffer 
235      * @param length length of buffer to fill 
236      * @throws IllegalArgumentException if the value will not fit in the buffer 
237      */ 
238     public static void formatUnsignedOctalString(final long value, byte[] buffer, 
239             final int offset, final int length) { 
240         int remaining = length; 
241         remaining--; 
242         if (value == 0) { 
243             buffer[offset + remaining--] = (byte) '0'; 
244         } else { 
245             long val = value; 
246             for (; remaining >= 0 && val != 0; --remaining) { 
247                 // CheckStyle:MagicNumber OFF 
248                 buffer[offset + remaining] = (byte) ((byte) '0' + (byte) (val & 7)); 
249                 val = val >>> 3; 
250                 // CheckStyle:MagicNumber ON 
251             } 
252             if (val != 0){ 
253                 throw new IllegalArgumentException 
254                 (value+"="+Long.toOctalString(value)+ " will not fit in octal number buffer of length "+length); 
255             } 
256         } 
257  
258         for (; remaining >= 0; --remaining) { // leading zeros 
259             buffer[offset + remaining] = (byte) '0'; 
260         } 
261     }

312     /** 
313      * Write an long integer into a buffer as an octal string if this 
314      * will fit, or as a binary number otherwise. 
315      *  
316      * Uses {@link #formatUnsignedOctalString} to format 
317      * the value as an octal string with leading zeros. 
318      * The converted number is followed by a space. 
319      *  
320      * @param value The value to write into the buffer. 
321      * @param buf The destination buffer. 
322      * @param offset The starting offset into the buffer. 
323      * @param length The length of the buffer. 
324      * @return The updated offset. 
325      * @throws IllegalArgumentException if the value (and trailer) 
326      * will not fit in the buffer. 
327      * @since Apache Commons Compress 1.4 
328      */ 
329     public static int formatLongOctalOrBinaryBytes( 
330         final long value, byte[] buf, final int offset, final int length) { 
331  
332         // Check whether we are dealing with UID/GID or SIZE field 
333         final long maxAsOctalChar = length == TarConstants.UIDLEN ? TarConstants.MAXID : TarConstants.MAXSIZE; 
334  
335         if (value <= maxAsOctalChar) { // OK to store as octal chars 
336             return formatLongOctalBytes(value, buf, offset, length); 
337         } 
338  
339         long val = value; 
340         for (int i = offset + length - 1; i >= offset; i--) { 
341             buf[i] = (byte) val; 
342             val >>= 8; 
343         } 
344  
345         if (val != 0 || (buf[offset] & 0x80) != 0) { 
346             throw new IllegalArgumentException("Value " + value + 
347                 " is too large for " + length + " byte field."); 
348         } 
349  
350         buf[offset] |= 0x80; 
351         return offset + length; 
352     }

371         formatUnsignedOctalString(value, buf, offset, idx);

379     /** 
380      * Compute the checksum of a tar entry header. 
381      * 
382      * @param buf The tar entry's header buffer. 
383      * @return The computed checksum. 
384      */ 
385     public static long computeCheckSum(final byte[] buf) { 
386         long sum = 0; 
387  
388         for (int i = 0; i < buf.length; ++i) { 
389             sum += BYTE_MASK & buf[i]; 
390         } 
391  
392         return sum;

290      * @param offset The starting offset into the buffer. 
291      * @param length The length of the buffer 
292      * @return The updated offset 
293      * @throws IllegalArgumentException if the value (and trailer) will not fit in the buffer 
294      */ 
295     public static int formatLongOctalBytes(final long value, byte[] buf, final int offset, final int length) { 
296  
297         int idx=length-1; // For space 
298          
299         formatUnsignedOctalString(value, buf, offset, idx); 
300         buf[offset + idx] = (byte) ' '; // Trailing space 
301  
302         return offset + length; 
303     }

305     /** 
306      * Write an long integer into a buffer as an octal string if this 
307      * will fit, or as a binary number otherwise. 
308      *  
309      * Uses {@link #formatUnsignedOctalString} to format 
310      * the value as an octal string with leading zeros. 
311      * The converted number is followed by a space. 
312      *  
313      * @param value The value to write into the buffer. 
314      * @param buf The destination buffer. 
315      * @param offset The starting offset into the buffer. 
316      * @param length The length of the buffer. 
317      * @return The updated offset. 
318      * @throws IllegalArgumentException if the value (and trailer) 
319      * will not fit in the buffer. 
320      * @since Apache Commons Compress 1.4 
321      */ 
322     public static int formatLongOctalOrBinaryBytes( 
323         final long value, byte[] buf, final int offset, final int length) { 
324  
325         // Check whether we are dealing with UID/GID or SIZE field 
326         final long maxAsOctalChar = length == TarConstants.UIDLEN ? TarConstants.MAXID : TarConstants.MAXSIZE; 
327  
328         if (value <= maxAsOctalChar) { // OK to store as octal chars 
329             return formatLongOctalBytes(value, buf, offset, length); 
330         } 
331  
332         long val = value; 
333         for (int i = offset + length - 1; i >= offset; i--) { 
334             buf[i] = (byte) val; 
335             val >>= 8; 
336         } 
337  
338         if (val != 0 || (buf[offset] & 0x80) != 0) { 
339             throw new IllegalArgumentException("Value " + value + 
340                 " is too large for " + length + " byte field."); 
341         } 
342  
343         buf[offset] |= 0x80; 
344         return offset + length; 
345     }

347     /** 
348      * Writes an octal value into a buffer. 
349      *  
350      * Uses {@link #formatUnsignedOctalString} to format 
351      * the value as an octal string with leading zeros. 
352      * The converted number is followed by NUL and then space. 
353      * 
354      * @param value The value to convert 
355      * @param buf The destination buffer 
356      * @param offset The starting offset into the buffer. 
357      * @param length The size of the buffer. 
358      * @return The updated value of offset, i.e. offset+length 
359      * @throws IllegalArgumentException if the value (and trailer) will not fit in the buffer 
360      */ 
361     public static int formatCheckSumOctalBytes(final long value, byte[] buf, final int offset, final int length) { 
362  
363         int idx=length-2; // for NUL and space 
364         formatUnsignedOctalString(value, buf, offset, idx); 
365  
366         buf[offset + idx++]   = 0; // Trailing null 
367         buf[offset + idx]     = (byte) ' '; // Trailing space 
368  
369         return offset + length; 
370     }

372     /** 
373      * Compute the checksum of a tar entry header. 
374      * 
375      * @param buf The tar entry's header buffer. 
376      * @return The computed checksum. 
377      */ 
378     public static long computeCheckSum(final byte[] buf) { 
379         long sum = 0; 
380  
381         for (int i = 0; i < buf.length; ++i) { 
382             sum += BYTE_MASK & buf[i]; 
383         } 
384  
385         return sum; 
386     }
