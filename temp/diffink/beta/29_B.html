 332     def _apply_params(self, **kw): 
 333         for name, target in [("gridOn", self.gridline), 
 334                                ("tick1On", self.tick1line), 
 335                                ("tick2On", self.tick2line), 
 336                                ("label1On", self.label1), 
 337                                ("label2On", self.label2)]: 
 338             if name in kw: 
 339                 target.set_visible(kw.pop(name)) 
 340         if any(k in kw for k in ['size', 'width', 'pad', 'tickdir']): 
 341             self._size = kw.pop('size', self._size) 
 342             # Width could be handled outside this block, but it is 
 343             # convenient to leave it here. 
 344             self._width = kw.pop('width', self._width) 
 345             self._base_pad = kw.pop('pad', self._base_pad) 
 346             # apply_tickdir uses _size and _base_pad to make _pad, 
 347             # and also makes _tickmarkers. 
 348             self.apply_tickdir(kw.pop('tickdir', self._tickdir)) 
 349             self.tick1line.set_marker(self._tickmarkers[0]) 
 350             self.tick2line.set_marker(self._tickmarkers[1]) 
 351             for line in (self.tick1line, self.tick2line): 
 352                 line.set_markersize(self._size) 
 353                 line.set_markeredgewidth(self._width) 
 354             # _get_text1_transform uses _pad from apply_tickdir. 
 355             trans = self._get_text1_transform()[0] 
 356             self.label1.set_transform(trans) 
 357             trans = self._get_text2_transform()[0] 
 358             self.label2.set_transform(trans) 
 359         tick_kw = {k: v for k, v in kw.items() if k in ['color', 'zorder']} 
 360         self.tick1line.set(**tick_kw) 
 361         self.tick2line.set(**tick_kw) 
 362         for k, v in tick_kw.items(): 
 363             setattr(self, '_' + k, v) 
 364  
 365         if 'labelrotation' in kw: 
 366             self._set_labelrotation(kw.pop('labelrotation')) 
 367             self.label1.set(rotation=self._labelrotation[1]) 
 368             self.label2.set(rotation=self._labelrotation[1]) 
 369  
 370         label_kw = {k[5:]: v for k, v in kw.items() 
 371                     if k in ['labelsize', 'labelcolor']} 
 372         self.label1.set(**label_kw) 
 373         self.label2.set(**label_kw) 
 374         for k, v in label_kw.items(): 
 375             # for labelsize the text objects covert str ('small') 
 376             # -> points. grab the integer from the `Text` object 
 377             # instead of saving the string representation 
 378             v = getattr(self.label1, 'get_' + k)() 
 379             setattr(self, '_label' + k, v) 
 380  
 381         grid_kw = {k[5:]: v for k, v in kw.items() 
 382                    if k in _gridline_param_names} 
 383         self.gridline.set(**grid_kw) 
 384         for k, v in grid_kw.items(): 
 385             setattr(self, '_grid_' + k, v)

 864         Set appearance parameters for ticks, ticklabels, and gridlines.

1074     def _update_ticks(self): 
1075         """ 
1076         Update ticks (position and labels) using the current data interval of 
1077         the axes.  Return the list of ticks that will be drawn. 
1078         """ 
1079         major_locs = self.get_majorticklocs() 
1080         major_labels = self.major.formatter.format_ticks(major_locs) 
1081         major_ticks = self.get_major_ticks(len(major_locs)) 
1082         self.major.formatter.set_locs(major_locs) 
1083         for tick, loc, label in zip(major_ticks, major_locs, major_labels): 
1084             tick.update_position(loc) 
1085             tick.set_label1(label) 
1086             tick.set_label2(label) 
1087         minor_locs = self.get_minorticklocs() 
1088         minor_labels = self.minor.formatter.format_ticks(minor_locs) 
1089         minor_ticks = self.get_minor_ticks(len(minor_locs)) 
1090         self.minor.formatter.set_locs(minor_locs) 
1091         for tick, loc, label in zip(minor_ticks, minor_locs, minor_labels): 
1092             tick.update_position(loc) 
1093             tick.set_label1(label) 
1094             tick.set_label2(label) 
1095         ticks = [*major_ticks, *minor_ticks] 
1096  
1097         view_low, view_high = self.get_view_interval() 
1098         if view_low > view_high: 
1099             view_low, view_high = view_high, view_low 
1100  
1101         if self._smart_bounds and ticks: 
1102             # handle inverted limits 
1103             data_low, data_high = sorted(self.get_data_interval()) 
1104             locs = np.sort([tick.get_loc() for tick in ticks]) 
1105             if data_low <= view_low: 
1106                 # data extends beyond view, take view as limit 
1107                 ilow = view_low 
1108             else: 
1109                 # data stops within view, take best tick 
1110                 good_locs = locs[locs <= data_low] 
1111                 if len(good_locs): 
1112                     # last tick prior or equal to first data point 
1113                     ilow = good_locs[-1] 
1114                 else: 
1115                     # No ticks (why not?), take first tick 
1116                     ilow = locs[0] 
1117             if data_high >= view_high: 
1118                 # data extends beyond view, take view as limit 
1119                 ihigh = view_high 
1120             else: 
1121                 # data stops within view, take best tick 
1122                 good_locs = locs[locs >= data_high] 
1123                 if len(good_locs): 
1124                     # first tick after or equal to last data point 
1125                     ihigh = good_locs[0] 
1126                 else: 
1127                     # No ticks (why not?), take last tick 
1128                     ihigh = locs[-1] 
1129             ticks = [tick for tick in ticks if ilow <= tick.get_loc() <= ihigh] 
1130  
1131         interval_t = self.get_transform().transform([view_low, view_high]) 
1132  
1133         ticks_to_draw = [] 
1134         for tick in ticks: 
1135             try: 
1136                 loc_t = self.get_transform().transform(tick.get_loc()) 
1137             except AssertionError: 
1138                 # transforms.transform doesn't allow masked values but 
1139                 # some scales might make them, so we need this try/except. 
1140                 pass 
1141             else: 
1142                 if mtransforms._interval_contains_close(interval_t, loc_t): 
1143                     ticks_to_draw.append(tick) 
1144  
1145         return ticks_to_draw

1155         """ 
1156         Return a bounding box that encloses the axis. It only accounts 
1157         tick labels, axis label, and offsetText. 
1158         """

1161  
1162         ticks_to_draw = self._update_ticks()

1168                     ticks_to_draw, renderer)

1175               for a in [self.label, self.offsetText] 
1176               if a.get_visible()), 
1177             *ticklabelBoxes, 
1178             *ticklabelBoxes2,

 862     def set_tick_params(self, which='major', reset=False, **kw): 
 863         """ 
 864         Set appearance parameters for ticks, ticklabels, and gridlines. 
 865  
 866         For documentation of keyword arguments, see 
 867         :meth:`matplotlib.axes.Axes.tick_params`. 
 868         """ 
 869         dicts = []

 877         # automatically get them

 891                 for tick in self.minorTicks:

1074     def _update_ticks(self): 
1075         """ 
1076         Update ticks (position and labels) using the current data interval of 
1077         the axes.  Return the list of ticks that will be drawn. 
1078         """ 
1079         major_locs = self.get_majorticklocs() 
1080         major_labels = self.major.formatter.format_ticks(major_locs) 
1081         major_ticks = self.get_major_ticks(len(major_locs)) 
1082         self.major.formatter.set_locs(major_locs) 
1083         for tick, loc, label in zip(major_ticks, major_locs, major_labels): 
1084             tick.update_position(loc) 
1085             tick.set_label1(label) 
1086             tick.set_label2(label) 
1087         minor_locs = self.get_minorticklocs() 
1088         minor_labels = self.minor.formatter.format_ticks(minor_locs) 
1089         minor_ticks = self.get_minor_ticks(len(minor_locs)) 
1090         self.minor.formatter.set_locs(minor_locs) 
1091         for tick, loc, label in zip(minor_ticks, minor_locs, minor_labels): 
1092             tick.update_position(loc) 
1093             tick.set_label1(label) 
1094             tick.set_label2(label) 
1095         ticks = [*major_ticks, *minor_ticks] 
1096  
1097         view_low, view_high = self.get_view_interval() 
1098         if view_low > view_high: 
1099             view_low, view_high = view_high, view_low 
1100  
1101         if self._smart_bounds and ticks: 
1102             # handle inverted limits 
1103             data_low, data_high = sorted(self.get_data_interval()) 
1104             locs = np.sort([tick.get_loc() for tick in ticks]) 
1105             if data_low <= view_low: 
1106                 # data extends beyond view, take view as limit 
1107                 ilow = view_low 
1108             else: 
1109                 # data stops within view, take best tick 
1110                 good_locs = locs[locs <= data_low] 
1111                 if len(good_locs): 
1112                     # last tick prior or equal to first data point 
1113                     ilow = good_locs[-1] 
1114                 else: 
1115                     # No ticks (why not?), take first tick 
1116                     ilow = locs[0] 
1117             if data_high >= view_high: 
1118                 # data extends beyond view, take view as limit 
1119                 ihigh = view_high 
1120             else: 
1121                 # data stops within view, take best tick 
1122                 good_locs = locs[locs >= data_high] 
1123                 if len(good_locs): 
1124                     # first tick after or equal to last data point 
1125                     ihigh = good_locs[0] 
1126                 else: 
1127                     # No ticks (why not?), take last tick 
1128                     ihigh = locs[-1] 
1129             ticks = [tick for tick in ticks if ilow <= tick.get_loc() <= ihigh] 
1130  
1131         interval_t = self.get_transform().transform([view_low, view_high]) 
1132  
1133         ticks_to_draw = [] 
1134         for tick in ticks: 
1135             try: 
1136                 loc_t = self.get_transform().transform(tick.get_loc()) 
1137             except AssertionError: 
1138                 # transforms.transform doesn't allow masked values but 
1139                 # some scales might make them, so we need this try/except. 
1140                 pass 
1141             else: 
1142                 if mtransforms._interval_contains_close(interval_t, loc_t): 
1143                     ticks_to_draw.append(tick) 
1144  
1145         return ticks_to_draw

1155         """ 
1156         Return a bounding box that encloses the axis. It only accounts 
1157         tick labels, axis label, and offsetText. 
1158         """ 
1159         if not self.get_visible(): 
1160             return 
1161  
1162         ticks_to_draw = self._update_ticks() 
1163  
1164         self._update_label_position(renderer) 
1165  
1166         # go back to just this axis's tick labels 
1167         ticklabelBoxes, ticklabelBoxes2 = self._get_tick_bboxes( 
1168                     ticks_to_draw, renderer) 
1169  
1170         self._update_offset_text_position(ticklabelBoxes, ticklabelBoxes2) 
1171         self.offsetText.set_text(self.major.formatter.get_offset())

1173         bboxes = [ 
1174             *(a.get_window_extent(renderer) 
1175               for a in [self.label, self.offsetText] 
1176               if a.get_visible()), 
1177             *ticklabelBoxes, 
1178             *ticklabelBoxes2, 
1179         ] 
1180         bboxes = [b for b in bboxes 
1181                   if 0 < b.width < np.inf and 0 < b.height < np.inf] 
1182         if bboxes: 
1183             return mtransforms.Bbox.union(bboxes) 
1184         else: 
1185             return None
