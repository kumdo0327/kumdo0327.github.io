  11 njs_function_t * 
  12 njs_function_alloc(njs_vm_t *vm, njs_function_lambda_t *lambda, 
  13     njs_bool_t async) 
  14 { 
  15     size_t          size; 
  16     njs_object_t    *proto; 
  17     njs_function_t  *function; 
  18  
  19     size = sizeof(njs_function_t) + lambda->nclosures * sizeof(njs_value_t *); 
  20  
  21     function = njs_mp_zalloc(vm->mem_pool, size); 
  22     if (njs_slow_path(function == NULL)) { 
  23         goto fail; 
  24     } 
  25  
  26     /* 
  27      * njs_mp_zalloc() does also: 
  28      *   njs_lvlhsh_init(&function->object.hash); 
  29      *   function->object.__proto__ = NULL; 
  30      */ 
  31  
  32     function->ctor = lambda->ctor; 
  33     function->args_offset = 1; 
  34     function->u.lambda = lambda; 
  35  
  36     if (function->ctor) { 
  37         function->object.shared_hash = vm->shared->function_instance_hash; 
  38  
  39     } else if (async) { 
  40         function->object.shared_hash = vm->shared->async_function_instance_hash; 
  41  
  42     } else { 
  43         function->object.shared_hash = vm->shared->arrow_instance_hash; 
  44     } 
  45  
  46     if (async) { 
  47         proto = &vm->prototypes[NJS_OBJ_TYPE_ASYNC_FUNCTION].object; 
  48  
  49     } else { 
  50         proto = &vm->prototypes[NJS_OBJ_TYPE_FUNCTION].object; 
  51     } 
  52  
  53     function->object.__proto__ = proto; 
  54     function->object.type = NJS_FUNCTION; 
  55  
  56     function->object.extensible = 1; 
  57  
  58     return function; 
  59  
  60 fail: 
  61  
  62     njs_memory_error(vm); 
  63  
  64     return NULL; 
  65 }

 123 njs_int_t 
 124 njs_function_name_set(njs_vm_t *vm, njs_function_t *function, 
 125     njs_value_t *name, const char *prefix) 
 126 { 
 127     u_char              *p; 
 128     size_t              len, symbol; 
 129     njs_int_t           ret; 
 130     njs_value_t         value; 
 131     njs_string_prop_t   string; 
 132     njs_object_prop_t   *prop; 
 133     njs_lvlhsh_query_t  lhq; 
 134  
 135     prop = njs_object_prop_alloc(vm, &njs_string_name, name, 0); 
 136     if (njs_slow_path(prop == NULL)) { 
 137         return NJS_ERROR; 
 138     } 
 139  
 140     symbol = 0; 
 141  
 142     if (njs_is_symbol(&prop->value)) { 
 143         symbol = 2; 
 144         prop->value = *njs_symbol_description(&prop->value); 
 145     } 
 146  
 147     if (prefix != NULL || symbol != 0) { 
 148         value = prop->value; 
 149         (void) njs_string_prop(&string, &value); 
 150  
 151         len = (prefix != NULL) ? njs_strlen(prefix) + 1: 0; 
 152         p = njs_string_alloc(vm, &prop->value, string.size + len + symbol, 
 153                              string.length + len + symbol); 
 154         if (njs_slow_path(p == NULL)) { 
 155             return NJS_ERROR; 
 156         } 
 157  
 158         if (len != 0) { 
 159             p = njs_cpymem(p, prefix, len - 1); 
 160             *p++ = ' '; 
 161         } 
 162  
 163         if (symbol != 0) { 
 164             *p++ = '['; 
 165         } 
 166  
 167         p = njs_cpymem(p, string.start, string.size); 
 168  
 169         if (symbol != 0) { 
 170             *p++ = ']'; 
 171         } 
 172     } 
 173  
 174     prop->configurable = 1; 
 175  
 176     lhq.key_hash = NJS_NAME_HASH; 
 177     lhq.key = njs_str_value("name"); 
 178     lhq.replace = 0; 
 179     lhq.value = prop; 
 180     lhq.proto = &njs_object_hash_proto; 
 181     lhq.pool = vm->mem_pool; 
 182  
 183     ret = njs_lvlhsh_insert(&function->object.hash, &lhq); 
 184     if (njs_slow_path(ret != NJS_OK)) { 
 185         njs_internal_error(vm, "lvlhsh insert failed"); 
 186         return NJS_ERROR; 
 187     } 
 188  
 189     return NJS_OK; 
 190 }

 194 njs_function_copy(njs_vm_t *vm, njs_function_t *function) 
 195 { 
 196     size_t             size, n; 
 197     njs_value_t        **from, **to; 
 198     njs_function_t     *copy; 
 199     njs_object_type_t  type; 
 200  
 201     n = (function->native) ? 0 : function->u.lambda->nclosures; 
 202  
 203     size = sizeof(njs_function_t) + n * sizeof(njs_value_t *); 
 204  
 205     copy = njs_mp_alloc(vm->mem_pool, size); 
 206     if (njs_slow_path(copy == NULL)) { 
 207         return NULL; 
 208     } 
 209  
 210     *copy = *function; 
 211  
 212     type = njs_function_object_type(vm, function); 
 213  
 214     copy->object.__proto__ = &vm->prototypes[type].object; 
 215     copy->object.shared = 0; 
 216  
 217     if (n == 0) { 
 218         return copy; 
 219     } 
 220  
 221     from = njs_function_closures(function); 
 222     to = njs_function_closures(copy); 
 223  
 224     do { 
 225         n--; 
 226  
 227         to[n] = from[n]; 
 228  
 229     } while (n != 0); 
 230  
 231     return copy; 
 232 }

 236 njs_function_arguments_object_init(njs_vm_t *vm, njs_native_frame_t *frame) 
 237 { 
 238     njs_int_t           ret; 
 239     njs_uint_t          nargs, n; 
 240     njs_value_t         value; 
 241     njs_object_t        *arguments; 
 242     njs_object_prop_t   *prop; 
 243     njs_lvlhsh_query_t  lhq; 
 244  
 245     static const njs_value_t  njs_string_length = njs_string("length"); 
 246  
 247     arguments = njs_object_alloc(vm); 
 248     if (njs_slow_path(arguments == NULL)) { 
 249         return NJS_ERROR; 
 250     } 
 251  
 252     arguments->shared_hash = vm->shared->arguments_object_instance_hash; 
 253  
 254     nargs = frame->nargs; 
 255  
 256     njs_set_number(&value, nargs); 
 257  
 258     prop = njs_object_prop_alloc(vm, &njs_string_length, &value, 1); 
 259     if (njs_slow_path(prop == NULL)) { 
 260         return NJS_ERROR; 
 261     } 
 262  
 263     prop->enumerable = 0; 
 264  
 265     lhq.value = prop; 
 266     lhq.key_hash = NJS_LENGTH_HASH; 
 267     njs_string_get(&prop->name, &lhq.key); 
 268  
 269     lhq.replace = 0; 
 270     lhq.pool = vm->mem_pool; 
 271     lhq.proto = &njs_object_hash_proto; 
 272  
 273     ret = njs_lvlhsh_insert(&arguments->hash, &lhq); 
 274     if (njs_slow_path(ret != NJS_OK)) { 
 275         njs_internal_error(vm, "lvlhsh insert failed"); 
 276         return NJS_ERROR; 
 277     } 
 278  
 279     for (n = 0; n < nargs; n++) { 
 280         njs_uint32_to_string(&value, n); 
 281  
 282         prop = njs_object_prop_alloc(vm, &value, &frame->arguments[n], 1); 
 283         if (njs_slow_path(prop == NULL)) { 
 284             return NJS_ERROR; 
 285         } 
 286  
 287         lhq.value = prop; 
 288         njs_string_get(&prop->name, &lhq.key); 
 289         lhq.key_hash = njs_djb_hash(lhq.key.start, lhq.key.length); 
 290  
 291         ret = njs_lvlhsh_insert(&arguments->hash, &lhq); 
 292         if (njs_slow_path(ret != NJS_OK)) { 
 293             njs_internal_error(vm, "lvlhsh insert failed"); 
 294             return NJS_ERROR; 
 295         } 
 296     } 
 297  
 298     frame->arguments_object = arguments; 
 299  
 300     return NJS_OK; 
 301 }

 305 njs_function_rest_parameters_init(njs_vm_t *vm, njs_native_frame_t *frame) 
 306 { 
 307     uint32_t     length; 
 308     njs_uint_t   nargs, n, i; 
 309     njs_array_t  *array; 
 310     njs_value_t  *rest_arguments; 
 311  
 312     nargs = frame->nargs; 
 313     n = frame->function->u.lambda->nargs; 
 314     length = (nargs >= n) ? (nargs - n + 1) : 0; 
 315  
 316     array = njs_array_alloc(vm, 1, length, 0); 
 317     if (njs_slow_path(array == NULL)) { 
 318         return NJS_ERROR; 
 319     } 
 320  
 321     for (i = 0; i < length; i++) { 
 322         array->start[i] = frame->arguments[i + n - 1]; 
 323     } 
 324  
 325     rest_arguments = njs_mp_alloc(vm->mem_pool, sizeof(njs_value_t)); 
 326     if (njs_slow_path(rest_arguments == NULL)) { 
 327         return NJS_ERROR; 
 328     } 
 329  
 330     /* GC: retain. */ 
 331     njs_set_array(rest_arguments, array); 
 332  
 333     vm->top_frame->local[n] = rest_arguments; 
 334  
 335     return NJS_OK; 
 336 }

 368 njs_int_t 
 369 njs_function_native_frame(njs_vm_t *vm, njs_function_t *function, 
 370     const njs_value_t *this, const njs_value_t *args, njs_uint_t nargs, 
 371     njs_bool_t ctor) 
 372 { 
 373     size_t              size; 
 374     njs_uint_t          n; 
 375     njs_value_t         *value, *bound; 
 376     njs_native_frame_t  *frame; 
 377  
 378     size = NJS_NATIVE_FRAME_SIZE 
 379            + (function->args_offset + nargs) * sizeof(njs_value_t); 
 380  
 381     frame = njs_function_frame_alloc(vm, size); 
 382     if (njs_slow_path(frame == NULL)) { 
 383         return NJS_ERROR; 
 384     } 
 385  
 386     frame->function = function; 
 387     frame->nargs = function->args_offset + nargs; 
 388     frame->ctor = ctor; 
 389     frame->native = 1; 
 390     frame->pc = NULL; 
 391  
 392     value = (njs_value_t *) ((u_char *) frame + NJS_NATIVE_FRAME_SIZE); 
 393  
 394     frame->arguments = value; 
 395     frame->arguments_offset = value + function->args_offset; 
 396  
 397     bound = function->bound; 
 398  
 399     if (bound == NULL) { 
 400         /* GC: njs_retain(this); */ 
 401         *value++ = *this; 
 402  
 403     } else { 
 404         n = function->args_offset; 
 405  
 406         do { 
 407             /* GC: njs_retain(bound); */ 
 408             *value++ = *bound++; 
 409             n--; 
 410         } while (n != 0); 
 411     } 
 412  
 413     if (args != NULL) { 
 414         memcpy(value, args, nargs * sizeof(njs_value_t)); 
 415     } 
 416  
 417     return NJS_OK; 
 418 }

 421 njs_int_t 
 422 njs_function_lambda_frame(njs_vm_t *vm, njs_function_t *function, 
 423     const njs_value_t *this, const njs_value_t *args, njs_uint_t nargs, 
 424     njs_bool_t ctor) 
 425 { 
 426     size_t                 n, frame_size; 
 427     uint32_t               args_count, value_count, value_size, temp_size; 
 428     njs_value_t            *value, *bound, **new, **temp; 
 429     njs_frame_t            *frame; 
 430     njs_function_t         *target; 
 431     njs_native_frame_t     *native_frame; 
 432     njs_function_lambda_t  *lambda; 
 433  
 434     bound = function->bound; 
 435  
 436     if (njs_fast_path(bound == NULL)) { 
 437         lambda = function->u.lambda; 
 438         target = function; 
 439  
 440     } else { 
 441         target = function->u.bound_target; 
 442  
 443         if (njs_slow_path(target->bound != NULL)) { 
 444  
 445             /* 
 446              * FIXME: bound functions should call target function with 
 447              * bound "this" and bound args. 
 448              */ 
 449  
 450             njs_internal_error(vm, "chain of bound function are not supported"); 
 451             return NJS_ERROR; 
 452         } 
 453  
 454         lambda = target->u.lambda; 
 455     } 
 456  
 457     args_count = function->args_offset + njs_max(nargs, lambda->nargs); 
 458     value_count = args_count + njs_max(args_count, lambda->nlocal); 
 459  
 460     value_size = value_count * sizeof(njs_value_t *); 
 461     temp_size = lambda->temp * sizeof(njs_value_t *); 
 462  
 463     frame_size = value_size + temp_size 
 464                         + ((value_count + lambda->temp) * sizeof(njs_value_t)); 
 465  
 466     native_frame = njs_function_frame_alloc(vm, NJS_FRAME_SIZE + frame_size); 
 467     if (njs_slow_path(native_frame == NULL)) { 
 468         return NJS_ERROR; 
 469     } 
 470  
 471     /* Local */ 
 472  
 473     new = (njs_value_t **) ((u_char *) native_frame + NJS_FRAME_SIZE); 
 474     value = (njs_value_t *) ((u_char *) new + value_size + temp_size); 
 475  
 476     n = value_count + lambda->temp; 
 477  
 478     while (n != 0) { 
 479         n--; 
 480         new[n] = &value[n]; 
 481         njs_set_invalid(new[n]); 
 482     } 
 483  
 484     /* Temp */ 
 485  
 486     temp = (njs_value_t **) ((u_char *) native_frame + NJS_FRAME_SIZE 
 487                                                      + value_size); 
 488  
 489     native_frame->arguments = value; 
 490     native_frame->arguments_offset = value + (function->args_offset - 1); 
 491     native_frame->local = new + args_count; 
 492     native_frame->temp = temp; 
 493     native_frame->function = target; 
 494     native_frame->nargs = nargs; 
 495     native_frame->ctor = ctor; 
 496     native_frame->native = 0; 
 497     native_frame->pc = NULL; 
 498  
 499     /* Set this and bound arguments. */ 
 500     *native_frame->local[0] = *this; 
 501  
 502     if (njs_slow_path(function->global_this 
 503                       && njs_is_null_or_undefined(this))) 
 504     { 
 505         njs_set_object(native_frame->local[0], &vm->global_object); 
 506     } 
 507  
 508     if (bound != NULL) { 
 509         n = function->args_offset; 
 510         native_frame->nargs += n - 1; 
 511  
 512         if (!ctor) { 
 513             *native_frame->local[0] = *bound; 
 514         } 
 515  
 516         bound++; 
 517         n--; 
 518  
 519         while (n != 0) { 
 520             *value++ = *bound++; 
 521             n--; 
 522         }; 
 523     } 
 524  
 525     /* Copy arguments. */ 
 526  
 527     if (args != NULL) { 
 528         while (nargs != 0) { 
 529             *value++ = *args++; 
 530             nargs--; 
 531         } 
 532     } 
 533  
 534     frame = (njs_frame_t *) native_frame; 
 535     frame->exception.catch = NULL; 
 536     frame->exception.next = NULL; 
 537     frame->previous_active_frame = vm->active_frame; 
 538  
 539     return NJS_OK; 
 540 }

 543 njs_native_frame_t * 
 544 njs_function_frame_alloc(njs_vm_t *vm, size_t size) 
 545 { 
 546     size_t              spare_size, chunk_size; 
 547     njs_native_frame_t  *frame; 
 548  
 549     spare_size = vm->top_frame ? vm->top_frame->free_size : 0; 
 550  
 551     if (njs_fast_path(size <= spare_size)) { 
 552         frame = (njs_native_frame_t *) vm->top_frame->free; 
 553         chunk_size = 0; 
 554  
 555     } else { 
 556         spare_size = size + NJS_FRAME_SPARE_SIZE; 
 557         spare_size = njs_align_size(spare_size, NJS_FRAME_SPARE_SIZE); 
 558  
 559         if (vm->stack_size + spare_size > NJS_MAX_STACK_SIZE) { 
 560             njs_range_error(vm, "Maximum call stack size exceeded"); 
 561             return NULL; 
 562         } 
 563  
 564         frame = njs_mp_align(vm->mem_pool, sizeof(njs_value_t), spare_size); 
 565         if (njs_slow_path(frame == NULL)) { 
 566             njs_memory_error(vm); 
 567             return NULL; 
 568         } 
 569  
 570         chunk_size = spare_size; 
 571         vm->stack_size += spare_size; 
 572     } 
 573  
 574     njs_memzero(frame, sizeof(njs_native_frame_t)); 
 575  
 576     frame->size = chunk_size; 
 577     frame->free_size = spare_size - size; 
 578     frame->free = (u_char *) frame + size; 
 579  
 580     frame->previous = vm->top_frame; 
 581     vm->top_frame = frame; 
 582  
 583     return frame; 
 584 }

 610 njs_int_t 
 611 njs_function_lambda_call(njs_vm_t *vm, void *promise_cap, void *async_ctx) 
 612 { 
 613     uint32_t               n; 
 614     njs_int_t              ret; 
 615     njs_frame_t            *frame; 
 616     njs_value_t            *args, **local, *value; 
 617     njs_value_t            **cur_local, **cur_closures, **cur_temp; 
 618     njs_function_t         *function; 
 619     njs_declaration_t      *declr; 
 620     njs_function_lambda_t  *lambda; 
 621  
 622     frame = (njs_frame_t *) vm->top_frame; 
 623     function = frame->native.function; 
 624  
 625     njs_assert(function->context == NULL); 
 626  
 627     if (function->global && !function->closure_copied) { 
 628         ret = njs_function_capture_global_closures(vm, function); 
 629         if (njs_slow_path(ret != NJS_OK)) { 
 630             return NJS_ERROR; 
 631         } 
 632     } 
 633  
 634     lambda = function->u.lambda; 
 635  
 636     args = vm->top_frame->arguments; 
 637     local = vm->top_frame->local + function->args_offset; 
 638  
 639     /* Move all arguments. */ 
 640  
 641     for (n = 0; n < function->args_count; n++) { 
 642         if (!njs_is_valid(args)) { 
 643             njs_set_undefined(args); 
 644         } 
 645  
 646         *local++ = args++; 
 647     } 
 648  
 649     /* Store current level. */ 
 650  
 651     cur_local = vm->levels[NJS_LEVEL_LOCAL]; 
 652     cur_closures = vm->levels[NJS_LEVEL_CLOSURE]; 
 653     cur_temp = vm->levels[NJS_LEVEL_TEMP]; 
 654  
 655     /* Replace current level. */ 
 656  
 657     vm->levels[NJS_LEVEL_LOCAL] = vm->top_frame->local; 
 658     vm->levels[NJS_LEVEL_CLOSURE] = njs_function_closures(function); 
 659     vm->levels[NJS_LEVEL_TEMP] = frame->native.temp; 
 660  
 661     if (lambda->rest_parameters) { 
 662         ret = njs_function_rest_parameters_init(vm, &frame->native); 
 663         if (njs_slow_path(ret != NJS_OK)) { 
 664             return NJS_ERROR; 
 665         } 
 666     } 
 667  
 668     /* Self */ 
 669  
 670     if (lambda->self != NJS_INDEX_NONE) { 
 671         value = njs_scope_value(vm, lambda->self); 
 672  
 673         if (!njs_is_valid(value)) { 
 674             njs_set_function(value, function); 
 675         } 
 676     } 
 677  
 678     vm->active_frame = frame; 
 679  
 680     /* Closures */ 
 681  
 682     n = lambda->ndeclarations; 
 683  
 684     while (n != 0) { 
 685         n--; 
 686  
 687         declr = &lambda->declarations[n]; 
 688         value = njs_scope_value(vm, declr->index); 
 689  
 690         *value = *declr->value; 
 691  
 692         function = njs_function_value_copy(vm, value); 
 693         if (njs_slow_path(function == NULL)) { 
 694             return NJS_ERROR; 
 695         } 
 696  
 697         ret = njs_function_capture_closure(vm, function, function->u.lambda);

 715 njs_function_native_call(njs_vm_t *vm) 
 716 { 
 717     njs_int_t              ret; 
 718     njs_function_t         *function, *target; 
 719     njs_native_frame_t     *native, *previous; 
 720     njs_function_native_t  call; 
 721  
 722     native = vm->top_frame; 
 723     function = native->function; 
 724  
 725     if (njs_fast_path(function->bound == NULL)) { 
 726         call = function->u.native; 
 727  
 728     } else { 
 729         target = function->u.bound_target; 
 730  
 731         if (njs_slow_path(target->bound != NULL)) { 
 732             njs_internal_error(vm, "chain of bound function are not supported"); 
 733             return NJS_ERROR; 
 734         } 
 735  
 736         call = target->u.native; 
 737     } 
 738  
 739     ret = call(vm, native->arguments, native->nargs, function->magic8); 
 740     if (njs_slow_path(ret == NJS_ERROR)) { 
 741         return ret; 
 742     } 
 743  
 744     if (ret == NJS_DECLINED) { 
 745         return NJS_OK; 
 746     } 
 747  
 748     previous = njs_function_previous_frame(native); 
 749  
 750     njs_vm_scopes_restore(vm, native, previous); 
 751  
 752     if (!native->skip) { 
 753         *native->retval = vm->retval; 
 754     } 
 755  
 756     njs_function_frame_free(vm, native); 
 757  
 758     return NJS_OK; 
 759 }

 874     njs_function_lambda_t *lambda)

 877     uint32_t            n; 
 878     njs_value_t         *value, **closure; 
 879     njs_native_frame_t  *frame; 
 880  
 881     if (lambda->nclosures == 0) { 
 882         return NJS_OK; 
 883     }

 885     frame = &vm->active_frame->native; 
 886  
 887     while (frame->native) { 
 888         frame = frame->previous; 
 889     }

 893

 905                 return NJS_ERROR; 
 906             } 
 907         }

 910

1044 /* 
1045  * The "prototype" property of user defined functions is created on 
1046  * demand in private hash of the functions by the "prototype" getter. 
1047  * The getter creates a copy of function which is private to nJSVM, 
1048  * adds a "prototype" object property to the copy, and then adds a 
1049  * "constructor" property in the prototype object.  The "constructor" 
1050  * property points to the copy of function: 
1051  *   "F.prototype.constructor === F" 
1052  */ 
1053  
1054 njs_int_t 
1055 njs_function_prototype_create(njs_vm_t *vm, njs_object_prop_t *prop, 
1056     njs_value_t *value, njs_value_t *setval, njs_value_t *retval) 
1057 { 
1058     njs_value_t     *proto, proto_value, *cons; 
1059     njs_object_t    *prototype; 
1060     njs_function_t  *function; 
1061  
1062     if (setval == NULL) { 
1063         prototype = njs_object_alloc(vm); 
1064         if (njs_slow_path(prototype == NULL)) { 
1065             return NJS_ERROR; 
1066         } 
1067  
1068         njs_set_object(&proto_value, prototype); 
1069  
1070         setval = &proto_value; 
1071     } 
1072  
1073     function = njs_function_value_copy(vm, value); 
1074     if (njs_slow_path(function == NULL)) { 
1075         return NJS_ERROR; 
1076     } 
1077  
1078     proto = njs_function_property_prototype_set(vm, njs_object_hash(value), 
1079                                                 setval); 
1080     if (njs_slow_path(proto == NULL)) { 
1081         return NJS_ERROR; 
1082     } 
1083  
1084     if (setval == &proto_value && njs_is_object(proto)) { 
1085         /* Only in getter context. */ 
1086         cons = njs_property_constructor_set(vm, njs_object_hash(proto), value); 
1087         if (njs_slow_path(cons == NULL)) { 
1088             return NJS_ERROR; 
1089         } 
1090     } 
1091  
1092     *retval = *proto; 
1093  
1094     return NJS_OK; 
1095 }

1098 njs_int_t 
1099 njs_function_constructor(njs_vm_t *vm, njs_value_t *args, njs_uint_t nargs, 
1100     njs_index_t async) 
1101 { 
1102     njs_chb_t               chain; 
1103     njs_int_t               ret; 
1104     njs_str_t               str, file; 
1105     njs_uint_t              i; 
1106     njs_parser_t            parser; 
1107     njs_vm_code_t           *code; 
1108     njs_function_t          *function; 
1109     njs_generator_t         generator; 
1110     njs_parser_node_t       *node; 
1111     njs_function_lambda_t   *lambda; 
1112     const njs_token_type_t  *type; 
1113  
1114     static const njs_token_type_t  safe_ast[] = { 
1115         NJS_TOKEN_END, 
1116         NJS_TOKEN_FUNCTION_EXPRESSION, 
1117         NJS_TOKEN_STATEMENT, 
1118         NJS_TOKEN_RETURN, 
1119         NJS_TOKEN_THIS, 
1120         NJS_TOKEN_ILLEGAL 
1121     }; 
1122  
1123     static const njs_token_type_t  safe_ast_async[] = { 
1124         NJS_TOKEN_END, 
1125         NJS_TOKEN_ASYNC_FUNCTION_EXPRESSION, 
1126         NJS_TOKEN_STATEMENT, 
1127         NJS_TOKEN_RETURN, 
1128         NJS_TOKEN_THIS, 
1129         NJS_TOKEN_ILLEGAL 
1130     }; 
1131  
1132     if (!vm->options.unsafe && nargs != 2) { 
1133         goto fail; 
1134     } 
1135  
1136     njs_chb_init(&chain, vm->mem_pool); 
1137  
1138     if (async) { 
1139         njs_chb_append_literal(&chain, "(async function("); 
1140  
1141     } else { 
1142         njs_chb_append_literal(&chain, "(function("); 
1143     } 
1144  
1145     for (i = 1; i < nargs - 1; i++) { 
1146         ret = njs_value_to_chain(vm, &chain, njs_argument(args, i)); 
1147         if (njs_slow_path(ret < NJS_OK)) { 
1148             return ret; 
1149         } 
1150  
1151         if (i != (nargs - 2)) { 
1152             njs_chb_append_literal(&chain, ","); 
1153         } 
1154     } 
1155  
1156     njs_chb_append_literal(&chain, "){"); 
1157  
1158     ret = njs_value_to_chain(vm, &chain, njs_argument(args, nargs - 1)); 
1159     if (njs_slow_path(ret < NJS_OK)) { 
1160         return ret; 
1161     } 
1162  
1163     njs_chb_append_literal(&chain, "})"); 
1164  
1165     ret = njs_chb_join(&chain, &str); 
1166     if (njs_slow_path(ret != NJS_OK)) { 
1167         njs_memory_error(vm); 
1168         return NJS_ERROR; 
1169     } 
1170  
1171     file = njs_str_value("runtime"); 
1172  
1173     ret = njs_parser_init(vm, &parser, NULL, &file, str.start, 
1174                           str.start + str.length, 1); 
1175     if (njs_slow_path(ret != NJS_OK)) { 
1176         return ret; 
1177     } 
1178  
1179     ret = njs_parser(vm, &parser); 
1180     if (njs_slow_path(ret != NJS_OK)) { 
1181         return ret; 
1182     } 
1183  
1184     if (!vm->options.unsafe) { 
1185         /* 
1186          * Safe mode exception: 
1187          * "(new Function('return this'))" is often used to get 
1188          * the global object in a portable way. 
1189          */ 
1190  
1191         node = parser.node; 
1192         type = (async) ? &safe_ast_async[0] : &safe_ast[0]; 
1193  
1194         for (; *type != NJS_TOKEN_ILLEGAL; type++, node = node->right) { 
1195             if (node == NULL) { 
1196                 goto fail; 
1197             } 
1198  
1199             if (node->left != NULL 
1200                 && node->token_type != NJS_TOKEN_FUNCTION_EXPRESSION 
1201                 && node->left->token_type != NJS_TOKEN_NAME) 
1202             { 
1203                 goto fail; 
1204             } 
1205  
1206             if (node->token_type != *type) { 
1207                 goto fail; 
1208             } 
1209         } 
1210     } 
1211  
1212     ret = njs_generator_init(&generator, &file, 0, 1); 
1213     if (njs_slow_path(ret != NJS_OK)) { 
1214         njs_internal_error(vm, "njs_generator_init() failed"); 
1215         return NJS_ERROR; 
1216     } 
1217  
1218     code = njs_generate_scope(vm, &generator, parser.scope, 
1219                               &njs_entry_anonymous); 
1220     if (njs_slow_path(code == NULL)) { 
1221         if (!njs_is_error(&vm->retval)) { 
1222             njs_internal_error(vm, "njs_generate_scope() failed"); 
1223         } 
1224  
1225         return NJS_ERROR; 
1226     } 
1227  
1228     njs_chb_destroy(&chain); 
1229  
1230     lambda = ((njs_vmcode_function_t *) generator.code_start)->lambda; 
1231  
1232     function = njs_function_alloc(vm, lambda, (njs_bool_t) async); 
1233     if (njs_slow_path(function == NULL)) { 
1234         return NJS_ERROR; 
1235     } 
1236  
1237     function->global = 1; 
1238     function->global_this = 1; 
1239     function->args_count = lambda->nargs - lambda->rest_parameters; 
1240  
1241     njs_set_function(&vm->retval, function); 
1242  
1243     return NJS_OK; 
1244  
1245 fail: 
1246  
1247     njs_type_error(vm, "function constructor is disabled in \"safe\" mode"); 
1248     return NJS_ERROR; 
1249 }

 399     if (bound == NULL) { 
 400         /* GC: njs_retain(this); */ 
 401         *value++ = *this; 
 402  
 403     } else { 
 404         n = function->args_offset; 
 405  
 406         do { 
 407             /* GC: njs_retain(bound); */ 
 408             *value++ = *bound++; 
 409             n--; 
 410         } while (n != 0); 
 411     } 
 412  
 413     if (args != NULL) { 
 414         memcpy(value, args, nargs * sizeof(njs_value_t)); 
 415     } 
 416  
 417     return NJS_OK; 
 418 }

 421 njs_int_t 
 422 njs_function_lambda_frame(njs_vm_t *vm, njs_function_t *function, 
 423     const njs_value_t *this, const njs_value_t *args, njs_uint_t nargs, 
 424     njs_bool_t ctor) 
 425 { 
 426     size_t                 n, frame_size; 
 427     uint32_t               args_count, value_count, value_size, temp_size; 
 428     njs_value_t            *value, *bound, **new, **temp; 
 429     njs_frame_t            *frame; 
 430     njs_function_t         *target; 
 431     njs_native_frame_t     *native_frame; 
 432     njs_function_lambda_t  *lambda; 
 433  
 434     bound = function->bound; 
 435  
 436     if (njs_fast_path(bound == NULL)) { 
 437         lambda = function->u.lambda; 
 438         target = function; 
 439  
 440     } else { 
 441         target = function->u.bound_target; 
 442  
 443         if (njs_slow_path(target->bound != NULL)) { 
 444  
 445             /* 
 446              * FIXME: bound functions should call target function with 
 447              * bound "this" and bound args. 
 448              */ 
 449  
 450             njs_internal_error(vm, "chain of bound function are not supported"); 
 451             return NJS_ERROR; 
 452         } 
 453  
 454         lambda = target->u.lambda; 
 455     } 
 456  
 457     args_count = function->args_offset + njs_max(nargs, lambda->nargs); 
 458     value_count = args_count + njs_max(args_count, lambda->nlocal); 
 459  
 460     value_size = value_count * sizeof(njs_value_t *); 
 461     temp_size = lambda->temp * sizeof(njs_value_t *); 
 462  
 463     frame_size = value_size + temp_size 
 464                         + ((value_count + lambda->temp) * sizeof(njs_value_t)); 
 465  
 466     native_frame = njs_function_frame_alloc(vm, NJS_FRAME_SIZE + frame_size); 
 467     if (njs_slow_path(native_frame == NULL)) { 
 468         return NJS_ERROR; 
 469     } 
 470  
 471     /* Local */ 
 472  
 473     new = (njs_value_t **) ((u_char *) native_frame + NJS_FRAME_SIZE); 
 474     value = (njs_value_t *) ((u_char *) new + value_size + temp_size); 
 475  
 476     n = value_count + lambda->temp; 
 477  
 478     while (n != 0) { 
 479         n--; 
 480         new[n] = &value[n]; 
 481         njs_set_invalid(new[n]); 
 482     } 
 483  
 484     /* Temp */ 
 485  
 486     temp = (njs_value_t **) ((u_char *) native_frame + NJS_FRAME_SIZE 
 487                                                      + value_size); 
 488  
 489     native_frame->arguments = value; 
 490     native_frame->arguments_offset = value + (function->args_offset - 1); 
 491     native_frame->local = new + args_count; 
 492     native_frame->temp = temp; 
 493     native_frame->function = target; 
 494     native_frame->nargs = nargs; 
 495     native_frame->ctor = ctor; 
 496     native_frame->native = 0; 
 497     native_frame->pc = NULL; 
 498  
 499     /* Set this and bound arguments. */ 
 500     *native_frame->local[0] = *this; 
 501  
 502     if (njs_slow_path(function->global_this 
 503                       && njs_is_null_or_undefined(this))) 
 504     { 
 505         njs_set_object(native_frame->local[0], &vm->global_object); 
 506     } 
 507  
 508     if (bound != NULL) { 
 509         n = function->args_offset; 
 510         native_frame->nargs += n - 1; 
 511  
 512         if (!ctor) { 
 513             *native_frame->local[0] = *bound; 
 514         } 
 515  
 516         bound++; 
 517         n--; 
 518  
 519         while (n != 0) { 
 520             *value++ = *bound++; 
 521             n--; 
 522         }; 
 523     } 
 524  
 525     /* Copy arguments. */ 
 526  
 527     if (args != NULL) { 
 528         while (nargs != 0) { 
 529             *value++ = *args++; 
 530             nargs--; 
 531         } 
 532     } 
 533  
 534     frame = (njs_frame_t *) native_frame; 
 535     frame->exception.catch = NULL; 
 536     frame->exception.next = NULL; 
 537     frame->previous_active_frame = vm->active_frame; 
 538  
 539     return NJS_OK; 
 540 }

 543 njs_native_frame_t * 
 544 njs_function_frame_alloc(njs_vm_t *vm, size_t size) 
 545 { 
 546     size_t              spare_size, chunk_size; 
 547     njs_native_frame_t  *frame; 
 548  
 549     spare_size = vm->top_frame ? vm->top_frame->free_size : 0; 
 550  
 551     if (njs_fast_path(size <= spare_size)) { 
 552         frame = (njs_native_frame_t *) vm->top_frame->free; 
 553         chunk_size = 0; 
 554  
 555     } else { 
 556         spare_size = size + NJS_FRAME_SPARE_SIZE; 
 557         spare_size = njs_align_size(spare_size, NJS_FRAME_SPARE_SIZE); 
 558  
 559         if (vm->stack_size + spare_size > NJS_MAX_STACK_SIZE) { 
 560             njs_range_error(vm, "Maximum call stack size exceeded"); 
 561             return NULL; 
 562         } 
 563  
 564         frame = njs_mp_align(vm->mem_pool, sizeof(njs_value_t), spare_size); 
 565         if (njs_slow_path(frame == NULL)) { 
 566             njs_memory_error(vm); 
 567             return NULL; 
 568         } 
 569  
 570         chunk_size = spare_size; 
 571         vm->stack_size += spare_size; 
 572     } 
 573  
 574     njs_memzero(frame, sizeof(njs_native_frame_t)); 
 575  
 576     frame->size = chunk_size; 
 577     frame->free_size = spare_size - size; 
 578     frame->free = (u_char *) frame + size; 
 579  
 580     frame->previous = vm->top_frame; 
 581     vm->top_frame = frame; 
 582  
 583     return frame; 
 584 }

 610 njs_int_t 
 611 njs_function_lambda_call(njs_vm_t *vm, void *promise_cap, void *async_ctx) 
 612 { 
 613     uint32_t               n; 
 614     njs_int_t              ret; 
 615     njs_frame_t            *frame; 
 616     njs_value_t            *args, **local, *value; 
 617     njs_value_t            **cur_local, **cur_closures, **cur_temp; 
 618     njs_function_t         *function; 
 619     njs_declaration_t      *declr; 
 620     njs_function_lambda_t  *lambda; 
 621  
 622     frame = (njs_frame_t *) vm->top_frame; 
 623     function = frame->native.function; 
 624  
 625     njs_assert(function->context == NULL); 
 626  
 627     if (function->global && !function->closure_copied) { 
 628         ret = njs_function_capture_global_closures(vm, function); 
 629         if (njs_slow_path(ret != NJS_OK)) { 
 630             return NJS_ERROR; 
 631         } 
 632     } 
 633  
 634     lambda = function->u.lambda; 
 635  
 636     args = vm->top_frame->arguments; 
 637     local = vm->top_frame->local + function->args_offset; 
 638  
 639     /* Move all arguments. */ 
 640  
 641     for (n = 0; n < function->args_count; n++) { 
 642         if (!njs_is_valid(args)) { 
 643             njs_set_undefined(args); 
 644         } 
 645  
 646         *local++ = args++; 
 647     } 
 648  
 649     /* Store current level. */ 
 650  
 651     cur_local = vm->levels[NJS_LEVEL_LOCAL]; 
 652     cur_closures = vm->levels[NJS_LEVEL_CLOSURE]; 
 653     cur_temp = vm->levels[NJS_LEVEL_TEMP]; 
 654  
 655     /* Replace current level. */ 
 656  
 657     vm->levels[NJS_LEVEL_LOCAL] = vm->top_frame->local; 
 658     vm->levels[NJS_LEVEL_CLOSURE] = njs_function_closures(function); 
 659     vm->levels[NJS_LEVEL_TEMP] = frame->native.temp; 
 660  
 661     if (lambda->rest_parameters) { 
 662         ret = njs_function_rest_parameters_init(vm, &frame->native); 
 663         if (njs_slow_path(ret != NJS_OK)) { 
 664             return NJS_ERROR; 
 665         } 
 666     } 
 667  
 668     /* Self */ 
 669  
 670     if (lambda->self != NJS_INDEX_NONE) { 
 671         value = njs_scope_value(vm, lambda->self); 
 672  
 673         if (!njs_is_valid(value)) { 
 674             njs_set_function(value, function); 
 675         } 
 676     } 
 677  
 678     vm->active_frame = frame; 
 679  
 680     /* Closures */ 
 681  
 682     n = lambda->ndeclarations; 
 683  
 684     while (n != 0) { 
 685         n--; 
 686  
 687         declr = &lambda->declarations[n]; 
 688         value = njs_scope_value(vm, declr->index); 
 689  
 690         *value = *declr->value; 
 691  
 692         function = njs_function_value_copy(vm, value); 
 693         if (njs_slow_path(function == NULL)) { 
 694             return NJS_ERROR; 
 695         } 
 696  
 697         ret = njs_function_capture_closure(vm, function, function->u.lambda); 
 698         if (njs_slow_path(ret != NJS_OK)) { 
 699             return ret; 
 700         } 
 701     } 
 702  
 703     ret = njs_vmcode_interpreter(vm, lambda->start, promise_cap, async_ctx); 
 704  
 705     /* Restore current level. */ 
 706     vm->levels[NJS_LEVEL_LOCAL] = cur_local; 
 707     vm->levels[NJS_LEVEL_CLOSURE] = cur_closures; 
 708     vm->levels[NJS_LEVEL_TEMP] = cur_temp; 
 709  
 710     return ret; 
 711 }

 714 njs_int_t 
 715 njs_function_native_call(njs_vm_t *vm) 
 716 { 
 717     njs_int_t              ret; 
 718     njs_function_t         *function, *target; 
 719     njs_native_frame_t     *native, *previous; 
 720     njs_function_native_t  call; 
 721  
 722     native = vm->top_frame; 
 723     function = native->function; 
 724  
 725     if (njs_fast_path(function->bound == NULL)) { 
 726         call = function->u.native; 
 727  
 728     } else { 
 729         target = function->u.bound_target; 
 730  
 731         if (njs_slow_path(target->bound != NULL)) { 
 732             njs_internal_error(vm, "chain of bound function are not supported"); 
 733             return NJS_ERROR; 
 734         } 
 735  
 736         call = target->u.native; 
 737     } 
 738  
 739     ret = call(vm, native->arguments, native->nargs, function->magic8); 
 740     if (njs_slow_path(ret == NJS_ERROR)) { 
 741         return ret; 
 742     } 
 743  
 744     if (ret == NJS_DECLINED) { 
 745         return NJS_OK; 
 746     } 
 747  
 748     previous = njs_function_previous_frame(native); 
 749  
 750     njs_vm_scopes_restore(vm, native, previous); 
 751  
 752     if (!native->skip) { 
 753         *native->retval = vm->retval; 
 754     } 
 755  
 756     njs_function_frame_free(vm, native); 
 757  
 758     return NJS_OK; 
 759 }

 806 njs_function_frame_save(njs_vm_t *vm, njs_frame_t *frame, u_char *pc) 
 807 { 
 808     size_t              value_count, n; 
 809     njs_value_t         *start, *end, *p, **new, *value, **local; 
 810     njs_function_t      *function; 
 811     njs_native_frame_t  *active, *native; 
 812  
 813     *frame = *vm->active_frame; 
 814  
 815     frame->previous_active_frame = NULL; 
 816  
 817     native = &frame->native; 
 818     native->size = 0; 
 819     native->free = NULL; 
 820     native->free_size = 0; 
 821  
 822     active = &vm->active_frame->native; 
 823     value_count = njs_function_frame_value_count(active); 
 824  
 825     function = active->function; 
 826

 828     value = (njs_value_t *) (new + value_count 
 829                              + function->u.lambda->temp); 
 830

 833     native->arguments_offset = value + (function->args_offset - 1); 
 834     native->local = new + njs_function_frame_args_count(active); 
 835     native->temp = new + value_count; 
 836     native->pc = pc;

 842         *p = *start++; 
 843         *new++ = p++; 
 844     }

 847  
 848     p = native->arguments; 
 849     local = native->local + function->args_offset; 
 850

 852         if (!njs_is_valid(p)) { 
 853             njs_set_undefined(p); 
 854         } 
 855  
 856         *local++ = p++; 
 857     } 
 858  
 859     return NJS_OK; 
 860 }

 876 njs_int_t 
 877 njs_function_capture_closure(njs_vm_t *vm, njs_function_t *function, 
 878     njs_function_lambda_t *lambda) 
 879 { 
 880     void                *start, *end; 
 881     uint32_t            n; 
 882     njs_value_t         *value, **closure; 
 883     njs_native_frame_t  *frame; 
 884  
 885     if (lambda->nclosures == 0) { 
 886         return NJS_OK; 
 887     } 
 888  
 889     frame = &vm->active_frame->native; 
 890  
 891     while (frame->native) { 
 892         frame = frame->previous; 
 893     } 
 894  
 895     start = frame; 
 896     end = frame->free; 
 897  
 898     closure = njs_function_closures(function); 
 899     n = lambda->nclosures; 
 900  
 901     do { 
 902         n--; 
 903  
 904         value = njs_scope_value(vm, lambda->closures[n]); 
 905  
 906         if (start <= (void *) value && (void *) value < end) { 
 907             value = njs_scope_value_clone(vm, lambda->closures[n], value); 
 908             if (njs_slow_path(value == NULL)) { 
 909                 return NJS_ERROR; 
 910             } 
 911         } 
 912  
 913         closure[n] = value; 
 914  
 915     } while (n != 0); 
 916  
 917     return NJS_OK; 
 918 }

 922 njs_function_closure_value(njs_vm_t *vm, njs_value_t **scope, njs_index_t index, 
 923     void *start, void *end) 
 924 { 
 925     njs_value_t  *value, *newval; 
 926

 928

 930         newval = njs_mp_alloc(vm->mem_pool, sizeof(njs_value_t)); 
 931         if (njs_slow_path(newval == NULL)) { 
 932             njs_memory_error(vm); 
 933             return NULL; 
 934         } 
 935  
 936         *newval = *value; 
 937         value = newval; 
 938     } 
 939  
 940     scope[njs_scope_index_value(index)] = value; 
 941  
 942     return value;

 946 njs_int_t 
 947 njs_function_capture_global_closures(njs_vm_t *vm, njs_function_t *function) 
 948 { 
 949     void                   *start, *end; 
 950     uint32_t               n; 
 951     njs_value_t            *value, **refs, **global; 
 952     njs_index_t            *indexes, index; 
 953     njs_native_frame_t     *native; 
 954     njs_function_lambda_t  *lambda; 
 955  
 956     lambda = function->u.lambda; 
 957  
 958     if (lambda->nclosures == 0) { 
 959         return NJS_OK; 
 960     } 
 961  
 962     native = vm->top_frame; 
 963  
 964     while (native->previous->function != NULL) { 
 965         native = native->previous; 
 966     } 
 967  
 968     start = native; 
 969     end = native->free; 
 970  
 971     indexes = lambda->closures; 
 972     refs = njs_function_closures(function); 
 973  
 974     global = vm->levels[NJS_LEVEL_GLOBAL]; 
 975  
 976     n = lambda->nclosures; 
 977  
 978     while (n > 0) { 
 979         n--; 
 980  
 981         index = indexes[n]; 
 982  
 983         switch (njs_scope_index_type(index)) { 
 984         case NJS_LEVEL_LOCAL: 
 985             value = njs_function_closure_value(vm, native->local, index, 
 986                                                start, end); 
 987             break; 
 988  
 989         case NJS_LEVEL_GLOBAL: 
 990             value = njs_function_closure_value(vm, global, index, start, end); 
 991             break; 
 992  
 993         default: 
 994             njs_type_error(vm, "unexpected value type for closure \"%uD\"", 
 995                            njs_scope_index_type(index)); 
 996             return NJS_ERROR; 
 997         } 
 998  
 999         if (njs_slow_path(value == NULL)) { 
1000             return NJS_ERROR; 
1001         } 
1002  
1003         refs[n] = value; 
1004     } 
1005  
1006     function->closure_copied = 1; 
1007  
1008     return NJS_OK; 
1009 }

1048 /* 
1049  * The "prototype" property of user defined functions is created on 
1050  * demand in private hash of the functions by the "prototype" getter. 
1051  * The getter creates a copy of function which is private to nJSVM, 
1052  * adds a "prototype" object property to the copy, and then adds a 
1053  * "constructor" property in the prototype object.  The "constructor" 
1054  * property points to the copy of function: 
1055  *   "F.prototype.constructor === F" 
1056  */ 
1057  
1058 njs_int_t 
1059 njs_function_prototype_create(njs_vm_t *vm, njs_object_prop_t *prop, 
1060     njs_value_t *value, njs_value_t *setval, njs_value_t *retval) 
1061 { 
1062     njs_value_t     *proto, proto_value, *cons; 
1063     njs_object_t    *prototype; 
1064     njs_function_t  *function; 
1065  
1066     if (setval == NULL) { 
1067         prototype = njs_object_alloc(vm); 
1068         if (njs_slow_path(prototype == NULL)) { 
1069             return NJS_ERROR; 
1070         } 
1071  
1072         njs_set_object(&proto_value, prototype); 
1073  
1074         setval = &proto_value; 
1075     } 
1076  
1077     function = njs_function_value_copy(vm, value); 
1078     if (njs_slow_path(function == NULL)) { 
1079         return NJS_ERROR; 
1080     } 
1081  
1082     proto = njs_function_property_prototype_set(vm, njs_object_hash(value), 
1083                                                 setval); 
1084     if (njs_slow_path(proto == NULL)) { 
1085         return NJS_ERROR; 
1086     } 
1087  
1088     if (setval == &proto_value && njs_is_object(proto)) { 
1089         /* Only in getter context. */ 
1090         cons = njs_property_constructor_set(vm, njs_object_hash(proto), value); 
1091         if (njs_slow_path(cons == NULL)) { 
1092             return NJS_ERROR; 
1093         } 
1094     } 
1095  
1096     *retval = *proto; 
1097  
1098     return NJS_OK; 
1099 }

1104     njs_index_t async)

1109     njs_uint_t              i; 
1110     njs_parser_t            parser; 
1111     njs_vm_code_t           *code; 
1112     njs_function_t          *function; 
1113     njs_generator_t         generator;

1115     njs_function_lambda_t   *lambda; 
1116     const njs_token_type_t  *type; 
1117  
1118     static const njs_token_type_t  safe_ast[] = { 
1119         NJS_TOKEN_END, 
1120         NJS_TOKEN_FUNCTION_EXPRESSION, 
1121         NJS_TOKEN_STATEMENT, 
1122         NJS_TOKEN_RETURN, 
1123         NJS_TOKEN_THIS, 
1124         NJS_TOKEN_ILLEGAL 
1125     }; 
1126  
1127     static const njs_token_type_t  safe_ast_async[] = { 
1128         NJS_TOKEN_END, 
1129         NJS_TOKEN_ASYNC_FUNCTION_EXPRESSION, 
1130         NJS_TOKEN_STATEMENT, 
1131         NJS_TOKEN_RETURN, 
1132         NJS_TOKEN_THIS, 
1133         NJS_TOKEN_ILLEGAL 
1134     }; 
1135  
1136     if (!vm->options.unsafe && nargs != 2) { 
1137         goto fail; 
1138     } 
1139  
1140     njs_chb_init(&chain, vm->mem_pool); 
1141  
1142     if (async) { 
1143         njs_chb_append_literal(&chain, "(async function("); 
1144  
1145     } else { 
1146         njs_chb_append_literal(&chain, "(function("); 
1147     } 
1148  
1149     for (i = 1; i < nargs - 1; i++) { 
1150         ret = njs_value_to_chain(vm, &chain, njs_argument(args, i)); 
1151         if (njs_slow_path(ret < NJS_OK)) { 
1152             return ret; 
1153         } 
1154  
1155         if (i != (nargs - 2)) { 
1156             njs_chb_append_literal(&chain, ","); 
1157         } 
1158     } 
1159  
1160     njs_chb_append_literal(&chain, "){"); 
1161  
1162     ret = njs_value_to_chain(vm, &chain, njs_argument(args, nargs - 1)); 
1163     if (njs_slow_path(ret < NJS_OK)) { 
1164         return ret; 
1165     } 
1166  
1167     njs_chb_append_literal(&chain, "})"); 
1168  
1169     ret = njs_chb_join(&chain, &str); 
1170     if (njs_slow_path(ret != NJS_OK)) { 
1171         njs_memory_error(vm); 
1172         return NJS_ERROR; 
1173     } 
1174  
1175     file = njs_str_value("runtime"); 
1176  
1177     ret = njs_parser_init(vm, &parser, NULL, &file, str.start, 
1178                           str.start + str.length, 1); 
1179     if (njs_slow_path(ret != NJS_OK)) { 
1180         return ret; 
1181     } 
1182  
1183     ret = njs_parser(vm, &parser); 
1184     if (njs_slow_path(ret != NJS_OK)) { 
1185         return ret; 
1186     } 
1187  
1188     if (!vm->options.unsafe) { 
1189         /* 
1190          * Safe mode exception: 
1191          * "(new Function('return this'))" is often used to get 
1192          * the global object in a portable way. 
1193          */ 
1194  
1195         node = parser.node; 
1196         type = (async) ? &safe_ast_async[0] : &safe_ast[0]; 
1197  
1198         for (; *type != NJS_TOKEN_ILLEGAL; type++, node = node->right) { 
1199             if (node == NULL) { 
1200                 goto fail; 
1201             } 
1202  
1203             if (node->left != NULL 
1204                 && node->token_type != NJS_TOKEN_FUNCTION_EXPRESSION 
1205                 && node->left->token_type != NJS_TOKEN_NAME) 
1206             { 
1207                 goto fail; 
1208             } 
1209  
1210             if (node->token_type != *type) { 
1211                 goto fail; 
1212             } 
1213         } 
1214     } 
1215  
1216     ret = njs_generator_init(&generator, &file, 0, 1); 
1217     if (njs_slow_path(ret != NJS_OK)) { 
1218         njs_internal_error(vm, "njs_generator_init() failed"); 
1219         return NJS_ERROR; 
1220     } 
1221  
1222     code = njs_generate_scope(vm, &generator, parser.scope, 
1223                               &njs_entry_anonymous); 
1224     if (njs_slow_path(code == NULL)) { 
1225         if (!njs_is_error(&vm->retval)) { 
1226             njs_internal_error(vm, "njs_generate_scope() failed"); 
1227         } 
1228  
1229         return NJS_ERROR; 
1230     } 
1231  
1232     njs_chb_destroy(&chain); 
1233  
1234     lambda = ((njs_vmcode_function_t *) generator.code_start)->lambda; 
1235  
1236     function = njs_function_alloc(vm, lambda, (njs_bool_t) async); 
1237     if (njs_slow_path(function == NULL)) { 
1238         return NJS_ERROR; 
1239     } 
1240  
1241     function->global = 1; 
1242     function->global_this = 1; 
1243     function->args_count = lambda->nargs - lambda->rest_parameters; 
1244  
1245     njs_set_function(&vm->retval, function); 
1246  
1247     return NJS_OK; 
1248  
1249 fail: 
1250  
1251     njs_type_error(vm, "function constructor is disabled in \"safe\" mode"); 
1252     return NJS_ERROR; 
1253 }
