  61 static bool isAutoDealloc(const Variable *var) 
  62 { 
  63     if (var->valueType() && var->valueType()->type != ValueType::Type::RECORD && var->valueType()->type != ValueType::Type::UNKNOWN_TYPE) 
  64         return false; 
  65  
  66     // return false if the type is a simple record type without side effects 
  67     // a type that has no side effects (no constructors and no members with constructors) 
  68     /** @todo false negative: check base class for side effects */ 
  69     /** @todo false negative: check constructors for side effects */ 
  70     if (var->typeScope() && var->typeScope()->numConstructors == 0 && 
  71         (var->typeScope()->varlist.empty() || var->type()->needInitialization == Type::NeedInitialization::True) && 
  72         var->type()->derivedFrom.empty()) 
  73         return false; 
  74  
  75     return true; 
  76 }

 175 void CheckLeakAutoVar::check() 
 176 { 
 177     if (mSettings->clang) 
 178         return; 
 179  
 180     const SymbolDatabase *symbolDatabase = mTokenizer->getSymbolDatabase(); 
 181  
 182     // Local variables that are known to be non-zero. 
 183     const std::set<int> notzero; 
 184  
 185     // Check function scopes 
 186     for (const Scope * scope : symbolDatabase->functionScopes) { 
 187         if (scope->hasInlineOrLambdaFunction()) 
 188             continue; 
 189  
 190         // Empty variable info 
 191         VarInfo varInfo; 
 192  
 193         checkScope(scope->bodyStart, &varInfo, notzero, 0); 
 194  
 195         varInfo.conditionalAlloc.clear(); 
 196  
 197         // Clear reference arguments from varInfo.. 
 198         std::map<int, VarInfo::AllocInfo>::iterator it = varInfo.alloctype.begin(); 
 199         while (it != varInfo.alloctype.end()) { 
 200             const Variable *var = symbolDatabase->getVariableFromVarId(it->first); 
 201             if (!var || 
 202                 (var->isArgument() && var->isReference()) || 
 203                 (!var->isArgument() && !var->isLocal())) 
 204                 varInfo.alloctype.erase(it++); 
 205             else 
 206                 ++it; 
 207         } 
 208  
 209         ret(scope->bodyEnd, varInfo); 
 210     } 
 211 }

 225 { 
 226     for (const Token *tok = startToken; tok && tok != endToken; tok = tok->next()) {

 230             return true;

 232             return false;

 237 static bool isLocalVarNoAutoDealloc(const Token *varTok, const bool isCpp) 
 238 { 
 239     // not a local variable nor argument? 
 240     const Variable *var = varTok->variable(); 
 241     if (!var) 
 242         return true; 
 243     if (!var->isArgument() && (!var->isLocal() || var->isStatic())) 
 244         return false; 
 245  
 246     // Don't check reference variables 
 247     if (var->isReference()) 
 248         return false; 
 249  
 250     // non-pod variable 
 251     if (isCpp) { 
 252         // Possibly automatically deallocated memory 
 253         if (isAutoDealloc(var) && Token::Match(varTok, "%var% = new")) 
 254             return false; 
 255         if (!var->isPointer() && !var->typeStartToken()->isStandardType()) 
 256             return false; 
 257     } 
 258     return true; 
 259 }

 261 /** checks if nameToken is a name of a function in a function call: 
 262 *     func(arg)

 266 * @return opening parenthesis token or NULL if not a function call

 269 static const Token * isFunctionCall(const Token * nameToken) 
 270 { 
 271     if (nameToken->isName()) { 
 272         nameToken = nameToken->next(); 
 273         // check if function is a template 
 274         if (nameToken && nameToken->link() && nameToken->str() == "<") { 
 275             // skip template arguments 
 276             nameToken = nameToken->link()->next(); 
 277         } 
 278         // check for '(' 
 279         if (nameToken && nameToken->link() && nameToken->str() == "(") { 
 280             // returning opening parenthesis pointer 
 281             return nameToken; 
 282         } 
 283     } 
 284     return nullptr; 
 285 }

 293     static const nonneg int recursiveLimit = 300;

 295     static const nonneg int recursiveLimit = 1000;

 298         throw InternalError(startToken, "Internal limit: CheckLeakAutoVar::checkScope() Maximum recursive count of 1000 reached.", InternalError::LIMIT);

 304     // Parse all tokens 
 305     const Token * const endToken = startToken->link();

 307         if (!tok->scope()->isExecutable()) { 
 308             tok = tok->scope()->bodyEnd; 
 309             if (!tok) // Ticket #6666 (crash upon invalid code) 
 310                 break; 
 311         } 
 312  
 313         // check each token

 315             const Token * nextTok = checkTokenInsideExpression(tok, varInfo); 
 316             if (nextTok) { 
 317                 tok = nextTok; 
 318                 continue; 
 319             } 
 320         } 
 321  
 322  
 323         // look for end of statement 
 324         if (!Token::Match(tok, "[;{},]") || Token::Match(tok->next(), "[;{},]")) 
 325             continue; 
 326  
 327         tok = tok->next(); 
 328         if (!tok || tok == endToken) 
 329             break; 
 330  
 331         // parse statement, skip to last member 
 332         const Token *varTok = tok; 
 333         while (Token::Match(varTok, "%name% ::|. %name% !!(")) 
 334             varTok = varTok->tokAt(2); 
 335  
 336         const Token *ftok = tok; 
 337         if (ftok->str() == "::") 
 338             ftok = ftok->next(); 
 339         while (Token::Match(ftok, "%name% :: %name%")) 
 340             ftok = ftok->tokAt(2); 
 341  
 342         // assignment.. 
 343         if (Token::Match(varTok, "%var% =")) { 
 344             const Token* const tokAssignOp = varTok->next(); 
 345  
 346             // taking address of another variable.. 
 347             if (Token::Match(tokAssignOp, "= %var% [+;]")) { 
 348                 if (varTok->tokAt(2)->varId() != varTok->varId()) { 
 349                     // If variable points at allocated memory => error

 352                     // no multivariable checking currently => bail out for rhs variables 
 353                     for (const Token *tok2 = varTok; tok2; tok2 = tok2->next()) { 
 354                         if (tok2->str() == ";") { 
 355                             break; 
 356                         } 
 357                         if (tok2->varId()) { 
 358                             varInfo->erase(tok2->varId()); 
 359                         } 
 360                     } 
 361                 } 
 362             }

 364             // right ast part (after `=` operator) 
 365             const Token* tokRightAstOperand = tokAssignOp->astOperand2();

 367                 tokRightAstOperand = tokRightAstOperand->astOperand2() ? tokRightAstOperand->astOperand2() : tokRightAstOperand->astOperand1(); 
 368  
 369             // is variable used in rhs? 
 370             if (isVarUsedInTree(tokRightAstOperand, varTok->varId()))

 374             if (conditionalAlloc.find(varTok->varId()) == conditionalAlloc.end())

 378             if (!isLocalVarNoAutoDealloc(varTok, mTokenizer->isCPP())) 
 379                 continue;

 381             // allocation? 
 382             const Token *const fTok = tokRightAstOperand ? tokRightAstOperand->previous() : nullptr; 
 383             if (Token::Match(fTok, "%type% (")) { 
 384                 const Library::AllocFunc* f = mSettings->library.getAllocFuncInfo(fTok); 
 385                 if (f && f->arg == -1) {

 388                     varAlloc.status = VarInfo::ALLOC;

 403             // track the execution for a later if condition.

 405                 notzero.insert(varTok->varId());

 574                 alloctype.insert(varInfo2.alloctype.begin(), varInfo2.alloctype.end()); 
 575  
 576                 possibleUsage.insert(varInfo1.possibleUsage.begin(), varInfo1.possibleUsage.end()); 
 577                 possibleUsage.insert(varInfo2.possibleUsage.begin(), varInfo2.possibleUsage.end()); 
 578             } 
 579         } 
 580  
 581         // unknown control.. (TODO: handle loops) 
 582         else if ((Token::Match(tok, "%type% (") && Token::simpleMatch(tok->linkAt(1), ") {")) || Token::simpleMatch(tok, "do {")) { 
 583             varInfo->clear(); 
 584             break; 
 585         } 
 586  
 587         // return 
 588         else if (tok->str() == "return") { 
 589             ret(tok, *varInfo); 
 590             varInfo->clear(); 
 591         } 
 592  
 593         // throw 
 594         else if (mTokenizer->isCPP() && tok->str() == "throw") { 
 595             bool tryFound = false; 
 596             const Scope* scope = tok->scope(); 
 597             while (scope && scope->isExecutable()) { 
 598                 if (scope->type == Scope::eTry) 
 599                     tryFound = true; 
 600                 scope = scope->nestedIn; 
 601             } 
 602             // If the execution leaves the function then treat it as return 
 603             if (!tryFound) 
 604                 ret(tok, *varInfo); 
 605             varInfo->clear(); 
 606         } 
 607  
 608         // delete 
 609         else if (mTokenizer->isCPP() && tok->str() == "delete") { 
 610             const Token * delTok = tok; 
 611             const bool arrayDelete = Token::simpleMatch(tok->next(), "[ ]"); 
 612             if (arrayDelete) 
 613                 tok = tok->tokAt(3); 
 614             else 
 615                 tok = tok->next(); 
 616             if (tok->str() == "(") 
 617                 tok = tok->next(); 
 618             while (Token::Match(tok, "%name% ::|.")) 
 619                 tok = tok->tokAt(2); 
 620             const bool isnull = tok->hasKnownIntValue() && tok->values().front().intvalue == 0; 
 621             if (!isnull && tok->varId() && tok->strAt(1) != "[") { 
 622                 const VarInfo::AllocInfo allocation(arrayDelete ? NEW_ARRAY : NEW, VarInfo::DEALLOC, delTok); 
 623                 changeAllocStatus(varInfo, allocation, tok, tok); 
 624             } 
 625         } 
 626  
 627         // Function call.. 
 628         else if (isFunctionCall(ftok)) { 
 629             const Token * openingPar = isFunctionCall(ftok); 
 630             const Library::AllocFunc* af = mSettings->library.getDeallocFuncInfo(ftok); 
 631             VarInfo::AllocInfo allocation(af ? af->groupId : 0, VarInfo::DEALLOC, ftok); 
 632             if (allocation.type == 0) 
 633                 allocation.status = VarInfo::NOALLOC; 
 634             functionCall(ftok, openingPar, varInfo, allocation, af); 
 635  
 636             tok = ftok->next()->link(); 
 637  
 638             // Handle scopes that might be noreturn 
 639             if (allocation.status == VarInfo::NOALLOC && Token::simpleMatch(tok, ") ; }")) { 
 640                 const std::string &functionName(tok->link()->previous()->str()); 
 641                 bool unknown = false; 
 642                 if (mTokenizer->isScopeNoReturn(tok->tokAt(2), &unknown)) { 
 643                     if (!unknown) 
 644                         varInfo->clear(); 
 645                     else if (!mSettings->library.isLeakIgnore(functionName) && !mSettings->library.isUse(functionName)) 
 646                         varInfo->possibleUsageAll(functionName); 
 647                 } 
 648             } 
 649  
 650             continue; 
 651         } 
 652  
 653         // goto => weird execution path 
 654         else if (tok->str() == "goto") { 
 655             varInfo->clear(); 
 656         } 
 657  
 658         // continue/break 
 659         else if (Token::Match(tok, "continue|break ;")) { 
 660             varInfo->clear(); 
 661         } 
 662  
 663         // Check smart pointer 
 664         else if (Token::Match(ftok, "%name% <") && mSettings->library.isSmartPointer(tok)) { 
 665             const Token * typeEndTok = ftok->linkAt(1); 
 666             if (!Token::Match(typeEndTok, "> %var% {|( %var% ,|)|}")) 
 667                 continue; 
 668  
 669             tok = typeEndTok->linkAt(2); 
 670  
 671             const int varid = typeEndTok->next()->varId(); 
 672             if (isPointerReleased(typeEndTok->tokAt(2), endToken, varid)) 
 673                 continue; 
 674  
 675             bool arrayDelete = false; 
 676             if (Token::findsimplematch(ftok->next(), "[ ]", typeEndTok)) 
 677                 arrayDelete = true; 
 678  
 679             // Check deleter 
 680             const Token * deleterToken = nullptr; 
 681             const Token * endDeleterToken = nullptr; 
 682             const Library::AllocFunc* af = nullptr; 
 683             if (Token::Match(ftok, "unique_ptr < %type% ,")) { 
 684                 deleterToken = ftok->tokAt(4); 
 685                 endDeleterToken = typeEndTok; 
 686             } else if (Token::Match(typeEndTok, "> %var% {|( %var% ,")) { 
 687                 deleterToken = typeEndTok->tokAt(5); 
 688                 endDeleterToken = typeEndTok->linkAt(2); 
 689             } 
 690             if (deleterToken) { 
 691                 // Skip the decaying plus in expressions like +[](T*){} 
 692                 if (deleterToken->str() == "+") { 
 693                     deleterToken = deleterToken->next(); 
 694                 } 
 695                 // Check if its a pointer to a function 
 696                 const Token * dtok = Token::findmatch(deleterToken, "& %name%", endDeleterToken); 
 697                 if (dtok) { 
 698                     af = mSettings->library.getDeallocFuncInfo(dtok->tokAt(1)); 
 699                 } else { 
 700                     const Token * tscopeStart = nullptr; 
 701                     const Token * tscopeEnd = nullptr; 
 702                     // If the deleter is a lambda, check if it calls the dealloc function 
 703                     if (deleterToken->str() == "[" && 
 704                         Token::simpleMatch(deleterToken->link(), "] (") && 
 705                         // TODO: Check for mutable keyword 
 706                         Token::simpleMatch(deleterToken->link()->linkAt(1), ") {")) { 
 707                         tscopeStart = deleterToken->link()->linkAt(1)->tokAt(1); 
 708                         tscopeEnd = tscopeStart->link(); 
 709                         // If the deleter is a class, check if class calls the dealloc function 
 710                     } else if ((dtok = Token::findmatch(deleterToken, "%type%", endDeleterToken)) && dtok->type()) { 
 711                         const Scope * tscope = dtok->type()->classScope; 
 712                         if (tscope) { 
 713                             tscopeStart = tscope->bodyStart; 
 714                             tscopeEnd = tscope->bodyEnd; 
 715                         } 
 716                     } 
 717  
 718                     if (tscopeStart && tscopeEnd) { 
 719                         for (const Token *tok2 = tscopeStart; tok2 != tscopeEnd; tok2 = tok2->next()) { 
 720                             af = mSettings->library.getDeallocFuncInfo(tok2); 
 721                             if (af) 
 722                                 break; 
 723                         } 
 724                     } 
 725                 } 
 726             } 
 727  
 728             const Token * vtok = typeEndTok->tokAt(3); 
 729             const VarInfo::AllocInfo allocation(af ? af->groupId : (arrayDelete ? NEW_ARRAY : NEW), VarInfo::OWNED, ftok); 
 730             changeAllocStatus(varInfo, allocation, vtok, vtok); 
 731         } 
 732     } 
 733 }

 736 const Token * CheckLeakAutoVar::checkTokenInsideExpression(const Token * const tok, VarInfo *varInfo) 
 737 { 
 738     // Deallocation and then dereferencing pointer.. 
 739     if (tok->varId() > 0) { 
 740         // TODO : Write a separate checker for this that uses valueFlowForward. 
 741         const std::map<int, VarInfo::AllocInfo>::const_iterator var = varInfo->alloctype.find(tok->varId()); 
 742         if (var != varInfo->alloctype.end()) { 
 743             bool unknown = false; 
 744             if (var->second.status == VarInfo::DEALLOC && CheckNullPointer::isPointerDeRef(tok, unknown, mSettings) && !unknown) { 
 745                 deallocUseError(tok, tok->str()); 
 746             } else if (Token::simpleMatch(tok->tokAt(-2), "= &")) { 
 747                 varInfo->erase(tok->varId()); 
 748             } else { 
 749                 // check if tok is assigned into another variable 
 750                 const Token *rhs = tok; 
 751                 while (rhs->astParent()) { 
 752                     if (rhs->astParent()->str() == "=") 
 753                         break; 
 754                     rhs = rhs->astParent(); 
 755                 } 
 756                 if (rhs->varId() == tok->varId()) { 
 757                     // simple assignment 
 758                     varInfo->erase(tok->varId()); 
 759                 } else if (rhs->str() == "(" && mSettings->library.returnValue(rhs->astOperand1()) != emptyString) { 
 760                     // #9298, assignment through return value of a function 
 761                     const std::string &returnValue = mSettings->library.returnValue(rhs->astOperand1()); 
 762                     if (returnValue.compare(0, 3, "arg") == 0) { 
 763                         int argn; 
 764                         const Token *func = getTokenArgumentFunction(tok, argn); 
 765                         if (func) { 
 766                             const std::string arg = "arg" + std::to_string(argn + 1); 
 767                             if (returnValue == arg) { 
 768                                 varInfo->erase(tok->varId()); 
 769                             } 
 770                         } 
 771                     } 
 772                 } 
 773             } 
 774         } else if (Token::Match(tok->previous(), "& %name% = %var% ;")) { 
 775             varInfo->referenced.insert(tok->tokAt(2)->varId()); 
 776         } 
 777     } 
 778  
 779     // check for function call 
 780     const Token * const openingPar = isFunctionCall(tok); 
 781     if (openingPar) { 
 782         const Library::AllocFunc* allocFunc = mSettings->library.getDeallocFuncInfo(tok); 
 783         VarInfo::AllocInfo alloc(allocFunc ? allocFunc->groupId : 0, VarInfo::DEALLOC, tok); 
 784         if (alloc.type == 0) 
 785             alloc.status = VarInfo::NOALLOC; 
 786         functionCall(tok, openingPar, varInfo, alloc, nullptr); 
 787         const std::string &returnValue = mSettings->library.returnValue(tok); 
 788         if (returnValue.compare(0, 3, "arg") == 0) 
 789             // the function returns one of its argument, we need to process a potential assignment 
 790             return openingPar; 
 791         return openingPar->link(); 
 792     } 
 793  
 794     return nullptr; 
 795 }

 799 { 
 800     const Library::AllocFunc* f = mSettings->library.getReallocFuncInfo(fTok); 
 801     if (f && f->arg == -1 && f->reallocArg > 0 && f->reallocArg <= numberOfArguments(fTok)) { 
 802         const Token* argTok = getArguments(fTok).at(f->reallocArg - 1); 
 803         VarInfo::AllocInfo& argAlloc = alloctype[argTok->varId()]; 
 804         VarInfo::AllocInfo& retAlloc = alloctype[retTok->varId()]; 
 805         if (argAlloc.type != 0 && argAlloc.type != f->groupId) 
 806             mismatchError(fTok, argAlloc.allocTok, argTok->str()); 
 807         argAlloc.status = VarInfo::DEALLOC; 
 808         argAlloc.allocTok = fTok; 
 809         retAlloc.type = f->groupId; 
 810         retAlloc.status = VarInfo::ALLOC; 
 811         retAlloc.allocTok = fTok; 
 812     }

 816 void CheckLeakAutoVar::changeAllocStatus(VarInfo *varInfo, const VarInfo::AllocInfo& allocation, const Token* tok, const Token* arg) 
 817 { 
 818     std::map<int, VarInfo::AllocInfo> &alloctype = varInfo->alloctype; 
 819     const std::map<int, VarInfo::AllocInfo>::iterator var = alloctype.find(arg->varId()); 
 820     if (var != alloctype.end()) { 
 821         if (allocation.status == VarInfo::NOALLOC) { 
 822             // possible usage 
 823             varInfo->possibleUsage[arg->varId()] = tok->str(); 
 824             if (var->second.status == VarInfo::DEALLOC && arg->previous()->str() == "&") 
 825                 varInfo->erase(arg->varId()); 
 826         } else if (var->second.managed()) { 
 827             doubleFreeError(tok, var->second.allocTok, arg->str(), allocation.type); 
 828         } else if (var->second.type != allocation.type) { 
 829             // mismatching allocation and deallocation 
 830             mismatchError(tok, var->second.allocTok, arg->str()); 
 831             varInfo->erase(arg->varId()); 
 832         } else { 
 833             // deallocation 
 834             var->second.status = allocation.status; 
 835             var->second.type = allocation.type; 
 836             var->second.allocTok = allocation.allocTok; 
 837         } 
 838     } else if (allocation.status != VarInfo::NOALLOC) { 
 839         alloctype[arg->varId()].status = VarInfo::DEALLOC; 
 840         alloctype[arg->varId()].allocTok = tok; 
 841     } 
 842 }

 844 void CheckLeakAutoVar::functionCall(const Token *tokName, const Token *tokOpeningPar, VarInfo *varInfo, const VarInfo::AllocInfo& allocation, const Library::AllocFunc* af) 
 845 { 
 846     // Ignore function call? 
 847     if (mSettings->library.isLeakIgnore(tokName->str())) 
 848         return; 
 849     if (mSettings->library.getReallocFuncInfo(tokName)) 
 850         return; 
 851  
 852     const Token * const tokFirstArg = tokOpeningPar->next(); 
 853     if (!tokFirstArg || tokFirstArg->str() == ")") { 
 854         // no arguments 
 855         return; 
 856     } 
 857  
 858     int argNr = 1; 
 859     for (const Token *arg = tokFirstArg; arg; arg = arg->nextArgument()) { 
 860         if (mTokenizer->isCPP() && arg->str() == "new") { 
 861             arg = arg->next(); 
 862             if (Token::simpleMatch(arg, "( std :: nothrow )")) 
 863                 arg = arg->tokAt(5); 
 864         } 
 865  
 866         // Skip casts 
 867         while (arg && arg->isCast()) 
 868             arg = arg->astOperand2() ? arg->astOperand2() : arg->astOperand1(); 
 869         const Token * const argTypeStartTok = arg; 
 870  
 871         while (Token::Match(arg, "%name% .|:: %name%")) 
 872             arg = arg->tokAt(2); 
 873  
 874         if (Token::Match(arg, "%var% [-,)] !!.") || Token::Match(arg, "& %var%")) { 
 875             // goto variable 
 876             if (arg->str() == "&") 
 877                 arg = arg->next(); 
 878  
 879             const bool isnull = arg->hasKnownIntValue() && arg->values().front().intvalue == 0; 
 880  
 881             // Is variable allocated? 
 882             if (!isnull && (!af || af->arg == argNr)) 
 883                 changeAllocStatus(varInfo, allocation, tokName, arg); 
 884         } 
 885         // Check smart pointer 
 886         else if (Token::Match(arg, "%name% < %type%") && mSettings->library.isSmartPointer(argTypeStartTok)) { 
 887             const Token * typeEndTok = arg->linkAt(1); 
 888             const Token * allocTok = nullptr; 
 889             if (!Token::Match(typeEndTok, "> {|( %var% ,|)|}")) 
 890                 continue; 
 891  
 892             bool arrayDelete = false; 
 893             if (Token::findsimplematch(arg->next(), "[ ]", typeEndTok)) 
 894                 arrayDelete = true; 
 895  
 896             // Check deleter 
 897             const Token * deleterToken = nullptr; 
 898             const Token * endDeleterToken = nullptr; 
 899             const Library::AllocFunc* sp_af = nullptr; 
 900             if (Token::Match(arg, "unique_ptr < %type% ,")) { 
 901                 deleterToken = arg->tokAt(4); 
 902                 endDeleterToken = typeEndTok; 
 903             } else if (Token::Match(typeEndTok, "> {|( %var% ,")) { 
 904                 deleterToken = typeEndTok->tokAt(4); 
 905                 endDeleterToken = typeEndTok->linkAt(1); 
 906             } 
 907             if (deleterToken) { 
 908                 // Check if its a pointer to a function 
 909                 const Token * dtok = Token::findmatch(deleterToken, "& %name%", endDeleterToken); 
 910                 if (dtok) { 
 911                     sp_af = mSettings->library.getDeallocFuncInfo(dtok->tokAt(1)); 
 912                 } else { 
 913                     // If the deleter is a class, check if class calls the dealloc function 
 914                     dtok = Token::findmatch(deleterToken, "%type%", endDeleterToken); 
 915                     if (dtok && dtok->type()) { 
 916                         const Scope * tscope = dtok->type()->classScope; 
 917                         for (const Token *tok2 = tscope->bodyStart; tok2 != tscope->bodyEnd; tok2 = tok2->next()) { 
 918                             sp_af = mSettings->library.getDeallocFuncInfo(tok2); 
 919                             if (sp_af) { 
 920                                 allocTok = tok2; 
 921                                 break; 
 922                             } 
 923                         } 
 924                     } 
 925                 } 
 926             } 
 927  
 928             const Token * vtok = typeEndTok->tokAt(2); 
 929             const VarInfo::AllocInfo sp_allocation(sp_af ? sp_af->groupId : (arrayDelete ? NEW_ARRAY : NEW), VarInfo::OWNED, allocTok); 
 930             changeAllocStatus(varInfo, sp_allocation, vtok, vtok); 
 931         } else { 
 932             checkTokenInsideExpression(arg, varInfo); 
 933         } 
 934         // TODO: check each token in argument expression (could contain multiple variables) 
 935         argNr++; 
 936     } 
 937 }

 957 void CheckLeakAutoVar::ret(const Token *tok, const VarInfo &varInfo) 
 958 { 
 959     const std::map<int, VarInfo::AllocInfo> &alloctype = varInfo.alloctype; 
 960     const std::map<int, std::string> &possibleUsage = varInfo.possibleUsage; 
 961  
 962     const SymbolDatabase *symbolDatabase = mTokenizer->getSymbolDatabase(); 
 963     for (std::map<int, VarInfo::AllocInfo>::const_iterator it = alloctype.begin(); it != alloctype.end(); ++it) { 
 964         // don't warn if variable is conditionally allocated 
 965         if (!it->second.managed() && varInfo.conditionalAlloc.find(it->first) != varInfo.conditionalAlloc.end()) 
 966             continue; 
 967  
 968         // don't warn if there is a reference of the variable 
 969         if (varInfo.referenced.find(it->first) != varInfo.referenced.end()) 
 970             continue; 
 971  
 972         const int varid = it->first; 
 973         const Variable *var = symbolDatabase->getVariableFromVarId(varid); 
 974         if (var) { 
 975             bool used = false; 
 976             for (const Token *tok2 = tok; tok2; tok2 = tok2->next()) { 
 977                 if (tok2->str() == ";") 
 978                     break; 
 979                 if (!Token::Match(tok2, "return|(|{|,")) 
 980                     continue; 
 981  
 982                 const Token* tok3 = tok2->next(); 
 983                 while (tok3 && tok3->isCast() && tok3->valueType() && 
 984                        (tok3->valueType()->pointer || 
 985                         (tok3->valueType()->typeSize(*mSettings) == 0) || 
 986                         (tok3->valueType()->typeSize(*mSettings) >= mSettings->sizeof_pointer))) 
 987                     tok3 = tok3->astOperand2() ? tok3->astOperand2() : tok3->astOperand1(); 
 988                 if (Token::Match(tok3, "%varid%", varid)) 
 989                     tok2 = tok3->next(); 
 990                 else if (Token::Match(tok3, "& %varid% . %name%", varid)) 
 991                     tok2 = tok3->tokAt(4); 
 992                 else 
 993                     continue; 
 994                 if (Token::Match(tok2, "[});,]")) { 
 995                     used = true; 
 996                     break; 
 997                 } 
 998             } 
 999  
1000             // return deallocated pointer 
1001             if (used && it->second.status == VarInfo::DEALLOC) 
1002                 deallocReturnError(tok, it->second.allocTok, var->name()); 
1003  
1004             else if (!used && !it->second.managed()) { 
1005                 const std::map<int, std::string>::const_iterator use = possibleUsage.find(varid); 
1006                 if (use == possibleUsage.end()) { 
1007                     leakError(tok, var->name(), it->second.type); 
1008                 } else { 
1009                     configurationInfo(tok, use->second); 
1010                 } 
1011             } 
1012         } 
1013     }

 175 void CheckLeakAutoVar::check() 
 176 { 
 177     if (mSettings->clang) 
 178         return; 
 179  
 180     const SymbolDatabase *symbolDatabase = mTokenizer->getSymbolDatabase(); 
 181  
 182     // Local variables that are known to be non-zero. 
 183     const std::set<int> notzero; 
 184  
 185     // Check function scopes 
 186     for (const Scope * scope : symbolDatabase->functionScopes) { 
 187         if (scope->hasInlineOrLambdaFunction()) 
 188             continue; 
 189  
 190         // Empty variable info 
 191         VarInfo varInfo; 
 192  
 193         checkScope(scope->bodyStart, &varInfo, notzero, 0); 
 194  
 195         varInfo.conditionalAlloc.clear(); 
 196  
 197         // Clear reference arguments from varInfo.. 
 198         std::map<int, VarInfo::AllocInfo>::iterator it = varInfo.alloctype.begin(); 
 199         while (it != varInfo.alloctype.end()) { 
 200             const Variable *var = symbolDatabase->getVariableFromVarId(it->first); 
 201             if (!var || 
 202                 (var->isArgument() && var->isReference()) || 
 203                 (!var->isArgument() && !var->isLocal())) 
 204                 varInfo.alloctype.erase(it++); 
 205             else 
 206                 ++it; 
 207         } 
 208  
 209         ret(scope->bodyEnd, varInfo); 
 210     } 
 211 }

 213 static bool isVarUsedInTree(const Token *tok, nonneg int varid) 
 214 { 
 215     if (!tok) 
 216         return false; 
 217     if (tok->varId() == varid) 
 218         return true; 
 219     if (tok->str() == "(" && Token::simpleMatch(tok->astOperand1(), "sizeof")) 
 220         return false; 
 221     return isVarUsedInTree(tok->astOperand1(), varid) || isVarUsedInTree(tok->astOperand2(), varid); 
 222 } 
 223  
 224 static bool isPointerReleased(const Token *startToken, const Token *endToken, nonneg int varid) 
 225 { 
 226     for (const Token *tok = startToken; tok && tok != endToken; tok = tok->next()) { 
 227         if (tok->varId() != varid) 
 228             continue; 
 229         if (Token::Match(tok, "%var% . release ( )")) 
 230             return true; 
 231         if (Token::Match(tok, "%var% =")) 
 232             return false; 
 233     } 
 234     return false; 
 235 } 
 236  
 237 static bool isLocalVarNoAutoDealloc(const Token *varTok, const bool isCpp) 
 238 { 
 239     // not a local variable nor argument? 
 240     const Variable *var = varTok->variable(); 
 241     if (!var) 
 242         return true; 
 243     if (!var->isArgument() && (!var->isLocal() || var->isStatic())) 
 244         return false; 
 245  
 246     // Don't check reference variables 
 247     if (var->isReference()) 
 248         return false; 
 249  
 250     // non-pod variable 
 251     if (isCpp) { 
 252         // Possibly automatically deallocated memory 
 253         if (isAutoDealloc(var) && Token::Match(varTok, "%var% = new")) 
 254             return false; 
 255         if (!var->isPointer() && !var->typeStartToken()->isStandardType()) 
 256             return false; 
 257     } 
 258     return true; 
 259 }

 261 /** checks if nameToken is a name of a function in a function call: 
 262 *     func(arg) 
 263 * or 
 264 *     func<temp1_arg>(arg) 
 265 * @param nameToken Function name token 
 266 * @return opening parenthesis token or NULL if not a function call 
 267 */ 
 268  
 269 static const Token * isFunctionCall(const Token * nameToken) 
 270 { 
 271     if (nameToken->isName()) { 
 272         nameToken = nameToken->next(); 
 273         // check if function is a template 
 274         if (nameToken && nameToken->link() && nameToken->str() == "<") { 
 275             // skip template arguments 
 276             nameToken = nameToken->link()->next(); 
 277         } 
 278         // check for '(' 
 279         if (nameToken && nameToken->link() && nameToken->str() == "(") { 
 280             // returning opening parenthesis pointer 
 281             return nameToken; 
 282         } 
 283     } 
 284     return nullptr; 
 285 } 
 286  
 287 void CheckLeakAutoVar::checkScope(const Token * const startToken, 
 288                                   VarInfo *varInfo, 
 289                                   std::set<int> notzero, 
 290                                   nonneg int recursiveCount) 
 291 { 
 292 #if ASAN 
 293     static const nonneg int recursiveLimit = 300; 
 294 #else 
 295     static const nonneg int recursiveLimit = 1000; 
 296 #endif 
 297     if (++recursiveCount > recursiveLimit)    // maximum number of "else if ()" 
 298         throw InternalError(startToken, "Internal limit: CheckLeakAutoVar::checkScope() Maximum recursive count of 1000 reached.", InternalError::LIMIT); 
 299  
 300     std::map<int, VarInfo::AllocInfo> &alloctype = varInfo->alloctype; 
 301     std::map<int, std::string> &possibleUsage = varInfo->possibleUsage; 
 302     const std::set<int> conditionalAlloc(varInfo->conditionalAlloc); 
 303  
 304     // Parse all tokens 
 305     const Token * const endToken = startToken->link(); 
 306     for (const Token *tok = startToken; tok && tok != endToken; tok = tok->next()) { 
 307         if (!tok->scope()->isExecutable()) { 
 308             tok = tok->scope()->bodyEnd; 
 309             if (!tok) // Ticket #6666 (crash upon invalid code) 
 310                 break; 
 311         } 
 312  
 313         // check each token 
 314         { 
 315             const Token * nextTok = checkTokenInsideExpression(tok, varInfo); 
 316             if (nextTok) { 
 317                 tok = nextTok; 
 318                 continue; 
 319             } 
 320         } 
 321  
 322  
 323         // look for end of statement 
 324         if (!Token::Match(tok, "[;{},]") || Token::Match(tok->next(), "[;{},]")) 
 325             continue; 
 326  
 327         tok = tok->next(); 
 328         if (!tok || tok == endToken) 
 329             break; 
 330  
 331         // parse statement, skip to last member 
 332         const Token *varTok = tok; 
 333         while (Token::Match(varTok, "%name% ::|. %name% !!(")) 
 334             varTok = varTok->tokAt(2); 
 335  
 336         const Token *ftok = tok; 
 337         if (ftok->str() == "::") 
 338             ftok = ftok->next(); 
 339         while (Token::Match(ftok, "%name% :: %name%")) 
 340             ftok = ftok->tokAt(2); 
 341  
 342         // assignment.. 
 343         if (Token::Match(varTok, "%var% =")) { 
 344             const Token* const tokAssignOp = varTok->next(); 
 345  
 346             // taking address of another variable.. 
 347             if (Token::Match(tokAssignOp, "= %var% [+;]")) { 
 348                 if (varTok->tokAt(2)->varId() != varTok->varId()) { 
 349                     // If variable points at allocated memory => error 
 350                     leakIfAllocated(varTok, *varInfo); 
 351  
 352                     // no multivariable checking currently => bail out for rhs variables 
 353                     for (const Token *tok2 = varTok; tok2; tok2 = tok2->next()) { 
 354                         if (tok2->str() == ";") { 
 355                             break; 
 356                         } 
 357                         if (tok2->varId()) { 
 358                             varInfo->erase(tok2->varId()); 
 359                         } 
 360                     } 
 361                 } 
 362             } 
 363  
 364             // right ast part (after `=` operator) 
 365             const Token* tokRightAstOperand = tokAssignOp->astOperand2(); 
 366             while (tokRightAstOperand && tokRightAstOperand->isCast()) 
 367                 tokRightAstOperand = tokRightAstOperand->astOperand2() ? tokRightAstOperand->astOperand2() : tokRightAstOperand->astOperand1(); 
 368  
 369             // is variable used in rhs? 
 370             if (isVarUsedInTree(tokRightAstOperand, varTok->varId())) 
 371                 continue; 
 372  
 373             // Variable has already been allocated => error 
 374             if (conditionalAlloc.find(varTok->varId()) == conditionalAlloc.end()) 
 375                 leakIfAllocated(varTok, *varInfo); 
 376             varInfo->erase(varTok->varId()); 
 377  
 378             if (!isLocalVarNoAutoDealloc(varTok, mTokenizer->isCPP())) 
 379                 continue; 
 380  
 381             // allocation? 
 382             const Token *const fTok = tokRightAstOperand ? tokRightAstOperand->previous() : nullptr; 
 383             if (Token::Match(fTok, "%type% (")) { 
 384                 const Library::AllocFunc* f = mSettings->library.getAllocFuncInfo(fTok); 
 385                 if (f && f->arg == -1) { 
 386                     VarInfo::AllocInfo& varAlloc = alloctype[varTok->varId()]; 
 387                     varAlloc.type = f->groupId; 
 388                     varAlloc.status = VarInfo::ALLOC; 
 389                     varAlloc.allocTok = fTok; 
 390                 } 
 391  
 392                 changeAllocStatusIfRealloc(alloctype, fTok, varTok); 
 393             } else if (mTokenizer->isCPP() && Token::Match(varTok->tokAt(2), "new !!(")) { 
 394                 const Token* tok2 = varTok->tokAt(2)->astOperand1(); 
 395                 const bool arrayNew = (tok2 && (tok2->str() == "[" || (tok2->str() == "(" && tok2->astOperand1() && tok2->astOperand1()->str() == "["))); 
 396                 VarInfo::AllocInfo& varAlloc = alloctype[varTok->varId()]; 
 397                 varAlloc.type = arrayNew ? NEW_ARRAY : NEW; 
 398                 varAlloc.status = VarInfo::ALLOC; 
 399                 varAlloc.allocTok = varTok->tokAt(2); 
 400             } 
 401  
 402             // Assigning non-zero value variable. It might be used to 
 403             // track the execution for a later if condition. 
 404             if (Token::Match(varTok->tokAt(2), "%num% ;") && MathLib::toLongNumber(varTok->strAt(2)) != 0) 
 405                 notzero.insert(varTok->varId()); 
 406             else if (Token::Match(varTok->tokAt(2), "- %type% ;") && varTok->tokAt(3)->isUpperCaseName()) 
 407                 notzero.insert(varTok->varId()); 
 408             else 
 409                 notzero.erase(varTok->varId()); 
 410         } 
 411  
 412         // if/else 
 413         else if (Token::simpleMatch(tok, "if (")) { 
 414             // Parse function calls inside the condition 
 415  
 416             const Token * closingParenthesis = tok->linkAt(1); 
 417             for (const Token *innerTok = tok->tokAt(2); innerTok && innerTok != closingParenthesis; innerTok = innerTok->next()) { 
 418                 // TODO: replace with checkTokenInsideExpression() 
 419  
 420                 if (!isLocalVarNoAutoDealloc(innerTok, mTokenizer->isCPP())) 
 421                     continue; 
 422  
 423                 if (Token::Match(innerTok, "%var% =") && innerTok->astParent() == innerTok->next()) { 
 424                     // allocation? 
 425                     // right ast part (after `=` operator) 
 426                     const Token* tokRightAstOperand = innerTok->next()->astOperand2(); 
 427                     while (tokRightAstOperand && tokRightAstOperand->isCast()) 
 428                         tokRightAstOperand = tokRightAstOperand->astOperand2() ? tokRightAstOperand->astOperand2() : tokRightAstOperand->astOperand1(); 
 429                     if (tokRightAstOperand && Token::Match(tokRightAstOperand->previous(), "%type% (")) { 
 430                         const Library::AllocFunc* f = mSettings->library.getAllocFuncInfo(tokRightAstOperand->previous()); 
 431                         if (f && f->arg == -1) { 
 432                             VarInfo::AllocInfo& varAlloc = alloctype[innerTok->varId()]; 
 433                             varAlloc.type = f->groupId; 
 434                             varAlloc.status = VarInfo::ALLOC; 
 435                             varAlloc.allocTok = tokRightAstOperand->previous(); 
 436                         } else { 
 437                             // Fixme: warn about leak 
 438                             alloctype.erase(innerTok->varId()); 
 439                         } 
 440  
 441                         changeAllocStatusIfRealloc(alloctype, innerTok->tokAt(2), varTok); 
 442                     } else if (mTokenizer->isCPP() && Token::Match(innerTok->tokAt(2), "new !!(")) { 
 443                         const Token* tok2 = innerTok->tokAt(2)->astOperand1(); 
 444                         const bool arrayNew = (tok2 && (tok2->str() == "[" || (tok2->str() == "(" && tok2->astOperand1() && tok2->astOperand1()->str() == "["))); 
 445                         VarInfo::AllocInfo& varAlloc = alloctype[innerTok->varId()]; 
 446                         varAlloc.type = arrayNew ? NEW_ARRAY : NEW; 
 447                         varAlloc.status = VarInfo::ALLOC; 
 448                         varAlloc.allocTok = innerTok->tokAt(2); 
 449                     } 
 450                 } 
 451  
 452                 // check for function call 
 453                 const Token * const openingPar = isFunctionCall(innerTok); 
 454                 if (openingPar) { 
 455                     // innerTok is a function name 
 456                     const VarInfo::AllocInfo allocation(0, VarInfo::NOALLOC); 
 457                     functionCall(innerTok, openingPar, varInfo, allocation, nullptr); 
 458                     innerTok = openingPar->link(); 
 459                 } 
 460             } 
 461  
 462             if (Token::simpleMatch(closingParenthesis, ") {")) { 
 463                 VarInfo varInfo1(*varInfo);  // VarInfo for if code 
 464                 VarInfo varInfo2(*varInfo);  // VarInfo for else code 
 465  
 466                 // Recursively scan variable comparisons in condition 
 467                 std::stack<const Token *> tokens; 
 468                 tokens.push(tok->next()->astOperand2()); 
 469                 while (!tokens.empty()) { 
 470                     const Token *tok3 = tokens.top(); 
 471                     tokens.pop(); 
 472                     if (!tok3) 
 473                         continue; 
 474                     if (tok3->str() == "&&" || tok3->str() == "||") { 
 475                         // FIXME: handle && ! || better 
 476                         tokens.push(tok3->astOperand1()); 
 477                         tokens.push(tok3->astOperand2()); 
 478                         continue; 
 479                     } 
 480                     if (tok3->str() == "(" && Token::Match(tok3->astOperand1(), "UNLIKELY|LIKELY")) { 
 481                         tokens.push(tok3->astOperand2()); 
 482                         continue; 
 483                     } else if (tok3->str() == "(" && Token::Match(tok3->previous(), "%name%")) { 
 484                         const std::vector<const Token *> params = getArguments(tok3->previous()); 
 485                         for (const Token *par : params) { 
 486                             if (!par->isComparisonOp()) 
 487                                 continue; 
 488                             const Token *vartok = nullptr; 
 489                             if (astIsVariableComparison(par, "!=", "0", &vartok) || 
 490                                 astIsVariableComparison(par, "==", "0", &vartok) || 
 491                                 astIsVariableComparison(par, "<", "0", &vartok) || 
 492                                 astIsVariableComparison(par, ">", "0", &vartok) || 
 493                                 astIsVariableComparison(par, "==", "-1", &vartok) || 
 494                                 astIsVariableComparison(par, "!=", "-1", &vartok)) { 
 495                                 varInfo1.erase(vartok->varId()); 
 496                                 varInfo2.erase(vartok->varId()); 
 497                             } 
 498                         } 
 499                         continue; 
 500                     } 
 501  
 502                     const Token *vartok = nullptr; 
 503                     if (astIsVariableComparison(tok3, "!=", "0", &vartok)) { 
 504                         varInfo2.erase(vartok->varId()); 
 505                         if (notzero.find(vartok->varId()) != notzero.end()) 
 506                             varInfo2.clear(); 
 507                     } else if (astIsVariableComparison(tok3, "==", "0", &vartok)) { 
 508                         varInfo1.erase(vartok->varId()); 
 509                     } else if (astIsVariableComparison(tok3, "<", "0", &vartok)) { 
 510                         varInfo1.erase(vartok->varId()); 
 511                     } else if (astIsVariableComparison(tok3, ">", "0", &vartok)) { 
 512                         varInfo2.erase(vartok->varId()); 
 513                     } else if (astIsVariableComparison(tok3, "==", "-1", &vartok)) { 
 514                         varInfo1.erase(vartok->varId()); 
 515                     } 
 516                 } 
 517  
 518                 checkScope(closingParenthesis->next(), &varInfo1, notzero, recursiveCount); 
 519                 closingParenthesis = closingParenthesis->linkAt(1); 
 520                 if (Token::simpleMatch(closingParenthesis, "} else {")) { 
 521                     checkScope(closingParenthesis->tokAt(2), &varInfo2, notzero, recursiveCount); 
 522                     tok = closingParenthesis->linkAt(2)->previous(); 
 523                 } else { 
 524                     tok = closingParenthesis->previous(); 
 525                 } 
 526  
 527                 VarInfo old; 
 528                 old.swap(*varInfo); 
 529  
 530                 std::map<int, VarInfo::AllocInfo>::const_iterator it; 
 531  
 532                 for (it = old.alloctype.begin(); it != old.alloctype.end(); ++it) { 
 533                     const int varId = it->first; 
 534                     if (old.conditionalAlloc.find(varId) == old.conditionalAlloc.end()) 
 535                         continue; 
 536                     if (varInfo1.alloctype.find(varId) == varInfo1.alloctype.end() || 
 537                         varInfo2.alloctype.find(varId) == varInfo2.alloctype.end()) { 
 538                         varInfo1.erase(varId); 
 539                         varInfo2.erase(varId); 
 540                     } 
 541                 } 
 542  
 543                 // Conditional allocation in varInfo1 
 544                 for (it = varInfo1.alloctype.begin(); it != varInfo1.alloctype.end(); ++it) { 
 545                     if (varInfo2.alloctype.find(it->first) == varInfo2.alloctype.end() && 
 546                         old.alloctype.find(it->first) == old.alloctype.end()) { 
 547                         varInfo->conditionalAlloc.insert(it->first); 
 548                     } 
 549                 } 
 550  
 551                 // Conditional allocation in varInfo2 
 552                 for (it = varInfo2.alloctype.begin(); it != varInfo2.alloctype.end(); ++it) { 
 553                     if (varInfo1.alloctype.find(it->first) == varInfo1.alloctype.end() && 
 554                         old.alloctype.find(it->first) == old.alloctype.end()) { 
 555                         varInfo->conditionalAlloc.insert(it->first); 
 556                     } 
 557                 } 
 558  
 559                 // Conditional allocation/deallocation 
 560                 for (it = varInfo1.alloctype.begin(); it != varInfo1.alloctype.end(); ++it) { 
 561                     if (it->second.managed() && conditionalAlloc.find(it->first) != conditionalAlloc.end()) { 
 562                         varInfo->conditionalAlloc.erase(it->first); 
 563                         varInfo2.erase(it->first); 
 564                     } 
 565                 } 
 566                 for (it = varInfo2.alloctype.begin(); it != varInfo2.alloctype.end(); ++it) { 
 567                     if (it->second.managed() && conditionalAlloc.find(it->first) != conditionalAlloc.end()) { 
 568                         varInfo->conditionalAlloc.erase(it->first); 
 569                         varInfo1.erase(it->first); 
 570                     } 
 571                 } 
 572  
 573                 alloctype.insert(varInfo1.alloctype.begin(), varInfo1.alloctype.end()); 
 574                 alloctype.insert(varInfo2.alloctype.begin(), varInfo2.alloctype.end()); 
 575  
 576                 possibleUsage.insert(varInfo1.possibleUsage.begin(), varInfo1.possibleUsage.end()); 
 577                 possibleUsage.insert(varInfo2.possibleUsage.begin(), varInfo2.possibleUsage.end()); 
 578             } 
 579         } 
 580  
 581         // unknown control.. (TODO: handle loops) 
 582         else if ((Token::Match(tok, "%type% (") && Token::simpleMatch(tok->linkAt(1), ") {")) || Token::simpleMatch(tok, "do {")) { 
 583             varInfo->clear(); 
 584             break; 
 585         } 
 586  
 587         // return 
 588         else if (tok->str() == "return") { 
 589             ret(tok, *varInfo); 
 590             varInfo->clear(); 
 591         } 
 592  
 593         // throw 
 594         else if (mTokenizer->isCPP() && tok->str() == "throw") { 
 595             bool tryFound = false; 
 596             const Scope* scope = tok->scope(); 
 597             while (scope && scope->isExecutable()) { 
 598                 if (scope->type == Scope::eTry) 
 599                     tryFound = true; 
 600                 scope = scope->nestedIn; 
 601             } 
 602             // If the execution leaves the function then treat it as return 
 603             if (!tryFound) 
 604                 ret(tok, *varInfo); 
 605             varInfo->clear(); 
 606         } 
 607  
 608         // delete 
 609         else if (mTokenizer->isCPP() && tok->str() == "delete") { 
 610             const Token * delTok = tok; 
 611             const bool arrayDelete = Token::simpleMatch(tok->next(), "[ ]"); 
 612             if (arrayDelete) 
 613                 tok = tok->tokAt(3); 
 614             else 
 615                 tok = tok->next(); 
 616             if (tok->str() == "(") 
 617                 tok = tok->next(); 
 618             while (Token::Match(tok, "%name% ::|.")) 
 619                 tok = tok->tokAt(2); 
 620             const bool isnull = tok->hasKnownIntValue() && tok->values().front().intvalue == 0; 
 621             if (!isnull && tok->varId() && tok->strAt(1) != "[") { 
 622                 const VarInfo::AllocInfo allocation(arrayDelete ? NEW_ARRAY : NEW, VarInfo::DEALLOC, delTok); 
 623                 changeAllocStatus(varInfo, allocation, tok, tok); 
 624             } 
 625         } 
 626  
 627         // Function call.. 
 628         else if (isFunctionCall(ftok)) { 
 629             const Token * openingPar = isFunctionCall(ftok); 
 630             const Library::AllocFunc* af = mSettings->library.getDeallocFuncInfo(ftok); 
 631             VarInfo::AllocInfo allocation(af ? af->groupId : 0, VarInfo::DEALLOC, ftok); 
 632             if (allocation.type == 0) 
 633                 allocation.status = VarInfo::NOALLOC; 
 634             functionCall(ftok, openingPar, varInfo, allocation, af); 
 635  
 636             tok = ftok->next()->link(); 
 637  
 638             // Handle scopes that might be noreturn 
 639             if (allocation.status == VarInfo::NOALLOC && Token::simpleMatch(tok, ") ; }")) { 
 640                 const std::string &functionName(tok->link()->previous()->str()); 
 641                 bool unknown = false; 
 642                 if (mTokenizer->isScopeNoReturn(tok->tokAt(2), &unknown)) { 
 643                     if (!unknown) 
 644                         varInfo->clear(); 
 645                     else if (!mSettings->library.isLeakIgnore(functionName) && !mSettings->library.isUse(functionName)) 
 646                         varInfo->possibleUsageAll(functionName); 
 647                 } 
 648             } 
 649  
 650             continue; 
 651         } 
 652  
 653         // goto => weird execution path 
 654         else if (tok->str() == "goto") { 
 655             varInfo->clear(); 
 656         } 
 657  
 658         // continue/break 
 659         else if (Token::Match(tok, "continue|break ;")) { 
 660             varInfo->clear(); 
 661         } 
 662  
 663         // Check smart pointer 
 664         else if (Token::Match(ftok, "%name% <") && mSettings->library.isSmartPointer(tok)) { 
 665             const Token * typeEndTok = ftok->linkAt(1); 
 666             if (!Token::Match(typeEndTok, "> %var% {|( %var% ,|)|}")) 
 667                 continue; 
 668  
 669             tok = typeEndTok->linkAt(2); 
 670  
 671             const int varid = typeEndTok->next()->varId(); 
 672             if (isPointerReleased(typeEndTok->tokAt(2), endToken, varid)) 
 673                 continue; 
 674  
 675             bool arrayDelete = false; 
 676             if (Token::findsimplematch(ftok->next(), "[ ]", typeEndTok)) 
 677                 arrayDelete = true; 
 678  
 679             // Check deleter 
 680             const Token * deleterToken = nullptr; 
 681             const Token * endDeleterToken = nullptr; 
 682             const Library::AllocFunc* af = nullptr; 
 683             if (Token::Match(ftok, "unique_ptr < %type% ,")) { 
 684                 deleterToken = ftok->tokAt(4); 
 685                 endDeleterToken = typeEndTok; 
 686             } else if (Token::Match(typeEndTok, "> %var% {|( %var% ,")) { 
 687                 deleterToken = typeEndTok->tokAt(5); 
 688                 endDeleterToken = typeEndTok->linkAt(2); 
 689             } 
 690             if (deleterToken) { 
 691                 // Skip the decaying plus in expressions like +[](T*){} 
 692                 if (deleterToken->str() == "+") { 
 693                     deleterToken = deleterToken->next(); 
 694                 } 
 695                 // Check if its a pointer to a function 
 696                 const Token * dtok = Token::findmatch(deleterToken, "& %name%", endDeleterToken); 
 697                 if (dtok) { 
 698                     af = mSettings->library.getDeallocFuncInfo(dtok->tokAt(1)); 
 699                 } else { 
 700                     const Token * tscopeStart = nullptr; 
 701                     const Token * tscopeEnd = nullptr; 
 702                     // If the deleter is a lambda, check if it calls the dealloc function 
 703                     if (deleterToken->str() == "[" && 
 704                         Token::simpleMatch(deleterToken->link(), "] (") && 
 705                         // TODO: Check for mutable keyword 
 706                         Token::simpleMatch(deleterToken->link()->linkAt(1), ") {")) { 
 707                         tscopeStart = deleterToken->link()->linkAt(1)->tokAt(1); 
 708                         tscopeEnd = tscopeStart->link(); 
 709                         // If the deleter is a class, check if class calls the dealloc function 
 710                     } else if ((dtok = Token::findmatch(deleterToken, "%type%", endDeleterToken)) && dtok->type()) { 
 711                         const Scope * tscope = dtok->type()->classScope; 
 712                         if (tscope) { 
 713                             tscopeStart = tscope->bodyStart; 
 714                             tscopeEnd = tscope->bodyEnd; 
 715                         } 
 716                     } 
 717  
 718                     if (tscopeStart && tscopeEnd) { 
 719                         for (const Token *tok2 = tscopeStart; tok2 != tscopeEnd; tok2 = tok2->next()) { 
 720                             af = mSettings->library.getDeallocFuncInfo(tok2); 
 721                             if (af) 
 722                                 break; 
 723                         } 
 724                     } 
 725                 } 
 726             } 
 727  
 728             const Token * vtok = typeEndTok->tokAt(3); 
 729             const VarInfo::AllocInfo allocation(af ? af->groupId : (arrayDelete ? NEW_ARRAY : NEW), VarInfo::OWNED, ftok); 
 730             changeAllocStatus(varInfo, allocation, vtok, vtok); 
 731         } 
 732     } 
 733 }

 736 const Token * CheckLeakAutoVar::checkTokenInsideExpression(const Token * const tok, VarInfo *varInfo) 
 737 { 
 738     // Deallocation and then dereferencing pointer.. 
 739     if (tok->varId() > 0) { 
 740         // TODO : Write a separate checker for this that uses valueFlowForward. 
 741         const std::map<int, VarInfo::AllocInfo>::const_iterator var = varInfo->alloctype.find(tok->varId()); 
 742         if (var != varInfo->alloctype.end()) { 
 743             bool unknown = false; 
 744             if (var->second.status == VarInfo::DEALLOC && CheckNullPointer::isPointerDeRef(tok, unknown, mSettings) && !unknown) { 
 745                 deallocUseError(tok, tok->str()); 
 746             } else if (Token::simpleMatch(tok->tokAt(-2), "= &")) { 
 747                 varInfo->erase(tok->varId()); 
 748             } else { 
 749                 // check if tok is assigned into another variable 
 750                 const Token *rhs = tok; 
 751                 while (rhs->astParent()) { 
 752                     if (rhs->astParent()->str() == "=") 
 753                         break; 
 754                     rhs = rhs->astParent(); 
 755                 } 
 756                 if (rhs->varId() == tok->varId()) { 
 757                     // simple assignment 
 758                     varInfo->erase(tok->varId()); 
 759                 } else if (rhs->str() == "(" && mSettings->library.returnValue(rhs->astOperand1()) != emptyString) { 
 760                     // #9298, assignment through return value of a function 
 761                     const std::string &returnValue = mSettings->library.returnValue(rhs->astOperand1()); 
 762                     if (returnValue.compare(0, 3, "arg") == 0) { 
 763                         int argn; 
 764                         const Token *func = getTokenArgumentFunction(tok, argn); 
 765                         if (func) { 
 766                             const std::string arg = "arg" + std::to_string(argn + 1); 
 767                             if (returnValue == arg) { 
 768                                 varInfo->erase(tok->varId()); 
 769                             } 
 770                         } 
 771                     } 
 772                 } 
 773             } 
 774         } else if (Token::Match(tok->previous(), "& %name% = %var% ;")) { 
 775             varInfo->referenced.insert(tok->tokAt(2)->varId()); 
 776         } 
 777     } 
 778  
 779     // check for function call 
 780     const Token * const openingPar = isFunctionCall(tok); 
 781     if (openingPar) { 
 782         const Library::AllocFunc* allocFunc = mSettings->library.getDeallocFuncInfo(tok); 
 783         VarInfo::AllocInfo alloc(allocFunc ? allocFunc->groupId : 0, VarInfo::DEALLOC, tok); 
 784         if (alloc.type == 0) 
 785             alloc.status = VarInfo::NOALLOC; 
 786         functionCall(tok, openingPar, varInfo, alloc, nullptr); 
 787         const std::string &returnValue = mSettings->library.returnValue(tok); 
 788         if (returnValue.compare(0, 3, "arg") == 0) 
 789             // the function returns one of its argument, we need to process a potential assignment 
 790             return openingPar; 
 791         return openingPar->link(); 
 792     } 
 793  
 794     return nullptr; 
 795 }

 798 void CheckLeakAutoVar::changeAllocStatusIfRealloc(std::map<int, VarInfo::AllocInfo> &alloctype, const Token *fTok, const Token *retTok) 
 799 { 
 800     const Library::AllocFunc* f = mSettings->library.getReallocFuncInfo(fTok); 
 801     if (f && f->arg == -1 && f->reallocArg > 0 && f->reallocArg <= numberOfArguments(fTok)) { 
 802         const Token* argTok = getArguments(fTok).at(f->reallocArg - 1); 
 803         VarInfo::AllocInfo& argAlloc = alloctype[argTok->varId()]; 
 804         VarInfo::AllocInfo& retAlloc = alloctype[retTok->varId()]; 
 805         if (argAlloc.type != 0 && argAlloc.type != f->groupId) 
 806             mismatchError(fTok, argAlloc.allocTok, argTok->str()); 
 807         argAlloc.status = VarInfo::DEALLOC; 
 808         argAlloc.allocTok = fTok; 
 809         retAlloc.type = f->groupId; 
 810         retAlloc.status = VarInfo::ALLOC; 
 811         retAlloc.allocTok = fTok; 
 812     } 
 813 }

 816 void CheckLeakAutoVar::changeAllocStatus(VarInfo *varInfo, const VarInfo::AllocInfo& allocation, const Token* tok, const Token* arg) 
 817 { 
 818     std::map<int, VarInfo::AllocInfo> &alloctype = varInfo->alloctype; 
 819     const std::map<int, VarInfo::AllocInfo>::iterator var = alloctype.find(arg->varId()); 
 820     if (var != alloctype.end()) { 
 821         if (allocation.status == VarInfo::NOALLOC) { 
 822             // possible usage 
 823             varInfo->possibleUsage[arg->varId()] = tok->str(); 
 824             if (var->second.status == VarInfo::DEALLOC && arg->previous()->str() == "&") 
 825                 varInfo->erase(arg->varId()); 
 826         } else if (var->second.managed()) { 
 827             doubleFreeError(tok, var->second.allocTok, arg->str(), allocation.type); 
 828         } else if (var->second.type != allocation.type) { 
 829             // mismatching allocation and deallocation 
 830             mismatchError(tok, var->second.allocTok, arg->str()); 
 831             varInfo->erase(arg->varId()); 
 832         } else { 
 833             // deallocation 
 834             var->second.status = allocation.status; 
 835             var->second.type = allocation.type; 
 836             var->second.allocTok = allocation.allocTok; 
 837         } 
 838     } else if (allocation.status != VarInfo::NOALLOC) { 
 839         alloctype[arg->varId()].status = VarInfo::DEALLOC; 
 840         alloctype[arg->varId()].allocTok = tok; 
 841     } 
 842 }

 844 void CheckLeakAutoVar::functionCall(const Token *tokName, const Token *tokOpeningPar, VarInfo *varInfo, const VarInfo::AllocInfo& allocation, const Library::AllocFunc* af) 
 845 { 
 846     // Ignore function call? 
 847     if (mSettings->library.isLeakIgnore(tokName->str())) 
 848         return; 
 849     if (mSettings->library.getReallocFuncInfo(tokName)) 
 850         return; 
 851  
 852     const Token * const tokFirstArg = tokOpeningPar->next(); 
 853     if (!tokFirstArg || tokFirstArg->str() == ")") { 
 854         // no arguments 
 855         return; 
 856     } 
 857  
 858     int argNr = 1; 
 859     for (const Token *funcArg = tokFirstArg; funcArg; funcArg = funcArg->nextArgument()) { 
 860         const Token* arg = funcArg; 
 861         if (mTokenizer->isCPP() && arg->str() == "new") { 
 862             arg = arg->next(); 
 863             if (Token::simpleMatch(arg, "( std :: nothrow )")) 
 864                 arg = arg->tokAt(5); 
 865         } 
 866  
 867         // Skip casts 
 868         while (arg && arg->isCast()) 
 869             arg = arg->astOperand2() ? arg->astOperand2() : arg->astOperand1(); 
 870         const Token * const argTypeStartTok = arg; 
 871  
 872         while (Token::Match(arg, "%name% .|:: %name%")) 
 873             arg = arg->tokAt(2); 
 874  
 875         if (Token::Match(arg, "%var% [-,)] !!.") || Token::Match(arg, "& %var%")) { 
 876             // goto variable 
 877             if (arg->str() == "&") 
 878                 arg = arg->next(); 
 879  
 880             const bool isnull = arg->hasKnownIntValue() && arg->values().front().intvalue == 0; 
 881  
 882             // Is variable allocated? 
 883             if (!isnull && (!af || af->arg == argNr)) 
 884                 changeAllocStatus(varInfo, allocation, tokName, arg); 
 885         } 
 886         // Check smart pointer 
 887         else if (Token::Match(arg, "%name% < %type%") && mSettings->library.isSmartPointer(argTypeStartTok)) { 
 888             const Token * typeEndTok = arg->linkAt(1); 
 889             const Token * allocTok = nullptr; 
 890             if (!Token::Match(typeEndTok, "> {|( %var% ,|)|}")) 
 891                 continue; 
 892  
 893             bool arrayDelete = false; 
 894             if (Token::findsimplematch(arg->next(), "[ ]", typeEndTok)) 
 895                 arrayDelete = true; 
 896  
 897             // Check deleter 
 898             const Token * deleterToken = nullptr; 
 899             const Token * endDeleterToken = nullptr; 
 900             const Library::AllocFunc* sp_af = nullptr; 
 901             if (Token::Match(arg, "unique_ptr < %type% ,")) { 
 902                 deleterToken = arg->tokAt(4); 
 903                 endDeleterToken = typeEndTok; 
 904             } else if (Token::Match(typeEndTok, "> {|( %var% ,")) { 
 905                 deleterToken = typeEndTok->tokAt(4); 
 906                 endDeleterToken = typeEndTok->linkAt(1); 
 907             } 
 908             if (deleterToken) { 
 909                 // Check if its a pointer to a function 
 910                 const Token * dtok = Token::findmatch(deleterToken, "& %name%", endDeleterToken); 
 911                 if (dtok) { 
 912                     sp_af = mSettings->library.getDeallocFuncInfo(dtok->tokAt(1)); 
 913                 } else { 
 914                     // If the deleter is a class, check if class calls the dealloc function 
 915                     dtok = Token::findmatch(deleterToken, "%type%", endDeleterToken); 
 916                     if (dtok && dtok->type()) { 
 917                         const Scope * tscope = dtok->type()->classScope; 
 918                         for (const Token *tok2 = tscope->bodyStart; tok2 != tscope->bodyEnd; tok2 = tok2->next()) { 
 919                             sp_af = mSettings->library.getDeallocFuncInfo(tok2); 
 920                             if (sp_af) { 
 921                                 allocTok = tok2; 
 922                                 break; 
 923                             } 
 924                         } 
 925                     } 
 926                 } 
 927             } 
 928  
 929             const Token * vtok = typeEndTok->tokAt(2); 
 930             const VarInfo::AllocInfo sp_allocation(sp_af ? sp_af->groupId : (arrayDelete ? NEW_ARRAY : NEW), VarInfo::OWNED, allocTok); 
 931             changeAllocStatus(varInfo, sp_allocation, vtok, vtok); 
 932         } else { 
 933             checkTokenInsideExpression(arg, varInfo); 
 934         } 
 935         // TODO: check each token in argument expression (could contain multiple variables) 
 936         argNr++; 
 937     } 
 938 }

 958 void CheckLeakAutoVar::ret(const Token *tok, const VarInfo &varInfo) 
 959 { 
 960     const std::map<int, VarInfo::AllocInfo> &alloctype = varInfo.alloctype; 
 961     const std::map<int, std::string> &possibleUsage = varInfo.possibleUsage; 
 962  
 963     const SymbolDatabase *symbolDatabase = mTokenizer->getSymbolDatabase(); 
 964     for (std::map<int, VarInfo::AllocInfo>::const_iterator it = alloctype.begin(); it != alloctype.end(); ++it) { 
 965         // don't warn if variable is conditionally allocated 
 966         if (!it->second.managed() && varInfo.conditionalAlloc.find(it->first) != varInfo.conditionalAlloc.end()) 
 967             continue; 
 968  
 969         // don't warn if there is a reference of the variable 
 970         if (varInfo.referenced.find(it->first) != varInfo.referenced.end()) 
 971             continue; 
 972  
 973         const int varid = it->first; 
 974         const Variable *var = symbolDatabase->getVariableFromVarId(varid); 
 975         if (var) { 
 976             bool used = false; 
 977             for (const Token *tok2 = tok; tok2; tok2 = tok2->next()) { 
 978                 if (tok2->str() == ";") 
 979                     break; 
 980                 if (!Token::Match(tok2, "return|(|{|,")) 
 981                     continue; 
 982  
 983                 const Token* tok3 = tok2->next(); 
 984                 while (tok3 && tok3->isCast() && tok3->valueType() && 
 985                        (tok3->valueType()->pointer || 
 986                         (tok3->valueType()->typeSize(*mSettings) == 0) || 
 987                         (tok3->valueType()->typeSize(*mSettings) >= mSettings->sizeof_pointer))) 
 988                     tok3 = tok3->astOperand2() ? tok3->astOperand2() : tok3->astOperand1(); 
 989                 if (Token::Match(tok3, "%varid%", varid)) 
 990                     tok2 = tok3->next(); 
 991                 else if (Token::Match(tok3, "& %varid% . %name%", varid)) 
 992                     tok2 = tok3->tokAt(4); 
 993                 else 
 994                     continue; 
 995                 if (Token::Match(tok2, "[});,]")) { 
 996                     used = true; 
 997                     break; 
 998                 } 
 999             } 
1000  
1001             // return deallocated pointer 
1002             if (used && it->second.status == VarInfo::DEALLOC) 
1003                 deallocReturnError(tok, it->second.allocTok, var->name()); 
1004  
1005             else if (!used && !it->second.managed()) { 
1006                 const std::map<int, std::string>::const_iterator use = possibleUsage.find(varid); 
1007                 if (use == possibleUsage.end()) { 
1008                     leakError(tok, var->name(), it->second.type); 
1009                 } else { 
1010                     configurationInfo(tok, use->second); 
1011                 } 
1012             } 
1013         } 
1014     } 
1015 }
