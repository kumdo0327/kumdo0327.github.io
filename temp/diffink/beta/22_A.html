 75     verify_integrity: bool = False, 
 76     sort: bool = False, 
 77     copy: bool = True, 
 78 ) -> FrameOrSeriesUnion: 
 79     """ 
 80     Concatenate pandas objects along a particular axis with optional set logic 
 81     along the other axes. 
 82  
 83     Can also add a layer of hierarchical indexing on the concatenation axis, 
 84     which may be useful if the labels are the same (or overlapping) on 
 85     the passed axis number. 
 86  
 87     Parameters 
 88     ---------- 
 89     objs : a sequence or mapping of Series or DataFrame objects 
 90         If a dict is passed, the sorted keys will be used as the `keys` 
 91         argument, unless it is passed, in which case the values will be 
 92         selected (see below). Any None objects will be dropped silently unless 
 93         they are all None in which case a ValueError will be raised. 
 94     axis : {0/'index', 1/'columns'}, default 0 
 95         The axis to concatenate along. 
 96     join : {'inner', 'outer'}, default 'outer' 
 97         How to handle indexes on other axis (or axes). 
 98     ignore_index : bool, default False 
 99         If True, do not use the index values along the concatenation axis. The 
100         resulting axis will be labeled 0, ..., n - 1. This is useful if you are 
101         concatenating objects where the concatenation axis does not have 
102         meaningful indexing information. Note the index values on the other 
103         axes are still respected in the join. 
104     keys : sequence, default None 
105         If multiple levels passed, should contain tuples. Construct 
106         hierarchical index using the passed keys as the outermost level. 
107     levels : list of sequences, default None 
108         Specific levels (unique values) to use for constructing a 
109         MultiIndex. Otherwise they will be inferred from the keys. 
110     names : list, default None 
111         Names for the levels in the resulting hierarchical index. 
112     verify_integrity : bool, default False 
113         Check whether the new concatenated axis contains duplicates. This can 
114         be very expensive relative to the actual data concatenation. 
115     sort : bool, default False 
116         Sort non-concatenation axis if it is not already aligned when `join` 
117         is 'outer'. 
118         This has no effect when ``join='inner'``, which already preserves 
119         the order of the non-concatenation axis. 
120  
121         .. versionadded:: 0.23.0 
122         .. versionchanged:: 1.0.0 
123  
124            Changed to not sort by default. 
125  
126     copy : bool, default True 
127         If False, do not copy data unnecessarily. 
128  
129     Returns 
130     ------- 
131     object, type of objs 
132         When concatenating all ``Series`` along the index (axis=0), a 
133         ``Series`` is returned. When ``objs`` contains at least one 
134         ``DataFrame``, a ``DataFrame`` is returned. When concatenating along 
135         the columns (axis=1), a ``DataFrame`` is returned. 
136  
137     See Also 
138     -------- 
139     Series.append : Concatenate Series. 
140     DataFrame.append : Concatenate DataFrames. 
141     DataFrame.join : Join DataFrames using indexes. 
142     DataFrame.merge : Merge DataFrames by indexes or columns. 
143  
144     Notes 
145     ----- 
146     The keys, levels, and names arguments are all optional. 
147  
148     A walkthrough of how this method fits in with other tools for combining 
149     pandas objects can be found `here 
150     <https://pandas.pydata.org/pandas-docs/stable/user_guide/merging.html>`__. 
151  
152     Examples 
153     -------- 
154     Combine two ``Series``. 
155  
156     >>> s1 = pd.Series(['a', 'b']) 
157     >>> s2 = pd.Series(['c', 'd']) 
158     >>> pd.concat([s1, s2]) 
159     0    a 
160     1    b 
161     0    c 
162     1    d 
163     dtype: object 
164  
165     Clear the existing index and reset it in the result 
166     by setting the ``ignore_index`` option to ``True``. 
167  
168     >>> pd.concat([s1, s2], ignore_index=True) 
169     0    a 
170     1    b 
171     2    c 
172     3    d 
173     dtype: object 
174  
175     Add a hierarchical index at the outermost level of 
176     the data with the ``keys`` option. 
177  
178     >>> pd.concat([s1, s2], keys=['s1', 's2']) 
179     s1  0    a 
180         1    b 
181     s2  0    c 
182         1    d 
183     dtype: object 
184  
185     Label the index keys you create with the ``names`` option. 
186  
187     >>> pd.concat([s1, s2], keys=['s1', 's2'], 
188     ...           names=['Series name', 'Row ID']) 
189     Series name  Row ID 
190     s1           0         a 
191                  1         b 
192     s2           0         c 
193                  1         d 
194     dtype: object 
195  
196     Combine two ``DataFrame`` objects with identical columns. 
197  
198     >>> df1 = pd.DataFrame([['a', 1], ['b', 2]], 
199     ...                    columns=['letter', 'number']) 
200     >>> df1 
201       letter  number 
202     0      a       1 
203     1      b       2 
204     >>> df2 = pd.DataFrame([['c', 3], ['d', 4]], 
205     ...                    columns=['letter', 'number']) 
206     >>> df2 
207       letter  number 
208     0      c       3 
209     1      d       4 
210     >>> pd.concat([df1, df2]) 
211       letter  number 
212     0      a       1 
213     1      b       2 
214     0      c       3 
215     1      d       4 
216  
217     Combine ``DataFrame`` objects with overlapping columns 
218     and return everything. Columns outside the intersection will 
219     be filled with ``NaN`` values. 
220  
221     >>> df3 = pd.DataFrame([['c', 3, 'cat'], ['d', 4, 'dog']], 
222     ...                    columns=['letter', 'number', 'animal']) 
223     >>> df3 
224       letter  number animal 
225     0      c       3    cat 
226     1      d       4    dog 
227     >>> pd.concat([df1, df3], sort=False) 
228       letter  number animal 
229     0      a       1    NaN 
230     1      b       2    NaN 
231     0      c       3    cat 
232     1      d       4    dog 
233  
234     Combine ``DataFrame`` objects with overlapping columns 
235     and return only those that are shared by passing ``inner`` to 
236     the ``join`` keyword argument. 
237  
238     >>> pd.concat([df1, df3], join="inner") 
239       letter  number 
240     0      a       1 
241     1      b       2 
242     0      c       3 
243     1      d       4 
244  
245     Combine ``DataFrame`` objects horizontally along the x axis by 
246     passing in ``axis=1``. 
247  
248     >>> df4 = pd.DataFrame([['bird', 'polly'], ['monkey', 'george']], 
249     ...                    columns=['animal', 'name']) 
250     >>> pd.concat([df1, df4], axis=1) 
251       letter  number  animal    name 
252     0      a       1    bird   polly 
253     1      b       2  monkey  george 
254  
255     Prevent the result from including duplicate index values with the 
256     ``verify_integrity`` option. 
257  
258     >>> df5 = pd.DataFrame([1], index=['a']) 
259     >>> df5 
260        0 
261     a  1 
262     >>> df6 = pd.DataFrame([2], index=['a']) 
263     >>> df6 
264        0 
265     a  2 
266     >>> pd.concat([df5, df6], verify_integrity=True) 
267     Traceback (most recent call last): 
268         ... 
269     ValueError: Indexes have overlapping values: ['a'] 
270     """ 
271     op = _Concatenator( 
272         objs, 
273         axis=axis, 
274         ignore_index=ignore_index, 
275         join=join, 
276         keys=keys, 
277         levels=levels, 
278         names=names, 
279         verify_integrity=verify_integrity, 
280         copy=copy, 
281         sort=sort, 
282     ) 
283  
284     return op.get_result()

292     def __init__( 
293         self, 
294         objs, 
295         axis=0, 
296         join: str = "outer", 
297         keys=None, 
298         levels=None, 
299         names=None, 
300         ignore_index: bool = False, 
301         verify_integrity: bool = False, 
302         copy: bool = True, 
303         sort=False, 
304     ): 
305         if isinstance(objs, (NDFrame, str)): 
306             raise TypeError( 
307                 "first argument must be an iterable of pandas " 
308                 f'objects, you passed an object of type "{type(objs).__name__}"' 
309             ) 
310  
311         if join == "outer": 
312             self.intersect = False 
313         elif join == "inner": 
314             self.intersect = True 
315         else:  # pragma: no cover 
316             raise ValueError( 
317                 "Only can inner (intersect) or outer (union) join the other axis" 
318             ) 
319  
320         if isinstance(objs, dict): 
321             if keys is None: 
322                 keys = list(objs.keys()) 
323             objs = [objs[k] for k in keys] 
324         else: 
325             objs = list(objs) 
326  
327         if len(objs) == 0: 
328             raise ValueError("No objects to concatenate") 
329  
330         if keys is None: 
331             objs = list(com.not_none(*objs)) 
332         else: 
333             # #1649 
334             clean_keys = [] 
335             clean_objs = [] 
336             for k, v in zip(keys, objs): 
337                 if v is None: 
338                     continue 
339                 clean_keys.append(k) 
340                 clean_objs.append(v) 
341             objs = clean_objs 
342             name = getattr(keys, "name", None) 
343             keys = Index(clean_keys, name=name) 
344  
345         if len(objs) == 0: 
346             raise ValueError("All objects passed were None") 
347  
348         # consolidate data & figure out what our result ndim is going to be 
349         ndims = set() 
350         for obj in objs: 
351             if not isinstance(obj, (Series, DataFrame)): 
352                 msg = ( 
353                     "cannot concatenate object of type '{typ}'; " 
354                     "only Series and DataFrame objs are valid".format(typ=type(obj)) 
355                 ) 
356                 raise TypeError(msg) 
357  
358             # consolidate 
359             obj._consolidate(inplace=True) 
360             ndims.add(obj.ndim) 
361  
362         # get the sample 
363         # want the highest ndim that we have, and must be non-empty 
364         # unless all objs are empty 
365         sample = None 
366         if len(ndims) > 1: 
367             max_ndim = max(ndims) 
368             for obj in objs: 
369                 if obj.ndim == max_ndim and np.sum(obj.shape): 
370                     sample = obj 
371                     break 
372  
373         else: 
374             # filter out the empties if we have not multi-index possibilities 
375             # note to keep empty Series as it affect to result columns / name 
376             non_empties = [ 
377                 obj for obj in objs if sum(obj.shape) > 0 or isinstance(obj, Series) 
378             ] 
379  
380             if len(non_empties) and ( 
381                 keys is None and names is None and levels is None and not self.intersect 
382             ): 
383                 objs = non_empties 
384                 sample = objs[0] 
385  
386         if sample is None: 
387             sample = objs[0] 
388         self.objs = objs 
389  
390         # Standardize axis parameter to int 
391         if isinstance(sample, Series): 
392             axis = DataFrame._get_axis_number(axis) 
393         else: 
394             axis = sample._get_axis_number(axis) 
395  
396         # Need to flip BlockManager axis in the DataFrame special case 
397         self._is_frame = isinstance(sample, DataFrame) 
398         if self._is_frame: 
399             axis = 1 if axis == 0 else 0 
400  
401         self._is_series = isinstance(sample, Series) 
402         if not 0 <= axis <= sample.ndim: 
403             raise AssertionError( 
404                 "axis must be between 0 and {ndim}, input was " 
405                 "{axis}".format(ndim=sample.ndim, axis=axis) 
406             ) 
407  
408         # if we have mixed ndims, then convert to highest ndim 
409         # creating column numbers as needed 
410         if len(ndims) > 1: 
411             current_column = 0 
412             max_ndim = sample.ndim 
413             self.objs, objs = [], self.objs 
414             for obj in objs: 
415  
416                 ndim = obj.ndim 
417                 if ndim == max_ndim: 
418                     pass 
419  
420                 elif ndim != max_ndim - 1: 
421                     raise ValueError( 
422                         "cannot concatenate unaligned mixed " 
423                         "dimensional NDFrame objects" 
424                     ) 
425  
426                 else: 
427                     name = getattr(obj, "name", None) 
428                     if ignore_index or name is None: 
429                         name = current_column 
430                         current_column += 1 
431  
432                     # doing a row-wise concatenation so need everything 
433                     # to line up 
434                     if self._is_frame and axis == 1: 
435                         name = 0 
436                     obj = sample._constructor({name: obj}) 
437  
438                 self.objs.append(obj) 
439  
440         # note: this is the BlockManager axis (since DataFrame is transposed) 
441         self.axis = axis 
442         self.keys = keys 
443         self.names = names or getattr(keys, "names", None) 
444         self.levels = levels 
445         self.sort = sort 
446  
447         self.ignore_index = ignore_index 
448         self.verify_integrity = verify_integrity 
449         self.copy = copy 
450  
451         self.new_axes = self._get_new_axes()

453     def get_result(self): 
454  
455         # series only 
456         if self._is_series: 
457  
458             # stack blocks 
459             if self.axis == 0: 
460                 name = com.consensus_name_attr(self.objs) 
461  
462                 mgr = self.objs[0]._data.concat( 
463                     [x._data for x in self.objs], self.new_axes 
464                 ) 
465                 cons = self.objs[0]._constructor 
466                 return cons(mgr, name=name).__finalize__(self, method="concat") 
467  
468             # combine as columns in a frame 
469             else: 
470                 data = dict(zip(range(len(self.objs)), self.objs)) 
471                 cons = DataFrame 
472  
473                 index, columns = self.new_axes 
474                 df = cons(data, index=index) 
475                 df.columns = columns 
476                 return df.__finalize__(self, method="concat") 
477  
478         # combine block managers 
479         else: 
480             mgrs_indexers = [] 
481             for obj in self.objs: 
482                 mgr = obj._data 
483                 indexers = {} 
484                 for ax, new_labels in enumerate(self.new_axes): 
485                     if ax == self.axis: 
486                         # Suppress reindexing on concat axis 
487                         continue 
488  
489                     obj_labels = mgr.axes[ax] 
490                     if not new_labels.equals(obj_labels): 
491                         indexers[ax] = obj_labels.reindex(new_labels)[1] 
492  
493                 mgrs_indexers.append((obj._data, indexers)) 
494  
495             new_data = concatenate_block_managers( 
496                 mgrs_indexers, self.new_axes, concat_axis=self.axis, copy=self.copy 
497             ) 
498             if not self.copy: 
499                 new_data._consolidate_inplace() 
500  
501             cons = self.objs[0]._constructor 
502             return cons(new_data).__finalize__(self, method="concat")

504     def _get_result_dim(self) -> int: 
505         if self._is_series and self.axis == 1: 
506             return 2 
507         else: 
508             return self.objs[0].ndim

510     def _get_new_axes(self) -> List[Index]: 
511         ndim = self._get_result_dim() 
512         return [ 
513             self._get_concat_axis() if i == self.axis else self._get_comb_axis(i) 
514             for i in range(ndim)

518         data_axis = self.objs[0]._get_block_manager_axis(i) 
519         return get_objs_combined_axis(

523     def _get_concat_axis(self) -> Index: 
524         """ 
525         Return index to be used along concatenation axis. 
526         """ 
527         if self._is_series: 
528             if self.axis == 0: 
529                 indexes = [x.index for x in self.objs] 
530             elif self.ignore_index: 
531                 idx = ibase.default_index(len(self.objs)) 
532                 return idx 
533             elif self.keys is None: 
534                 names: List[Optional[Hashable]] = [None] * len(self.objs) 
535                 num = 0 
536                 has_names = False 
537                 for i, x in enumerate(self.objs): 
538                     if not isinstance(x, Series): 
539                         raise TypeError( 
540                             f"Cannot concatenate type 'Series' with " 
541                             f"object of type '{type(x).__name__}'" 
542                         ) 
543                     if x.name is not None: 
544                         names[i] = x.name 
545                         has_names = True 
546                     else: 
547                         names[i] = num 
548                         num += 1 
549                 if has_names: 
550                     return Index(names) 
551                 else: 
552                     return ibase.default_index(len(self.objs)) 
553             else: 
554                 return ensure_index(self.keys).set_names(self.names) 
555         else: 
556             indexes = [x._data.axes[self.axis] for x in self.objs] 
557  
558         if self.ignore_index: 
559             idx = ibase.default_index(sum(len(i) for i in indexes)) 
560             return idx 
561  
562         if self.keys is None: 
563             concat_axis = _concat_indexes(indexes) 
564         else: 
565             concat_axis = _make_concat_multiindex( 
566                 indexes, self.keys, self.levels, self.names 
567             ) 
568  
569         self._maybe_check_integrity(concat_axis) 
570  
571         return concat_axis

573     def _maybe_check_integrity(self, concat_index: Index): 
574         if self.verify_integrity: 
575             if not concat_index.is_unique: 
576                 overlap = concat_index[concat_index.duplicated()].unique() 
577                 raise ValueError(f"Indexes have overlapping values: {overlap}")

581     return indexes[0].append(indexes[1:])

584 def _make_concat_multiindex(indexes, keys, levels=None, names=None) -> MultiIndex: 
585  
586     if (levels is None and isinstance(keys[0], tuple)) or ( 
587         levels is not None and len(levels) > 1 
588     ): 
589         zipped = list(zip(*keys)) 
590         if names is None: 
591             names = [None] * len(zipped) 
592  
593         if levels is None: 
594             _, levels = factorize_from_iterables(zipped) 
595         else: 
596             levels = [ensure_index(x) for x in levels] 
597     else: 
598         zipped = [keys] 
599         if names is None: 
600             names = [None] 
601  
602         if levels is None: 
603             levels = [ensure_index(keys)] 
604         else: 
605             levels = [ensure_index(x) for x in levels] 
606  
607     if not all_indexes_same(indexes): 
608         codes_list = [] 
609  
610         # things are potentially different sizes, so compute the exact codes 
611         # for each level and pass those to MultiIndex.from_arrays 
612  
613         for hlevel, level in zip(zipped, levels): 
614             to_concat = [] 
615             for key, index in zip(hlevel, indexes): 
616                 try: 
617                     i = level.get_loc(key) 
618                 except KeyError: 
619                     raise ValueError( 
620                         "Key {key!s} not in level {level!s}".format( 
621                             key=key, level=level 
622                         ) 
623                     ) 
624  
625                 to_concat.append(np.repeat(i, len(index))) 
626             codes_list.append(np.concatenate(to_concat)) 
627  
628         concat_index = _concat_indexes(indexes) 
629  
630         # these go at the end 
631         if isinstance(concat_index, MultiIndex): 
632             levels.extend(concat_index.levels) 
633             codes_list.extend(concat_index.codes) 
634         else: 
635             codes, categories = factorize_from_iterable(concat_index) 
636             levels.append(categories) 
637             codes_list.append(codes) 
638  
639         if len(names) == len(levels): 
640             names = list(names) 
641         else: 
642             # make sure that all of the passed indices have the same nlevels 
643             if not len({idx.nlevels for idx in indexes}) == 1: 
644                 raise AssertionError( 
645                     "Cannot concat indices that do not have the same number of levels" 
646                 ) 
647  
648             # also copies 
649             names = names + get_consensus_names(indexes) 
650  
651         return MultiIndex( 
652             levels=levels, codes=codes_list, names=names, verify_integrity=False 
653         ) 
654  
655     new_index = indexes[0] 
656     n = len(new_index) 
657     kpieces = len(indexes) 
658  
659     # also copies 
660     new_names = list(names) 
661     new_levels = list(levels) 
662  
663     # construct codes 
664     new_codes = [] 
665  
666     # do something a bit more speedy 
667  
668     for hlevel, level in zip(zipped, levels): 
669         hlevel = ensure_index(hlevel) 
670         mapped = level.get_indexer(hlevel) 
671  
672         mask = mapped == -1 
673         if mask.any(): 
674             raise ValueError( 
675                 "Values not found in passed level: {hlevel!s}".format( 
676                     hlevel=hlevel[mask] 
677                 ) 
678             ) 
679  
680         new_codes.append(np.repeat(mapped, n)) 
681  
682     if isinstance(new_index, MultiIndex): 
683         new_levels.extend(new_index.levels) 
684         new_codes.extend([np.tile(lab, kpieces) for lab in new_index.codes]) 
685     else: 
686         new_levels.append(new_index) 
687         new_codes.append(np.tile(np.arange(n), kpieces)) 
688  
689     if len(new_names) < len(new_levels): 
690         new_names.extend(new_index.names) 
691  
692     return MultiIndex( 
693         levels=new_levels, codes=new_codes, names=new_names, verify_integrity=False 
694     )

 65 def concat( 
 66     objs: Union[ 
 67         Iterable[FrameOrSeriesUnion], Mapping[Optional[Hashable], FrameOrSeriesUnion] 
 68     ], 
 69     axis=0, 
 70     join="outer", 
 71     ignore_index: bool = False, 
 72     keys=None, 
 73     levels=None, 
 74     names=None, 
 75     verify_integrity: bool = False, 
 76     sort: bool = False, 
 77     copy: bool = True, 
 78 ) -> FrameOrSeriesUnion: 
 79     """ 
 80     Concatenate pandas objects along a particular axis with optional set logic 
 81     along the other axes. 
 82  
 83     Can also add a layer of hierarchical indexing on the concatenation axis, 
 84     which may be useful if the labels are the same (or overlapping) on 
 85     the passed axis number. 
 86  
 87     Parameters 
 88     ---------- 
 89     objs : a sequence or mapping of Series or DataFrame objects 
 90         If a dict is passed, the sorted keys will be used as the `keys` 
 91         argument, unless it is passed, in which case the values will be 
 92         selected (see below). Any None objects will be dropped silently unless 
 93         they are all None in which case a ValueError will be raised. 
 94     axis : {0/'index', 1/'columns'}, default 0 
 95         The axis to concatenate along. 
 96     join : {'inner', 'outer'}, default 'outer' 
 97         How to handle indexes on other axis (or axes). 
 98     ignore_index : bool, default False 
 99         If True, do not use the index values along the concatenation axis. The 
100         resulting axis will be labeled 0, ..., n - 1. This is useful if you are 
101         concatenating objects where the concatenation axis does not have 
102         meaningful indexing information. Note the index values on the other 
103         axes are still respected in the join. 
104     keys : sequence, default None 
105         If multiple levels passed, should contain tuples. Construct 
106         hierarchical index using the passed keys as the outermost level. 
107     levels : list of sequences, default None 
108         Specific levels (unique values) to use for constructing a 
109         MultiIndex. Otherwise they will be inferred from the keys. 
110     names : list, default None 
111         Names for the levels in the resulting hierarchical index. 
112     verify_integrity : bool, default False 
113         Check whether the new concatenated axis contains duplicates. This can 
114         be very expensive relative to the actual data concatenation. 
115     sort : bool, default False 
116         Sort non-concatenation axis if it is not already aligned when `join` 
117         is 'outer'. 
118         This has no effect when ``join='inner'``, which already preserves 
119         the order of the non-concatenation axis. 
120  
121         .. versionadded:: 0.23.0 
122         .. versionchanged:: 1.0.0 
123  
124            Changed to not sort by default. 
125  
126     copy : bool, default True 
127         If False, do not copy data unnecessarily. 
128  
129     Returns 
130     ------- 
131     object, type of objs 
132         When concatenating all ``Series`` along the index (axis=0), a 
133         ``Series`` is returned. When ``objs`` contains at least one 
134         ``DataFrame``, a ``DataFrame`` is returned. When concatenating along 
135         the columns (axis=1), a ``DataFrame`` is returned. 
136  
137     See Also 
138     -------- 
139     Series.append : Concatenate Series. 
140     DataFrame.append : Concatenate DataFrames. 
141     DataFrame.join : Join DataFrames using indexes. 
142     DataFrame.merge : Merge DataFrames by indexes or columns. 
143  
144     Notes 
145     ----- 
146     The keys, levels, and names arguments are all optional. 
147  
148     A walkthrough of how this method fits in with other tools for combining 
149     pandas objects can be found `here 
150     <https://pandas.pydata.org/pandas-docs/stable/user_guide/merging.html>`__. 
151  
152     Examples 
153     -------- 
154     Combine two ``Series``. 
155  
156     >>> s1 = pd.Series(['a', 'b']) 
157     >>> s2 = pd.Series(['c', 'd']) 
158     >>> pd.concat([s1, s2]) 
159     0    a 
160     1    b 
161     0    c 
162     1    d 
163     dtype: object 
164  
165     Clear the existing index and reset it in the result 
166     by setting the ``ignore_index`` option to ``True``. 
167  
168     >>> pd.concat([s1, s2], ignore_index=True) 
169     0    a 
170     1    b 
171     2    c 
172     3    d 
173     dtype: object 
174  
175     Add a hierarchical index at the outermost level of 
176     the data with the ``keys`` option. 
177  
178     >>> pd.concat([s1, s2], keys=['s1', 's2']) 
179     s1  0    a 
180         1    b 
181     s2  0    c 
182         1    d 
183     dtype: object 
184  
185     Label the index keys you create with the ``names`` option. 
186  
187     >>> pd.concat([s1, s2], keys=['s1', 's2'], 
188     ...           names=['Series name', 'Row ID']) 
189     Series name  Row ID 
190     s1           0         a 
191                  1         b 
192     s2           0         c 
193                  1         d 
194     dtype: object 
195  
196     Combine two ``DataFrame`` objects with identical columns. 
197  
198     >>> df1 = pd.DataFrame([['a', 1], ['b', 2]], 
199     ...                    columns=['letter', 'number']) 
200     >>> df1 
201       letter  number 
202     0      a       1 
203     1      b       2 
204     >>> df2 = pd.DataFrame([['c', 3], ['d', 4]], 
205     ...                    columns=['letter', 'number']) 
206     >>> df2 
207       letter  number 
208     0      c       3 
209     1      d       4 
210     >>> pd.concat([df1, df2]) 
211       letter  number 
212     0      a       1 
213     1      b       2 
214     0      c       3 
215     1      d       4 
216  
217     Combine ``DataFrame`` objects with overlapping columns 
218     and return everything. Columns outside the intersection will 
219     be filled with ``NaN`` values. 
220  
221     >>> df3 = pd.DataFrame([['c', 3, 'cat'], ['d', 4, 'dog']], 
222     ...                    columns=['letter', 'number', 'animal']) 
223     >>> df3 
224       letter  number animal 
225     0      c       3    cat 
226     1      d       4    dog 
227     >>> pd.concat([df1, df3], sort=False) 
228       letter  number animal 
229     0      a       1    NaN 
230     1      b       2    NaN 
231     0      c       3    cat 
232     1      d       4    dog 
233  
234     Combine ``DataFrame`` objects with overlapping columns 
235     and return only those that are shared by passing ``inner`` to 
236     the ``join`` keyword argument. 
237  
238     >>> pd.concat([df1, df3], join="inner") 
239       letter  number 
240     0      a       1 
241     1      b       2 
242     0      c       3 
243     1      d       4 
244  
245     Combine ``DataFrame`` objects horizontally along the x axis by 
246     passing in ``axis=1``. 
247  
248     >>> df4 = pd.DataFrame([['bird', 'polly'], ['monkey', 'george']], 
249     ...                    columns=['animal', 'name']) 
250     >>> pd.concat([df1, df4], axis=1) 
251       letter  number  animal    name 
252     0      a       1    bird   polly 
253     1      b       2  monkey  george 
254  
255     Prevent the result from including duplicate index values with the 
256     ``verify_integrity`` option. 
257  
258     >>> df5 = pd.DataFrame([1], index=['a']) 
259     >>> df5 
260        0 
261     a  1 
262     >>> df6 = pd.DataFrame([2], index=['a']) 
263     >>> df6 
264        0 
265     a  2 
266     >>> pd.concat([df5, df6], verify_integrity=True) 
267     Traceback (most recent call last): 
268         ... 
269     ValueError: Indexes have overlapping values: ['a'] 
270     """ 
271     op = _Concatenator( 
272         objs, 
273         axis=axis, 
274         ignore_index=ignore_index, 
275         join=join, 
276         keys=keys, 
277         levels=levels, 
278         names=names, 
279         verify_integrity=verify_integrity, 
280         copy=copy, 
281         sort=sort, 
282     ) 
283  
284     return op.get_result()

287 class _Concatenator: 
288     """ 
289     Orchestrates a concatenation operation for BlockManagers 
290     """ 
291  
292     def __init__( 
293         self, 
294         objs, 
295         axis=0, 
296         join: str = "outer", 
297         keys=None, 
298         levels=None, 
299         names=None, 
300         ignore_index: bool = False, 
301         verify_integrity: bool = False, 
302         copy: bool = True, 
303         sort=False, 
304     ): 
305         if isinstance(objs, (NDFrame, str)): 
306             raise TypeError( 
307                 "first argument must be an iterable of pandas " 
308                 f'objects, you passed an object of type "{type(objs).__name__}"' 
309             ) 
310  
311         if join == "outer": 
312             self.intersect = False 
313         elif join == "inner": 
314             self.intersect = True 
315         else:  # pragma: no cover 
316             raise ValueError( 
317                 "Only can inner (intersect) or outer (union) join the other axis" 
318             ) 
319  
320         if isinstance(objs, dict): 
321             if keys is None: 
322                 keys = list(objs.keys()) 
323             objs = [objs[k] for k in keys] 
324         else: 
325             objs = list(objs) 
326  
327         if len(objs) == 0: 
328             raise ValueError("No objects to concatenate") 
329  
330         if keys is None: 
331             objs = list(com.not_none(*objs)) 
332         else: 
333             # #1649 
334             clean_keys = [] 
335             clean_objs = [] 
336             for k, v in zip(keys, objs): 
337                 if v is None: 
338                     continue 
339                 clean_keys.append(k) 
340                 clean_objs.append(v) 
341             objs = clean_objs 
342             name = getattr(keys, "name", None) 
343             keys = Index(clean_keys, name=name) 
344  
345         if len(objs) == 0: 
346             raise ValueError("All objects passed were None") 
347  
348         # consolidate data & figure out what our result ndim is going to be 
349         ndims = set() 
350         for obj in objs: 
351             if not isinstance(obj, (Series, DataFrame)): 
352                 msg = ( 
353                     "cannot concatenate object of type '{typ}'; " 
354                     "only Series and DataFrame objs are valid".format(typ=type(obj)) 
355                 ) 
356                 raise TypeError(msg) 
357  
358             # consolidate 
359             obj._consolidate(inplace=True) 
360             ndims.add(obj.ndim) 
361  
362         # get the sample 
363         # want the highest ndim that we have, and must be non-empty 
364         # unless all objs are empty 
365         sample = None 
366         if len(ndims) > 1: 
367             max_ndim = max(ndims) 
368             for obj in objs: 
369                 if obj.ndim == max_ndim and np.sum(obj.shape): 
370                     sample = obj 
371                     break 
372  
373         else: 
374             # filter out the empties if we have not multi-index possibilities 
375             # note to keep empty Series as it affect to result columns / name 
376             non_empties = [ 
377                 obj for obj in objs if sum(obj.shape) > 0 or isinstance(obj, Series) 
378             ] 
379  
380             if len(non_empties) and ( 
381                 keys is None and names is None and levels is None and not self.intersect 
382             ): 
383                 objs = non_empties 
384                 sample = objs[0] 
385  
386         if sample is None: 
387             sample = objs[0] 
388         self.objs = objs 
389  
390         # Standardize axis parameter to int 
391         if isinstance(sample, Series): 
392             axis = DataFrame._get_axis_number(axis) 
393         else: 
394             axis = sample._get_axis_number(axis) 
395  
396         # Need to flip BlockManager axis in the DataFrame special case 
397         self._is_frame = isinstance(sample, DataFrame) 
398         if self._is_frame: 
399             axis = 1 if axis == 0 else 0 
400  
401         self._is_series = isinstance(sample, Series) 
402         if not 0 <= axis <= sample.ndim: 
403             raise AssertionError( 
404                 "axis must be between 0 and {ndim}, input was " 
405                 "{axis}".format(ndim=sample.ndim, axis=axis) 
406             ) 
407  
408         # if we have mixed ndims, then convert to highest ndim 
409         # creating column numbers as needed 
410         if len(ndims) > 1: 
411             current_column = 0 
412             max_ndim = sample.ndim 
413             self.objs, objs = [], self.objs 
414             for obj in objs: 
415  
416                 ndim = obj.ndim 
417                 if ndim == max_ndim: 
418                     pass 
419  
420                 elif ndim != max_ndim - 1: 
421                     raise ValueError( 
422                         "cannot concatenate unaligned mixed " 
423                         "dimensional NDFrame objects" 
424                     ) 
425  
426                 else: 
427                     name = getattr(obj, "name", None) 
428                     if ignore_index or name is None: 
429                         name = current_column 
430                         current_column += 1 
431  
432                     # doing a row-wise concatenation so need everything 
433                     # to line up 
434                     if self._is_frame and axis == 1: 
435                         name = 0 
436                     obj = sample._constructor({name: obj}) 
437  
438                 self.objs.append(obj) 
439  
440         # note: this is the BlockManager axis (since DataFrame is transposed) 
441         self.axis = axis 
442         self.keys = keys 
443         self.names = names or getattr(keys, "names", None) 
444         self.levels = levels 
445         self.sort = sort 
446  
447         self.ignore_index = ignore_index 
448         self.verify_integrity = verify_integrity 
449         self.copy = copy 
450  
451         self.new_axes = self._get_new_axes() 
452  
453     def get_result(self): 
454  
455         # series only 
456         if self._is_series: 
457  
458             # stack blocks 
459             if self.axis == 0: 
460                 name = com.consensus_name_attr(self.objs) 
461  
462                 mgr = self.objs[0]._data.concat( 
463                     [x._data for x in self.objs], self.new_axes 
464                 ) 
465                 cons = self.objs[0]._constructor 
466                 return cons(mgr, name=name).__finalize__(self, method="concat") 
467  
468             # combine as columns in a frame 
469             else: 
470                 data = dict(zip(range(len(self.objs)), self.objs)) 
471                 cons = DataFrame 
472  
473                 index, columns = self.new_axes 
474                 df = cons(data, index=index) 
475                 df.columns = columns 
476                 return df.__finalize__(self, method="concat") 
477  
478         # combine block managers 
479         else: 
480             mgrs_indexers = [] 
481             for obj in self.objs: 
482                 mgr = obj._data 
483                 indexers = {} 
484                 for ax, new_labels in enumerate(self.new_axes): 
485                     if ax == self.axis: 
486                         # Suppress reindexing on concat axis 
487                         continue 
488  
489                     obj_labels = mgr.axes[ax] 
490                     if not new_labels.equals(obj_labels): 
491                         indexers[ax] = obj_labels.reindex(new_labels)[1] 
492  
493                 mgrs_indexers.append((obj._data, indexers)) 
494  
495             new_data = concatenate_block_managers( 
496                 mgrs_indexers, self.new_axes, concat_axis=self.axis, copy=self.copy 
497             ) 
498             if not self.copy: 
499                 new_data._consolidate_inplace() 
500  
501             cons = self.objs[0]._constructor 
502             return cons(new_data).__finalize__(self, method="concat") 
503  
504     def _get_result_dim(self) -> int: 
505         if self._is_series and self.axis == 1: 
506             return 2 
507         else: 
508             return self.objs[0].ndim

510     def _get_new_axes(self) -> List[Index]: 
511         ndim = self._get_result_dim() 
512         return [ 
513             self._get_concat_axis() if i == self.axis else self._get_comb_axis(i) 
514             for i in range(ndim) 
515         ]

517     def _get_comb_axis(self, i: int) -> Index: 
518         data_axis = self.objs[0]._get_block_manager_axis(i) 
519         return get_objs_combined_axis( 
520             self.objs, 
521             axis=data_axis, 
522             intersect=self.intersect, 
523             sort=self.sort, 
524             copy=self.copy, 
525         )

527     def _get_concat_axis(self) -> Index: 
528         """ 
529         Return index to be used along concatenation axis. 
530         """ 
531         if self._is_series: 
532             if self.axis == 0: 
533                 indexes = [x.index for x in self.objs] 
534             elif self.ignore_index: 
535                 idx = ibase.default_index(len(self.objs)) 
536                 return idx 
537             elif self.keys is None: 
538                 names: List[Optional[Hashable]] = [None] * len(self.objs) 
539                 num = 0 
540                 has_names = False 
541                 for i, x in enumerate(self.objs): 
542                     if not isinstance(x, Series): 
543                         raise TypeError( 
544                             f"Cannot concatenate type 'Series' with " 
545                             f"object of type '{type(x).__name__}'" 
546                         ) 
547                     if x.name is not None: 
548                         names[i] = x.name 
549                         has_names = True 
550                     else: 
551                         names[i] = num 
552                         num += 1 
553                 if has_names: 
554                     return Index(names) 
555                 else: 
556                     return ibase.default_index(len(self.objs)) 
557             else: 
558                 return ensure_index(self.keys).set_names(self.names) 
559         else: 
560             indexes = [x._data.axes[self.axis] for x in self.objs] 
561  
562         if self.ignore_index: 
563             idx = ibase.default_index(sum(len(i) for i in indexes)) 
564             return idx 
565  
566         if self.keys is None: 
567             concat_axis = _concat_indexes(indexes) 
568         else: 
569             concat_axis = _make_concat_multiindex( 
570                 indexes, self.keys, self.levels, self.names 
571             ) 
572  
573         self._maybe_check_integrity(concat_axis) 
574  
575         return concat_axis 
576  
577     def _maybe_check_integrity(self, concat_index: Index): 
578         if self.verify_integrity: 
579             if not concat_index.is_unique: 
580                 overlap = concat_index[concat_index.duplicated()].unique() 
581                 raise ValueError(f"Indexes have overlapping values: {overlap}")

584 def _concat_indexes(indexes) -> Index: 
585     return indexes[0].append(indexes[1:])

588 def _make_concat_multiindex(indexes, keys, levels=None, names=None) -> MultiIndex: 
589  
590     if (levels is None and isinstance(keys[0], tuple)) or ( 
591         levels is not None and len(levels) > 1 
592     ): 
593         zipped = list(zip(*keys)) 
594         if names is None: 
595             names = [None] * len(zipped) 
596  
597         if levels is None: 
598             _, levels = factorize_from_iterables(zipped) 
599         else: 
600             levels = [ensure_index(x) for x in levels] 
601     else: 
602         zipped = [keys] 
603         if names is None: 
604             names = [None] 
605  
606         if levels is None: 
607             levels = [ensure_index(keys)] 
608         else: 
609             levels = [ensure_index(x) for x in levels] 
610  
611     if not all_indexes_same(indexes): 
612         codes_list = [] 
613  
614         # things are potentially different sizes, so compute the exact codes 
615         # for each level and pass those to MultiIndex.from_arrays 
616  
617         for hlevel, level in zip(zipped, levels): 
618             to_concat = [] 
619             for key, index in zip(hlevel, indexes): 
620                 try: 
621                     i = level.get_loc(key) 
622                 except KeyError: 
623                     raise ValueError( 
624                         "Key {key!s} not in level {level!s}".format( 
625                             key=key, level=level 
626                         ) 
627                     ) 
628  
629                 to_concat.append(np.repeat(i, len(index))) 
630             codes_list.append(np.concatenate(to_concat)) 
631  
632         concat_index = _concat_indexes(indexes) 
633  
634         # these go at the end 
635         if isinstance(concat_index, MultiIndex): 
636             levels.extend(concat_index.levels) 
637             codes_list.extend(concat_index.codes) 
638         else: 
639             codes, categories = factorize_from_iterable(concat_index) 
640             levels.append(categories) 
641             codes_list.append(codes) 
642  
643         if len(names) == len(levels): 
644             names = list(names) 
645         else: 
646             # make sure that all of the passed indices have the same nlevels 
647             if not len({idx.nlevels for idx in indexes}) == 1: 
648                 raise AssertionError( 
649                     "Cannot concat indices that do not have the same number of levels" 
650                 ) 
651  
652             # also copies 
653             names = names + get_consensus_names(indexes) 
654  
655         return MultiIndex( 
656             levels=levels, codes=codes_list, names=names, verify_integrity=False 
657         ) 
658  
659     new_index = indexes[0] 
660     n = len(new_index) 
661     kpieces = len(indexes) 
662  
663     # also copies 
664     new_names = list(names) 
665     new_levels = list(levels) 
666  
667     # construct codes 
668     new_codes = [] 
669  
670     # do something a bit more speedy 
671  
672     for hlevel, level in zip(zipped, levels): 
673         hlevel = ensure_index(hlevel) 
674         mapped = level.get_indexer(hlevel) 
675  
676         mask = mapped == -1 
677         if mask.any(): 
678             raise ValueError( 
679                 "Values not found in passed level: {hlevel!s}".format( 
680                     hlevel=hlevel[mask] 
681                 ) 
682             ) 
683  
684         new_codes.append(np.repeat(mapped, n)) 
685  
686     if isinstance(new_index, MultiIndex): 
687         new_levels.extend(new_index.levels) 
688         new_codes.extend([np.tile(lab, kpieces) for lab in new_index.codes]) 
689     else: 
690         new_levels.append(new_index) 
691         new_codes.append(np.tile(np.arange(n), kpieces)) 
692  
693     if len(new_names) < len(new_levels): 
694         new_names.extend(new_index.names) 
695  
696     return MultiIndex( 
697         levels=new_levels, codes=new_codes, names=new_names, verify_integrity=False 
698     )
