133 def safe_cast(values, dtype, copy): 
134     """ 
135     Safely cast the values to the dtype if they 
136     are equivalent, meaning floats must be equivalent to the 
137     ints. 
138  
139     """ 
140  
141     try: 
142         return values.astype(dtype, casting="safe", copy=copy) 
143     except TypeError: 
144

146         if (casted == values).all(): 
147             return casted

149         raise TypeError(

154 def coerce_to_array(values, dtype, mask=None, copy=False): 
155     """ 
156     Coerce the input values array to numpy arrays with a mask 
157  
158     Parameters 
159     ---------- 
160     values : 1D list-like 
161     dtype : integer dtype 
162     mask : bool 1D array, optional 
163     copy : bool, default False 
164         if True, copy the input 
165  
166     Returns 
167     ------- 
168     tuple of (values, mask) 
169     """ 
170     # if values is integer numpy array, preserve it's dtype 
171     if dtype is None and hasattr(values, "dtype"): 
172         if is_integer_dtype(values.dtype): 
173             dtype = values.dtype 
174  
175     if dtype is not None: 
176         if isinstance(dtype, str) and ( 
177             dtype.startswith("Int") or dtype.startswith("UInt") 
178         ): 
179             # Avoid DeprecationWarning from NumPy about np.dtype("Int64") 
180             # https://github.com/numpy/numpy/pull/7476 
181             dtype = dtype.lower() 
182  
183         if not issubclass(type(dtype), _IntegerDtype): 
184             try: 
185                 dtype = _dtypes[str(np.dtype(dtype))] 
186             except KeyError: 
187                 raise ValueError(f"invalid dtype specified {dtype}") 
188  
189     if isinstance(values, IntegerArray): 
190         values, mask = values._data, values._mask 
191         if dtype is not None: 
192             values = values.astype(dtype.numpy_dtype, copy=False) 
193  
194         if copy: 
195             values = values.copy() 
196             mask = mask.copy() 
197         return values, mask 
198  
199     values = np.array(values, copy=copy) 
200     if is_object_dtype(values): 
201         inferred_type = lib.infer_dtype(values, skipna=True) 
202         if inferred_type == "empty": 
203             values = np.empty(len(values)) 
204             values.fill(np.nan) 
205         elif inferred_type not in [ 
206             "floating", 
207             "integer", 
208             "mixed-integer", 
209             "integer-na", 
210             "mixed-integer-float", 
211         ]: 
212             raise TypeError(f"{values.dtype} cannot be converted to an IntegerDtype") 
213  
214     elif is_bool_dtype(values) and is_integer_dtype(dtype): 
215         values = np.array(values, dtype=int, copy=copy) 
216  
217     elif not (is_integer_dtype(values) or is_float_dtype(values)): 
218         raise TypeError(f"{values.dtype} cannot be converted to an IntegerDtype") 
219  
220     if mask is None: 
221         mask = isna(values) 
222     else: 
223         assert len(mask) == len(values) 
224  
225     if not values.ndim == 1: 
226         raise TypeError("values must be a 1D list-like") 
227     if not mask.ndim == 1: 
228         raise TypeError("mask must be a 1D list-like") 
229  
230     # infer dtype if needed 
231     if dtype is None: 
232         dtype = np.dtype("int64") 
233     else: 
234         dtype = dtype.type 
235  
236     # if we are float, let's make sure that we can 
237     # safely cast 
238  
239     # we copy as need to coerce here 
240     if mask.any(): 
241         values = values.copy() 
242         values[mask] = 1 
243         values = safe_cast(values, dtype, copy=False) 
244     else: 
245         values = safe_cast(values, dtype, copy=False) 
246  
247     return values, mask

250 class IntegerArray(BaseMaskedArray): 
251     """ 
252     Array of integer (optional missing) values. 
253  
254     .. versionadded:: 0.24.0 
255  
256     .. versionchanged:: 1.0.0 
257  
258        Now uses :attr:`pandas.NA` as the missing value rather 
259        than :attr:`numpy.nan`. 
260  
261     .. warning:: 
262  
263        IntegerArray is currently experimental, and its API or internal 
264        implementation may change without warning. 
265  
266     We represent an IntegerArray with 2 numpy arrays: 
267  
268     - data: contains a numpy integer array of the appropriate dtype 
269     - mask: a boolean array holding a mask on the data, True is missing 
270  
271     To construct an IntegerArray from generic array-like input, use 
272     :func:`pandas.array` with one of the integer dtypes (see examples). 
273  
274     See :ref:`integer_na` for more. 
275  
276     Parameters 
277     ---------- 
278     values : numpy.ndarray 
279         A 1-d integer-dtype array. 
280     mask : numpy.ndarray 
281         A 1-d boolean-dtype array indicating missing values.

285     Attributes 
286     ---------- 
287     None 
288  
289     Methods 
290     ------- 
291     None 
292  
293     Returns 
294     ------- 
295     IntegerArray 
296  
297     Examples 
298     -------- 
299     Create an IntegerArray with :func:`pandas.array`. 
300  
301     >>> int_array = pd.array([1, None, 3], dtype=pd.Int32Dtype()) 
302     >>> int_array 
303     <IntegerArray> 
304     [1, <NA>, 3] 
305     Length: 3, dtype: Int32 
306  
307     String aliases for the dtypes are also available. They are capitalized. 
308  
309     >>> pd.array([1, None, 3], dtype='Int32') 
310     <IntegerArray> 
311     [1, <NA>, 3] 
312     Length: 3, dtype: Int32

361     def __array_ufunc__(self, ufunc, method, *inputs, **kwargs): 
362         # For IntegerArray inputs, we apply the ufunc to ._data 
363         # and mask the result. 
364         if method == "reduce": 
365             # Not clear how to handle missing values in reductions. Raise. 
366             raise NotImplementedError("The 'reduce' method is not supported.") 
367         out = kwargs.get("out", ()) 
368  
369         for x in inputs + out: 
370             if not isinstance(x, self._HANDLED_TYPES + (IntegerArray,)): 
371                 return NotImplemented 
372  
373         # for binary ops, use our custom dunder methods 
374         result = ops.maybe_dispatch_ufunc_to_dunder_op( 
375             self, ufunc, method, *inputs, **kwargs 
376         ) 
377         if result is not NotImplemented: 
378             return result 
379  
380         mask = np.zeros(len(self), dtype=bool) 
381         inputs2 = [] 
382         for x in inputs: 
383             if isinstance(x, IntegerArray): 
384                 mask |= x._mask 
385                 inputs2.append(x._data) 
386             else: 
387                 inputs2.append(x) 
388  
389         def reconstruct(x): 
390             # we don't worry about scalar `x` here, since we 
391             # raise for reduce up above. 
392  
393             if is_integer_dtype(x.dtype): 
394                 m = mask.copy() 
395                 return IntegerArray(x, m) 
396             else: 
397                 x[mask] = np.nan 
398             return x 
399  
400         result = getattr(ufunc, method)(*inputs2, **kwargs) 
401         if isinstance(result, tuple): 
402             tuple(reconstruct(x) for x in result) 
403         else: 
404             return reconstruct(result)

492     @classmethod 
493     def _create_comparison_method(cls, op): 
494         op_name = op.__name__ 
495  
496         @unpack_zerodim_and_defer(op.__name__) 
497         def cmp_method(self, other): 
498             from pandas.arrays import BooleanArray 
499  
500             mask = None 
501  
502             if isinstance(other, (BooleanArray, IntegerArray)): 
503                 other, mask = other._data, other._mask 
504  
505             elif is_list_like(other): 
506                 other = np.asarray(other) 
507                 if other.ndim > 1: 
508                     raise NotImplementedError( 
509                         "can only perform ops with 1-d structures" 
510                     ) 
511                 if len(self) != len(other): 
512                     raise ValueError("Lengths must match to compare") 
513  
514             if other is libmissing.NA: 
515                 # numpy does not handle pd.NA well as "other" scalar (it returns 
516                 # a scalar False instead of an array) 
517                 # This may be fixed by NA.__array_ufunc__. Revisit this check 
518                 # once that's implemented. 
519                 result = np.zeros(self._data.shape, dtype="bool") 
520                 mask = np.ones(self._data.shape, dtype="bool") 
521             else: 
522                 with warnings.catch_warnings(): 
523                     # numpy may show a FutureWarning: 
524                     #     elementwise comparison failed; returning scalar instead, 
525                     #     but in the future will perform elementwise comparison 
526                     # before returning NotImplemented. We fall back to the correct 
527                     # behavior today, so that should be fine to ignore. 
528                     warnings.filterwarnings("ignore", "elementwise", FutureWarning) 
529                     with np.errstate(all="ignore"): 
530                         method = getattr(self._data, f"__{op_name}__") 
531                         result = method(other) 
532  
533                     if result is NotImplemented: 
534                         result = invalid_comparison(self._data, other, op) 
535  
536             # nans propagate 
537             if mask is None: 
538                 mask = self._mask.copy() 
539             else: 
540                 mask = self._mask | mask 
541  
542             return BooleanArray(result, mask) 
543  
544         name = f"__{op.__name__}__" 
545         return set_function_name(cmp_method, name, cls)

547     def _reduce(self, name, skipna=True, **kwargs): 
548         data = self._data 
549         mask = self._mask 
550  
551         # coerce to a nan-aware float if needed 
552         # (we explicitly use NaN within reductions) 
553         if self._hasna: 
554             data = self.to_numpy("float64", na_value=np.nan) 
555  
556         op = getattr(nanops, "nan" + name) 
557         result = op(data, axis=0, skipna=skipna, mask=mask, **kwargs) 
558  
559         if np.isnan(result): 
560             return libmissing.NA 
561  
562         # if we have a boolean op, don't coerce 
563         if name in ["any", "all"]: 
564             pass 
565  
566         # if we have a preservable numeric op, 
567         # provide coercion back to an integer type if possible 
568         elif name in ["sum", "min", "max", "prod"]: 
569             int_result = int(result) 
570             if int_result == result: 
571                 result = int_result 
572  
573         return result

575     def _maybe_mask_result(self, result, mask, other, op_name): 
576         """ 
577         Parameters 
578         ---------- 
579         result : array-like 
580         mask : array-like bool 
581         other : scalar or array-like 
582         op_name : str 
583         """ 
584  
585         # if we have a float operand we are by-definition 
586         # a float result 
587         # or our op is a divide 
588         if (is_float_dtype(other) or is_float(other)) or ( 
589             op_name in ["rtruediv", "truediv"]

596     @classmethod 
597     def _create_arithmetic_method(cls, op): 
598         op_name = op.__name__ 
599  
600         @unpack_zerodim_and_defer(op.__name__) 
601         def integer_arithmetic_method(self, other): 
602  
603             omask = None 
604  
605             if getattr(other, "ndim", 0) > 1: 
606                 raise NotImplementedError("can only perform ops with 1-d structures") 
607  
608             if isinstance(other, IntegerArray): 
609                 other, omask = other._data, other._mask 
610  
611             elif is_list_like(other): 
612                 other = np.asarray(other) 
613                 if other.ndim > 1: 
614                     raise NotImplementedError( 
615                         "can only perform ops with 1-d structures" 
616                     ) 
617                 if len(self) != len(other): 
618                     raise ValueError("Lengths must match") 
619                 if not (is_float_dtype(other) or is_integer_dtype(other)): 
620                     raise TypeError("can only perform ops with numeric values") 
621  
622             else: 
623                 if not (is_float(other) or is_integer(other) or other is libmissing.NA): 
624                     raise TypeError("can only perform ops with numeric values") 
625  
626             if omask is None: 
627                 mask = self._mask.copy() 
628                 if other is libmissing.NA: 
629                     mask |= True 
630             else: 
631                 mask = self._mask | omask 
632  
633             if op_name == "pow": 
634                 # 1 ** x is 1. 
635                 mask = np.where((self._data == 1) & ~self._mask, False, mask) 
636                 # x ** 0 is 1. 
637                 if omask is not None: 
638                     mask = np.where((other == 0) & ~omask, False, mask) 
639                 elif other is not libmissing.NA: 
640                     mask = np.where(other == 0, False, mask) 
641  
642             elif op_name == "rpow": 
643                 # 1 ** x is 1. 
644                 if omask is not None: 
645                     mask = np.where((other == 1) & ~omask, False, mask) 
646                 elif other is not libmissing.NA: 
647                     mask = np.where(other == 1, False, mask) 
648                 # x ** 0 is 1. 
649                 mask = np.where((self._data == 0) & ~self._mask, False, mask) 
650  
651             if other is libmissing.NA: 
652                 result = np.ones_like(self._data) 
653             else: 
654                 with np.errstate(all="ignore"): 
655                     result = op(self._data, other) 
656  
657             # divmod returns a tuple 
658             if op_name == "divmod": 
659                 div, mod = result 
660                 return ( 
661                     self._maybe_mask_result(div, mask, other, "floordiv"), 
662                     self._maybe_mask_result(mod, mask, other, "mod"), 
663                 ) 
664  
665             return self._maybe_mask_result(result, mask, other, op_name) 
666  
667         name = f"__{op.__name__}__" 
668         return set_function_name(integer_arithmetic_method, name, cls)

155 def coerce_to_array(values, dtype, mask=None, copy=False): 
156     """ 
157     Coerce the input values array to numpy arrays with a mask 
158  
159     Parameters 
160     ---------- 
161     values : 1D list-like 
162     dtype : integer dtype 
163     mask : bool 1D array, optional 
164     copy : bool, default False 
165         if True, copy the input 
166  
167     Returns 
168     ------- 
169     tuple of (values, mask) 
170     """ 
171     # if values is integer numpy array, preserve it's dtype 
172     if dtype is None and hasattr(values, "dtype"): 
173         if is_integer_dtype(values.dtype): 
174             dtype = values.dtype 
175  
176     if dtype is not None: 
177         if isinstance(dtype, str) and ( 
178             dtype.startswith("Int") or dtype.startswith("UInt") 
179         ): 
180             # Avoid DeprecationWarning from NumPy about np.dtype("Int64") 
181             # https://github.com/numpy/numpy/pull/7476 
182             dtype = dtype.lower() 
183  
184         if not issubclass(type(dtype), _IntegerDtype): 
185             try: 
186                 dtype = _dtypes[str(np.dtype(dtype))] 
187             except KeyError: 
188                 raise ValueError(f"invalid dtype specified {dtype}") 
189  
190     if isinstance(values, IntegerArray): 
191         values, mask = values._data, values._mask 
192         if dtype is not None: 
193             values = values.astype(dtype.numpy_dtype, copy=False) 
194  
195         if copy: 
196             values = values.copy() 
197             mask = mask.copy() 
198         return values, mask 
199  
200     values = np.array(values, copy=copy) 
201     if is_object_dtype(values): 
202         inferred_type = lib.infer_dtype(values, skipna=True) 
203         if inferred_type == "empty": 
204             values = np.empty(len(values)) 
205             values.fill(np.nan) 
206         elif inferred_type not in [ 
207             "floating", 
208             "integer", 
209             "mixed-integer", 
210             "integer-na", 
211             "mixed-integer-float", 
212         ]: 
213             raise TypeError(f"{values.dtype} cannot be converted to an IntegerDtype") 
214  
215     elif is_bool_dtype(values) and is_integer_dtype(dtype): 
216         values = np.array(values, dtype=int, copy=copy) 
217  
218     elif not (is_integer_dtype(values) or is_float_dtype(values)): 
219         raise TypeError(f"{values.dtype} cannot be converted to an IntegerDtype") 
220  
221     if mask is None: 
222         mask = isna(values) 
223     else: 
224         assert len(mask) == len(values) 
225  
226     if not values.ndim == 1: 
227         raise TypeError("values must be a 1D list-like") 
228     if not mask.ndim == 1: 
229         raise TypeError("mask must be a 1D list-like") 
230  
231     # infer dtype if needed 
232     if dtype is None: 
233         dtype = np.dtype("int64") 
234     else: 
235         dtype = dtype.type 
236  
237     # if we are float, let's make sure that we can 
238     # safely cast 
239  
240     # we copy as need to coerce here 
241     if mask.any(): 
242         values = values.copy() 
243         values[mask] = 1 
244         values = safe_cast(values, dtype, copy=False) 
245     else: 
246         values = safe_cast(values, dtype, copy=False) 
247  
248     return values, mask

502  
503         @unpack_zerodim_and_defer(op.__name__) 
504         def cmp_method(self, other): 
505             from pandas.arrays import BooleanArray 
506  
507             mask = None 
508  
509             if isinstance(other, (BooleanArray, IntegerArray)): 
510                 other, mask = other._data, other._mask

546             else:

554     def _reduce(self, name, skipna=True, **kwargs): 
555         data = self._data 
556         mask = self._mask 
557  
558         # coerce to a nan-aware float if needed 
559         # (we explicitly use NaN within reductions) 
560         if self._hasna: 
561             data = self.to_numpy("float64", na_value=np.nan) 
562  
563         op = getattr(nanops, "nan" + name) 
564         result = op(data, axis=0, skipna=skipna, mask=mask, **kwargs) 
565  
566         if np.isnan(result): 
567             return libmissing.NA 
568  
569         # if we have a boolean op, don't coerce 
570         if name in ["any", "all"]: 
571             pass 
572  
573         # if we have a preservable numeric op, 
574         # provide coercion back to an integer type if possible 
575         elif name in ["sum", "min", "max", "prod"]: 
576             int_result = int(result) 
577             if int_result == result: 
578                 result = int_result

604     def _create_arithmetic_method(cls, op): 
605         op_name = op.__name__ 
606  
607         @unpack_zerodim_and_defer(op.__name__) 
608         def integer_arithmetic_method(self, other): 
609  
610             omask = None 
611  
612             if getattr(other, "ndim", 0) > 1: 
613                 raise NotImplementedError("can only perform ops with 1-d structures") 
614  
615             if isinstance(other, IntegerArray): 
616                 other, omask = other._data, other._mask 
617  
618             elif is_list_like(other): 
619                 other = np.asarray(other) 
620                 if other.ndim > 1: 
621                     raise NotImplementedError( 
622                         "can only perform ops with 1-d structures" 
623                     ) 
624                 if len(self) != len(other): 
625                     raise ValueError("Lengths must match") 
626                 if not (is_float_dtype(other) or is_integer_dtype(other)): 
627                     raise TypeError("can only perform ops with numeric values") 
628  
629             else: 
630                 if not (is_float(other) or is_integer(other) or other is libmissing.NA): 
631                     raise TypeError("can only perform ops with numeric values") 
632  
633             if omask is None: 
634                 mask = self._mask.copy() 
635                 if other is libmissing.NA: 
636                     mask |= True 
637             else: 
638                 mask = self._mask | omask 
639  
640             if op_name == "pow": 
641                 # 1 ** x is 1. 
642                 mask = np.where((self._data == 1) & ~self._mask, False, mask) 
643                 # x ** 0 is 1. 
644                 if omask is not None: 
645                     mask = np.where((other == 0) & ~omask, False, mask) 
646                 elif other is not libmissing.NA: 
647                     mask = np.where(other == 0, False, mask) 
648  
649             elif op_name == "rpow": 
650                 # 1 ** x is 1. 
651                 if omask is not None: 
652                     mask = np.where((other == 1) & ~omask, False, mask) 
653                 elif other is not libmissing.NA: 
654                     mask = np.where(other == 1, False, mask) 
655                 # x ** 0 is 1. 
656                 mask = np.where((self._data == 0) & ~self._mask, False, mask) 
657  
658             if other is libmissing.NA: 
659                 result = np.ones_like(self._data) 
660             else: 
661                 with np.errstate(all="ignore"): 
662                     result = op(self._data, other) 
663  
664             # divmod returns a tuple 
665             if op_name == "divmod": 
666                 div, mod = result 
667                 return ( 
668                     self._maybe_mask_result(div, mask, other, "floordiv"), 
669                     self._maybe_mask_result(mod, mask, other, "mod"), 
670                 ) 
671  
672             return self._maybe_mask_result(result, mask, other, op_name) 
673  
674         name = f"__{op.__name__}__" 
675         return set_function_name(integer_arithmetic_method, name, cls)
