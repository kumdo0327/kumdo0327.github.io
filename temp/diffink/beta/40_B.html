 93 class CrawlerRunner(object): 
 94     """ 
 95     This is a convenient helper class that keeps track of, manages and runs 
 96     crawlers inside an already setup Twisted `reactor`_. 
 97  
 98     The CrawlerRunner object must be instantiated with a 
 99     :class:`~scrapy.settings.Settings` object. 
100  
101     This class shouldn't be needed (since Scrapy is responsible of using it 
102     accordingly) unless writing scripts that manually handle the crawling 
103     process. See :ref:`run-from-script` for an example. 
104     """ 
105  
106     crawlers = property( 
107         lambda self: self._crawlers, 
108         doc="Set of :class:`crawlers <scrapy.crawler.Crawler>` started by " 
109             ":meth:`crawl` and managed by this class." 
110     ) 
111  
112     def __init__(self, settings): 
113         if isinstance(settings, dict): 
114             settings = Settings(settings) 
115         self.settings = settings 
116         self.spider_loader = _get_spider_loader(settings) 
117         self._crawlers = set() 
118         self._active = set() 
119  
120     @property 
121     def spiders(self): 
122         warnings.warn("CrawlerRunner.spiders attribute is renamed to " 
123                       "CrawlerRunner.spider_loader.", 
124                       category=ScrapyDeprecationWarning, stacklevel=2) 
125         return self.spider_loader 
126  
127     def crawl(self, crawler_or_spidercls, *args, **kwargs): 
128         """ 
129         Run a crawler with the provided arguments. 
130  
131         It will call the given Crawler's :meth:`~Crawler.crawl` method, while 
132         keeping track of it so it can be stopped later. 
133  
134         If `crawler_or_spidercls` isn't a :class:`~scrapy.crawler.Crawler` 
135         instance, this method will try to create one using this parameter as 
136         the spider class given to it. 
137  
138         Returns a deferred that is fired when the crawling is finished. 
139  
140         :param crawler_or_spidercls: already created crawler, or a spider class 
141             or spider's name inside the project to create it 
142         :type crawler_or_spidercls: :class:`~scrapy.crawler.Crawler` instance, 
143             :class:`~scrapy.spiders.Spider` subclass or string 
144  
145         :param list args: arguments to initialize the spider 
146  
147         :param dict kwargs: keyword arguments to initialize the spider 
148         """ 
149         crawler = crawler_or_spidercls 
150         if not isinstance(crawler_or_spidercls, Crawler): 
151             crawler = self._create_crawler(crawler_or_spidercls) 
152  
153         self.crawlers.add(crawler) 
154         d = crawler.crawl(*args, **kwargs) 
155         self._active.add(d) 
156  
157         def _done(result): 
158             self.crawlers.discard(crawler) 
159             self._active.discard(d) 
160             return result 
161  
162         return d.addBoth(_done) 
163  
164     def _create_crawler(self, spidercls): 
165         if isinstance(spidercls, six.string_types): 
166             spidercls = self.spider_loader.load(spidercls) 
167         return Crawler(spidercls, self.settings) 
168  
169     def stop(self): 
170         """ 
171         Stops simultaneously all the crawling jobs taking place. 
172  
173         Returns a deferred that is fired when they all have ended. 
174         """ 
175         return defer.DeferredList([c.stop() for c in list(self.crawlers)]) 
176  
177     @defer.inlineCallbacks 
178     def join(self): 
179         """ 
180         join() 
181  
182         Returns a deferred that is fired when all managed :attr:`crawlers` have 
183         completed their executions. 
184         """ 
185         while self._active: 
186             yield defer.DeferredList(self._active)

189 class CrawlerProcess(CrawlerRunner): 
190     """ 
191     A class to run multiple scrapy crawlers in a process simultaneously. 
192  
193     This class extends :class:`~scrapy.crawler.CrawlerRunner` by adding support 
194     for starting a Twisted `reactor`_ and handling shutdown signals, like the 
195     keyboard interrupt command Ctrl-C. It also configures top-level logging. 
196  
197     This utility should be a better fit than 
198     :class:`~scrapy.crawler.CrawlerRunner` if you aren't running another 
199     Twisted `reactor`_ within your application. 
200  
201     The CrawlerProcess object must be instantiated with a 
202     :class:`~scrapy.settings.Settings` object. 
203  
204     This class shouldn't be needed (since Scrapy is responsible of using it 
205     accordingly) unless writing scripts that manually handle the crawling 
206     process. See :ref:`run-from-script` for an example. 
207     """ 
208  
209     def __init__(self, settings): 
210         super(CrawlerProcess, self).__init__(settings) 
211         install_shutdown_handlers(self._signal_shutdown) 
212         configure_logging(settings) 
213         log_scrapy_info(settings) 
214  
215     def _signal_shutdown(self, signum, _): 
216         install_shutdown_handlers(self._signal_kill) 
217         signame = signal_names[signum] 
218         logger.info("Received %(signame)s, shutting down gracefully. Send again to force ", 
219                     {'signame': signame}) 
220         reactor.callFromThread(self.stop) 
221  
222     def _signal_kill(self, signum, _): 
223         install_shutdown_handlers(signal.SIG_IGN) 
224         signame = signal_names[signum] 
225         logger.info('Received %(signame)s twice, forcing unclean shutdown', 
226                     {'signame': signame}) 
227         reactor.callFromThread(self._stop_reactor) 
228  
229     def start(self, stop_after_crawl=True): 
230         """ 
231         This method starts a Twisted `reactor`_, adjusts its pool size to 
232         :setting:`REACTOR_THREADPOOL_MAXSIZE`, and installs a DNS cache based 
233         on :setting:`DNSCACHE_ENABLED` and :setting:`DNSCACHE_SIZE`. 
234  
235         If `stop_after_crawl` is True, the reactor will be stopped after all 
236         crawlers have finished, using :meth:`join`. 
237  
238         :param boolean stop_after_crawl: stop or not the reactor when all 
239             crawlers have finished 
240         """ 
241         if stop_after_crawl: 
242             d = self.join() 
243             # Don't start the reactor if the deferreds are already fired 
244             if d.called: 
245                 return 
246             d.addBoth(lambda _: self._stop_reactor()) 
247  
248         cache_size = self.settings.getint('DNSCACHE_SIZE') if self.settings.getbool('DNSCACHE_ENABLED') else 0 
249         reactor.installResolver(CachingThreadedResolver(reactor, cache_size, 
250                                                             self.settings.getfloat('DNS_TIMEOUT'))) 
251         tp = reactor.getThreadPool() 
252         tp.adjustPoolsize(maxthreads=self.settings.getint('REACTOR_THREADPOOL_MAXSIZE')) 
253         reactor.addSystemEventTrigger('before', 'shutdown', self.stop) 
254         reactor.run(installSignalHandlers=False)  # blocking call 
255  
256     def _stop_reactor(self, _=None): 
257         try: 
258             reactor.stop() 
259         except RuntimeError:  # raised if already stopped or in shutdown stage 
260             pass

263 def _get_spider_loader(settings): 
264     """ Get SpiderLoader instance from settings """ 
265     if settings.get('SPIDER_MANAGER_CLASS'): 
266         warnings.warn( 
267             'SPIDER_MANAGER_CLASS option is deprecated. ' 
268             'Please use SPIDER_LOADER_CLASS.', 
269             category=ScrapyDeprecationWarning, stacklevel=2 
270         ) 
271     cls_path = settings.get('SPIDER_MANAGER_CLASS', 
272                             settings.get('SPIDER_LOADER_CLASS')) 
273     loader_cls = load_object(cls_path) 
274     try: 
275         verifyClass(ISpiderLoader, loader_cls) 
276     except DoesNotImplement: 
277         warnings.warn( 
278             'SPIDER_LOADER_CLASS (previously named SPIDER_MANAGER_CLASS) does ' 
279             'not fully implement scrapy.interfaces.ISpiderLoader interface. ' 
280             'Please add all missing methods to avoid unexpected runtime errors.', 
281             category=ScrapyDeprecationWarning, stacklevel=2 
282         ) 
283     return loader_cls.from_settings(settings.frozencopy())
