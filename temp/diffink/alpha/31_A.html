 115         if self.center is not None and not is_bool(self.center): 
 116             raise ValueError("center must be a boolean") 
 117         if self.min_periods is not None and not is_integer(self.min_periods):

 120             "right",

 125             raise ValueError("closed must be 'right', 'left', 'both' or 'neither'")

 252     def _prep_values(self, values: Optional[np.ndarray] = None) -> np.ndarray: 
 253         """Convert input to numpy arrays for Cython routines""" 
 254         if values is None: 
 255             values = getattr(self._selected_obj, "values", self._selected_obj) 
 256  
 257         # GH #12373 : rolling functions error on float32 data 
 258         # make sure the data is coerced to float64 
 259         if is_float_dtype(values.dtype): 
 260             values = ensure_float64(values) 
 261         elif is_integer_dtype(values.dtype): 
 262             values = ensure_float64(values) 
 263         elif needs_i8_conversion(values.dtype): 
 264             raise NotImplementedError( 
 265                 f"ops for {self._window_type} for this " 
 266                 f"dtype {values.dtype} are not implemented" 
 267             ) 
 268         else: 
 269             try: 
 270                 values = ensure_float64(values) 
 271             except (ValueError, TypeError) as err: 
 272                 raise TypeError(f"cannot handle this type -> {values.dtype}") from err 
 273  
 274         # Convert inf to nan for C funcs 
 275         inf = np.isinf(values) 
 276         if inf.any(): 
 277             values = np.where(inf, np.nan, values) 
 278  
 279         return values

 299     def _wrap_results(self, results, blocks, obj, exclude=None) -> FrameOrSeries: 
 300         """ 
 301         Wrap the results. 
 302  
 303         Parameters 
 304         ---------- 
 305         results : list of ndarrays 
 306         blocks : list of blocks 
 307         obj : conformed data (may be resampled) 
 308         exclude: list of columns to exclude, default to None 
 309         """ 
 310         from pandas import Series, concat 
 311  
 312         final = [] 
 313         for result, block in zip(results, blocks): 
 314  
 315             result = self._wrap_result(result, block=block, obj=obj) 
 316             if result.ndim == 1: 
 317                 return result 
 318             final.append(result) 
 319  
 320         # if we have an 'on' column 
 321         # we want to put it back into the results 
 322         # in the same location 
 323         columns = self._selected_obj.columns 
 324         if self.on is not None and not self._on.equals(obj.index): 
 325  
 326             name = self._on.name 
 327             final.append(Series(self._on, index=obj.index, name=name)) 
 328  
 329             if self._selection is not None: 
 330  
 331                 selection = ensure_index(self._selection) 
 332  
 333                 # need to reorder to include original location of 
 334                 # the on column (if its not already there) 
 335                 if name not in selection: 
 336                     columns = self.obj.columns 
 337                     indexer = columns.get_indexer(selection.tolist() + [name]) 
 338                     columns = columns.take(sorted(indexer)) 
 339  
 340         # exclude nuisance columns so that they are not reindexed 
 341         if exclude is not None and exclude: 
 342             columns = [c for c in columns if c not in exclude] 
 343  
 344             if not columns: 
 345                 raise DataError("No numeric types to aggregate") 
 346  
 347         if not len(final): 
 348             return obj.astype("float64") 
 349         return concat(final, axis=1).reindex(columns=columns, copy=False)

 351     def _center_window(self, result, window) -> np.ndarray: 
 352         """ 
 353         Center the result in the window. 
 354         """ 
 355         if self.axis > result.ndim - 1: 
 356             raise ValueError("Requested axis is larger then no. of argument dimensions") 
 357  
 358         offset = calculate_center_offset(window) 
 359         if offset > 0: 
 360             lead_indexer = [slice(None)] * result.ndim 
 361             lead_indexer[self.axis] = slice(offset, None) 
 362             result = np.copy(result[tuple(lead_indexer)]) 
 363         return result

 406     def _apply( 
 407         self, 
 408         func: Callable, 
 409         center: bool, 
 410         require_min_periods: int = 0, 
 411         floor: int = 1, 
 412         is_weighted: bool = False, 
 413         name: Optional[str] = None, 
 414         use_numba_cache: bool = False, 
 415         **kwargs, 
 416     ): 
 417         """ 
 418         Rolling statistical measure using supplied function. 
 419  
 420         Designed to be used with passed-in Cython array-based functions. 
 421  
 422         Parameters 
 423         ---------- 
 424         func : callable function to apply 
 425         center : bool 
 426         require_min_periods : int 
 427         floor : int 
 428         is_weighted : bool 
 429         name : str, 
 430             compatibility with groupby.rolling 
 431         use_numba_cache : bool 
 432             whether to cache a numba compiled function. Only available for numba 
 433             enabled methods (so far only apply) 
 434         **kwargs 
 435             additional arguments for rolling function and window function 
 436  
 437         Returns 
 438         ------- 
 439         y : type of input 
 440         """ 
 441         win_type = self._get_win_type(kwargs) 
 442         window = self._get_window(win_type=win_type) 
 443  
 444         blocks, obj = self._create_blocks() 
 445         block_list = list(blocks) 
 446         window_indexer = self._get_window_indexer(window, name) 
 447  
 448         results = [] 
 449         exclude: List[Scalar] = [] 
 450         for i, b in enumerate(blocks): 
 451             try: 
 452                 values = self._prep_values(b.values) 
 453  
 454             except (TypeError, NotImplementedError) as err: 
 455                 if isinstance(obj, ABCDataFrame): 
 456                     exclude.extend(b.columns) 
 457                     del block_list[i] 
 458                     continue 
 459                 else: 
 460                     raise DataError("No numeric types to aggregate") from err 
 461  
 462             if values.size == 0: 
 463                 results.append(values.copy()) 
 464                 continue 
 465  
 466             # calculation function 
 467             offset = calculate_center_offset(window) if center else 0 
 468             additional_nans = np.array([np.nan] * offset) 
 469  
 470             if not is_weighted: 
 471  
 472                 def calc(x): 
 473                     x = np.concatenate((x, additional_nans)) 
 474                     if not isinstance(window, BaseIndexer): 
 475                         min_periods = calculate_min_periods( 
 476                             window, self.min_periods, len(x), require_min_periods, floor 
 477                         ) 
 478                     else: 
 479                         min_periods = calculate_min_periods( 
 480                             self.min_periods or 1, 
 481                             self.min_periods, 
 482                             len(x), 
 483                             require_min_periods, 
 484                             floor, 
 485                         ) 
 486                     start, end = window_indexer.get_window_bounds( 
 487                         num_values=len(x), 
 488                         min_periods=self.min_periods, 
 489                         center=self.center, 
 490                         closed=self.closed, 
 491                     ) 
 492                     return func(x, start, end, min_periods) 
 493  
 494             else: 
 495  
 496                 def calc(x): 
 497                     x = np.concatenate((x, additional_nans)) 
 498                     return func(x, window, self.min_periods) 
 499  
 500             with np.errstate(all="ignore"): 
 501                 if values.ndim > 1: 
 502                     result = np.apply_along_axis(calc, self.axis, values) 
 503                 else: 
 504                     result = calc(values) 
 505                     result = np.asarray(result) 
 506  
 507             if use_numba_cache: 
 508                 self._numba_func_cache[name] = func 
 509  
 510             if center: 
 511                 result = self._center_window(result, window) 
 512  
 513             results.append(result) 
 514  
 515         return self._wrap_results(results, block_list, obj, exclude)

 518         result, how = self._aggregate(func, *args, **kwargs) 
 519         if result is None:

 774     ----------

 969     def _get_win_type(self, kwargs: Dict) -> Union[str, Tuple]: 
 970         """ 
 971         Extract arguments for the window type, provide validation for it 
 972         and return the validated window type. 
 973  
 974         Parameters 
 975         ---------- 
 976         kwargs : dict 
 977  
 978         Returns 
 979         ------- 
 980         win_type : str, or tuple 
 981         """ 
 982         # the below may pop from kwargs 
 983         def _validate_win_type(win_type, kwargs): 
 984             arg_map = { 
 985                 "kaiser": ["beta"], 
 986                 "gaussian": ["std"], 
 987                 "general_gaussian": ["power", "width"], 
 988                 "slepian": ["width"], 
 989                 "exponential": ["tau"], 
 990             } 
 991  
 992             if win_type in arg_map: 
 993                 win_args = _pop_args(win_type, arg_map[win_type], kwargs) 
 994                 if win_type == "exponential": 
 995                     # exponential window requires the first arg (center) 
 996                     # to be set to None (necessary for symmetric window) 
 997                     win_args.insert(0, None) 
 998  
 999                 return tuple([win_type] + win_args) 
1000  
1001             return win_type 
1002  
1003         def _pop_args(win_type, arg_names, kwargs): 
1004             all_args = [] 
1005             for n in arg_names: 
1006                 if n not in kwargs: 
1007                     raise ValueError(f"{win_type} window requires {n}") 
1008                 all_args.append(kwargs.pop(n)) 
1009             return all_args 
1010  
1011         return _validate_win_type(self.win_type, kwargs)

 114     def validate(self) -> None: 
 115         if self.center is not None and not is_bool(self.center): 
 116             raise ValueError("center must be a boolean") 
 117         if self.min_periods is not None and not is_integer(self.min_periods): 
 118             raise ValueError("min_periods must be an integer") 
 119         if self.closed is not None and self.closed not in [ 
 120             "right", 
 121             "both", 
 122             "left", 
 123             "neither", 
 124         ]: 
 125             raise ValueError("closed must be 'right', 'left', 'both' or 'neither'") 
 126         if not isinstance(self.obj, (ABCSeries, ABCDataFrame)): 
 127             raise TypeError(f"invalid type: {type(self)}") 
 128         if isinstance(self.window, BaseIndexer): 
 129             self._validate_get_window_bounds_signature(self.window)

 149     def _create_blocks(self): 
 150         """ 
 151         Split data into blocks & return conformed data. 
 152         """ 
 153         obj = self._selected_obj 
 154  
 155         # filter out the on from the object 
 156         if self.on is not None and not isinstance(self.on, Index): 
 157             if obj.ndim == 2: 
 158                 obj = obj.reindex(columns=obj.columns.difference([self.on]), copy=False) 
 159         blocks = obj._to_dict_of_blocks(copy=False).values() 
 160  
 161         return blocks, obj

 252     def _prep_values(self, values: Optional[np.ndarray] = None) -> np.ndarray: 
 253         """Convert input to numpy arrays for Cython routines""" 
 254         if values is None: 
 255             values = getattr(self._selected_obj, "values", self._selected_obj) 
 256  
 257         # GH #12373 : rolling functions error on float32 data 
 258         # make sure the data is coerced to float64 
 259         if is_float_dtype(values.dtype): 
 260             values = ensure_float64(values) 
 261         elif is_integer_dtype(values.dtype): 
 262             values = ensure_float64(values) 
 263         elif needs_i8_conversion(values.dtype): 
 264             raise NotImplementedError( 
 265                 f"ops for {self._window_type} for this " 
 266                 f"dtype {values.dtype} are not implemented" 
 267             ) 
 268         else: 
 269             try: 
 270                 values = ensure_float64(values) 
 271             except (ValueError, TypeError) as err: 
 272                 raise TypeError(f"cannot handle this type -> {values.dtype}") from err 
 273  
 274         # Convert inf to nan for C funcs 
 275         inf = np.isinf(values) 
 276         if inf.any(): 
 277             values = np.where(inf, np.nan, values) 
 278  
 279         return values

 281     def _wrap_result(self, result, block=None, obj=None): 
 282         """ 
 283         Wrap a single result. 
 284         """ 
 285         if obj is None: 
 286             obj = self._selected_obj 
 287         index = obj.index 
 288  
 289         if isinstance(result, np.ndarray): 
 290  
 291             if result.ndim == 1: 
 292                 from pandas import Series 
 293  
 294                 return Series(result, index, name=obj.name) 
 295  
 296             return type(obj)(result, index=index, columns=block.columns) 
 297         return result

 299     def _wrap_results(self, results, blocks, obj, exclude=None) -> FrameOrSeries: 
 300         """ 
 301         Wrap the results. 
 302  
 303         Parameters 
 304         ---------- 
 305         results : list of ndarrays 
 306         blocks : list of blocks 
 307         obj : conformed data (may be resampled) 
 308         exclude: list of columns to exclude, default to None 
 309         """ 
 310         from pandas import Series, concat 
 311  
 312         final = [] 
 313         for result, block in zip(results, blocks): 
 314  
 315             result = self._wrap_result(result, block=block, obj=obj) 
 316             if result.ndim == 1: 
 317                 return result 
 318             final.append(result) 
 319  
 320         # if we have an 'on' column 
 321         # we want to put it back into the results 
 322         # in the same location 
 323         columns = self._selected_obj.columns 
 324         if self.on is not None and not self._on.equals(obj.index): 
 325  
 326             name = self._on.name 
 327             final.append(Series(self._on, index=obj.index, name=name)) 
 328  
 329             if self._selection is not None: 
 330  
 331                 selection = ensure_index(self._selection) 
 332  
 333                 # need to reorder to include original location of 
 334                 # the on column (if its not already there) 
 335                 if name not in selection: 
 336                     columns = self.obj.columns 
 337                     indexer = columns.get_indexer(selection.tolist() + [name]) 
 338                     columns = columns.take(sorted(indexer)) 
 339  
 340         # exclude nuisance columns so that they are not reindexed 
 341         if exclude is not None and exclude: 
 342             columns = [c for c in columns if c not in exclude] 
 343  
 344             if not columns: 
 345                 raise DataError("No numeric types to aggregate") 
 346  
 347         if not len(final): 
 348             return obj.astype("float64") 
 349         return concat(final, axis=1).reindex(columns=columns, copy=False)

 351     def _center_window(self, result, window) -> np.ndarray: 
 352         """ 
 353         Center the result in the window. 
 354         """ 
 355         if self.axis > result.ndim - 1: 
 356             raise ValueError("Requested axis is larger then no. of argument dimensions") 
 357  
 358         offset = calculate_center_offset(window) 
 359         if offset > 0: 
 360             lead_indexer = [slice(None)] * result.ndim 
 361             lead_indexer[self.axis] = slice(offset, None) 
 362             result = np.copy(result[tuple(lead_indexer)]) 
 363         return result

 396         """

 398         """

 406     def _apply( 
 407         self, 
 408         func: Callable, 
 409         center: bool, 
 410         require_min_periods: int = 0, 
 411         floor: int = 1, 
 412         is_weighted: bool = False, 
 413         name: Optional[str] = None, 
 414         use_numba_cache: bool = False, 
 415         **kwargs, 
 416     ): 
 417         """ 
 418         Rolling statistical measure using supplied function. 
 419  
 420         Designed to be used with passed-in Cython array-based functions. 
 421  
 422         Parameters 
 423         ---------- 
 424         func : callable function to apply 
 425         center : bool 
 426         require_min_periods : int 
 427         floor : int 
 428         is_weighted : bool 
 429         name : str, 
 430             compatibility with groupby.rolling 
 431         use_numba_cache : bool 
 432             whether to cache a numba compiled function. Only available for numba 
 433             enabled methods (so far only apply) 
 434         **kwargs 
 435             additional arguments for rolling function and window function 
 436  
 437         Returns 
 438         ------- 
 439         y : type of input 
 440         """ 
 441         win_type = self._get_win_type(kwargs) 
 442         window = self._get_window(win_type=win_type) 
 443  
 444         blocks, obj = self._create_blocks() 
 445         block_list = list(blocks) 
 446         window_indexer = self._get_window_indexer(window, name) 
 447  
 448         results = [] 
 449         exclude: List[Scalar] = [] 
 450         for i, b in enumerate(blocks): 
 451             try: 
 452                 values = self._prep_values(b.values) 
 453  
 454             except (TypeError, NotImplementedError) as err: 
 455                 if isinstance(obj, ABCDataFrame): 
 456                     exclude.extend(b.columns) 
 457                     del block_list[i] 
 458                     continue 
 459                 else: 
 460                     raise DataError("No numeric types to aggregate") from err 
 461  
 462             if values.size == 0: 
 463                 results.append(values.copy()) 
 464                 continue 
 465  
 466             # calculation function 
 467             offset = calculate_center_offset(window) if center else 0 
 468             additional_nans = np.array([np.nan] * offset) 
 469  
 470             if not is_weighted: 
 471  
 472                 def calc(x): 
 473                     x = np.concatenate((x, additional_nans)) 
 474                     if not isinstance(window, BaseIndexer): 
 475                         min_periods = calculate_min_periods( 
 476                             window, self.min_periods, len(x), require_min_periods, floor 
 477                         ) 
 478                     else: 
 479                         min_periods = calculate_min_periods( 
 480                             self.min_periods or 1, 
 481                             self.min_periods, 
 482                             len(x), 
 483                             require_min_periods, 
 484                             floor, 
 485                         ) 
 486                     start, end = window_indexer.get_window_bounds( 
 487                         num_values=len(x), 
 488                         min_periods=self.min_periods, 
 489                         center=self.center, 
 490                         closed=self.closed, 
 491                     ) 
 492                     return func(x, start, end, min_periods) 
 493  
 494             else: 
 495  
 496                 def calc(x): 
 497                     x = np.concatenate((x, additional_nans)) 
 498                     return func(x, window, self.min_periods) 
 499  
 500             with np.errstate(all="ignore"): 
 501                 if values.ndim > 1: 
 502                     result = np.apply_along_axis(calc, self.axis, values) 
 503                 else: 
 504                     result = calc(values) 
 505                     result = np.asarray(result) 
 506  
 507             if use_numba_cache: 
 508                 self._numba_func_cache[name] = func 
 509  
 510             if center: 
 511                 result = self._center_window(result, window) 
 512  
 513             results.append(result) 
 514  
 515         return self._wrap_results(results, block_list, obj, exclude) 
 516  
 517     def aggregate(self, func, *args, **kwargs): 
 518         result, how = self._aggregate(func, *args, **kwargs) 
 519         if result is None: 
 520             return self.apply(func, raw=False, args=args, kwargs=kwargs) 
 521         return result

 769 class Window(_Window): 
 770     """ 
 771     Provide rolling window calculations. 
 772  
 773     Parameters 
 774     ---------- 
 775     window : int, offset, or BaseIndexer subclass 
 776         Size of the moving window. This is the number of observations used for 
 777         calculating the statistic. Each window will be a fixed size. 
 778  
 779         If its an offset then this will be the time period of each window. Each 
 780         window will be a variable sized based on the observations included in 
 781         the time-period. This is only valid for datetimelike indexes. 
 782  
 783         If a BaseIndexer subclass is passed, calculates the window boundaries 
 784         based on the defined ``get_window_bounds`` method. Additional rolling 
 785         keyword arguments, namely `min_periods`, `center`, and 
 786         `closed` will be passed to `get_window_bounds`. 
 787     min_periods : int, default None 
 788         Minimum number of observations in window required to have a value 
 789         (otherwise result is NA). For a window that is specified by an offset, 
 790         `min_periods` will default to 1. Otherwise, `min_periods` will default 
 791         to the size of the window. 
 792     center : bool, default False 
 793         Set the labels at the center of the window. 
 794     win_type : str, default None 
 795         Provide a window type. If ``None``, all points are evenly weighted. 
 796         See the notes below for further information. 
 797     on : str, optional 
 798         For a DataFrame, a datetime-like column or MultiIndex level on which 
 799         to calculate the rolling window, rather than the DataFrame's index. 
 800         Provided integer column is ignored and excluded from result since 
 801         an integer index is not used to calculate the rolling window. 
 802     axis : int or str, default 0 
 803     closed : str, default None 
 804         Make the interval closed on the 'right', 'left', 'both' or 
 805         'neither' endpoints. 
 806         For offset-based windows, it defaults to 'right'. 
 807         For fixed windows, defaults to 'both'. Remaining cases not implemented 
 808         for fixed windows. 
 809  
 810     Returns 
 811     ------- 
 812     a Window or Rolling sub-classed for the particular operation 
 813  
 814     See Also 
 815     -------- 
 816     expanding : Provides expanding transformations. 
 817     ewm : Provides exponential weighted functions. 
 818  
 819     Notes 
 820     ----- 
 821     By default, the result is set to the right edge of the window. This can be 
 822     changed to the center of the window by setting ``center=True``. 
 823  
 824     To learn more about the offsets & frequency strings, please see `this link 
 825     <https://pandas.pydata.org/pandas-docs/stable/user_guide/timeseries.html#offset-aliases>`__. 
 826  
 827     The recognized win_types are: 
 828  
 829     * ``boxcar`` 
 830     * ``triang`` 
 831     * ``blackman`` 
 832     * ``hamming`` 
 833     * ``bartlett`` 
 834     * ``parzen`` 
 835     * ``bohman`` 
 836     * ``blackmanharris`` 
 837     * ``nuttall`` 
 838     * ``barthann`` 
 839     * ``kaiser`` (needs beta) 
 840     * ``gaussian`` (needs std) 
 841     * ``general_gaussian`` (needs power, width) 
 842     * ``slepian`` (needs width) 
 843     * ``exponential`` (needs tau), center is set to None. 
 844  
 845     If ``win_type=None`` all points are evenly weighted. To learn more about 
 846     different window types see `scipy.signal window functions 
 847     <https://docs.scipy.org/doc/scipy/reference/signal.html#window-functions>`__. 
 848  
 849     Examples 
 850     -------- 
 851     >>> df = pd.DataFrame({'B': [0, 1, 2, np.nan, 4]}) 
 852     >>> df 
 853          B 
 854     0  0.0 
 855     1  1.0 
 856     2  2.0 
 857     3  NaN 
 858     4  4.0 
 859  
 860     Rolling sum with a window length of 2, using the 'triang' 
 861     window type. 
 862  
 863     >>> df.rolling(2, win_type='triang').sum() 
 864          B 
 865     0  NaN 
 866     1  0.5 
 867     2  1.5 
 868     3  NaN 
 869     4  NaN 
 870  
 871     Rolling sum with a window length of 2, using the 'gaussian' 
 872     window type (note how we need to specify std). 
 873  
 874     >>> df.rolling(2, win_type='gaussian').sum(std=3) 
 875               B 
 876     0       NaN 
 877     1  0.986207 
 878     2  2.958621 
 879     3       NaN 
 880     4       NaN 
 881  
 882     Rolling sum with a window length of 2, min_periods defaults 
 883     to the window length. 
 884  
 885     >>> df.rolling(2).sum() 
 886          B 
 887     0  NaN 
 888     1  1.0 
 889     2  3.0 
 890     3  NaN 
 891     4  NaN 
 892  
 893     Same as above, but explicitly set the min_periods 
 894  
 895     >>> df.rolling(2, min_periods=1).sum() 
 896          B 
 897     0  0.0 
 898     1  1.0 
 899     2  3.0 
 900     3  2.0 
 901     4  4.0 
 902  
 903     Same as above, but with forward-looking windows 
 904  
 905     >>> indexer = pd.api.indexers.FixedForwardWindowIndexer(window_size=2) 
 906     >>> df.rolling(window=indexer, min_periods=1).sum() 
 907          B 
 908     0  1.0 
 909     1  3.0 
 910     2  2.0 
 911     3  4.0 
 912     4  4.0 
 913  
 914     A ragged (meaning not-a-regular frequency), time-indexed DataFrame 
 915  
 916     >>> df = pd.DataFrame({'B': [0, 1, 2, np.nan, 4]},

 947         window = self.window

 969     def _get_win_type(self, kwargs: Dict) -> Union[str, Tuple]: 
 970         """ 
 971         Extract arguments for the window type, provide validation for it 
 972         and return the validated window type. 
 973  
 974         Parameters 
 975         ---------- 
 976         kwargs : dict 
 977  
 978         Returns 
 979         ------- 
 980         win_type : str, or tuple 
 981         """ 
 982         # the below may pop from kwargs 
 983         def _validate_win_type(win_type, kwargs): 
 984             arg_map = { 
 985                 "kaiser": ["beta"], 
 986                 "gaussian": ["std"], 
 987                 "general_gaussian": ["power", "width"], 
 988                 "slepian": ["width"], 
 989                 "exponential": ["tau"], 
 990             } 
 991  
 992             if win_type in arg_map: 
 993                 win_args = _pop_args(win_type, arg_map[win_type], kwargs) 
 994                 if win_type == "exponential": 
 995                     # exponential window requires the first arg (center) 
 996                     # to be set to None (necessary for symmetric window) 
 997                     win_args.insert(0, None) 
 998  
 999                 return tuple([win_type] + win_args) 
1000  
1001             return win_type 
1002  
1003         def _pop_args(win_type, arg_names, kwargs): 
1004             all_args = [] 
1005             for n in arg_names: 
1006                 if n not in kwargs: 
1007                     raise ValueError(f"{win_type} window requires {n}") 
1008                 all_args.append(kwargs.pop(n)) 
1009             return all_args 
1010  
1011         return _validate_win_type(self.win_type, kwargs)

1014         self, other=None, win_type: Optional[Union[str, Tuple]] = None 
1015     ) -> np.ndarray: 
1016         """ 
1017         Get the window, weights. 
1018  
1019         Parameters 
1020         ---------- 
1021         other : 
1022             ignored, exists for compatibility 
1023         win_type : str, or tuple 
1024             type of window to create 
1025  
1026         Returns 
1027         ------- 
1028         window : ndarray 
1029             the window, weights 
1030         """ 
1031         window = self.window 
1032         if isinstance(window, (list, tuple, np.ndarray)): 
1033             return com.asarray_tuplesafe(window).astype(float) 
1034         elif is_integer(window): 
1035             import scipy.signal as sig 
1036  
1037             # GH #15662. `False` makes symmetric window, rather than periodic. 
1038             return sig.get_window(win_type, window, False).astype(float)
