 144   /** 
 145    * Creates a scope with all types declared. Declares newly discovered types 
 146    * and type properties in the type registry. 
 147    */ 
 148   public Scope createScope(Node root, Scope parent) { 
 149     // Constructing the global scope is very different than constructing 
 150     // inner scopes, because only global scopes can contain named classes that 
 151     // show up in the type registry. 
 152     Scope newScope = null; 
 153     if (parent == null) { 
 154       // Find all the classes in the global scope. 
 155       newScope = createInitialScope(root); 
 156  
 157       GlobalScopeBuilder scopeBuilder = new GlobalScopeBuilder(newScope); 
 158       NodeTraversal.traverse(compiler, root, scopeBuilder); 
 159       scopeBuilder.resolveStubDeclarations(); 
 160       scopeBuilder.resolveTypes(); 
 161  
 162       // Gather the properties in each function that we found in the 
 163       // global scope, if that function has a @this type that we can 
 164       // build properties on. 
 165       for (Node functionNode : scopeBuilder.nonExternFunctions) { 
 166         JSType type = functionNode.getJSType(); 
 167         if (type != null && type instanceof FunctionType) { 
 168           FunctionType fnType = (FunctionType) type; 
 169           ObjectType fnThisType = fnType.getTypeOfThis(); 
 170           if (!fnThisType.isUnknownType()) { 
 171             NodeTraversal.traverse(compiler, functionNode.getLastChild(), 
 172                 scopeBuilder.new CollectProperties(fnThisType)); 
 173           } 
 174         } 
 175       } 
 176  
 177       codingConvention.defineDelegateProxyPrototypeProperties( 
 178           typeRegistry, newScope, delegateProxyPrototypes); 
 179     } else { 
 180       newScope = new Scope(parent, root); 
 181       LocalScopeBuilder scopeBuilder = new LocalScopeBuilder(newScope); 
 182       scopeBuilder.build(); 
 183       scopeBuilder.resolveTypes(); 
 184     } 
 185     return newScope; 
 186   }

 252     public void visit(NodeTraversal t, Node node, Node parent) { 
 253       Node nameNode = null; 
 254       switch (node.getType()) { 
 255         case Token.NAME: 
 256           nameNode = node; 
 257           break; 
 258         case Token.VAR: 
 259           if (node.hasOneChild()) { 
 260             nameNode = node.getFirstChild(); 
 261           } 
 262           break;

 271           registry.identifyEnumName(nameNode.getQualifiedName()); 
 272         } 
 273       }

 278    * Given a node, determines whether that node names a prototype 
 279    * property, and if so, returns the qualfied name node representing 
 280    * the owner of that property. Otherwise, returns null. 
 281    */ 
 282   private static Node getPrototypePropertyOwner(Node n) { 
 283     if (n.getType() == Token.GETPROP) { 
 284       Node firstChild = n.getFirstChild(); 
 285       if (firstChild.getType() == Token.GETPROP && 
 286           firstChild.getLastChild().getString().equals("prototype")) { 
 287         Node maybeOwner = firstChild.getFirstChild(); 
 288         if (maybeOwner.isQualifiedName()) { 
 289           return maybeOwner; 
 290         } 
 291       } 
 292     } 
 293     return null; 
 294   }

 320     void resolveTypes() { 
 321       // Resolve types and attach them to nodes. 
 322       for (DeferredSetType deferred : deferredSetTypes) { 
 323         deferred.resolve(scope); 
 324       } 
 325  
 326       // Resolve types and attach them to scope slots. 
 327       Iterator<Var> vars = scope.getVars(); 
 328       while (vars.hasNext()) { 
 329         vars.next().resolveType(typeParsingErrorReporter); 
 330       } 
 331  
 332       // Tell the type registry that any remaining types 
 333       // are unknown. 
 334       typeRegistry.resolveTypesInScope(scope); 
 335     }

 354     /** 
 355      * Returns the type specified in a JSDoc annotation near a GETPROP or NAME. 
 356      * 
 357      * Extracts type information from either the {@code @type} tag or from 
 358      * the {@code @return} and {@code @param} tags. 
 359      */

 361         NodeTraversal t, Node node, JSDocInfo info) { 
 362       return getDeclaredTypeInAnnotation(t.getSourceName(), node, info); 
 363     } 
 364  
 365     JSType getDeclaredTypeInAnnotation(String sourceName, 
 366         Node node, JSDocInfo info) { 
 367       JSType jsType = null; 
 368       Node objNode = node.getType() == Token.GETPROP ? 
 369           node.getFirstChild() : null; 
 370       if (info != null) { 
 371         if (info.hasType()) { 
 372           jsType = info.getType().evaluate(scope, typeRegistry); 
 373         } else if (FunctionTypeBuilder.isFunctionTypeDeclaration(info)) { 
 374           String fnName = node.getQualifiedName(); 
 375  
 376           // constructors are often handled separately. 
 377           if (info.isConstructor() && typeRegistry.getType(fnName) != null) { 
 378             return null; 
 379           } 
 380  
 381           FunctionTypeBuilder builder = 
 382               new FunctionTypeBuilder( 
 383                   fnName, compiler, node, sourceName, scope) 
 384               .inferTemplateTypeName(info) 
 385               .inferReturnType(info) 
 386               .inferParameterTypes(info) 
 387               .inferInheritance(info); 
 388  
 389           // Infer the context type. 
 390           boolean searchedForThisType = false; 
 391           if (objNode != null) { 
 392             if (objNode.getType() == Token.GETPROP && 
 393                 objNode.getLastChild().getString().equals("prototype")) { 
 394               builder.inferThisType(info, objNode.getFirstChild()); 
 395               searchedForThisType = true; 
 396             } else if (objNode.getType() == Token.THIS) { 
 397               builder.inferThisType(info, objNode.getJSType()); 
 398               searchedForThisType = true; 
 399             } 
 400           } 
 401  
 402           if (!searchedForThisType) { 
 403             builder.inferThisType(info, (Node) null); 
 404           } 
 405  
 406           jsType = builder.buildAndRegister(); 
 407         } 
 408       } 
 409       return jsType; 
 410     }

 412     /** 
 413      * Defines variable(s) or modifies types based on the content of the node 
 414      * {@code n}. A variable definition creates variables in the current scope, 
 415      * a function definition creates a binding, and an assignment updates the 
 416      * type of the namespaces on which the definition is made (e.g. 
 417      * {@code goog.FOO = 6}). 
 418      * @param n a {@link Token#VAR}, {@link Token#FUNCTION} or 
 419      *     {@link Token#ASSIGN} node 
 420      * @param parent {@code n}'s parent 
 421      */ 
 422     void define(Node n, Node parent) { 
 423       Preconditions.checkState(sourceName != null); 
 424  
 425       JSDocInfo info = n.getJSDocInfo(); 
 426       switch (n.getType()) { 
 427         case Token.CATCH: 
 428           Node catchName = n.getFirstChild(); 
 429           defineSlot(catchName, n, null); 
 430           break; 
 431  
 432         case Token.VAR: 
 433           if (n.getChildCount() > 1) { 
 434             if (info != null) { 
 435               // multiple children 
 436               compiler.report(JSError.make(sourceName, n, MULTIPLE_VAR_DEF)); 
 437             } 
 438             for (Node name : n.children()) { 
 439               defineName(name, n, parent, name.getJSDocInfo()); 
 440             } 
 441           } else { 
 442             Node name = n.getFirstChild(); 
 443             defineName(name, n, parent, 
 444                 (info != null) ? info : name.getJSDocInfo()); 
 445           } 
 446           break; 
 447  
 448         case Token.FUNCTION: 
 449           int parentType = parent.getType(); 
 450           Preconditions.checkState( 
 451               (scope.isLocal() || parentType != Token.ASSIGN) && 
 452               parentType != Token.NAME, 
 453               "function defined as standalone function when it is being " + 
 454               "assigned"); 
 455           String functionName = n.getFirstChild().getString(); 
 456           FunctionType functionType = getFunctionType(functionName, n, info, 
 457               null); 
 458           if (NodeUtil.isFunctionDeclaration(n)) { 
 459             defineSlot(n.getFirstChild(), n, functionType); 
 460           } 
 461           break; 
 462  
 463         case Token.ASSIGN: 
 464           // TODO(nicksantos): We should support direct assignment to a 
 465           // prototype, as in: 
 466           // Foo.prototype = { 
 467           //   a: function() { ... }, 
 468           //   b: function() { ... } 
 469           // }; 
 470           // Right now (6/23/08), we understand most of this syntax, but we 
 471           // don't tie the "a" and "b" methods to the context of Foo. 
 472  
 473           Node rvalue = n.getLastChild(); 
 474           Node lvalue = n.getFirstChild(); 
 475           info = (info != null) ? info : rvalue.getJSDocInfo(); 
 476           if (rvalue.getType() == Token.FUNCTION || 
 477               info != null && info.isConstructor()) { 
 478             getFunctionType(lvalue.getQualifiedName(), rvalue, info, 
 479                 lvalue); 
 480           } else if (info != null && info.hasEnumParameterType()) { 
 481             JSType type = getEnumType(lvalue.getQualifiedName(), n, rvalue, 
 482                 info.getEnumParameterType().evaluate(scope, typeRegistry)); 
 483             if (type != null) { 
 484               setDeferredType(lvalue, type); 
 485             } 
 486           } 
 487           break; 
 488  
 489         default: 
 490           throw new IllegalStateException(Integer.toString(n.getType())); 
 491       } 
 492     }

 494     /** 
 495      * Defines a variable based on the {@link Token#NAME} node passed. 
 496      * @param name The {@link Token#NAME} node. 
 497      * @param var The parent of the {@code name} node, which must be a 
 498      *     {@link Token#VAR} node. 
 499      * @param parent {@code var}'s parent. 
 500      * @param info the {@link JSDocInfo} information relating to this 
 501      *     {@code name} node. 
 502      */ 
 503     private void defineName(Node name, Node var, Node parent, JSDocInfo info) { 
 504       Node value = name.getFirstChild(); 
 505  
 506       if (value != null && value.getType() == Token.FUNCTION) { 
 507         // function 
 508         String functionName = name.getString(); 
 509         FunctionType functionType = 
 510             getFunctionType(functionName, value, info, null); 
 511         if (functionType.isReturnTypeInferred() && 
 512             scope.isLocal()) { 
 513           defineSlot(name, var, null); 
 514         } else { 
 515           defineSlot(name, var, functionType); 
 516         } 
 517       } else { 
 518         // variable's type 
 519         JSType type = null; 
 520         if (info == null) { 
 521           // the variable's type will be inferred 
 522           CompilerInput input = compiler.getInput(sourceName); 
 523           Preconditions.checkNotNull(input, sourceName); 
 524           type = input.isExtern() ? 
 525               typeRegistry.getNativeType(UNKNOWN_TYPE) : null; 
 526         } else if (info.hasEnumParameterType()) { 
 527           type = getEnumType(name.getString(), var, value, 
 528               info.getEnumParameterType().evaluate(scope, typeRegistry)); 
 529         } else if (info.isConstructor()) { 
 530           type = getFunctionType(name.getString(), value, info, name); 
 531         } else { 
 532           type = getDeclaredTypeInAnnotation(sourceName, name, info); 
 533         } 
 534  
 535         defineSlot(name, var, type); 
 536       } 
 537     }

 539     /** 
 540      * Gets the function type from the function node and its attached 
 541      * {@link JSDocInfo}. 
 542      * @param name the function's name 
 543      * @param rValue the function node. It must be a {@link Token#FUNCTION}. 
 544      * @param info the {@link JSDocInfo} attached to the function definition 
 545      * @param lvalueNode The node where this function is being 
 546      *     assigned. For example, {@code A.prototype.foo = ...} would be used to 
 547      *     determine that this function is a method of A.prototype. May be 
 548      *     null to indicate that this is not being assigned to a qualified name. 
 549      */ 
 550     private FunctionType getFunctionType(String name, 
 551         Node rValue, JSDocInfo info, @Nullable Node lvalueNode) { 
 552       FunctionType functionType = null; 
 553  
 554       // Handle function aliases. 
 555       if (rValue != null && rValue.isQualifiedName()) { 
 556         Var var = scope.getVar(rValue.getQualifiedName()); 
 557         if (var != null && var.getType() instanceof FunctionType) { 
 558           functionType = (FunctionType) var.getType(); 
 559           if (functionType != null && functionType.isConstructor()) { 
 560             typeRegistry.declareType(name, functionType.getInstanceType()); 
 561           } 
 562         } 
 563         return functionType; 
 564       } 
 565  
 566       Node owner = null; 
 567       if (lvalueNode != null) { 
 568         owner = getPrototypePropertyOwner(lvalueNode); 
 569       } 
 570  
 571       Node errorRoot = rValue == null ? lvalueNode : rValue; 
 572       boolean isFnLiteral = 
 573           rValue != null && rValue.getType() == Token.FUNCTION; 
 574       Node fnRoot = isFnLiteral ? rValue : null; 
 575       Node parametersNode = isFnLiteral ? 
 576           rValue.getFirstChild().getNext() : null; 
 577  
 578       if (functionType == null && info != null && info.hasType()) { 
 579         JSType type = info.getType().evaluate(scope, typeRegistry); 
 580  
 581         // Known to be not null since we have the FUNCTION token there. 
 582         type = type.restrictByNotNullOrUndefined(); 
 583         if (type.isFunctionType()) { 
 584           functionType = (FunctionType) type; 
 585           functionType.setJSDocInfo(info); 
 586         } 
 587       } 
 588  
 589       if (functionType == null) { 
 590         if (info == null || 
 591             !FunctionTypeBuilder.isFunctionTypeDeclaration(info)) { 
 592           // We don't really have any type information in the annotation. 
 593           // Before we give up on this function, look at the object we're 
 594           // assigning it to. For example, if the function looks like this: 
 595           // SubFoo.prototype.bar = function() { ... }; 
 596           // We can use type information on Foo.prototype.bar and apply it 
 597           // to this function. 
 598           if (lvalueNode != null && lvalueNode.getType() == Token.GETPROP && 
 599               lvalueNode.isQualifiedName()) { 
 600             Var var = scope.getVar( 
 601                 lvalueNode.getFirstChild().getQualifiedName()); 
 602             if (var != null) { 
 603               ObjectType ownerType = ObjectType.cast(var.getType()); 
 604               FunctionType propType = null; 
 605               if (ownerType != null) { 
 606                 String propName = lvalueNode.getLastChild().getString(); 
 607                 propType = findOverriddenFunction(ownerType, propName); 
 608               } 
 609  
 610               if (propType != null) { 
 611                 functionType = 
 612                     new FunctionTypeBuilder( 
 613                         name, compiler, errorRoot, sourceName, scope) 
 614                     .setSourceNode(fnRoot) 
 615                     .inferFromOverriddenFunction(propType, parametersNode) 
 616                     .inferThisType(info, owner) 
 617                     .buildAndRegister(); 
 618               } 
 619             } 
 620           } 
 621         } 
 622       } // end if (functionType == null) 
 623  
 624       if (functionType == null) { 
 625         functionType = 
 626             new FunctionTypeBuilder(name, compiler, errorRoot, sourceName, 
 627                 scope) 
 628             .setSourceNode(fnRoot) 
 629             .inferTemplateTypeName(info) 
 630             .inferReturnType(info) 
 631             .inferInheritance(info) 
 632             .inferThisType(info, owner) 
 633             .inferParameterTypes(parametersNode, info) 
 634             .buildAndRegister(); 
 635       } 
 636  
 637       // assigning the function type to the function node 
 638       if (rValue != null) { 
 639         setDeferredType(rValue, functionType); 
 640       } 
 641  
 642       // all done 
 643       return functionType; 
 644     }

 646     /** 
 647      * Find the function that's being overridden on this type, if any. 
 648      */ 
 649     private FunctionType findOverriddenFunction( 
 650         ObjectType ownerType, String propName) { 
 651       // First, check to see if the property is implemented 
 652       // on a superclass. 
 653       JSType propType = ownerType.getPropertyType(propName); 
 654       if (propType instanceof FunctionType) { 
 655         return (FunctionType) propType; 
 656       } else { 
 657         // If it's not, then check to see if it's implemented 
 658         // on an implemented interface. 
 659         for (ObjectType iface : 
 660                  ownerType.getCtorImplementedInterfaces()) { 
 661           propType = iface.getPropertyType(propName); 
 662           if (propType instanceof FunctionType) { 
 663             return (FunctionType) propType; 
 664           } 
 665         } 
 666       } 
 667  
 668       return null; 
 669     }

 671     /** 
 672      * Gets an enum type. If the definition is correct, the object literal used 
 673      * to define the enum is traversed to gather the elements name, and this 
 674      * method checks for duplicates. This method also enforces that all 
 675      * elements' name be syntactic constants according to the 
 676      * {@link CodingConvention} used. 
 677      * 
 678      * @param name the enum's name such as {@code HELLO} or {@code goog.foo.BAR} 
 679      * @param value the enum's original value. This value may be {@code null}. 
 680      * @param parent the value's parent 
 681      * @param elementsType the type of the elements of this enum 
 682      * @return the enum type 
 683      */ 
 684     private EnumType getEnumType(String name, Node parent, 
 685         Node value, JSType elementsType) { 
 686       EnumType enumType = null; 
 687  
 688       // no value with @enum 
 689       if (value != null) { 
 690         if (value.getType() == Token.OBJECTLIT) { 
 691           // collect enum elements 
 692           enumType = typeRegistry.createEnumType(name, elementsType); 
 693  
 694           // populate the enum type. 
 695           Node key = value.getFirstChild(); 
 696           while (key != null) { 
 697             String keyName = NodeUtil.getStringValue(key); 
 698  
 699             if (enumType.hasOwnProperty(keyName)) { 
 700               compiler.report(JSError.make(sourceName, key, ENUM_DUP, keyName)); 
 701             } else if (!codingConvention.isValidEnumKey(keyName)) { 
 702               compiler.report( 
 703                   JSError.make(sourceName, key, ENUM_NOT_CONSTANT, keyName)); 
 704             } else { 
 705               enumType.defineElement(keyName); 
 706             } 
 707             key = key.getNext(); 
 708             key = (key == null) ? null : key.getNext(); 
 709           } 
 710         } else if (value.isQualifiedName()) { 
 711           Var var = scope.getVar(value.getQualifiedName()); 
 712           if (var != null && var.getType() instanceof EnumType) { 
 713             enumType = (EnumType) var.getType(); 
 714           } 
 715         } 
 716       } 
 717  
 718       if (enumType == null) { 
 719         compiler.report(JSError.make(sourceName, parent, ENUM_INITIALIZER)); 
 720       } else if (scope.isGlobal()) { 
 721         if (name != null && !name.isEmpty()) { 
 722           typeRegistry.declareType(name, enumType.getElementsType()); 
 723         } 
 724       } 
 725  
 726       return enumType; 
 727     }

 741     /** 
 742      * Defines a typed variable. The defining node will be annotated with the 
 743      * variable's type of {@link JSTypeNative#UNKNOWN_TYPE} if its type is 
 744      * inferred. 
 745      * 
 746      * Slots may be any variable or any qualified name in the global scope. 
 747      * 
 748      * @param n the defining NAME or GETPROP node. 
 749      * @param parent the {@code n}'s parent. 
 750      * @param type the variable's type. It may be {@code null} if 
 751      *     {@code inferred} is {@code true}. 
 752      */ 
 753     void defineSlot(Node n, Node parent, JSType type, boolean inferred) { 
 754       Preconditions.checkArgument(inferred || type != null); 
 755  
 756       // Only allow declarations of NAMEs and qualfied names. 
 757       boolean shouldDeclareOnGlobalThis = false; 
 758       if (n.getType() == Token.NAME) { 
 759         Preconditions.checkArgument( 
 760             parent.getType() == Token.FUNCTION || 
 761             parent.getType() == Token.VAR || 
 762             parent.getType() == Token.LP || 
 763             parent.getType() == Token.CATCH); 
 764         shouldDeclareOnGlobalThis = scope.isGlobal() && 
 765             (parent.getType() == Token.VAR || 
 766              parent.getType() == Token.FUNCTION); 
 767       } else { 
 768         Preconditions.checkArgument( 
 769             n.getType() == Token.GETPROP && 
 770             (parent.getType() == Token.ASSIGN || 
 771              parent.getType() == Token.EXPR_RESULT)); 
 772       } 
 773       String variableName = n.getQualifiedName(); 
 774       Preconditions.checkArgument(!variableName.isEmpty()); 
 775  
 776       // declared in closest scope? 
 777       if (scope.isDeclared(variableName, false)) { 
 778         Var oldVar = scope.getVar(variableName); 
 779         validator.expectUndeclaredVariable( 
 780             sourceName, n, parent, oldVar, variableName, type); 
 781       } else { 
 782         if (!inferred) { 
 783           setDeferredType(n, type); 
 784         } 
 785         CompilerInput input = compiler.getInput(sourceName); 
 786         scope.declare(variableName, n, type, input, inferred); 
 787  
 788         if (shouldDeclareOnGlobalThis) { 
 789           ObjectType globalThis = 
 790               typeRegistry.getNativeObjectType(JSTypeNative.GLOBAL_THIS); 
 791           boolean isExtern = input.isExtern(); 
 792           if (inferred) { 
 793             globalThis.defineInferredProperty(variableName, 
 794                 type == null ? 
 795                     typeRegistry.getNativeType(JSTypeNative.NO_TYPE) : 
 796                     type, 
 797                 isExtern); 
 798           } else { 
 799             globalThis.defineDeclaredProperty(variableName, type, isExtern); 
 800           } 
 801         } 
 802  
 803         // If we're in the global scope, also declare var.prototype 
 804         // in the scope chain. 
 805         if (scope.isGlobal() && type instanceof FunctionType) { 
 806           FunctionType fnType = (FunctionType) type; 
 807           if (fnType.isConstructor() || fnType.isInterface()) { 
 808             FunctionType superClassCtor = fnType.getSuperClassConstructor(); 
 809             scope.declare(variableName + ".prototype", n, 
 810                 fnType.getPrototype(), compiler.getInput(sourceName), 
 811                 /* declared iff there's an explicit supertype */ 
 812                 superClassCtor == null || 
 813                 superClassCtor.getInstanceType().equals( 
 814                     typeRegistry.getNativeType(OBJECT_TYPE))); 
 815           } 
 816         } 
 817       } 
 818     }

 845     /** 
 846      * Visit a node in the global scope, and add anything it declares to the 
 847      * global symbol table. 
 848      * 
 849      * @param t The current traversal. 
 850      * @param n The node being visited. 
 851      * @param parent The parent of n 
 852      */ 
 853     @Override public void visit(NodeTraversal t, Node n, Node parent) { 
 854       switch (n.getType()) { 
 855         case Token.CALL: 
 856           SubclassRelationship relationship = 
 857               codingConvention.getClassesDefinedByCall(n); 
 858           if (relationship != null) { 
 859             ObjectType superClass = ObjectType.cast( 
 860                 typeRegistry.getType(relationship.superclassName)); 
 861             ObjectType subClass = ObjectType.cast( 
 862                 typeRegistry.getType(relationship.subclassName)); 
 863             if (superClass != null && subClass != null) { 
 864               FunctionType superCtor = superClass.getConstructor(); 
 865               FunctionType subCtor = subClass.getConstructor(); 
 866  
 867               if (relationship.type == SubclassType.INHERITS) { 
 868                 validator.expectSuperType(t, n, superClass, subClass); 
 869               } 
 870  
 871               if (superCtor != null && subCtor != null) { 
 872                 codingConvention.applySubclassRelationship( 
 873                     superCtor, subCtor, relationship.type); 
 874               } 
 875             } 
 876           } 
 877  
 878           String singletonGetterClassName = 
 879               codingConvention.getSingletonGetterClassName(n); 
 880           if (singletonGetterClassName != null) { 
 881             ObjectType objectType = ObjectType.cast( 
 882                 typeRegistry.getType(singletonGetterClassName)); 
 883             if (objectType != null) { 
 884               FunctionType functionType = objectType.getConstructor(); 
 885  
 886               if (functionType != null) { 
 887                 FunctionType getterType = 
 888                     typeRegistry.createFunctionType(objectType); 
 889                 codingConvention.applySingletonGetter(functionType, getterType, 
 890                                                       objectType); 
 891               } 
 892             } 
 893           } 
 894  
 895           DelegateRelationship delegateRelationship = 
 896               codingConvention.getDelegateRelationship(n); 
 897           if (delegateRelationship != null) { 
 898             applyDelegateRelationship(delegateRelationship); 
 899           } 
 900  
 901           ObjectLiteralCast objectLiteralCast = 
 902               codingConvention.getObjectLiteralCast(t, n); 
 903           if (objectLiteralCast != null) { 
 904             ObjectType type = ObjectType.cast( 
 905                 typeRegistry.getType(objectLiteralCast.typeName)); 
 906             if (type != null && type.getConstructor() != null) { 
 907               setDeferredType(objectLiteralCast.objectNode, type); 
 908             } else { 
 909               compiler.report(JSError.make(t.getSourceName(), n, 
 910                                            CONSTRUCTOR_EXPECTED)); 
 911             } 
 912           } 
 913           break; 
 914  
 915         case Token.FUNCTION: 
 916           if (!t.getInput().isExtern()) { 
 917             nonExternFunctions.add(n); 
 918           } 
 919  
 920           // VARs and ASSIGNs are handled separately. 
 921           if (parent.getType() == Token.ASSIGN || 
 922               parent.getType() == Token.NAME) { 
 923             return; 
 924           } 
 925           define(n, parent); 
 926           break; 
 927  
 928         case Token.ASSIGN: 
 929           // Handle constructor and enum definitions. 
 930           define(n, parent); 
 931  
 932           // Handle typedefs. 
 933           checkForOldStyleTypedef(t, n); 
 934  
 935           // Handle initialization of properties. 
 936           Node firstChild = n.getFirstChild(); 
 937           if (firstChild.getType() == Token.GETPROP && 
 938               firstChild.isQualifiedName()) { 
 939             maybeDeclareQualifiedName(t, n.getJSDocInfo(), 
 940                 firstChild, n, firstChild.getNext()); 
 941           } 
 942           break; 
 943  
 944         case Token.CATCH: 
 945           define(n, parent); 
 946           break; 
 947  
 948         case Token.VAR: 
 949           define(n, parent); 
 950  
 951           // Handle typedefs. 
 952           if (n.hasOneChild()) { 
 953             checkForOldStyleTypedef(t, n); 
 954             checkForTypedef(t, n.getFirstChild(), n.getJSDocInfo()); 
 955           } 
 956           break; 
 957  
 958         case Token.GETPROP: 
 959           // Handle stubbed properties. 
 960           if (parent.getType() == Token.EXPR_RESULT && 
 961               n.isQualifiedName()) { 
 962             checkForTypedef(t, n, n.getJSDocInfo()); 
 963             maybeDeclareQualifiedName(t, n.getJSDocInfo(), n, parent, null); 
 964           } 
 965           break; 
 966       } 
 967     }

 969     private void applyDelegateRelationship( 
 970         DelegateRelationship delegateRelationship) { 
 971       ObjectType delegatorObject = ObjectType.cast( 
 972           typeRegistry.getType(delegateRelationship.delegator)); 
 973       ObjectType delegateBaseObject = ObjectType.cast( 
 974           typeRegistry.getType(delegateRelationship.delegateBase)); 
 975       ObjectType delegateSuperObject = ObjectType.cast( 
 976           typeRegistry.getType(codingConvention.getDelegateSuperclassName())); 
 977       if (delegatorObject != null && 
 978           delegateBaseObject != null && 
 979           delegateSuperObject != null) { 
 980         FunctionType delegatorCtor = delegatorObject.getConstructor(); 
 981         FunctionType delegateBaseCtor = delegateBaseObject.getConstructor(); 
 982         FunctionType delegateSuperCtor = delegateSuperObject.getConstructor(); 
 983  
 984         if (delegatorCtor != null && delegateBaseCtor != null && 
 985             delegateSuperCtor != null) { 
 986           FunctionParamBuilder functionParamBuilder = 
 987               new FunctionParamBuilder(typeRegistry); 
 988           functionParamBuilder.addRequiredParams( 
 989               typeRegistry.getNativeType(U2U_CONSTRUCTOR_TYPE)); 
 990           FunctionType findDelegate = typeRegistry.createFunctionType( 
 991               typeRegistry.createDefaultObjectUnion(delegateBaseObject), 
 992               functionParamBuilder.build()); 
 993  
 994           // The suffix is necessary for the proxy to be distinguished from 
 995           // the base. 
 996           FunctionType delegateProxy = typeRegistry.createConstructorType( 
 997               delegateBaseObject.getReferenceName() + "(Proxy)", 
 998               null, null, null); 
 999           delegateProxy.setPrototypeBasedOn(delegateBaseObject); 
1000  
1001           codingConvention.applyDelegateRelationship( 
1002               delegateSuperObject, delegateBaseObject, delegatorObject, 
1003               delegateProxy, findDelegate); 
1004           delegateProxyPrototypes.add(delegateProxy.getPrototype()); 
1005         } 
1006       }

1079     /** 
1080      * Declare the symbol for a qualified name in the global scope. 
1081      * 
1082      * @param info The doc info for this property. 
1083      * @param n A top-level GETPROP node (it should not be contained inside 
1084      *     another GETPROP). 
1085      * @param parent The parent of {@code n}. 
1086      * @param rhsValue The node that {@code n} is being initialized to, 
1087      *     or {@code null} if this is a stub declaration. 
1088      */ 
1089     private void maybeDeclareQualifiedName(NodeTraversal t, JSDocInfo info, 
1090         Node n, Node parent, Node rhsValue) { 
1091       Node ownerNode = n.getFirstChild(); 
1092       String ownerName = ownerNode.getQualifiedName(); 
1093       String qName = n.getQualifiedName(); 
1094       String propName = n.getLastChild().getString(); 
1095       Preconditions.checkArgument(qName != null && ownerName != null); 
1096  
1097       // Function prototypes are special. 
1098       // It's a common JS idiom to do: 
1099       // F.prototype = { ... }; 
1100       // So if F does not have an explicitly declared super type, 
1101       // allow F.prototype to be redefined arbitrarily. 
1102       if ("prototype".equals(propName)) { 
1103         Var qVar = scope.getVar(qName); 
1104         if (qVar != null) { 
1105           if (!qVar.isTypeInferred()) { 
1106             // Just ignore assigns to declared prototypes. 
1107             return; 
1108           } 
1109           scope.undeclare(qVar); 
1110         } 
1111       } 
1112  
1113       // Precedence of type information on GETPROPs: 
1114       // 1) @type annotation / @enum annotation 
1115       // 2) ASSIGN to FUNCTION literal 
1116       // 3) @param/@return annotation (with no function literal) 
1117       // 4) ASSIGN to anything else 
1118       // 
1119       // 1 and 3 are declarations, 4 is inferred, and 2 is a declaration iff 
1120       // the function has not been declared before. 
1121       // 
1122       // FUNCTION literals are special because TypedScopeCreator is very smart 
1123       // about getting as much type information as possible for them. 
1124  
1125       // Determining type for #1 + #2 + #3 
1126       JSType valueType = getDeclaredGetPropType(t, info, n, rhsValue); 
1127       if (valueType == null && rhsValue != null) { 
1128         // Determining type for #4 
1129         valueType = rhsValue.getJSType(); 
1130       } 
1131  
1132       if (valueType == null) { 
1133         if (parent.getType() == Token.EXPR_RESULT) { 
1134           stubDeclarations.add(new StubDeclaration( 
1135               n, t.getInput().isExtern(), ownerName)); 
1136         } else if (rhsValue != null && 
1137                    rhsValue.getType() == Token.TRUE) { 
1138           // We declare these for delegate proxy method properties. 
1139           ObjectType ownerType = getObjectSlot(ownerName); 
1140           if (ownerType instanceof FunctionType) { 
1141             JSType ownerTypeOfThis = ((FunctionType) ownerType).getTypeOfThis(); 
1142             String delegateName = codingConvention.getDelegateSuperclassName(); 
1143             JSType delegateType = delegateName == null ? 
1144                 null : typeRegistry.getType(delegateName); 
1145             if (delegateType != null && 
1146                 ownerTypeOfThis.isSubtype(delegateType)) { 
1147               defineSlot(n, parent, typeRegistry.getNativeType(BOOLEAN_TYPE), 
1148                          true); 
1149             } 
1150           } 
1151         } 
1152  
1153         return; 
1154       } 
1155  
1156       boolean inferred = true; 
1157       if (info != null) { 
1158         // Determining declaration for #1 + #3 
1159         inferred = !(info.hasType() || info.hasEnumParameterType() || 
1160             FunctionTypeBuilder.isFunctionTypeDeclaration(info)); 
1161       } 
1162  
1163       if (inferred) { 
1164         // Determining declaration for #2 
1165         inferred = !(rhsValue != null && 
1166             rhsValue.getType() == Token.FUNCTION && 
1167             !scope.isDeclared(qName, false)); 
1168       } 
1169  
1170       if (!inferred) { 
1171         ObjectType ownerType = getObjectSlot(ownerName); 
1172         if (ownerType != null) { 
1173           // Only declare this as an official property if it has not been 
1174           // declared yet. 
1175           boolean isExtern = t.getInput().isExtern(); 
1176           if ((!ownerType.hasOwnProperty(propName) || 
1177                ownerType.isPropertyTypeInferred(propName)) && 
1178               ((isExtern && !ownerType.isNativeObjectType()) || 
1179                !ownerType.isInstanceType())) { 
1180             // If the property is undeclared or inferred, declare it now. 
1181             ownerType.defineDeclaredProperty(propName, valueType, isExtern); 
1182           } 
1183         } 
1184  
1185         // If the property is already declared, the error will be 
1186         // caught when we try to declare it in the current scope. 
1187         defineSlot(n, parent, valueType, inferred); 
1188       } 
1189     }

1229     /** 
1230      * Resolve any stub delcarations to unknown types if we could not 
1231      * find types for them during traversal. 
1232      */ 
1233     private void resolveStubDeclarations() { 
1234       for (StubDeclaration stub : stubDeclarations) { 
1235         Node n = stub.node; 
1236         Node parent = n.getParent(); 
1237         String qName = n.getQualifiedName(); 
1238         String propName = n.getLastChild().getString(); 
1239         String ownerName = stub.ownerName; 
1240         boolean isExtern = stub.isExtern; 
1241  
1242         if (scope.isDeclared(qName, false)) { 
1243           continue; 
1244         } 
1245  
1246         // If we see a stub property, make sure to register this property 
1247         // in the type registry. 
1248         ObjectType ownerType = getObjectSlot(ownerName); 
1249         ObjectType unknownType = typeRegistry.getNativeObjectType(UNKNOWN_TYPE); 
1250         defineSlot(n, parent, unknownType, true); 
1251  
1252         if (ownerType != null && 
1253             (isExtern || ownerType.isFunctionPrototypeType())) { 
1254           // If this is a stub for a prototype, just declare it

1260               propName, ownerType == null ? unknownType : ownerType);

1285         extends AbstractShallowCallback {

1294           switch (n.getType()) { 
1295             case Token.ASSIGN:

1305       private void maybeCollectMember(NodeTraversal t, 
1306           Node member, Node nodeWithJsDocInfo) { 
1307         JSDocInfo info = nodeWithJsDocInfo.getJSDocInfo(); 
1308  
1309         // Do nothing if there is no JSDoc type info, or 
1310         // if the node is not a member expression, or 
1311         // if the member expression is not of the form: this.someProperty. 
1312         if (info == null || 
1313             member.getType() != Token.GETPROP || 
1314             member.getFirstChild().getType() != Token.THIS) { 
1315           return; 
1316         } 
1317  
1318         member.getFirstChild().setJSType(thisType); 
1319         JSType jsType = getDeclaredTypeInAnnotation(t, member, info); 
1320         Node name = member.getLastChild(); 
1321         if (jsType != null && 
1322             (name.getType() == Token.NAME || name.getType() == Token.STRING)) { 
1323           thisType.defineDeclaredProperty( 
1324               name.getString(), 
1325               jsType, 
1326               false /* functions with implementations are not in externs */); 
1327         }

1333    * A shallow traversal of a local scope to find all arguments and 
1334    * local variables. 
1335    */ 
1336   private final class LocalScopeBuilder extends AbstractScopeBuilder { 
1337     /** 
1338      * @param scope The scope that we're builidng. 
1339      */ 
1340     private LocalScopeBuilder(Scope scope) { 
1341       super(scope); 
1342     } 
1343  
1344     /** 
1345      * Traverse the scope root and build it. 
1346      */ 
1347     void build() { 
1348       NodeTraversal.traverse(compiler, scope.getRootNode(), this); 
1349     } 
1350  
1351     /** 
1352      * Visit a node in a local scope, and add any local variables or catch 
1353      * parameters into the local symbol table. 
1354      * 
1355      * @param t The node traversal. 
1356      * @param n The node being visited. 
1357      * @param parent The parent of n 
1358      */ 
1359     @Override public void visit(NodeTraversal t, Node n, Node parent) { 
1360       if (n == scope.getRootNode()) return; 
1361  
1362       if (n.getType() == Token.LP && parent == scope.getRootNode()) { 
1363         handleFunctionInputs(parent); 
1364         return; 
1365       } 
1366  
1367       switch (n.getType()) { 
1368         case Token.FUNCTION: 
1369           // VARs and ASSIGNs are handled separately. 
1370           if (parent.getType() == Token.NAME) { 
1371             return; 
1372           } 
1373           define(n, parent); 
1374           break; 
1375  
1376         case Token.CATCH: 
1377         case Token.VAR: 
1378           define(n, parent); 
1379           break; 
1380       } 
1381     }

1383     /** Handle bleeding functions and function parameters. */ 
1384     private void handleFunctionInputs(Node fnNode) { 
1385       // Handle bleeding functions. 
1386       Node fnNameNode = fnNode.getFirstChild(); 
1387       String fnName = fnNameNode.getString(); 
1388       if (!fnName.isEmpty()) { 
1389         Scope.Var fnVar = scope.getVar(fnName); 
1390         if (fnVar == null || 
1391             // Make sure we're not touching a native function. Native 
1392             // functions aren't bleeding, but may not have a declaration 
1393             // node. 
1394             (fnVar.getNameNode() != null && 
1395                 // Make sure that the function is actually bleeding by checking 
1396                 // if has already been declared. 
1397                 fnVar.getInitialValue() != fnNode)) { 
1398           defineSlot(fnNameNode, fnNode, fnNode.getJSType(), false); 
1399         } 
1400       } 
1401  
1402       declareArguments(fnNode); 
1403     } 
1404  
1405     /** 
1406      * Declares all of a function's arguments. 
1407      */ 
1408     private void declareArguments(Node functionNode) { 
1409       Node astParameters = functionNode.getFirstChild().getNext(); 
1410       Node body = astParameters.getNext(); 
1411       FunctionType functionType = (FunctionType) functionNode.getJSType(); 
1412       if (functionType != null) { 
1413         Node jsDocParameters = functionType.getParametersNode(); 
1414         if (jsDocParameters != null) { 
1415           Node jsDocParameter = jsDocParameters.getFirstChild(); 
1416           for (Node astParameter : astParameters.children()) { 
1417             if (jsDocParameter != null) { 
1418               defineSlot(astParameter, functionNode, 
1419                   jsDocParameter.getJSType(), true); 
1420               jsDocParameter = jsDocParameter.getNext(); 
1421             } else { 
1422               defineSlot(astParameter, functionNode, null, true); 
1423             } 
1424           } 
1425         } 
1426       } 
1427     } // end declareArguments 
1428   } // end LocalScopeBuilder

 144   /** 
 145    * Creates a scope with all types declared. Declares newly discovered types 
 146    * and type properties in the type registry. 
 147    */ 
 148   public Scope createScope(Node root, Scope parent) { 
 149     // Constructing the global scope is very different than constructing 
 150     // inner scopes, because only global scopes can contain named classes that 
 151     // show up in the type registry. 
 152     Scope newScope = null; 
 153     if (parent == null) { 
 154       // Find all the classes in the global scope. 
 155       newScope = createInitialScope(root); 
 156  
 157       GlobalScopeBuilder scopeBuilder = new GlobalScopeBuilder(newScope); 
 158       NodeTraversal.traverse(compiler, root, scopeBuilder); 
 159       scopeBuilder.resolveStubDeclarations(); 
 160       scopeBuilder.resolveTypes(); 
 161  
 162       // Gather the properties in each function that we found in the 
 163       // global scope, if that function has a @this type that we can 
 164       // build properties on. 
 165       for (Node functionNode : scopeBuilder.nonExternFunctions) { 
 166         JSType type = functionNode.getJSType(); 
 167         if (type != null && type instanceof FunctionType) { 
 168           FunctionType fnType = (FunctionType) type; 
 169           ObjectType fnThisType = fnType.getTypeOfThis(); 
 170           if (!fnThisType.isUnknownType()) { 
 171             NodeTraversal.traverse(compiler, functionNode.getLastChild(), 
 172                 scopeBuilder.new CollectProperties(fnThisType)); 
 173           } 
 174         } 
 175       } 
 176  
 177       codingConvention.defineDelegateProxyPrototypeProperties( 
 178           typeRegistry, newScope, delegateProxyPrototypes); 
 179     } else { 
 180       newScope = new Scope(parent, root); 
 181       LocalScopeBuilder scopeBuilder = new LocalScopeBuilder(newScope); 
 182       scopeBuilder.build(); 
 183       scopeBuilder.resolveTypes(); 
 184     } 
 185     return newScope; 
 186   }

 313       this.scope = scope;

 316     void setDeferredType(Node node, JSType type) { 
 317       deferredSetTypes.add(new DeferredSetType(node, type));

 320     void resolveTypes() { 
 321       // Resolve types and attach them to nodes. 
 322       for (DeferredSetType deferred : deferredSetTypes) { 
 323         deferred.resolve(scope); 
 324       } 
 325  
 326       // Resolve types and attach them to scope slots. 
 327       Iterator<Var> vars = scope.getVars(); 
 328       while (vars.hasNext()) { 
 329         vars.next().resolveType(typeParsingErrorReporter); 
 330       }

 332       // Tell the type registry that any remaining types 
 333       // are unknown. 
 334       typeRegistry.resolveTypesInScope(scope); 
 335     }

 339         Node parent) { 
 340       if (n.getType() == Token.FUNCTION || 
 341           n.getType() == Token.SCRIPT) { 
 342         sourceName = (String) n.getProp(Node.SOURCENAME_PROP); 
 343       }

 345       // We do want to traverse the name of a named function, but we don't 
 346       // want to traverse the arguments or body. 
 347       return parent == null || parent.getType() != Token.FUNCTION || 
 348           n == parent.getFirstChild() || parent == scope.getRootNode();

 354     /** 
 355      * Returns the type specified in a JSDoc annotation near a GETPROP or NAME. 
 356      * 
 357      * Extracts type information from either the {@code @type} tag or from 
 358      * the {@code @return} and {@code @param} tags. 
 359      */ 
 360     JSType getDeclaredTypeInAnnotation( 
 361         NodeTraversal t, Node node, JSDocInfo info) { 
 362       return getDeclaredTypeInAnnotation(t.getSourceName(), node, info); 
 363     } 
 364  
 365     JSType getDeclaredTypeInAnnotation(String sourceName, 
 366         Node node, JSDocInfo info) { 
 367       JSType jsType = null; 
 368       Node objNode = node.getType() == Token.GETPROP ? 
 369           node.getFirstChild() : null; 
 370       if (info != null) { 
 371         if (info.hasType()) { 
 372           jsType = info.getType().evaluate(scope, typeRegistry); 
 373         } else if (FunctionTypeBuilder.isFunctionTypeDeclaration(info)) { 
 374           String fnName = node.getQualifiedName(); 
 375  
 376           // constructors are often handled separately. 
 377           if (info.isConstructor() && typeRegistry.getType(fnName) != null) { 
 378             return null; 
 379           } 
 380  
 381           FunctionTypeBuilder builder = 
 382               new FunctionTypeBuilder( 
 383                   fnName, compiler, node, sourceName, scope) 
 384               .inferTemplateTypeName(info) 
 385               .inferReturnType(info) 
 386               .inferParameterTypes(info) 
 387               .inferInheritance(info); 
 388  
 389           // Infer the context type. 
 390           boolean searchedForThisType = false; 
 391           if (objNode != null) { 
 392             if (objNode.getType() == Token.GETPROP && 
 393                 objNode.getLastChild().getString().equals("prototype")) { 
 394               builder.inferThisType(info, objNode.getFirstChild()); 
 395               searchedForThisType = true; 
 396             } else if (objNode.getType() == Token.THIS) { 
 397               builder.inferThisType(info, objNode.getJSType()); 
 398               searchedForThisType = true; 
 399             } 
 400           } 
 401  
 402           if (!searchedForThisType) { 
 403             builder.inferThisType(info, (Node) null); 
 404           } 
 405  
 406           jsType = builder.buildAndRegister(); 
 407         } 
 408       } 
 409       return jsType; 
 410     }

 412     /** 
 413      * Defines variable(s) or modifies types based on the content of the node 
 414      * {@code n}. A variable definition creates variables in the current scope, 
 415      * a function definition creates a binding, and an assignment updates the 
 416      * type of the namespaces on which the definition is made (e.g. 
 417      * {@code goog.FOO = 6}). 
 418      * @param n a {@link Token#VAR}, {@link Token#FUNCTION} or 
 419      *     {@link Token#ASSIGN} node 
 420      * @param parent {@code n}'s parent 
 421      */ 
 422     void define(Node n, Node parent) { 
 423       Preconditions.checkState(sourceName != null); 
 424  
 425       JSDocInfo info = n.getJSDocInfo(); 
 426       switch (n.getType()) { 
 427         case Token.CATCH: 
 428           Node catchName = n.getFirstChild(); 
 429           defineSlot(catchName, n, null); 
 430           break; 
 431  
 432         case Token.VAR: 
 433           if (n.getChildCount() > 1) { 
 434             if (info != null) { 
 435               // multiple children 
 436               compiler.report(JSError.make(sourceName, n, MULTIPLE_VAR_DEF)); 
 437             } 
 438             for (Node name : n.children()) { 
 439               defineName(name, n, parent, name.getJSDocInfo()); 
 440             } 
 441           } else { 
 442             Node name = n.getFirstChild(); 
 443             defineName(name, n, parent, 
 444                 (info != null) ? info : name.getJSDocInfo()); 
 445           } 
 446           break; 
 447  
 448         case Token.FUNCTION: 
 449           int parentType = parent.getType(); 
 450           Preconditions.checkState( 
 451               (scope.isLocal() || parentType != Token.ASSIGN) && 
 452               parentType != Token.NAME, 
 453               "function defined as standalone function when it is being " + 
 454               "assigned"); 
 455           String functionName = n.getFirstChild().getString(); 
 456           FunctionType functionType = getFunctionType(functionName, n, info, 
 457               null); 
 458           if (NodeUtil.isFunctionDeclaration(n)) { 
 459             defineSlot(n.getFirstChild(), n, functionType); 
 460           } 
 461           break; 
 462  
 463         case Token.ASSIGN: 
 464           // TODO(nicksantos): We should support direct assignment to a 
 465           // prototype, as in: 
 466           // Foo.prototype = { 
 467           //   a: function() { ... }, 
 468           //   b: function() { ... } 
 469           // }; 
 470           // Right now (6/23/08), we understand most of this syntax, but we 
 471           // don't tie the "a" and "b" methods to the context of Foo. 
 472  
 473           Node rvalue = n.getLastChild(); 
 474           Node lvalue = n.getFirstChild(); 
 475           info = (info != null) ? info : rvalue.getJSDocInfo(); 
 476           if (rvalue.getType() == Token.FUNCTION || 
 477               info != null && info.isConstructor()) { 
 478             getFunctionType(lvalue.getQualifiedName(), rvalue, info, 
 479                 lvalue); 
 480           } else if (info != null && info.hasEnumParameterType()) { 
 481             JSType type = getEnumType(lvalue.getQualifiedName(), n, rvalue, 
 482                 info.getEnumParameterType().evaluate(scope, typeRegistry)); 
 483             if (type != null) { 
 484               setDeferredType(lvalue, type); 
 485             } 
 486           } 
 487           break; 
 488  
 489         default: 
 490           throw new IllegalStateException(Integer.toString(n.getType())); 
 491       } 
 492     }

 494     /** 
 495      * Defines a variable based on the {@link Token#NAME} node passed. 
 496      * @param name The {@link Token#NAME} node. 
 497      * @param var The parent of the {@code name} node, which must be a 
 498      *     {@link Token#VAR} node. 
 499      * @param parent {@code var}'s parent. 
 500      * @param info the {@link JSDocInfo} information relating to this 
 501      *     {@code name} node. 
 502      */ 
 503     private void defineName(Node name, Node var, Node parent, JSDocInfo info) { 
 504       Node value = name.getFirstChild(); 
 505  
 506       if (value != null && value.getType() == Token.FUNCTION) { 
 507         // function 
 508         String functionName = name.getString(); 
 509         FunctionType functionType = 
 510             getFunctionType(functionName, value, info, null); 
 511         if (functionType.isReturnTypeInferred() && 
 512             scope.isLocal()) { 
 513           defineSlot(name, var, null); 
 514         } else { 
 515           defineSlot(name, var, functionType); 
 516         } 
 517       } else { 
 518         // variable's type 
 519         JSType type = null; 
 520         if (info == null) { 
 521           // the variable's type will be inferred 
 522           CompilerInput input = compiler.getInput(sourceName); 
 523           Preconditions.checkNotNull(input, sourceName); 
 524           type = input.isExtern() ? 
 525               typeRegistry.getNativeType(UNKNOWN_TYPE) : null; 
 526         } else if (info.hasEnumParameterType()) { 
 527           type = getEnumType(name.getString(), var, value, 
 528               info.getEnumParameterType().evaluate(scope, typeRegistry)); 
 529         } else if (info.isConstructor()) { 
 530           type = getFunctionType(name.getString(), value, info, name); 
 531         } else { 
 532           type = getDeclaredTypeInAnnotation(sourceName, name, info); 
 533         } 
 534  
 535         defineSlot(name, var, type); 
 536       } 
 537     }

 539     /** 
 540      * Gets the function type from the function node and its attached 
 541      * {@link JSDocInfo}. 
 542      * @param name the function's name 
 543      * @param rValue the function node. It must be a {@link Token#FUNCTION}. 
 544      * @param info the {@link JSDocInfo} attached to the function definition 
 545      * @param lvalueNode The node where this function is being 
 546      *     assigned. For example, {@code A.prototype.foo = ...} would be used to 
 547      *     determine that this function is a method of A.prototype. May be 
 548      *     null to indicate that this is not being assigned to a qualified name. 
 549      */ 
 550     private FunctionType getFunctionType(String name, 
 551         Node rValue, JSDocInfo info, @Nullable Node lvalueNode) { 
 552       FunctionType functionType = null; 
 553  
 554       // Handle function aliases. 
 555       if (rValue != null && rValue.isQualifiedName()) { 
 556         Var var = scope.getVar(rValue.getQualifiedName()); 
 557         if (var != null && var.getType() instanceof FunctionType) { 
 558           functionType = (FunctionType) var.getType(); 
 559           if (functionType != null && functionType.isConstructor()) { 
 560             typeRegistry.declareType(name, functionType.getInstanceType()); 
 561           } 
 562         } 
 563         return functionType; 
 564       } 
 565  
 566       Node owner = null; 
 567       if (lvalueNode != null) { 
 568         owner = getPrototypePropertyOwner(lvalueNode); 
 569       } 
 570  
 571       Node errorRoot = rValue == null ? lvalueNode : rValue; 
 572       boolean isFnLiteral = 
 573           rValue != null && rValue.getType() == Token.FUNCTION; 
 574       Node fnRoot = isFnLiteral ? rValue : null; 
 575       Node parametersNode = isFnLiteral ? 
 576           rValue.getFirstChild().getNext() : null; 
 577       Node fnBlock = isFnLiteral ? parametersNode.getNext() : null; 
 578  
 579       if (functionType == null && info != null && info.hasType()) { 
 580         JSType type = info.getType().evaluate(scope, typeRegistry); 
 581  
 582         // Known to be not null since we have the FUNCTION token there. 
 583         type = type.restrictByNotNullOrUndefined(); 
 584         if (type.isFunctionType()) { 
 585           functionType = (FunctionType) type; 
 586           functionType.setJSDocInfo(info); 
 587         } 
 588       } 
 589  
 590       if (functionType == null) { 
 591         if (info == null || 
 592             !FunctionTypeBuilder.isFunctionTypeDeclaration(info)) { 
 593           // We don't really have any type information in the annotation. 
 594           // Before we give up on this function, look at the object we're 
 595           // assigning it to. For example, if the function looks like this: 
 596           // SubFoo.prototype.bar = function() { ... }; 
 597           // We can use type information on Foo.prototype.bar and apply it 
 598           // to this function. 
 599           if (lvalueNode != null && lvalueNode.getType() == Token.GETPROP && 
 600               lvalueNode.isQualifiedName()) { 
 601             Var var = scope.getVar( 
 602                 lvalueNode.getFirstChild().getQualifiedName()); 
 603             if (var != null) { 
 604               ObjectType ownerType = ObjectType.cast(var.getType()); 
 605               FunctionType propType = null; 
 606               if (ownerType != null) { 
 607                 String propName = lvalueNode.getLastChild().getString(); 
 608                 propType = findOverriddenFunction(ownerType, propName); 
 609               } 
 610  
 611               if (propType != null) { 
 612                 functionType = 
 613                     new FunctionTypeBuilder( 
 614                         name, compiler, errorRoot, sourceName, scope) 
 615                     .setSourceNode(fnRoot) 
 616                     .inferFromOverriddenFunction(propType, parametersNode) 
 617                     .inferThisType(info, owner) 
 618                     .inferReturnStatements(fnBlock) 
 619                     .buildAndRegister(); 
 620               } 
 621             } 
 622           } 
 623         } 
 624       } // end if (functionType == null) 
 625  
 626       if (functionType == null) { 
 627         functionType = 
 628             new FunctionTypeBuilder(name, compiler, errorRoot, sourceName, 
 629                 scope) 
 630             .setSourceNode(fnRoot) 
 631             .inferTemplateTypeName(info) 
 632             .inferReturnType(info) 
 633             .inferInheritance(info) 
 634             .inferThisType(info, owner) 
 635             .inferParameterTypes(parametersNode, info) 
 636             .inferReturnStatements(fnBlock) 
 637             .buildAndRegister(); 
 638       } 
 639  
 640       // assigning the function type to the function node 
 641       if (rValue != null) { 
 642         setDeferredType(rValue, functionType); 
 643       } 
 644  
 645       // all done 
 646       return functionType; 
 647     }

 649     /** 
 650      * Find the function that's being overridden on this type, if any. 
 651      */ 
 652     private FunctionType findOverriddenFunction( 
 653         ObjectType ownerType, String propName) { 
 654       // First, check to see if the property is implemented 
 655       // on a superclass. 
 656       JSType propType = ownerType.getPropertyType(propName); 
 657       if (propType instanceof FunctionType) { 
 658         return (FunctionType) propType; 
 659       } else { 
 660         // If it's not, then check to see if it's implemented 
 661         // on an implemented interface. 
 662         for (ObjectType iface : 
 663                  ownerType.getCtorImplementedInterfaces()) { 
 664           propType = iface.getPropertyType(propName); 
 665           if (propType instanceof FunctionType) { 
 666             return (FunctionType) propType; 
 667           } 
 668         } 
 669       } 
 670  
 671       return null; 
 672     }

 674     /** 
 675      * Gets an enum type. If the definition is correct, the object literal used 
 676      * to define the enum is traversed to gather the elements name, and this 
 677      * method checks for duplicates. This method also enforces that all 
 678      * elements' name be syntactic constants according to the 
 679      * {@link CodingConvention} used. 
 680      * 
 681      * @param name the enum's name such as {@code HELLO} or {@code goog.foo.BAR} 
 682      * @param value the enum's original value. This value may be {@code null}. 
 683      * @param parent the value's parent 
 684      * @param elementsType the type of the elements of this enum 
 685      * @return the enum type 
 686      */ 
 687     private EnumType getEnumType(String name, Node parent, 
 688         Node value, JSType elementsType) { 
 689       EnumType enumType = null; 
 690  
 691       // no value with @enum 
 692       if (value != null) { 
 693         if (value.getType() == Token.OBJECTLIT) { 
 694           // collect enum elements 
 695           enumType = typeRegistry.createEnumType(name, elementsType); 
 696  
 697           // populate the enum type. 
 698           Node key = value.getFirstChild(); 
 699           while (key != null) { 
 700             String keyName = NodeUtil.getStringValue(key); 
 701  
 702             if (enumType.hasOwnProperty(keyName)) { 
 703               compiler.report(JSError.make(sourceName, key, ENUM_DUP, keyName)); 
 704             } else if (!codingConvention.isValidEnumKey(keyName)) { 
 705               compiler.report( 
 706                   JSError.make(sourceName, key, ENUM_NOT_CONSTANT, keyName)); 
 707             } else { 
 708               enumType.defineElement(keyName); 
 709             } 
 710             key = key.getNext(); 
 711             key = (key == null) ? null : key.getNext(); 
 712           } 
 713         } else if (value.isQualifiedName()) { 
 714           Var var = scope.getVar(value.getQualifiedName()); 
 715           if (var != null && var.getType() instanceof EnumType) { 
 716             enumType = (EnumType) var.getType(); 
 717           } 
 718         } 
 719       } 
 720  
 721       if (enumType == null) { 
 722         compiler.report(JSError.make(sourceName, parent, ENUM_INITIALIZER)); 
 723       } else if (scope.isGlobal()) { 
 724         if (name != null && !name.isEmpty()) { 
 725           typeRegistry.declareType(name, enumType.getElementsType()); 
 726         } 
 727       } 
 728  
 729       return enumType; 
 730     }

 744     /** 
 745      * Defines a typed variable. The defining node will be annotated with the 
 746      * variable's type of {@link JSTypeNative#UNKNOWN_TYPE} if its type is 
 747      * inferred. 
 748      * 
 749      * Slots may be any variable or any qualified name in the global scope. 
 750      * 
 751      * @param n the defining NAME or GETPROP node. 
 752      * @param parent the {@code n}'s parent. 
 753      * @param type the variable's type. It may be {@code null} if 
 754      *     {@code inferred} is {@code true}. 
 755      */ 
 756     void defineSlot(Node n, Node parent, JSType type, boolean inferred) { 
 757       Preconditions.checkArgument(inferred || type != null); 
 758  
 759       // Only allow declarations of NAMEs and qualfied names. 
 760       boolean shouldDeclareOnGlobalThis = false; 
 761       if (n.getType() == Token.NAME) { 
 762         Preconditions.checkArgument( 
 763             parent.getType() == Token.FUNCTION || 
 764             parent.getType() == Token.VAR || 
 765             parent.getType() == Token.LP || 
 766             parent.getType() == Token.CATCH); 
 767         shouldDeclareOnGlobalThis = scope.isGlobal() && 
 768             (parent.getType() == Token.VAR || 
 769              parent.getType() == Token.FUNCTION); 
 770       } else { 
 771         Preconditions.checkArgument( 
 772             n.getType() == Token.GETPROP && 
 773             (parent.getType() == Token.ASSIGN || 
 774              parent.getType() == Token.EXPR_RESULT)); 
 775       } 
 776       String variableName = n.getQualifiedName(); 
 777       Preconditions.checkArgument(!variableName.isEmpty()); 
 778  
 779       // declared in closest scope? 
 780       if (scope.isDeclared(variableName, false)) { 
 781         Var oldVar = scope.getVar(variableName); 
 782         validator.expectUndeclaredVariable( 
 783             sourceName, n, parent, oldVar, variableName, type); 
 784       } else { 
 785         if (!inferred) { 
 786           setDeferredType(n, type); 
 787         } 
 788         CompilerInput input = compiler.getInput(sourceName); 
 789         scope.declare(variableName, n, type, input, inferred); 
 790  
 791         if (shouldDeclareOnGlobalThis) { 
 792           ObjectType globalThis = 
 793               typeRegistry.getNativeObjectType(JSTypeNative.GLOBAL_THIS); 
 794           boolean isExtern = input.isExtern(); 
 795           if (inferred) { 
 796             globalThis.defineInferredProperty(variableName, 
 797                 type == null ? 
 798                     typeRegistry.getNativeType(JSTypeNative.NO_TYPE) : 
 799                     type, 
 800                 isExtern); 
 801           } else { 
 802             globalThis.defineDeclaredProperty(variableName, type, isExtern); 
 803           } 
 804         } 
 805  
 806         // If we're in the global scope, also declare var.prototype 
 807         // in the scope chain. 
 808         if (scope.isGlobal() && type instanceof FunctionType) { 
 809           FunctionType fnType = (FunctionType) type; 
 810           if (fnType.isConstructor() || fnType.isInterface()) { 
 811             FunctionType superClassCtor = fnType.getSuperClassConstructor(); 
 812             scope.declare(variableName + ".prototype", n, 
 813                 fnType.getPrototype(), compiler.getInput(sourceName), 
 814                 /* declared iff there's an explicit supertype */ 
 815                 superClassCtor == null || 
 816                 superClassCtor.getInstanceType().equals( 
 817                     typeRegistry.getNativeType(OBJECT_TYPE))); 
 818           } 
 819         } 
 820       } 
 821     }

 832      */ 
 833     private final List<Node> nonExternFunctions = Lists.newArrayList();

 836      * Type-less stubs in the global scope.

 838      * If at the end of traversal, we still don't have types for these 
 839      * stubs, then we should declare UNKNOWN types.

 841     private final List<StubDeclaration> stubDeclarations =

 848     /** 
 849      * Visit a node in the global scope, and add anything it declares to the 
 850      * global symbol table. 
 851      * 
 852      * @param t The current traversal. 
 853      * @param n The node being visited. 
 854      * @param parent The parent of n 
 855      */ 
 856     @Override public void visit(NodeTraversal t, Node n, Node parent) { 
 857       switch (n.getType()) { 
 858         case Token.CALL: 
 859           SubclassRelationship relationship = 
 860               codingConvention.getClassesDefinedByCall(n); 
 861           if (relationship != null) { 
 862             ObjectType superClass = ObjectType.cast( 
 863                 typeRegistry.getType(relationship.superclassName)); 
 864             ObjectType subClass = ObjectType.cast( 
 865                 typeRegistry.getType(relationship.subclassName)); 
 866             if (superClass != null && subClass != null) { 
 867               FunctionType superCtor = superClass.getConstructor(); 
 868               FunctionType subCtor = subClass.getConstructor(); 
 869  
 870               if (relationship.type == SubclassType.INHERITS) { 
 871                 validator.expectSuperType(t, n, superClass, subClass); 
 872               } 
 873  
 874               if (superCtor != null && subCtor != null) { 
 875                 codingConvention.applySubclassRelationship( 
 876                     superCtor, subCtor, relationship.type); 
 877               } 
 878             } 
 879           } 
 880  
 881           String singletonGetterClassName = 
 882               codingConvention.getSingletonGetterClassName(n); 
 883           if (singletonGetterClassName != null) { 
 884             ObjectType objectType = ObjectType.cast( 
 885                 typeRegistry.getType(singletonGetterClassName)); 
 886             if (objectType != null) { 
 887               FunctionType functionType = objectType.getConstructor(); 
 888  
 889               if (functionType != null) { 
 890                 FunctionType getterType = 
 891                     typeRegistry.createFunctionType(objectType); 
 892                 codingConvention.applySingletonGetter(functionType, getterType, 
 893                                                       objectType); 
 894               } 
 895             } 
 896           } 
 897  
 898           DelegateRelationship delegateRelationship = 
 899               codingConvention.getDelegateRelationship(n); 
 900           if (delegateRelationship != null) { 
 901             applyDelegateRelationship(delegateRelationship); 
 902           } 
 903  
 904           ObjectLiteralCast objectLiteralCast = 
 905               codingConvention.getObjectLiteralCast(t, n); 
 906           if (objectLiteralCast != null) { 
 907             ObjectType type = ObjectType.cast( 
 908                 typeRegistry.getType(objectLiteralCast.typeName)); 
 909             if (type != null && type.getConstructor() != null) { 
 910               setDeferredType(objectLiteralCast.objectNode, type); 
 911             } else { 
 912               compiler.report(JSError.make(t.getSourceName(), n, 
 913                                            CONSTRUCTOR_EXPECTED)); 
 914             } 
 915           } 
 916           break; 
 917  
 918         case Token.FUNCTION: 
 919           if (!t.getInput().isExtern()) { 
 920             nonExternFunctions.add(n); 
 921           } 
 922  
 923           // VARs and ASSIGNs are handled separately. 
 924           if (parent.getType() == Token.ASSIGN || 
 925               parent.getType() == Token.NAME) { 
 926             return; 
 927           } 
 928           define(n, parent); 
 929           break; 
 930  
 931         case Token.ASSIGN: 
 932           // Handle constructor and enum definitions. 
 933           define(n, parent); 
 934  
 935           // Handle typedefs. 
 936           checkForOldStyleTypedef(t, n); 
 937  
 938           // Handle initialization of properties. 
 939           Node firstChild = n.getFirstChild(); 
 940           if (firstChild.getType() == Token.GETPROP && 
 941               firstChild.isQualifiedName()) { 
 942             maybeDeclareQualifiedName(t, n.getJSDocInfo(), 
 943                 firstChild, n, firstChild.getNext()); 
 944           } 
 945           break; 
 946  
 947         case Token.CATCH: 
 948           define(n, parent); 
 949           break; 
 950  
 951         case Token.VAR: 
 952           define(n, parent); 
 953  
 954           // Handle typedefs. 
 955           if (n.hasOneChild()) { 
 956             checkForOldStyleTypedef(t, n); 
 957             checkForTypedef(t, n.getFirstChild(), n.getJSDocInfo()); 
 958           } 
 959           break; 
 960  
 961         case Token.GETPROP: 
 962           // Handle stubbed properties. 
 963           if (parent.getType() == Token.EXPR_RESULT && 
 964               n.isQualifiedName()) { 
 965             checkForTypedef(t, n, n.getJSDocInfo()); 
 966             maybeDeclareQualifiedName(t, n.getJSDocInfo(), n, parent, null); 
 967           } 
 968           break; 
 969       } 
 970     }

 972     private void applyDelegateRelationship( 
 973         DelegateRelationship delegateRelationship) { 
 974       ObjectType delegatorObject = ObjectType.cast( 
 975           typeRegistry.getType(delegateRelationship.delegator)); 
 976       ObjectType delegateBaseObject = ObjectType.cast( 
 977           typeRegistry.getType(delegateRelationship.delegateBase)); 
 978       ObjectType delegateSuperObject = ObjectType.cast( 
 979           typeRegistry.getType(codingConvention.getDelegateSuperclassName())); 
 980       if (delegatorObject != null && 
 981           delegateBaseObject != null && 
 982           delegateSuperObject != null) { 
 983         FunctionType delegatorCtor = delegatorObject.getConstructor(); 
 984         FunctionType delegateBaseCtor = delegateBaseObject.getConstructor(); 
 985         FunctionType delegateSuperCtor = delegateSuperObject.getConstructor(); 
 986  
 987         if (delegatorCtor != null && delegateBaseCtor != null && 
 988             delegateSuperCtor != null) { 
 989           FunctionParamBuilder functionParamBuilder = 
 990               new FunctionParamBuilder(typeRegistry); 
 991           functionParamBuilder.addRequiredParams( 
 992               typeRegistry.getNativeType(U2U_CONSTRUCTOR_TYPE)); 
 993           FunctionType findDelegate = typeRegistry.createFunctionType( 
 994               typeRegistry.createDefaultObjectUnion(delegateBaseObject), 
 995               functionParamBuilder.build()); 
 996  
 997           // The suffix is necessary for the proxy to be distinguished from 
 998           // the base. 
 999           FunctionType delegateProxy = typeRegistry.createConstructorType( 
1000               delegateBaseObject.getReferenceName() + "(Proxy)", 
1001               null, null, null); 
1002           delegateProxy.setPrototypeBasedOn(delegateBaseObject); 
1003  
1004           codingConvention.applyDelegateRelationship( 
1005               delegateSuperObject, delegateBaseObject, delegatorObject, 
1006               delegateProxy, findDelegate); 
1007           delegateProxyPrototypes.add(delegateProxy.getPrototype()); 
1008         } 
1009       }

1082     /** 
1083      * Declare the symbol for a qualified name in the global scope. 
1084      * 
1085      * @param info The doc info for this property. 
1086      * @param n A top-level GETPROP node (it should not be contained inside 
1087      *     another GETPROP). 
1088      * @param parent The parent of {@code n}. 
1089      * @param rhsValue The node that {@code n} is being initialized to, 
1090      *     or {@code null} if this is a stub declaration. 
1091      */ 
1092     private void maybeDeclareQualifiedName(NodeTraversal t, JSDocInfo info, 
1093         Node n, Node parent, Node rhsValue) { 
1094       Node ownerNode = n.getFirstChild(); 
1095       String ownerName = ownerNode.getQualifiedName(); 
1096       String qName = n.getQualifiedName(); 
1097       String propName = n.getLastChild().getString(); 
1098       Preconditions.checkArgument(qName != null && ownerName != null); 
1099  
1100       // Function prototypes are special. 
1101       // It's a common JS idiom to do: 
1102       // F.prototype = { ... }; 
1103       // So if F does not have an explicitly declared super type, 
1104       // allow F.prototype to be redefined arbitrarily. 
1105       if ("prototype".equals(propName)) { 
1106         Var qVar = scope.getVar(qName); 
1107         if (qVar != null) { 
1108           if (!qVar.isTypeInferred()) { 
1109             // Just ignore assigns to declared prototypes. 
1110             return; 
1111           } 
1112           scope.undeclare(qVar); 
1113         } 
1114       } 
1115  
1116       // Precedence of type information on GETPROPs: 
1117       // 1) @type annotation / @enum annotation 
1118       // 2) ASSIGN to FUNCTION literal 
1119       // 3) @param/@return annotation (with no function literal) 
1120       // 4) ASSIGN to anything else 
1121       // 
1122       // 1 and 3 are declarations, 4 is inferred, and 2 is a declaration iff 
1123       // the function has not been declared before. 
1124       // 
1125       // FUNCTION literals are special because TypedScopeCreator is very smart 
1126       // about getting as much type information as possible for them. 
1127  
1128       // Determining type for #1 + #2 + #3 
1129       JSType valueType = getDeclaredGetPropType(t, info, n, rhsValue); 
1130       if (valueType == null && rhsValue != null) { 
1131         // Determining type for #4 
1132         valueType = rhsValue.getJSType(); 
1133       } 
1134  
1135       if (valueType == null) { 
1136         if (parent.getType() == Token.EXPR_RESULT) { 
1137           stubDeclarations.add(new StubDeclaration( 
1138               n, t.getInput().isExtern(), ownerName)); 
1139         } else if (rhsValue != null && 
1140                    rhsValue.getType() == Token.TRUE) { 
1141           // We declare these for delegate proxy method properties. 
1142           ObjectType ownerType = getObjectSlot(ownerName); 
1143           if (ownerType instanceof FunctionType) { 
1144             JSType ownerTypeOfThis = ((FunctionType) ownerType).getTypeOfThis(); 
1145             String delegateName = codingConvention.getDelegateSuperclassName(); 
1146             JSType delegateType = delegateName == null ? 
1147                 null : typeRegistry.getType(delegateName); 
1148             if (delegateType != null && 
1149                 ownerTypeOfThis.isSubtype(delegateType)) { 
1150               defineSlot(n, parent, typeRegistry.getNativeType(BOOLEAN_TYPE), 
1151                          true); 
1152             } 
1153           } 
1154         } 
1155  
1156         return; 
1157       } 
1158  
1159       boolean inferred = true; 
1160       if (info != null) { 
1161         // Determining declaration for #1 + #3 
1162         inferred = !(info.hasType() || info.hasEnumParameterType() || 
1163             FunctionTypeBuilder.isFunctionTypeDeclaration(info)); 
1164       } 
1165  
1166       if (inferred) { 
1167         // Determining declaration for #2 
1168         inferred = !(rhsValue != null && 
1169             rhsValue.getType() == Token.FUNCTION && 
1170             !scope.isDeclared(qName, false)); 
1171       } 
1172  
1173       if (!inferred) { 
1174         ObjectType ownerType = getObjectSlot(ownerName); 
1175         if (ownerType != null) { 
1176           // Only declare this as an official property if it has not been 
1177           // declared yet. 
1178           boolean isExtern = t.getInput().isExtern(); 
1179           if ((!ownerType.hasOwnProperty(propName) || 
1180                ownerType.isPropertyTypeInferred(propName)) && 
1181               ((isExtern && !ownerType.isNativeObjectType()) || 
1182                !ownerType.isInstanceType())) { 
1183             // If the property is undeclared or inferred, declare it now. 
1184             ownerType.defineDeclaredProperty(propName, valueType, isExtern); 
1185           } 
1186         } 
1187  
1188         // If the property is already declared, the error will be 
1189         // caught when we try to declare it in the current scope. 
1190         defineSlot(n, parent, valueType, inferred); 
1191       } 
1192     }

1194     /** 
1195      * Find the ObjectType associated with the given slot. 
1196      * @param slotName The name of the slot to find the type in. 
1197      * @return An object type, or null if this slot does not contain an object. 
1198      */ 
1199     private ObjectType getObjectSlot(String slotName) { 
1200       Var ownerVar = scope.getVar(slotName); 
1201       if (ownerVar != null) { 
1202         JSType ownerVarType = ownerVar.getType(); 
1203         return ObjectType.cast(ownerVarType == null ? 
1204             null : ownerVarType.restrictByNotNullOrUndefined()); 
1205       } 
1206       return null; 
1207     }

1209     /** 
1210      * Look for a type declaration on a GETPROP node. 
1211      * 
1212      * @param info The doc info for this property. 
1213      * @param n A top-level GETPROP node (it should not be contained inside 
1214      *     another GETPROP). 
1215      * @param rhsValue The node that {@code n} is being initialized to, 
1216      *     or {@code null} if this is a stub declaration. 
1217      */ 
1218     private JSType getDeclaredGetPropType(NodeTraversal t, JSDocInfo info, 
1219         Node n, Node rhsValue) { 
1220       if (info != null && info.hasType()) { 
1221         return getDeclaredTypeInAnnotation(t, n, info); 
1222       } else if (info != null && info.hasEnumParameterType()) { 
1223         return n.getJSType(); 
1224       } else if (rhsValue != null && 
1225           rhsValue.getType() == Token.FUNCTION) { 
1226         return rhsValue.getJSType(); 
1227       } else { 
1228         return getDeclaredTypeInAnnotation(t, n, info); 
1229       } 
1230     }

1232     /** 
1233      * Resolve any stub delcarations to unknown types if we could not 
1234      * find types for them during traversal. 
1235      */ 
1236     private void resolveStubDeclarations() { 
1237       for (StubDeclaration stub : stubDeclarations) { 
1238         Node n = stub.node; 
1239         Node parent = n.getParent(); 
1240         String qName = n.getQualifiedName(); 
1241         String propName = n.getLastChild().getString(); 
1242         String ownerName = stub.ownerName; 
1243         boolean isExtern = stub.isExtern; 
1244  
1245         if (scope.isDeclared(qName, false)) { 
1246           continue; 
1247         } 
1248  
1249         // If we see a stub property, make sure to register this property 
1250         // in the type registry. 
1251         ObjectType ownerType = getObjectSlot(ownerName); 
1252         ObjectType unknownType = typeRegistry.getNativeObjectType(UNKNOWN_TYPE); 
1253         defineSlot(n, parent, unknownType, true); 
1254  
1255         if (ownerType != null && 
1256             (isExtern || ownerType.isFunctionPrototypeType())) { 
1257           // If this is a stub for a prototype, just declare it 
1258           // as an unknown type. These are seen often in externs. 
1259           ownerType.defineInferredProperty( 
1260               propName, unknownType, isExtern); 
1261         } else { 
1262           typeRegistry.registerPropertyOnType( 
1263               propName, ownerType == null ? unknownType : ownerType); 
1264         } 
1265       } 
1266     }

1283     /** 
1284      * Collects all declared properties in a function, and 
1285      * resolves them relative to the global scope. 
1286      */ 
1287     private final class CollectProperties 
1288         extends AbstractShallowCallback { 
1289       private final ObjectType thisType; 
1290  
1291       CollectProperties(ObjectType thisType) { 
1292         this.thisType = thisType; 
1293       } 
1294  
1295       public void visit(NodeTraversal t, Node n, Node parent) { 
1296         if (parent != null && parent.getType() == Token.EXPR_RESULT) { 
1297           switch (n.getType()) { 
1298             case Token.ASSIGN: 
1299               maybeCollectMember(t, n.getFirstChild(), n); 
1300               break; 
1301             case Token.GETPROP: 
1302               maybeCollectMember(t, n, n); 
1303               break; 
1304           } 
1305         } 
1306       } 
1307  
1308       private void maybeCollectMember(NodeTraversal t, 
1309           Node member, Node nodeWithJsDocInfo) { 
1310         JSDocInfo info = nodeWithJsDocInfo.getJSDocInfo(); 
1311  
1312         // Do nothing if there is no JSDoc type info, or 
1313         // if the node is not a member expression, or 
1314         // if the member expression is not of the form: this.someProperty. 
1315         if (info == null || 
1316             member.getType() != Token.GETPROP || 
1317             member.getFirstChild().getType() != Token.THIS) { 
1318           return; 
1319         } 
1320  
1321         member.getFirstChild().setJSType(thisType); 
1322         JSType jsType = getDeclaredTypeInAnnotation(t, member, info); 
1323         Node name = member.getLastChild(); 
1324         if (jsType != null && 
1325             (name.getType() == Token.NAME || name.getType() == Token.STRING)) { 
1326           thisType.defineDeclaredProperty( 
1327               name.getString(), 
1328               jsType, 
1329               false /* functions with implementations are not in externs */); 
1330         } 
1331       } 
1332     } // end CollectProperties

1335   /** 
1336    * A shallow traversal of a local scope to find all arguments and 
1337    * local variables. 
1338    */ 
1339   private final class LocalScopeBuilder extends AbstractScopeBuilder { 
1340     /** 
1341      * @param scope The scope that we're builidng. 
1342      */ 
1343     private LocalScopeBuilder(Scope scope) { 
1344       super(scope); 
1345     } 
1346  
1347     /** 
1348      * Traverse the scope root and build it. 
1349      */ 
1350     void build() { 
1351       NodeTraversal.traverse(compiler, scope.getRootNode(), this); 
1352     } 
1353  
1354     /** 
1355      * Visit a node in a local scope, and add any local variables or catch 
1356      * parameters into the local symbol table. 
1357      * 
1358      * @param t The node traversal. 
1359      * @param n The node being visited. 
1360      * @param parent The parent of n 
1361      */ 
1362     @Override public void visit(NodeTraversal t, Node n, Node parent) { 
1363       if (n == scope.getRootNode()) return; 
1364  
1365       if (n.getType() == Token.LP && parent == scope.getRootNode()) { 
1366         handleFunctionInputs(parent); 
1367         return; 
1368       } 
1369  
1370       switch (n.getType()) { 
1371         case Token.FUNCTION: 
1372           // VARs and ASSIGNs are handled separately. 
1373           if (parent.getType() == Token.NAME) { 
1374             return; 
1375           } 
1376           define(n, parent); 
1377           break; 
1378  
1379         case Token.CATCH: 
1380         case Token.VAR: 
1381           define(n, parent); 
1382           break; 
1383       } 
1384     } 
1385  
1386     /** Handle bleeding functions and function parameters. */ 
1387     private void handleFunctionInputs(Node fnNode) { 
1388       // Handle bleeding functions. 
1389       Node fnNameNode = fnNode.getFirstChild(); 
1390       String fnName = fnNameNode.getString(); 
1391       if (!fnName.isEmpty()) { 
1392         Scope.Var fnVar = scope.getVar(fnName); 
1393         if (fnVar == null || 
1394             // Make sure we're not touching a native function. Native 
1395             // functions aren't bleeding, but may not have a declaration 
1396             // node. 
1397             (fnVar.getNameNode() != null && 
1398                 // Make sure that the function is actually bleeding by checking 
1399                 // if has already been declared. 
1400                 fnVar.getInitialValue() != fnNode)) { 
1401           defineSlot(fnNameNode, fnNode, fnNode.getJSType(), false); 
1402         } 
1403       } 
1404  
1405       declareArguments(fnNode); 
1406     } 
1407  
1408     /** 
1409      * Declares all of a function's arguments. 
1410      */ 
1411     private void declareArguments(Node functionNode) { 
1412       Node astParameters = functionNode.getFirstChild().getNext(); 
1413       Node body = astParameters.getNext(); 
1414       FunctionType functionType = (FunctionType) functionNode.getJSType(); 
1415       if (functionType != null) { 
1416         Node jsDocParameters = functionType.getParametersNode(); 
1417         if (jsDocParameters != null) { 
1418           Node jsDocParameter = jsDocParameters.getFirstChild(); 
1419           for (Node astParameter : astParameters.children()) { 
1420             if (jsDocParameter != null) { 
1421               defineSlot(astParameter, functionNode, 
1422                   jsDocParameter.getJSType(), true); 
1423               jsDocParameter = jsDocParameter.getNext(); 
1424             } else { 
1425               defineSlot(astParameter, functionNode, null, true); 
1426             } 
1427           } 
1428         } 
1429       } 
1430     } // end declareArguments 
1431   } // end LocalScopeBuilder
