 35 def _partition_tasks(worker): 
 36     """ 
 37     Takes a worker and sorts out tasks based on their status. 
 38     Still_pending_not_ext is only used to get upstream_failure, upstream_missing_dependency and run_by_other_worker 
 39     """ 
 40     task_history = worker._add_task_history 
 41     pending_tasks = {task for(task, status, ext) in task_history if status == 'PENDING'} 
 42     set_tasks = {} 
 43     set_tasks["completed"] = {task for (task, status, ext) in task_history if status == 'DONE' and task in pending_tasks} 
 44     set_tasks["already_done"] = {task for (task, status, ext) in task_history 
 45                                  if status == 'DONE' and task not in pending_tasks and task not in set_tasks["completed"]} 
 46     set_tasks["failed"] = {task for (task, status, ext) in task_history if status == 'FAILED'} 
 47     set_tasks["scheduling_error"] = {task for(task, status, ext) in task_history if status == 'UNKNOWN'} 
 48     set_tasks["still_pending_ext"] = {task for (task, status, ext) in task_history 
 49                                       if status == 'PENDING' and task not in set_tasks["failed"] and task not in set_tasks["completed"] and not ext} 
 50     set_tasks["still_pending_not_ext"] = {task for (task, status, ext) in task_history 
 51                                           if status == 'PENDING' and task not in set_tasks["failed"] and task not in set_tasks["completed"] and ext} 
 52     set_tasks["run_by_other_worker"] = set() 
 53     set_tasks["upstream_failure"] = set() 
 54     set_tasks["upstream_missing_dependency"] = set() 
 55     set_tasks["upstream_run_by_other_worker"] = set() 
 56     set_tasks["upstream_scheduling_error"] = set() 
 57     set_tasks["not_run"] = set() 
 58     return set_tasks

 77 def _depth_first_search(set_tasks, current_task, visited): 
 78     """ 
 79     This dfs checks why tasks are still pending. 
 80     """ 
 81     visited.add(current_task) 
 82     if current_task in set_tasks["still_pending_not_ext"]: 
 83         upstream_failure = False 
 84         upstream_missing_dependency = False 
 85         upstream_run_by_other_worker = False 
 86         upstream_scheduling_error = False 
 87         for task in current_task._requires(): 
 88             if task not in visited: 
 89                 _depth_first_search(set_tasks, task, visited) 
 90             if task in set_tasks["failed"] or task in set_tasks["upstream_failure"]: 
 91                 set_tasks["upstream_failure"].add(current_task) 
 92                 upstream_failure = True 
 93             if task in set_tasks["still_pending_ext"] or task in set_tasks["upstream_missing_dependency"]: 
 94                 set_tasks["upstream_missing_dependency"].add(current_task) 
 95                 upstream_missing_dependency = True 
 96             if task in set_tasks["run_by_other_worker"] or task in set_tasks["upstream_run_by_other_worker"]: 
 97                 set_tasks["upstream_run_by_other_worker"].add(current_task) 
 98                 upstream_run_by_other_worker = True 
 99             if task in set_tasks["scheduling_error"]: 
100                 set_tasks["upstream_scheduling_error"].add(current_task) 
101                 upstream_scheduling_error = True 
102         if not upstream_failure and not upstream_missing_dependency and \ 
103                 not upstream_run_by_other_worker and not upstream_scheduling_error and \ 
104                 current_task not in set_tasks["run_by_other_worker"]: 
105             set_tasks["not_run"].add(current_task)

108 def _get_str(task_dict, extra_indent): 
109     """ 
110     This returns a string for each status 
111     """ 
112     summary_length = execution_summary().summary_length 
113  
114     lines = [] 
115     task_names = sorted(task_dict.keys()) 
116     for task_family in task_names: 
117         tasks = task_dict[task_family] 
118         tasks = sorted(tasks, key=lambda x: str(x)) 
119         prefix_size = 8 if extra_indent else 4 
120         prefix = ' ' * prefix_size 
121  
122         line = None 
123  
124         if summary_length > 0 and len(lines) >= summary_length: 
125             line = prefix + "..." 
126             lines.append(line) 
127             break 
128         if len(tasks[0].get_params()) == 0: 
129             line = prefix + '- {0} {1}()'.format(len(tasks), str(task_family)) 
130         elif _get_len_of_params(tasks[0]) > 60 or len(str(tasks[0])) > 200 or \ 
131                 (len(tasks) == 2 and len(tasks[0].get_params()) > 1 and (_get_len_of_params(tasks[0]) > 40 or len(str(tasks[0])) > 100)): 
132             """ 
133             This is to make sure that there is no really long task in the output 
134             """ 
135             line = prefix + '- {0} {1}(...)'.format(len(tasks), task_family) 
136         elif len((tasks[0].get_params())) == 1: 
137             attributes = {getattr(task, tasks[0].get_params()[0][0]) for task in tasks} 
138             param_class = tasks[0].get_params()[0][1] 
139             first, last = _ranging_attributes(attributes, param_class) 
140             if first is not None and last is not None and len(attributes) > 3: 
141                 param_str = '{0}...{1}'.format(param_class.serialize(first), param_class.serialize(last)) 
142             else: 
143                 param_str = '{0}'.format(_get_str_one_parameter(tasks)) 
144             line = prefix + '- {0} {1}({2}={3})'.format(len(tasks), task_family, tasks[0].get_params()[0][0], param_str) 
145         else: 
146             ranging = False 
147             params = _get_set_of_params(tasks) 
148             unique_param_keys = list(_get_unique_param_keys(params)) 
149             if len(unique_param_keys) == 1: 
150                 unique_param, = unique_param_keys 
151                 attributes = params[unique_param] 
152                 param_class = unique_param[1] 
153                 first, last = _ranging_attributes(attributes, param_class) 
154                 if first is not None and last is not None and len(attributes) > 2: 
155                     ranging = True 
156                     line = prefix + '- {0} {1}({2}'.format(len(tasks), task_family, _get_str_ranging_multiple_parameters(first, last, tasks, unique_param)) 
157             if not ranging: 
158                 if len(tasks) == 1: 
159                     line = prefix + '- {0} {1}'.format(len(tasks), tasks[0]) 
160                 if len(tasks) == 2: 
161                     line = prefix + '- {0} {1} and {2}'.format(len(tasks), tasks[0], tasks[1]) 
162                 if len(tasks) > 2: 
163                     line = prefix + '- {0} {1} ...'.format(len(tasks), tasks[0]) 
164         lines.append(line) 
165     return '\n'.join(lines)

172 def _get_str_ranging_multiple_parameters(first, last, tasks, unique_param): 
173     row = '' 
174     str_unique_param = '{0}...{1}'.format(unique_param[1].serialize(first), unique_param[1].serialize(last)) 
175     for param in tasks[0].get_params(): 
176         row += '{0}='.format(param[0]) 
177         if param[0] == unique_param[0]: 
178             row += '{0}'.format(str_unique_param) 
179         else: 
180             row += '{0}'.format(param[1].serialize(getattr(tasks[0], param[0]))) 
181         if param != tasks[0].get_params()[-1]: 
182             row += ", " 
183     row += ')' 
184     return row

196         if len(param_values) > 1:

200 def _ranging_attributes(attributes, param_class): 
201     """ 
202     Checks if there is a continuous range 
203     """ 
204     next_attributes = {param_class.next_in_enumeration(attribute) for attribute in attributes} 
205     in_first = attributes.difference(next_attributes) 
206     in_second = next_attributes.difference(attributes) 
207     if len(in_first) == 1 and len(in_second) == 1: 
208         for x in attributes: 
209             if {param_class.next_in_enumeration(x)} == in_second: 
210                 return next(iter(in_first)), x 
211     return None, None

214 def _get_str_one_parameter(tasks): 
215     row = '' 
216     count = 0 
217     for task in tasks: 
218         if (len(row) >= 30 and count > 2 and count != len(tasks) - 1) or len(row) > 200: 
219             row += '...' 
220             break 
221         param = task.get_params()[0] 
222         row += '{0}'.format(param[1].serialize(getattr(task, param[0]))) 
223         if count < len(tasks) - 1: 
224             row += ',' 
225         count += 1 
226     return row

244 def _get_comments(group_tasks): 
245     """ 
246     Get the human readable comments and quantities for the task types. 
247     """ 
248     comments = {} 
249     for status, human in _COMMENTS: 
250         num_tasks = _get_number_of_tasks_for(status, group_tasks) 
251         if num_tasks: 
252             space = "    " if status in _PENDING_SUB_STATUSES else "" 
253             comments[status] = '{space}* {num_tasks} {human}:\n'.format( 
254                 space=space, 
255                 num_tasks=num_tasks, 
256                 human=human) 
257     return comments

287     ("upstream_scheduling_error", 'had dependencies whose scheduling failed'), 
288     ("not_run", 'was not granted run permission by the scheduler'),

292 def _get_run_by_other_worker(worker): 
293     """ 
294     This returns a set of the tasks that are being run by other worker 
295     """ 
296     task_sets = _get_external_workers(worker).values() 
297     return functools.reduce(lambda a, b: a | b, task_sets, set())

300 def _get_external_workers(worker): 
301     """ 
302     This returns a dict with a set of tasks for all of the other workers 
303     """ 
304     worker_that_blocked_task = collections.defaultdict(set) 
305     get_work_response_history = worker._get_work_response_history 
306     for get_work_response in get_work_response_history: 
307         if get_work_response['task_id'] is None: 
308             for running_task in get_work_response['running_tasks']: 
309                 other_worker_id = running_task['worker'] 
310                 other_task_id = running_task['task_id'] 
311                 other_task = worker._scheduled_tasks.get(other_task_id) 
312                 if other_worker_id == worker._id or not other_task: 
313                     continue 
314                 worker_that_blocked_task[other_worker_id].add(other_task) 
315     return worker_that_blocked_task

318 def _group_tasks_by_name_and_status(task_dict): 
319     """ 
320     Takes a dictionary with sets of tasks grouped by their status and 
321     returns a dictionary with dictionaries with an array of tasks grouped by 
322     their status and task name 
323     """ 
324     group_status = {} 
325     for task in task_dict: 
326         if task.task_family not in group_status: 
327             group_status[task.task_family] = [] 
328         group_status[task.task_family].append(task) 
329     return group_status

339 def _summary_format(set_tasks, worker): 
340     group_tasks = {} 
341     for status, task_dict in set_tasks.items(): 
342         group_tasks[status] = _group_tasks_by_name_and_status(task_dict) 
343     comments = _get_comments(group_tasks) 
344     num_all_tasks = sum([len(set_tasks["already_done"]), 
345                          len(set_tasks["completed"]), len(set_tasks["failed"]), 
346                          len(set_tasks["scheduling_error"]), 
347                          len(set_tasks["still_pending_ext"]), 
348                          len(set_tasks["still_pending_not_ext"])]) 
349     str_output = '' 
350     str_output += 'Scheduled {0} tasks of which:\n'.format(num_all_tasks) 
351     for status in _ORDERED_STATUSES: 
352         if status not in comments: 
353             continue 
354         str_output += '{0}'.format(comments[status]) 
355         if status != 'still_pending': 
356             str_output += '{0}\n'.format(_get_str(group_tasks[status], status in _PENDING_SUB_STATUSES)) 
357     ext_workers = _get_external_workers(worker) 
358     group_tasks_ext_workers = {} 
359     for ext_worker, task_dict in ext_workers.items(): 
360         group_tasks_ext_workers[ext_worker] = _group_tasks_by_name_and_status(task_dict) 
361     if len(ext_workers) > 0: 
362         str_output += "\nThe other workers were:\n" 
363         count = 0 
364         for ext_worker, task_dict in ext_workers.items(): 
365             if count > 3 and count < len(ext_workers) - 1: 
366                 str_output += "    and {0} other workers".format(len(ext_workers) - count) 
367                 break 
368             str_output += "    - {0} ran {1} tasks\n".format(ext_worker, len(task_dict)) 
369             count += 1 
370         str_output += '\n' 
371     if num_all_tasks == sum([len(set_tasks["already_done"]), 
372                              len(set_tasks["scheduling_error"]), 
373                              len(set_tasks["still_pending_ext"]), 
374                              len(set_tasks["still_pending_not_ext"])]): 
375         if len(ext_workers) == 0: 
376             str_output += '\n' 
377         str_output += 'Did not run any tasks' 
378     smiley = "" 
379     reason = "" 
380     if set_tasks["failed"]: 
381         smiley = ":(" 
382         reason = "there were failed tasks" 
383         if set_tasks["scheduling_error"]: 
384             reason += " and tasks whose scheduling failed" 
385     elif set_tasks["scheduling_error"]: 
386         smiley = ":(" 
387         reason = "there were tasks whose scheduling failed" 
388     elif set_tasks["not_run"]: 
389         smiley = ":|" 
390         reason = "there were tasks that were not granted run permission by the scheduler" 
391     elif set_tasks["still_pending_ext"]: 
392         smiley = ":|" 
393         reason = "there were missing external dependencies" 
394     else: 
395         smiley = ":)" 
396         reason = "there were no failed tasks or missing external dependencies" 
397     str_output += "\nThis progress looks {0} because {1}".format(smiley, reason) 
398     if num_all_tasks == 0: 
399         str_output = 'Did not schedule any tasks' 
400     return str_output

140             first, last = _ranging_attributes(attributes, param_class) 
141             if first is not None and last is not None and len(attributes) > 3: 
142                 param_str = '{0}...{1}'.format(param_class.serialize(first), param_class.serialize(last)) 
143             else: 
144                 param_str = '{0}'.format(_get_str_one_parameter(tasks)) 
145             line = prefix + '- {0} {1}({2}={3})'.format(len(tasks), task_family, tasks[0].get_params()[0][0], param_str) 
146         else: 
147             ranging = False 
148             params = _get_set_of_params(tasks) 
149             unique_param_keys = list(_get_unique_param_keys(params)) 
150             if len(unique_param_keys) == 1: 
151                 unique_param, = unique_param_keys 
152                 attributes = params[unique_param] 
153                 param_class = unique_param[1] 
154                 first, last = _ranging_attributes(attributes, param_class) 
155                 if first is not None and last is not None and len(attributes) > 2: 
156                     ranging = True 
157                     line = prefix + '- {0} {1}({2}'.format(len(tasks), task_family, _get_str_ranging_multiple_parameters(first, last, tasks, unique_param)) 
158             if not ranging: 
159                 if len(tasks) == 1: 
160                     line = prefix + '- {0} {1}'.format(len(tasks), tasks[0]) 
161                 if len(tasks) == 2: 
162                     line = prefix + '- {0} {1} and {2}'.format(len(tasks), tasks[0], tasks[1]) 
163                 if len(tasks) > 2: 
164                     line = prefix + '- {0} {1} ...'.format(len(tasks), tasks[0]) 
165         lines.append(line) 
166     return '\n'.join(lines)

173 def _get_str_ranging_multiple_parameters(first, last, tasks, unique_param): 
174     row = '' 
175     str_unique_param = '{0}...{1}'.format(unique_param[1].serialize(first), unique_param[1].serialize(last)) 
176     for param in tasks[0].get_params(): 
177         row += '{0}='.format(param[0]) 
178         if param[0] == unique_param[0]: 
179             row += '{0}'.format(str_unique_param) 
180         else: 
181             row += '{0}'.format(param[1].serialize(getattr(tasks[0], param[0]))) 
182         if param != tasks[0].get_params()[-1]: 
183             row += ", " 
184     row += ')' 
185     return row

201 def _ranging_attributes(attributes, param_class): 
202     """ 
203     Checks if there is a continuous range 
204     """ 
205     next_attributes = {param_class.next_in_enumeration(attribute) for attribute in attributes} 
206     in_first = attributes.difference(next_attributes) 
207     in_second = next_attributes.difference(attributes) 
208     if len(in_first) == 1 and len(in_second) == 1: 
209         for x in attributes: 
210             if {param_class.next_in_enumeration(x)} == in_second: 
211                 return next(iter(in_first)), x 
212     return None, None

215 def _get_str_one_parameter(tasks): 
216     row = '' 
217     count = 0 
218     for task in tasks: 
219         if (len(row) >= 30 and count > 2 and count != len(tasks) - 1) or len(row) > 200: 
220             row += '...' 
221             break 
222         param = task.get_params()[0] 
223         row += '{0}'.format(param[1].serialize(getattr(task, param[0]))) 
224         if count < len(tasks) - 1: 
225             row += ',' 
226         count += 1 
227     return row

245 def _get_comments(group_tasks): 
246     """ 
247     Get the human readable comments and quantities for the task types. 
248     """ 
249     comments = {} 
250     for status, human in _COMMENTS: 
251         num_tasks = _get_number_of_tasks_for(status, group_tasks) 
252         if num_tasks: 
253             space = "    " if status in _PENDING_SUB_STATUSES else "" 
254             comments[status] = '{space}* {num_tasks} {human}:\n'.format( 
255                 space=space, 
256                 num_tasks=num_tasks, 
257                 human=human) 
258     return comments

294 def _get_run_by_other_worker(worker): 
295     """ 
296     This returns a set of the tasks that are being run by other worker 
297     """ 
298     task_sets = _get_external_workers(worker).values() 
299     return functools.reduce(lambda a, b: a | b, task_sets, set())

302 def _get_external_workers(worker): 
303     """ 
304     This returns a dict with a set of tasks for all of the other workers 
305     """ 
306     worker_that_blocked_task = collections.defaultdict(set) 
307     get_work_response_history = worker._get_work_response_history 
308     for get_work_response in get_work_response_history: 
309         if get_work_response['task_id'] is None: 
310             for running_task in get_work_response['running_tasks']: 
311                 other_worker_id = running_task['worker'] 
312                 other_task_id = running_task['task_id'] 
313                 other_task = worker._scheduled_tasks.get(other_task_id) 
314                 if other_worker_id == worker._id or not other_task: 
315                     continue 
316                 worker_that_blocked_task[other_worker_id].add(other_task) 
317     return worker_that_blocked_task

320 def _group_tasks_by_name_and_status(task_dict): 
321     """ 
322     Takes a dictionary with sets of tasks grouped by their status and 
323     returns a dictionary with dictionaries with an array of tasks grouped by 
324     their status and task name

326     group_status = {}

330         group_status[task.task_family].append(task)

341 def _summary_format(set_tasks, worker): 
342     group_tasks = {} 
343     for status, task_dict in set_tasks.items(): 
344         group_tasks[status] = _group_tasks_by_name_and_status(task_dict) 
345     comments = _get_comments(group_tasks) 
346     num_all_tasks = sum([len(set_tasks["already_done"]), 
347                          len(set_tasks["completed"]), len(set_tasks["failed"]), 
348                          len(set_tasks["scheduling_error"]), 
349                          len(set_tasks["still_pending_ext"]), 
350                          len(set_tasks["still_pending_not_ext"])]) 
351     str_output = '' 
352     str_output += 'Scheduled {0} tasks of which:\n'.format(num_all_tasks) 
353     for status in _ORDERED_STATUSES: 
354         if status not in comments: 
355             continue 
356         str_output += '{0}'.format(comments[status]) 
357         if status != 'still_pending': 
358             str_output += '{0}\n'.format(_get_str(group_tasks[status], status in _PENDING_SUB_STATUSES)) 
359     ext_workers = _get_external_workers(worker) 
360     group_tasks_ext_workers = {} 
361     for ext_worker, task_dict in ext_workers.items(): 
362         group_tasks_ext_workers[ext_worker] = _group_tasks_by_name_and_status(task_dict) 
363     if len(ext_workers) > 0: 
364         str_output += "\nThe other workers were:\n" 
365         count = 0 
366         for ext_worker, task_dict in ext_workers.items(): 
367             if count > 3 and count < len(ext_workers) - 1: 
368                 str_output += "    and {0} other workers".format(len(ext_workers) - count) 
369                 break 
370             str_output += "    - {0} ran {1} tasks\n".format(ext_worker, len(task_dict)) 
371             count += 1 
372         str_output += '\n' 
373     if num_all_tasks == sum([len(set_tasks["already_done"]), 
374                              len(set_tasks["scheduling_error"]), 
375                              len(set_tasks["still_pending_ext"]), 
376                              len(set_tasks["still_pending_not_ext"])]): 
377         if len(ext_workers) == 0: 
378             str_output += '\n' 
379         str_output += 'Did not run any tasks' 
380     smiley = "" 
381     reason = "" 
382     if set_tasks["ever_failed"]: 
383         if not set_tasks["failed"]: 
384             smiley = ":)" 
385             reason = "there were failed tasks but they all suceeded in a retry" 
386         else: 
387             smiley = ":(" 
388             reason = "there were failed tasks" 
389             if set_tasks["scheduling_error"]: 
390                 reason += " and tasks whose scheduling failed" 
391     elif set_tasks["scheduling_error"]: 
392         smiley = ":(" 
393         reason = "there were tasks whose scheduling failed" 
394     elif set_tasks["not_run"]: 
395         smiley = ":|" 
396         reason = "there were tasks that were not granted run permission by the scheduler" 
397     elif set_tasks["still_pending_ext"]: 
398         smiley = ":|" 
399         reason = "there were missing external dependencies" 
400     else: 
401         smiley = ":)" 
402         reason = "there were no failed tasks or missing external dependencies" 
403     str_output += "\nThis progress looks {0} because {1}".format(smiley, reason) 
404     if num_all_tasks == 0: 
405         str_output = 'Did not schedule any tasks' 
406     return str_output
