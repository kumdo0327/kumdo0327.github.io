 46 /** 
 47  * A builder for {@link JSDocInfo} objects. This builder abstracts the 
 48  * construction process of {@link JSDocInfo} objects whilst minimizing the 
 49  * number of instances of {@link JSDocInfo} objects. It provides early 
 50  * incompatibility detection among properties stored on the {@code JSDocInfo} 
 51  * object being created. 
 52  * 
 53 * 
 54  */ 
 55 final public class JSDocInfoBuilder { 
 56   // the current JSDoc which is being populated 
 57   private JSDocInfo currentInfo; 
 58  
 59   // whether the current JSDocInfo has valuable information 
 60   private boolean populated = false; 
 61  
 62   // whether to include the documentation itself when parsing the JsDoc 
 63   private boolean parseDocumentation = false; 
 64  
 65   // the current marker, if any. 
 66   private JSDocInfo.Marker currentMarker = null; 
 67  
 68   public JSDocInfoBuilder(boolean parseDocumentation) { 
 69     this.currentInfo = new JSDocInfo(parseDocumentation); 
 70     this.parseDocumentation = parseDocumentation; 
 71   }

 97   /** 
 98    * Builds a {@link JSDocInfo} object based on the populated information and 
 99    * returns it. Once this method is called, the builder can be reused to build 
100    * another {@link JSDocInfo} object. 
101    * 
102    * @param sourceName The source file containing the JSDoc. 
103    * @return a {@link JSDocInfo} object populated with the values given to this 
104    *     builder. If no value was populated, this method simply returns 
105    *     {@code null} 
106    */ 
107   public JSDocInfo build(String sourceName) { 
108     if (populated) { 
109       JSDocInfo built = currentInfo; 
110       built.setSourceName(sourceName); 
111       populateDefaults(built); 
112       populated = false; 
113       currentInfo = new JSDocInfo(this.parseDocumentation); 
114       return built; 
115     } else { 
116       return null; 
117     } 
118   }

120   /** Generate defaults when certain parameters are not specified. */ 
121   private static void populateDefaults(JSDocInfo info) { 
122     if (info.getVisibility() == null) { 
123       info.setVisibility(Visibility.INHERITED); 
124     } 
125   }

127   /** 
128    * Adds a marker to the current JSDocInfo and populates the marker with the 
129    * annotation information. 
130    */ 
131   public void markAnnotation(String annotation, int lineno, int charno) { 
132     JSDocInfo.Marker marker = currentInfo.addMarker(); 
133  
134     if (marker != null) { 
135       marker.annotation = new JSDocInfo.StringPosition(); 
136       marker.annotation.setItem(annotation); 
137       marker.annotation.setPositionInformation(lineno, charno, lineno, 
138                                                charno + annotation.length()); 
139     } 
140  
141     currentMarker = marker; 
142   }

159    */ 
160   public void markTypeNode(Node typeNode, int lineno, int startCharno, int endCharno, 
161                     boolean hasLC) {

164       currentMarker.type.setItem(typeNode);

167                                                 lineno, endCharno); 
168     }

745   /** 
746    * Whether the current doc info has other type tags, like 
747    * {@code @param} or {@code @return} or {@code @type} or etc. 
748    */ 
749   private boolean hasAnyTypeRelatedTags() { 
750     return currentInfo.isConstructor() || 
751         currentInfo.isInterface() || 
752         currentInfo.getParameterCount() > 0 || 
753         currentInfo.hasReturnType() || 
754         currentInfo.hasBaseType() || 
755         currentInfo.hasThisType() || 
756         hasAnySingletonTypeTags(); 
757   } 
758  
759   /** 
760    * Whether the current doc info has any of the singleton type 
761    * tags that may not appear with other type tags, like 
762    * {@code @type} or {@code @typedef}. 
763    */ 
764   private boolean hasAnySingletonTypeTags() { 
765     return currentInfo.hasType() || 
766         currentInfo.hasTypedefType() || 
767         currentInfo.hasEnumParameterType(); 
768   }

 46 /** 
 47  * A builder for {@link JSDocInfo} objects. This builder abstracts the 
 48  * construction process of {@link JSDocInfo} objects whilst minimizing the 
 49  * number of instances of {@link JSDocInfo} objects. It provides early 
 50  * incompatibility detection among properties stored on the {@code JSDocInfo} 
 51  * object being created. 
 52  * 
 53 * 
 54  */ 
 55 final public class JSDocInfoBuilder { 
 56   // the current JSDoc which is being populated 
 57   private JSDocInfo currentInfo; 
 58  
 59   // whether the current JSDocInfo has valuable information 
 60   private boolean populated = false; 
 61  
 62   // whether to include the documentation itself when parsing the JsDoc 
 63   private boolean parseDocumentation = false; 
 64  
 65   // the current marker, if any. 
 66   private JSDocInfo.Marker currentMarker = null; 
 67  
 68   public JSDocInfoBuilder(boolean parseDocumentation) { 
 69     this.currentInfo = new JSDocInfo(parseDocumentation); 
 70     this.parseDocumentation = parseDocumentation; 
 71   } 
 72  
 73   /** 
 74    * Returns whether this builder is populated with information that can be 
 75    * used to {@link #build} a {@link JSDocInfo} object. 
 76    */ 
 77   public boolean isPopulated() { 
 78     return populated; 
 79   } 
 80  
 81   /** 
 82    * Returns whether this builder is populated with information that can be 
 83    * used to {@link #build} a {@link JSDocInfo} object that has a 
 84    * fileoverview tag. 
 85    */ 
 86   public boolean isPopulatedWithFileOverview() { 
 87     return isPopulated() && currentInfo.hasFileOverview(); 
 88   }

 90   /** 
 91    * Returns whether this builder recorded a description. 
 92    */ 
 93   public boolean isDescriptionRecorded() { 
 94     return currentInfo.getDescription() != null; 
 95   }

 97   /** 
 98    * Builds a {@link JSDocInfo} object based on the populated information and 
 99    * returns it. Once this method is called, the builder can be reused to build 
100    * another {@link JSDocInfo} object. 
101    * 
102    * @param sourceName The source file containing the JSDoc. 
103    * @return a {@link JSDocInfo} object populated with the values given to this 
104    *     builder. If no value was populated, this method simply returns 
105    *     {@code null} 
106    */ 
107   public JSDocInfo build(String sourceName) { 
108     if (populated) { 
109       JSDocInfo built = currentInfo; 
110       built.setSourceName(sourceName); 
111       populateDefaults(built); 
112       populated = false; 
113       currentInfo = new JSDocInfo(this.parseDocumentation); 
114       return built; 
115     } else { 
116       return null; 
117     } 
118   }

120   /** Generate defaults when certain parameters are not specified. */ 
121   private static void populateDefaults(JSDocInfo info) { 
122     if (info.getVisibility() == null) { 
123       info.setVisibility(Visibility.INHERITED); 
124     } 
125   }

127   /** 
128    * Adds a marker to the current JSDocInfo and populates the marker with the 
129    * annotation information. 
130    */ 
131   public void markAnnotation(String annotation, int lineno, int charno) { 
132     JSDocInfo.Marker marker = currentInfo.addMarker(); 
133  
134     if (marker != null) { 
135       marker.annotation = new JSDocInfo.StringPosition(); 
136       marker.annotation.setItem(annotation); 
137       marker.annotation.setPositionInformation(lineno, charno, lineno, 
138                                                charno + annotation.length()); 
139     } 
140  
141     currentMarker = marker; 
142   } 
143  
144   /** 
145    * Adds a textual block to the current marker. 
146    */ 
147   public void markText(String text, int startLineno, int startCharno, int endLineno, 
148                 int endCharno) { 
149     if (currentMarker != null) { 
150       currentMarker.description = new JSDocInfo.StringPosition(); 
151       currentMarker.description.setItem(text); 
152       currentMarker.description.setPositionInformation(startLineno, startCharno, 
153                                                        endLineno, endCharno); 
154     } 
155   }

157   /** 
158    * Adds a type declaration to the current marker. 
159    */ 
160   public void markTypeNode(Node typeNode, int lineno, int startCharno, int endCharno, 
161                     boolean hasLC) { 
162     if (currentMarker != null) { 
163       currentMarker.type = new JSDocInfo.TypePosition(); 
164       currentMarker.type.setItem(typeNode); 
165       currentMarker.type.hasBrackets = hasLC; 
166       currentMarker.type.setPositionInformation(lineno, startCharno, 
167                                                 lineno, endCharno); 
168     } 
169   }

405   /** 
406    * Records the type of a define. 
407    * 
408    * 'Define' values are special constants that may be manipulated by 
409    * the compiler. They are designed to mimic the #define command in 
410    * the C preprocessor. 
411    */ 
412   public boolean recordDefineType(JSTypeExpression type) { 
413     if (type != null && 
414         !currentInfo.isConstant() && 
415         !currentInfo.isDefine() && 
416         recordType(type)) { 
417       currentInfo.setDefine(true); 
418       populated = true; 
419       return true; 
420     } else { 
421       return false; 
422     } 
423   }

743   /** 
744    * Whether the current doc info has other type tags, like 
745    * {@code @param} or {@code @return} or {@code @type} or etc. 
746    */ 
747   private boolean hasAnyTypeRelatedTags() { 
748     return currentInfo.isConstructor() || 
749         currentInfo.isInterface() || 
750         currentInfo.getParameterCount() > 0 || 
751         currentInfo.hasReturnType() || 
752         currentInfo.hasBaseType() || 
753         currentInfo.hasThisType() || 
754         hasAnySingletonTypeTags(); 
755   } 
756  
757   /** 
758    * Whether the current doc info has any of the singleton type 
759    * tags that may not appear with other type tags, like 
760    * {@code @type} or {@code @typedef}. 
761    */ 
762   private boolean hasAnySingletonTypeTags() { 
763     return currentInfo.hasType() || 
764         currentInfo.hasTypedefType() || 
765         currentInfo.hasEnumParameterType(); 
766   }
