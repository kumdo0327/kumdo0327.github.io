 83     def __init__(self, settings): 
 84         if isinstance(settings, dict): 
 85             settings = Settings(settings) 
 86         self.settings = settings 
 87         self.spider_loader = _get_spider_loader(settings) 
 88         self.crawlers = set() 
 89         self._active = set()

 92     def spiders(self):

 98     def crawl(self, crawler_or_spidercls, *args, **kwargs): 
 99         crawler = crawler_or_spidercls 
100         if not isinstance(crawler_or_spidercls, Crawler): 
101             crawler = self._create_crawler(crawler_or_spidercls) 
102             self._setup_crawler_logging(crawler) 
103  
104         self.crawlers.add(crawler) 
105         d = crawler.crawl(*args, **kwargs) 
106         self._active.add(d) 
107  
108         def _done(result): 
109             self.crawlers.discard(crawler) 
110             self._active.discard(d) 
111             return result 
112  
113         return d.addBoth(_done)

115     def _create_crawler(self, spidercls): 
116         if isinstance(spidercls, six.string_types): 
117             spidercls = self.spider_loader.load(spidercls) 
118         return Crawler(spidercls, self.settings)

120     def _setup_crawler_logging(self, crawler): 
121         log_observer = log.start_from_crawler(crawler) 
122         if log_observer: 
123             crawler.signals.connect(log_observer.stop, signals.engine_stopped)

135 class CrawlerProcess(CrawlerRunner): 
136     """A class to run multiple scrapy crawlers in a process simultaneously""" 
137  
138     def __init__(self, settings): 
139         super(CrawlerProcess, self).__init__(settings) 
140         install_shutdown_handlers(self._signal_shutdown) 
141         self.stopping = False 
142         self.log_observer = log.start_from_settings(self.settings) 
143         log.scrapy_info(settings) 
144  
145     def _signal_shutdown(self, signum, _): 
146         install_shutdown_handlers(self._signal_kill) 
147         signame = signal_names[signum] 
148         log.msg(format="Received %(signame)s, shutting down gracefully. Send again to force ", 
149                 level=log.INFO, signame=signame) 
150         reactor.callFromThread(self.stop) 
151  
152     def _signal_kill(self, signum, _): 
153         install_shutdown_handlers(signal.SIG_IGN) 
154         signame = signal_names[signum] 
155         log.msg(format='Received %(signame)s twice, forcing unclean shutdown', 
156                 level=log.INFO, signame=signame) 
157         self._stop_logging() 
158         reactor.callFromThread(self._stop_reactor) 
159  
160     def start(self, stop_after_crawl=True): 
161         if stop_after_crawl: 
162             d = self.join() 
163             # Don't start the reactor if the deferreds are already fired 
164             if d.called: 
165                 return 
166             d.addBoth(lambda _: self._stop_reactor()) 
167  
168         cache_size = self.settings.getint('DNSCACHE_SIZE') if self.settings.getbool('DNSCACHE_ENABLED') else 0 
169         reactor.installResolver(CachingThreadedResolver(reactor, cache_size, 
170                                                             self.settings.getfloat('DNS_TIMEOUT'))) 
171         tp = reactor.getThreadPool() 
172         tp.adjustPoolsize(maxthreads=self.settings.getint('REACTOR_THREADPOOL_MAXSIZE')) 
173         reactor.addSystemEventTrigger('before', 'shutdown', self.stop) 
174         reactor.run(installSignalHandlers=False)  # blocking call 
175  
176     def _stop_logging(self): 
177         if self.log_observer: 
178             self.log_observer.stop() 
179  
180     def _stop_reactor(self, _=None): 
181         try: 
182             reactor.stop() 
183         except RuntimeError:  # raised if already stopped or in shutdown stage 
184             pass

187 def _get_spider_loader(settings): 
188     """ Get SpiderLoader instance from settings """ 
189     if settings.get('SPIDER_MANAGER_CLASS'): 
190         warnings.warn( 
191             'SPIDER_MANAGER_CLASS option is deprecated. ' 
192             'Please use SPIDER_LOADER_CLASS.', 
193             category=ScrapyDeprecationWarning, stacklevel=2 
194         ) 
195     cls_path = settings.get('SPIDER_LOADER_CLASS', 
196                             settings.get('SPIDER_MANAGER_CLASS')) 
197     loader_cls = load_object(cls_path) 
198     verifyClass(ISpiderLoader, loader_cls) 
199     return loader_cls.from_settings(settings.frozencopy())
