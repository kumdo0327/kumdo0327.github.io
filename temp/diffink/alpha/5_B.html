501     /** 
502      * <p>Prints the usage statement for the specified application.</p> 
503      * 
504      * @param pw The PrintWriter to print the usage statement  
505      * @param width The number of characters to display per line 
506      * @param app The application name 
507      * @param options The command line Options 
508      * 
509      */ 
510     public void printUsage(PrintWriter pw, int width, String app,  
511                            Options options) 
512     { 
513         // initialise the string buffer 
514         StringBuffer buff = new StringBuffer(defaultSyntaxPrefix).append(app) 
515                                                                  .append(" "); 
516  
517         // create a list for processed option groups 
518         final Collection processedGroups = new ArrayList(); 
519  
520         // temp variable 
521         Option option; 
522  
523         List optList = new ArrayList(options.getOptions()); 
524         Collections.sort(optList, getOptionComparator() ); 
525         // iterate over the options 
526         for (Iterator i = optList.iterator(); i.hasNext();) 
527         { 
528             // get the next Option 
529             option = (Option) i.next(); 
530  
531             // check if the option is part of an OptionGroup 
532             OptionGroup group = options.getOptionGroup(option); 
533  
534             // if the option is part of a group  
535             if (group != null) 
536             { 
537                 // and if the group has not already been processed 
538                 if (!processedGroups.contains(group)) 
539                 { 
540                     // add the group to the processed list 
541                     processedGroups.add(group); 
542  
543  
544                     // add the usage clause 
545                     appendOptionGroup(buff, group); 
546                 } 
547  
548                 // otherwise the option was displayed in the group 
549                 // previously so ignore it. 
550             } 
551  
552             // if the Option is not part of an OptionGroup 
553             else 
554             { 
555                 appendOption(buff, option, option.isRequired()); 
556             } 
557  
558             if (i.hasNext()) 
559             { 
560                 buff.append(" "); 
561             } 
562         } 
563  
564  
565         // call printWrapped 
566         printWrapped(pw, width, buff.toString().indexOf(' ') + 1,  
567                      buff.toString()); 
568     }

570     /** 
571      * Appends the usage clause for an OptionGroup to a StringBuffer.   
572      * The clause is wrapped in square brackets if the group is required. 
573      * The display of the options is handled by appendOption 
574      * @param buff the StringBuffer to append to 
575      * @param group the group to append 
576      * @see #appendOption(StringBuffer,Option,boolean) 
577      */ 
578     private void appendOptionGroup(final StringBuffer buff,  
579                                           final OptionGroup group) 
580     { 
581         if (!group.isRequired()) 
582         { 
583             buff.append("["); 
584         } 
585  
586         List optList = new ArrayList(group.getOptions()); 
587         Collections.sort(optList, getOptionComparator() ); 
588         // for each option in the OptionGroup 
589         for (Iterator i = optList.iterator(); i.hasNext();) 
590         { 
591             // whether the option is required or not is handled at group level 
592             appendOption(buff, (Option) i.next(), true); 
593  
594             if (i.hasNext()) 
595             { 
596                 buff.append(" | "); 
597             } 
598         } 
599  
600         if (!group.isRequired()) 
601         { 
602             buff.append("]"); 
603         } 
604     }

606     /** 
607      * Appends the usage clause for an Option to a StringBuffer.   
608      * 
609      * @param buff the StringBuffer to append to 
610      * @param option the Option to append 
611      * @param required whether the Option is required or not 
612      */ 
613     private static void appendOption(final StringBuffer buff,  
614                                      final Option option,  
615                                      final boolean required) 
616     { 
617         if (!required) 
618         { 
619             buff.append("["); 
620         } 
621  
622         if (option.getOpt() != null) 
623         { 
624             buff.append("-").append(option.getOpt()); 
625         } 
626         else 
627         { 
628             buff.append("--").append(option.getLongOpt()); 
629         } 
630  
631         // if the Option has a value 
632         if (option.hasArg() && (option.getArgName() != null)) 
633         { 
634             buff.append(" <").append(option.getArgName()).append(">"); 
635         } 
636  
637         // if the Option is not a required option 
638         if (!required) 
639         { 
640             buff.append("]"); 
641         } 
642     }

713      * <p>Render the specified Options and return the rendered Options 
714      * in a StringBuffer.</p> 
715      * 
716      * @param sb The StringBuffer to place the rendered Options into. 
717      * @param width The number of characters to display per line 
718      * @param options The command line Options 
719      * @param leftPad the number of characters of padding to be prefixed 
720      * to each line 
721      * @param descPad the number of characters of padding to be prefixed 
722      * to each description line 
723      * 
724      * @return the StringBuffer with the rendered Options contents. 
725      */ 
726     protected StringBuffer renderOptions(StringBuffer sb, int width,  
727                                          Options options, int leftPad,  
728                                          int descPad) 
729     { 
730         final String lpad = createPadding(leftPad); 
731         final String dpad = createPadding(descPad); 
732  
733         // first create list containing only <lpad>-a,--aaa where  
734         // -a is opt and --aaa is long opt; in parallel look for  
735         // the longest opt string this list will be then used to  
736         // sort options ascending 
737         int max = 0; 
738         StringBuffer optBuf; 
739         List prefixList = new ArrayList(); 
740         Option option; 
741         List optList = options.helpOptions(); 
742  
743         Collections.sort(optList, getOptionComparator() ); 
744  
745         for (Iterator i = optList.iterator(); i.hasNext();) 
746         { 
747             option = (Option) i.next(); 
748             optBuf = new StringBuffer(8); 
749  
750             if (option.getOpt() == null) 
751             { 
752                 optBuf.append(lpad).append("   " + defaultLongOptPrefix) 
753                       .append(option.getLongOpt()); 
754             } 
755             else 
756             { 
757                 optBuf.append(lpad).append(defaultOptPrefix) 
758                       .append(option.getOpt()); 
759  
760                 if (option.hasLongOpt()) 
761                 { 
762                     optBuf.append(',').append(defaultLongOptPrefix) 
763                           .append(option.getLongOpt()); 
764                 } 
765             } 
766  
767             if (option.hasArg()) 
768             { 
769                 if (option.hasArgName()) 
770                 { 
771                     optBuf.append(" <").append(option.getArgName()).append(">"); 
772                 } 
773                 else 
774                 { 
775                     optBuf.append(' '); 
776                 } 
777             } 
778  
779             prefixList.add(optBuf); 
780             max = (optBuf.length() > max)       ? optBuf.length() : max; 
781         } 
782  
783         int x = 0; 
784  
785         for (Iterator i = optList.iterator(); i.hasNext();) 
786         { 
787             option = (Option) i.next(); 
788             optBuf = new StringBuffer(prefixList.get(x++).toString()); 
789  
790             if (optBuf.length() < max) 
791             { 
792                 optBuf.append(createPadding(max - optBuf.length())); 
793             } 
794  
795             optBuf.append(dpad); 
796  
797             int nextLineTabStop = max + descPad; 
798  
799             if (option.getDescription() != null) 
800             { 
801                 optBuf.append(option.getDescription()); 
802             } 
803  
804             renderWrappedText(sb, width, nextLineTabStop, optBuf.toString()); 
805  
806             if (i.hasNext()) 
807             { 
808                 sb.append(defaultNewLine); 
809             } 
810         } 
811  
812         return sb; 
813     }

815     /** 
816      * <p>Render the specified text and return the rendered Options 
817      * in a StringBuffer.</p> 
818      * 
819      * @param sb The StringBuffer to place the rendered text into. 
820      * @param width The number of characters to display per line 
821      * @param nextLineTabStop The position on the next line for the first tab. 
822      * @param text The text to be rendered. 
823      * 
824      * @return the StringBuffer with the rendered Options contents. 
825      */ 
826     protected StringBuffer renderWrappedText(StringBuffer sb, int width,  
827                                              int nextLineTabStop, String text) 
828     { 
829         int pos = findWrapPos(text, width, 0); 
830  
831         if (pos == -1) 
832         { 
833             sb.append(rtrim(text)); 
834  
835             return sb; 
836         } 
837         sb.append(rtrim(text.substring(0, pos))).append(defaultNewLine); 
838  
839         // all following lines must be padded with nextLineTabStop space  
840         // characters 
841         final String padding = createPadding(nextLineTabStop); 
842  
843         while (true) 
844         { 
845             text = padding + text.substring(pos).trim(); 
846             pos = findWrapPos(text, width, 0); 
847  
848             if (pos == -1) 
849             { 
850                 sb.append(text); 
851  
852                 return sb; 
853             } 
854  
855             sb.append(rtrim(text.substring(0, pos))).append(defaultNewLine); 
856         } 
857     }

859     /** 
860      * Finds the next text wrap position after <code>startPos</code> for the  
861      * text in <code>text</code> with the column width <code>width</code>. 
862      * The wrap point is the last postion before startPos+width having a  
863      * whitespace character (space, \n, \r). 
864      * 
865      * @param text The text being searched for the wrap position 
866      * @param width width of the wrapped text 
867      * @param startPos position from which to start the lookup whitespace  
868      * character 
869      * @return postion on which the text must be wrapped or -1 if the wrap  
870      * position is at the end of the text 
871      */ 
872     protected int findWrapPos(String text, int width, int startPos) 
873     { 
874         int pos = -1; 
875  
876         // the line ends before the max wrap pos or a new line char found 
877         if (((pos = text.indexOf('\n', startPos)) != -1 && pos <= width) 
878             || ((pos = text.indexOf('\t', startPos)) != -1 && pos <= width)) 
879         { 
880             return pos+1; 
881         } 
882         else if ((startPos + width) >= text.length()) 
883         { 
884             return -1; 
885         } 
886  
887  
888         // look for the last whitespace character before startPos+width 
889         pos = startPos + width; 
890  
891         char c; 
892  
893         while ((pos >= startPos) && ((c = text.charAt(pos)) != ' ') 
894                && (c != '\n') && (c != '\r')) 
895         { 
896             --pos; 
897         } 
898  
899         // if we found it - just return 
900         if (pos > startPos) 
901         { 
902             return pos; 
903         } 
904          
905         // must look for the first whitespace chearacter after startPos  
906         // + width 
907         pos = startPos + width; 
908  
909         while ((pos <= text.length()) && ((c = text.charAt(pos)) != ' ') 
910                && (c != '\n') && (c != '\r')) 
911         { 
912             ++pos; 
913         } 
914  
915         return (pos == text.length())        ? (-1) : pos; 
916     }

918     /** 
919      * <p>Return a String of padding of length <code>len</code>.</p> 
920      * 
921      * @param len The length of the String of padding to create. 
922      * 
923      * @return The String of padding 
924      */ 
925     protected String createPadding(int len) 
926     { 
927         StringBuffer sb = new StringBuffer(len); 
928  
929         for (int i = 0; i < len; ++i) 
930         { 
931             sb.append(' '); 
932         } 
933  
934         return sb.toString(); 
935     }

937     /** 
938      * <p>Remove the trailing whitespace from the specified String.</p> 
939      * 
940      * @param s The String to remove the trailing padding from. 
941      * 
942      * @return The String of without the trailing padding 
943      */ 
944     protected String rtrim(String s) 
945     { 
946         if ((s == null) || (s.length() == 0)) 
947         { 
948             return s; 
949         } 
950  
951         int pos = s.length(); 
952  
953         while ((pos > 0) && Character.isWhitespace(s.charAt(pos - 1))) 
954         { 
955             --pos; 
956         } 
957  
958         return s.substring(0, pos); 
959     }

501     /** 
502      * <p>Prints the usage statement for the specified application.</p> 
503      * 
504      * @param pw The PrintWriter to print the usage statement  
505      * @param width The number of characters to display per line 
506      * @param app The application name 
507      * @param options The command line Options 
508      * 
509      */ 
510     public void printUsage(PrintWriter pw, int width, String app,  
511                            Options options) 
512     { 
513         // initialise the string buffer 
514         StringBuffer buff = new StringBuffer(defaultSyntaxPrefix).append(app) 
515                                                                  .append(" "); 
516  
517         // create a list for processed option groups 
518         final Collection processedGroups = new ArrayList(); 
519  
520         // temp variable 
521         Option option; 
522  
523         List optList = new ArrayList(options.getOptions()); 
524         Collections.sort(optList, getOptionComparator() ); 
525         // iterate over the options 
526         for (Iterator i = optList.iterator(); i.hasNext();) 
527         { 
528             // get the next Option 
529             option = (Option) i.next(); 
530  
531             // check if the option is part of an OptionGroup 
532             OptionGroup group = options.getOptionGroup(option); 
533  
534             // if the option is part of a group  
535             if (group != null) 
536             { 
537                 // and if the group has not already been processed 
538                 if (!processedGroups.contains(group)) 
539                 { 
540                     // add the group to the processed list 
541                     processedGroups.add(group); 
542  
543  
544                     // add the usage clause 
545                     appendOptionGroup(buff, group); 
546                 } 
547  
548                 // otherwise the option was displayed in the group 
549                 // previously so ignore it. 
550             } 
551  
552             // if the Option is not part of an OptionGroup 
553             else 
554             { 
555                 appendOption(buff, option, option.isRequired()); 
556             } 
557  
558             if (i.hasNext()) 
559             { 
560                 buff.append(" "); 
561             } 
562         } 
563  
564  
565         // call printWrapped 
566         printWrapped(pw, width, buff.toString().indexOf(' ') + 1,  
567                      buff.toString()); 
568     }

574      * @param buff the StringBuffer to append to 
575      * @param group the group to append 
576      * @see #appendOption(StringBuffer,Option,boolean) 
577      */ 
578     private void appendOptionGroup(final StringBuffer buff,  
579                                           final OptionGroup group) 
580     { 
581         if (!group.isRequired()) 
582         { 
583             buff.append("["); 
584         } 
585  
586         List optList = new ArrayList(group.getOptions()); 
587         Collections.sort(optList, getOptionComparator() ); 
588         // for each option in the OptionGroup 
589         for (Iterator i = optList.iterator(); i.hasNext();) 
590         { 
591             // whether the option is required or not is handled at group level 
592             appendOption(buff, (Option) i.next(), true); 
593  
594             if (i.hasNext()) 
595             { 
596                 buff.append(" | "); 
597             } 
598         } 
599  
600         if (!group.isRequired()) 
601         { 
602             buff.append("]"); 
603         } 
604     }

712     /** 
713      * <p>Render the specified Options and return the rendered Options 
714      * in a StringBuffer.</p> 
715      * 
716      * @param sb The StringBuffer to place the rendered Options into. 
717      * @param width The number of characters to display per line 
718      * @param options The command line Options 
719      * @param leftPad the number of characters of padding to be prefixed 
720      * to each line 
721      * @param descPad the number of characters of padding to be prefixed 
722      * to each description line 
723      * 
724      * @return the StringBuffer with the rendered Options contents. 
725      */ 
726     protected StringBuffer renderOptions(StringBuffer sb, int width,  
727                                          Options options, int leftPad,  
728                                          int descPad) 
729     { 
730         final String lpad = createPadding(leftPad); 
731         final String dpad = createPadding(descPad); 
732  
733         // first create list containing only <lpad>-a,--aaa where  
734         // -a is opt and --aaa is long opt; in parallel look for  
735         // the longest opt string this list will be then used to  
736         // sort options ascending 
737         int max = 0; 
738         StringBuffer optBuf; 
739         List prefixList = new ArrayList(); 
740         Option option; 
741         List optList = options.helpOptions(); 
742  
743         Collections.sort(optList, getOptionComparator() ); 
744  
745         for (Iterator i = optList.iterator(); i.hasNext();) 
746         { 
747             option = (Option) i.next(); 
748             optBuf = new StringBuffer(8); 
749  
750             if (option.getOpt() == null) 
751             { 
752                 optBuf.append(lpad).append("   " + defaultLongOptPrefix) 
753                       .append(option.getLongOpt()); 
754             } 
755             else 
756             { 
757                 optBuf.append(lpad).append(defaultOptPrefix) 
758                       .append(option.getOpt()); 
759  
760                 if (option.hasLongOpt()) 
761                 { 
762                     optBuf.append(',').append(defaultLongOptPrefix) 
763                           .append(option.getLongOpt()); 
764                 } 
765             } 
766  
767             if (option.hasArg()) 
768             { 
769                 if (option.hasArgName()) 
770                 { 
771                     optBuf.append(" <").append(option.getArgName()).append(">"); 
772                 } 
773                 else 
774                 { 
775                     optBuf.append(' '); 
776                 } 
777             } 
778  
779             prefixList.add(optBuf); 
780             max = (optBuf.length() > max)       ? optBuf.length() : max; 
781         } 
782  
783         int x = 0; 
784  
785         for (Iterator i = optList.iterator(); i.hasNext();) 
786         { 
787             option = (Option) i.next(); 
788             optBuf = new StringBuffer(prefixList.get(x++).toString()); 
789  
790             if (optBuf.length() < max) 
791             { 
792                 optBuf.append(createPadding(max - optBuf.length())); 
793             } 
794  
795             optBuf.append(dpad); 
796  
797             int nextLineTabStop = max + descPad; 
798  
799             if (option.getDescription() != null) 
800             { 
801                 optBuf.append(option.getDescription()); 
802             } 
803  
804             renderWrappedText(sb, width, nextLineTabStop, optBuf.toString()); 
805  
806             if (i.hasNext()) 
807             { 
808                 sb.append(defaultNewLine); 
809             } 
810         } 
811  
812         return sb; 
813     }

815     /** 
816      * <p>Render the specified text and return the rendered Options 
817      * in a StringBuffer.</p> 
818      * 
819      * @param sb The StringBuffer to place the rendered text into. 
820      * @param width The number of characters to display per line 
821      * @param nextLineTabStop The position on the next line for the first tab. 
822      * @param text The text to be rendered. 
823      * 
824      * @return the StringBuffer with the rendered Options contents. 
825      */ 
826     protected StringBuffer renderWrappedText(StringBuffer sb, int width,  
827                                              int nextLineTabStop, String text) 
828     { 
829         int pos = findWrapPos(text, width, 0); 
830  
831         if (pos == -1) 
832         { 
833             sb.append(rtrim(text)); 
834  
835             return sb; 
836         } 
837         sb.append(rtrim(text.substring(0, pos))).append(defaultNewLine); 
838  
839         // all following lines must be padded with nextLineTabStop space  
840         // characters 
841         final String padding = createPadding(nextLineTabStop); 
842  
843         while (true) 
844         { 
845             text = padding + text.substring(pos).trim(); 
846             pos = findWrapPos(text, width, 0); 
847  
848             if (pos == -1) 
849             { 
850                 sb.append(text); 
851  
852                 return sb; 
853             } 
854  
855             sb.append(rtrim(text.substring(0, pos))).append(defaultNewLine); 
856         } 
857     }

859     /** 
860      * Finds the next text wrap position after <code>startPos</code> for the  
861      * text in <code>text</code> with the column width <code>width</code>. 
862      * The wrap point is the last postion before startPos+width having a  
863      * whitespace character (space, \n, \r). 
864      * 
865      * @param text The text being searched for the wrap position 
866      * @param width width of the wrapped text 
867      * @param startPos position from which to start the lookup whitespace  
868      * character 
869      * @return postion on which the text must be wrapped or -1 if the wrap  
870      * position is at the end of the text 
871      */ 
872     protected int findWrapPos(String text, int width, int startPos) 
873     { 
874         int pos = -1; 
875  
876         // the line ends before the max wrap pos or a new line char found 
877         if (((pos = text.indexOf('\n', startPos)) != -1 && pos <= width) 
878             || ((pos = text.indexOf('\t', startPos)) != -1 && pos <= width)) 
879         { 
880             return pos+1; 
881         } 
882         else if ((startPos + width) >= text.length()) 
883         { 
884             return -1; 
885         } 
886  
887  
888         // look for the last whitespace character before startPos+width 
889         pos = startPos + width; 
890  
891         char c; 
892  
893         while ((pos >= startPos) && ((c = text.charAt(pos)) != ' ') 
894                && (c != '\n') && (c != '\r')) 
895         { 
896             --pos; 
897         } 
898  
899         // if we found it - just return 
900         if (pos > startPos) 
901         { 
902             return pos; 
903         } 
904          
905         // must look for the first whitespace chearacter after startPos  
906         // + width 
907         pos = startPos + width; 
908  
909         while ((pos <= text.length()) && ((c = text.charAt(pos)) != ' ') 
910                && (c != '\n') && (c != '\r')) 
911         { 
912             ++pos; 
913         } 
914  
915         return (pos == text.length())        ? (-1) : pos; 
916     }

938      * <p>Remove the trailing whitespace from the specified String.</p> 
939      * 
940      * @param s The String to remove the trailing padding from. 
941      * 
942      * @return The String of without the trailing padding 
943      */ 
944     protected String rtrim(String s) 
945     { 
946         if ((s == null) || (s.length() == 0)) 
947         { 
948             return s; 
949         } 
950  
951         int pos = s.length(); 
952  
953         while ((pos > 0) && Character.isWhitespace(s.charAt(pos - 1))) 
954         { 
955             --pos; 
956         } 
957  
958         return s.substring(0, pos); 
959     }
