  75 void CheckCondition::assignIf() 
  76 { 
  77     if (!mSettings->isEnabled(Settings::STYLE)) 
  78         return; 
  79  
  80     for (const Token *tok = mTokenizer->tokens(); tok; tok = tok->next()) { 
  81         if (tok->str() != "=") 
  82             continue; 
  83  
  84         if (Token::Match(tok->tokAt(-2), "[;{}] %var% =")) { 
  85             const Variable *var = tok->previous()->variable(); 
  86             if (var == nullptr) 
  87                 continue; 
  88  
  89             char bitop = '\0'; 
  90             MathLib::bigint num = 0; 
  91  
  92             if (Token::Match(tok->next(), "%num% [&|]")) { 
  93                 bitop = tok->strAt(2).at(0); 
  94                 num = MathLib::toLongNumber(tok->next()->str()); 
  95             } else { 
  96                 const Token *endToken = Token::findsimplematch(tok, ";"); 
  97  
  98                 // Casting address 
  99                 if (endToken && Token::Match(endToken->tokAt(-4), "* ) & %any% ;")) 
 100                     endToken = nullptr; 
 101  
 102                 if (endToken && Token::Match(endToken->tokAt(-2), "[&|] %num% ;")) { 
 103                     bitop = endToken->strAt(-2).at(0); 
 104                     num = MathLib::toLongNumber(endToken->previous()->str()); 
 105                 } 
 106             } 
 107  
 108             if (bitop == '\0') 
 109                 continue; 
 110  
 111             if (num < 0 && bitop == '|') 
 112                 continue; 
 113  
 114             assignIfParseScope(tok, tok->tokAt(4), var->declarationId(), var->isLocal(), bitop, num); 
 115         } 
 116     } 
 117 }

 119 static bool isParameterChanged(const Token *partok) 
 120 { 
 121     bool addressOf = Token::Match(partok, "[(,] &"); 
 122     unsigned int argumentNumber = 0; 
 123     const Token *ftok; 
 124     for (ftok = partok; ftok && ftok->str() != "("; ftok = ftok->previous()) { 
 125         if (ftok->str() == ")") 
 126             ftok = ftok->link(); 
 127         else if (argumentNumber == 0U && ftok->str() == "&") 
 128             addressOf = true; 
 129         else if (ftok->str() == ",") 
 130             argumentNumber++; 
 131     } 
 132     ftok = ftok ? ftok->previous() : nullptr; 
 133     if (!(ftok && ftok->function())) 
 134         return true; 
 135     const Variable *par = ftok->function()->getArgumentVar(argumentNumber); 
 136     if (!par) 
 137         return true; 
 138     if (par->isConst()) 
 139         return false; 
 140     if (addressOf || par->isReference() || par->isPointer()) 
 141         return true; 
 142     return false; 
 143 }

 145 /** parse scopes recursively */ 
 146 bool CheckCondition::assignIfParseScope(const Token * const assignTok, 
 147                                         const Token * const startTok, 
 148                                         const unsigned int varid, 
 149                                         const bool islocal, 
 150                                         const char bitop, 
 151                                         const MathLib::bigint num) 
 152 { 
 153     bool ret = false; 
 154  
 155     for (const Token *tok2 = startTok; tok2; tok2 = tok2->next()) { 
 156         if ((bitop == '&') && Token::Match(tok2->tokAt(2), "%varid% %cop% %num% ;", varid) && tok2->strAt(3) == std::string(1U, bitop)) { 
 157             const MathLib::bigint num2 = MathLib::toLongNumber(tok2->strAt(4)); 
 158             if (0 == (num & num2)) 
 159                 mismatchingBitAndError(assignTok, num, tok2, num2); 
 160         } 
 161         if (Token::Match(tok2, "%varid% =", varid)) { 
 162             return true; 
 163         } 
 164         if (Token::Match(tok2, "++|-- %varid%", varid) || Token::Match(tok2, "%varid% ++|--", varid)) 
 165             return true; 
 166         if (Token::Match(tok2, "[(,] &| %varid% [,)]", varid) && isParameterChanged(tok2)) 
 167             return true; 
 168         if (tok2->str() == "}") 
 169             return false; 
 170         if (Token::Match(tok2, "break|continue|return")) 
 171             ret = true; 
 172         if (ret && tok2->str() == ";") 
 173             return false; 
 174         if (!islocal && Token::Match(tok2, "%name% (") && !Token::simpleMatch(tok2->next()->link(), ") {")) 
 175             return true; 
 176         if (Token::Match(tok2, "if|while (")) { 
 177             if (!islocal && tok2->str() == "while") 
 178                 continue; 
 179             if (tok2->str() == "while") { 
 180                 // is variable changed in loop? 
 181                 const Token *bodyStart = tok2->linkAt(1)->next(); 
 182                 const Token *bodyEnd   = bodyStart ? bodyStart->link() : nullptr; 
 183                 if (!bodyEnd || bodyEnd->str() != "}" || isVariableChanged(bodyStart, bodyEnd, varid, !islocal, mSettings, mTokenizer->isCPP())) 
 184                     continue; 
 185             } 
 186  
 187             // parse condition 
 188             const Token * const end = tok2->next()->link(); 
 189             for (; tok2 != end; tok2 = tok2->next()) { 
 190                 if (Token::Match(tok2, "[(,] &| %varid% [,)]", varid)) { 
 191                     return true; 
 192                 } 
 193                 if (Token::Match(tok2,"&&|%oror%|( %varid% ==|!= %num% &&|%oror%|)", varid)) { 
 194                     const Token *vartok = tok2->next(); 
 195                     const MathLib::bigint num2 = MathLib::toLongNumber(vartok->strAt(2)); 
 196                     if ((num & num2) != ((bitop=='&') ? num2 : num)) { 
 197                         const std::string& op(vartok->strAt(1)); 
 198                         const bool alwaysTrue = op == "!="; 
 199                         const std::string condition(vartok->str() + op + vartok->strAt(2)); 
 200                         assignIfError(assignTok, tok2, condition, alwaysTrue); 
 201                     } 
 202                 } 
 203                 if (Token::Match(tok2, "%varid% %op%", varid) && tok2->next()->isAssignmentOp()) { 
 204                     return true; 
 205                 } 
 206             } 
 207  
 208             const bool ret1 = assignIfParseScope(assignTok, end->tokAt(2), varid, islocal, bitop, num); 
 209             bool ret2 = false; 
 210             if (Token::simpleMatch(end->next()->link(), "} else {")) 
 211                 ret2 = assignIfParseScope(assignTok, end->next()->link()->tokAt(3), varid, islocal, bitop, num); 
 212             if (ret1 || ret2) 
 213                 return true; 
 214         } 
 215     } 
 216     return false; 
 217 }

 245 { 
 246     if (tok->astOperand1() && tok->astOperand1()->isNumber()) 
 247         numchildren.push_back(MathLib::toLongNumber(tok->astOperand1()->str())); 
 248     else if (tok->astOperand1() && tok->str() == tok->astOperand1()->str()) 
 249         getnumchildren(tok->astOperand1(), numchildren); 
 250     if (tok->astOperand2() && tok->astOperand2()->isNumber()) 
 251         numchildren.push_back(MathLib::toLongNumber(tok->astOperand2()->str())); 
 252     else if (tok->astOperand2() && tok->str() == tok->astOperand2()->str()) 
 253         getnumchildren(tok->astOperand2(), numchildren);

 256 /* Return whether tok is in the body for a function returning a boolean. */ 
 257 static bool inBooleanFunction(const Token *tok) 
 258 { 
 259     const Scope *scope = tok ? tok->scope() : nullptr; 
 260     while (scope && scope->isLocal()) 
 261         scope = scope->nestedIn; 
 262     if (scope && scope->type == Scope::eFunction) { 
 263         const Function *func = scope->function; 
 264         if (func) { 
 265             const Token *ret = func->retDef; 
 266             while (Token::Match(ret, "static|const")) 
 267                 ret = ret->next(); 
 268             return Token::Match(ret, "bool|_Bool"); 
 269         } 
 270     } 
 271     return false; 
 272 }

 274 void CheckCondition::checkBadBitmaskCheck() 
 275 { 
 276     if (!mSettings->isEnabled(Settings::WARNING)) 
 277         return; 
 278  
 279     for (const Token *tok = mTokenizer->tokens(); tok; tok = tok->next()) { 
 280         if (tok->str() == "|" && tok->astOperand1() && tok->astOperand2() && tok->astParent()) { 
 281             const Token* parent = tok->astParent(); 
 282             const bool isBoolean = Token::Match(parent, "&&|%oror%") || 
 283                                    (parent->str() == "?" && parent->astOperand1() == tok) || 
 284                                    (parent->str() == "=" && parent->astOperand2() == tok && parent->astOperand1() && parent->astOperand1()->variable() && Token::Match(parent->astOperand1()->variable()->typeStartToken(), "bool|_Bool")) || 
 285                                    (parent->str() == "(" && Token::Match(parent->astOperand1(), "if|while")) || 
 286                                    (parent->str() == "return" && parent->astOperand1() == tok && inBooleanFunction(tok)); 
 287  
 288             const bool isTrue = (tok->astOperand1()->hasKnownIntValue() && tok->astOperand1()->values().front().intvalue != 0) || 
 289                                 (tok->astOperand2()->hasKnownIntValue() && tok->astOperand2()->values().front().intvalue != 0); 
 290  
 291             if (isBoolean && isTrue) 
 292                 badBitmaskCheckError(tok); 
 293         } 
 294     } 
 295 }

 302 void CheckCondition::comparison() 
 303 { 
 304     if (!mSettings->isEnabled(Settings::STYLE)) 
 305         return; 
 306  
 307     for (const Token *tok = mTokenizer->tokens(); tok; tok = tok->next()) { 
 308         if (!tok->isComparisonOp()) 
 309             continue; 
 310  
 311         const Token *expr1 = tok->astOperand1(); 
 312         const Token *expr2 = tok->astOperand2(); 
 313         if (!expr1 || !expr2) 
 314             continue; 
 315         if (expr1->isNumber()) 
 316             std::swap(expr1,expr2); 
 317         if (!expr2->isNumber()) 
 318             continue; 
 319         const MathLib::bigint num2 = MathLib::toLongNumber(expr2->str()); 
 320         if (num2 < 0) 
 321             continue; 
 322         if (!Token::Match(expr1,"[&|]")) 
 323             continue; 
 324         std::list<MathLib::bigint> numbers; 
 325         getnumchildren(expr1, numbers); 
 326         for (const MathLib::bigint num1 : numbers) { 
 327             if (num1 < 0) 
 328                 continue; 
 329             if (Token::Match(tok, "==|!=")) { 
 330                 if ((expr1->str() == "&" && (num1 & num2) != num2) || 
 331                     (expr1->str() == "|" && (num1 | num2) != num2)) { 
 332                     const std::string& op(tok->str()); 
 333                     comparisonError(expr1, expr1->str(), num1, op, num2, op=="==" ? false : true); 
 334                 } 
 335             } else if (expr1->str() == "&") { 
 336                 const bool or_equal = Token::Match(tok, ">=|<="); 
 337                 const std::string& op(tok->str()); 
 338                 if ((Token::Match(tok, ">=|<")) && (num1 < num2)) { 
 339                     comparisonError(expr1, expr1->str(), num1, op, num2, or_equal ? false : true); 
 340                 } else if ((Token::Match(tok, "<=|>")) && (num1 <= num2)) { 
 341                     comparisonError(expr1, expr1->str(), num1, op, num2, or_equal ? true : false); 
 342                 } 
 343             } else if (expr1->str() == "|") { 
 344                 if ((expr1->astOperand1()->valueType()) && 
 345                     (expr1->astOperand1()->valueType()->sign == ValueType::Sign::UNSIGNED)) { 
 346                     const bool or_equal = Token::Match(tok, ">=|<="); 
 347                     const std::string& op(tok->str()); 
 348                     if ((Token::Match(tok, ">=|<")) && (num1 >= num2)) { 
 349                         //"(a | 0x07) >= 7U" is always true for unsigned a 
 350                         //"(a | 0x07) < 7U" is always false for unsigned a 
 351                         comparisonError(expr1, expr1->str(), num1, op, num2, or_equal ? true : false); 
 352                     } else if ((Token::Match(tok, "<=|>")) && (num1 > num2)) { 
 353                         //"(a | 0x08) <= 7U" is always false for unsigned a 
 354                         //"(a | 0x07) > 6U" is always true for unsigned a 
 355                         comparisonError(expr1, expr1->str(), num1, op, num2, or_equal ? false : true); 
 356                     } 
 357                 } 
 358             } 
 359         } 
 360     } 
 361 }

 377 bool CheckCondition::isOverlappingCond(const Token * const cond1, const Token * const cond2, bool pure) const 
 378 { 
 379     if (!cond1 || !cond2) 
 380         return false; 
 381  
 382     // same expressions 
 383     if (isSameExpression(mTokenizer->isCPP(), true, cond1, cond2, mSettings->library, pure, false)) 
 384         return true; 
 385  
 386     // bitwise overlap for example 'x&7' and 'x==1' 
 387     if (cond1->str() == "&" && cond1->astOperand1() && cond2->astOperand2()) { 
 388         const Token *expr1 = cond1->astOperand1(); 
 389         const Token *num1  = cond1->astOperand2(); 
 390         if (!num1) // unary operator& 
 391             return false; 
 392         if (!num1->isNumber()) 
 393             std::swap(expr1,num1); 
 394         if (!num1->isNumber() || MathLib::isNegative(num1->str())) 
 395             return false; 
 396  
 397         if (!Token::Match(cond2, "&|==") || !cond2->astOperand1() || !cond2->astOperand2()) 
 398             return false; 
 399         const Token *expr2 = cond2->astOperand1(); 
 400         const Token *num2  = cond2->astOperand2(); 
 401         if (!num2->isNumber()) 
 402             std::swap(expr2,num2); 
 403         if (!num2->isNumber() || MathLib::isNegative(num2->str())) 
 404             return false; 
 405  
 406         if (!isSameExpression(mTokenizer->isCPP(), true, expr1, expr2, mSettings->library, pure, false)) 
 407             return false; 
 408  
 409         const MathLib::bigint value1 = MathLib::toLongNumber(num1->str()); 
 410         const MathLib::bigint value2 = MathLib::toLongNumber(num2->str()); 
 411         if (cond2->str() == "&") 
 412             return ((value1 & value2) == value2); 
 413         return ((value1 & value2) > 0); 
 414     } 
 415     return false; 
 416 }

 418 void CheckCondition::duplicateCondition() 
 419 { 
 420     if (!mSettings->isEnabled(Settings::STYLE)) 
 421         return; 
 422  
 423     const SymbolDatabase *const symbolDatabase = mTokenizer->getSymbolDatabase(); 
 424  
 425     for (const Scope &scope : symbolDatabase->scopeList) { 
 426         if (scope.type != Scope::eIf) 
 427             continue; 
 428  
 429         const Token *cond1 = scope.classDef->next()->astOperand2(); 
 430         if (!cond1) 
 431             continue; 
 432         if (cond1->hasKnownIntValue()) 
 433             continue; 
 434  
 435         const Token *tok2 = scope.classDef->next(); 
 436         if (!tok2) 
 437             continue; 
 438         tok2 = tok2->link(); 
 439         if (!Token::simpleMatch(tok2, ") {")) 
 440             continue; 
 441         tok2 = tok2->linkAt(1); 
 442         if (!Token::simpleMatch(tok2, "} if (")) 
 443             continue; 
 444         const Token *cond2 = tok2->tokAt(2)->astOperand2(); 
 445         if (!cond2) 
 446             continue; 
 447  
 448         bool modified = false; 
 449         visitAstNodes(cond1, [&](const Token *tok3) { 
 450             if (tok3->varId() > 0 && 
 451                 isVariableChanged(scope.classDef->next(), cond2, tok3->varId(), false, mSettings, mTokenizer->isCPP())) { 
 452                 modified = true; 
 453                 return ChildrenToVisit::done; 
 454             } 
 455             return ChildrenToVisit::op1_and_op2; 
 456         }); 
 457         ErrorPath errorPath; 
 458         if (!modified && 
 459             isSameExpression(mTokenizer->isCPP(), true, cond1, cond2, mSettings->library, true, true, &errorPath)) 
 460             duplicateConditionError(cond1, cond2, errorPath); 
 461     } 
 462 }

 476 void CheckCondition::multiCondition() 
 477 { 
 478     if (!mSettings->isEnabled(Settings::STYLE)) 
 479         return; 
 480  
 481     const SymbolDatabase* const symbolDatabase = mTokenizer->getSymbolDatabase(); 
 482  
 483     for (const Scope &scope : symbolDatabase->scopeList) { 
 484         if (scope.type != Scope::eIf) 
 485             continue; 
 486  
 487         const Token * const cond1 = scope.classDef->next()->astOperand2(); 
 488  
 489         const Token * tok2 = scope.classDef->next(); 
 490         for (;;) { 
 491             tok2 = tok2->link(); 
 492             if (!Token::simpleMatch(tok2, ") {")) 
 493                 break; 
 494             tok2 = tok2->linkAt(1); 
 495             if (!Token::simpleMatch(tok2, "} else { if (")) 
 496                 break; 
 497             tok2 = tok2->tokAt(4); 
 498  
 499             if (cond1 && 
 500                 tok2->astOperand2() && 
 501                 !cond1->hasKnownIntValue() && 
 502                 !tok2->astOperand2()->hasKnownIntValue() && 
 503                 isOverlappingCond(cond1, tok2->astOperand2(), true)) 
 504                 multiConditionError(tok2, cond1->linenr()); 
 505         } 
 506     } 
 507 }

 524 static bool isNonConstFunctionCall(const Token *ftok, const Library &library) 
 525 { 
 526     if (library.isFunctionConst(ftok)) 
 527         return false; 
 528     const Token *obj = ftok->next()->astOperand1();

 530         obj = obj->astOperand1();

 532         return true; 
 533     else if (obj->variable() && obj->variable()->isConst()) 
 534         return false; 
 535     else if (ftok->function() && ftok->function()->isConst()) 
 536         return false; 
 537     return true; 
 538 }

 540 void CheckCondition::multiCondition2() 
 541 { 
 542     if (!mSettings->isEnabled(Settings::WARNING)) 
 543         return; 
 544  
 545     const SymbolDatabase *symbolDatabase = mTokenizer->getSymbolDatabase(); 
 546  
 547     for (const Scope &scope : symbolDatabase->scopeList) { 
 548         const Token *condTok = nullptr; 
 549         if (scope.type == Scope::eIf || scope.type == Scope::eWhile) 
 550             condTok = scope.classDef->next()->astOperand2(); 
 551         else if (scope.type == Scope::eFor) { 
 552             condTok = scope.classDef->next()->astOperand2(); 
 553             if (!condTok || condTok->str() != ";") 
 554                 continue; 
 555             condTok = condTok->astOperand2(); 
 556             if (!condTok || condTok->str() != ";") 
 557                 continue; 
 558             condTok = condTok->astOperand1(); 
 559         } 
 560         if (!condTok) 
 561             continue; 
 562         const Token * const cond1 = condTok; 
 563  
 564         if (!Token::simpleMatch(scope.classDef->linkAt(1), ") {")) 
 565             continue; 
 566  
 567         bool nonConstFunctionCall = false; 
 568         bool nonlocal = false; // nonlocal variable used in condition 
 569         std::set<unsigned int> vars; // variables used in condition 
 570         visitAstNodes(condTok, 
 571         [&](const Token *cond) { 
 572             if (Token::Match(cond, "%name% (")) { 
 573                 nonConstFunctionCall = isNonConstFunctionCall(cond, mSettings->library); 
 574                 if (nonConstFunctionCall) 
 575                     return ChildrenToVisit::done; 
 576             } 
 577  
 578             if (cond->varId()) { 
 579                 vars.insert(cond->varId()); 
 580                 const Variable *var = cond->variable(); 
 581                 if (!nonlocal && var) { 
 582                     if (!(var->isLocal() || var->isArgument())) 
 583                         nonlocal = true; 
 584                     else if ((var->isPointer() || var->isReference()) && !Token::Match(cond->astParent(), "%oror%|&&|!")) 
 585                         // TODO: if var is pointer check what it points at 
 586                         nonlocal = true; 
 587                 } 
 588             } else if (!nonlocal && cond->isName()) { 
 589                 // varid is 0. this is possibly a nonlocal variable.. 
 590                 nonlocal = Token::Match(cond->astParent(), "%cop%|(|[") || Token::Match(cond, "%name% .") || (mTokenizer->isCPP() && cond->str() == "this"); 
 591             } else { 
 592                 return ChildrenToVisit::op1_and_op2; 
 593             } 
 594             return ChildrenToVisit::none; 
 595         }); 
 596  
 597         if (nonConstFunctionCall) 
 598             continue; 
 599  
 600         // parse until second condition is reached.. 
 601         enum MULTICONDITIONTYPE { INNER, AFTER }; 
 602         const Token *tok; 
 603  
 604         // Parse inner condition first and then early return condition 
 605         std::vector<MULTICONDITIONTYPE> types = {MULTICONDITIONTYPE::INNER}; 
 606         if (Token::Match(scope.bodyStart, "{ return|throw|continue|break")) 
 607             types.push_back(MULTICONDITIONTYPE::AFTER); 
 608         for (MULTICONDITIONTYPE type:types) { 
 609             if (type == MULTICONDITIONTYPE::AFTER) { 
 610                 tok = scope.bodyEnd->next(); 
 611             } else { 
 612                 tok = scope.bodyStart; 
 613             } 
 614             const Token * const endToken = tok->scope()->bodyEnd; 
 615  
 616             for (; tok && tok != endToken; tok = tok->next()) { 
 617                 if (Token::Match(tok, "if|return")) { 
 618                     const Token * condStartToken = tok->str() == "if" ? tok->next() : tok; 
 619                     const Token * condEndToken = tok->str() == "if" ? condStartToken->link() : Token::findsimplematch(condStartToken, ";"); 
 620                     // Does condition modify tracked variables? 
 621                     if (const Token *op = Token::findmatch(tok, "++|--", condEndToken)) { 
 622                         bool bailout = false; 
 623                         while (op) { 
 624                             if (vars.find(op->astOperand1()->varId()) != vars.end()) { 
 625                                 bailout = true; 
 626                                 break; 
 627                             } 
 628                             if (nonlocal && op->astOperand1()->varId() == 0) { 
 629                                 bailout = true; 
 630                                 break; 
 631                             } 
 632                             op = Token::findmatch(op->next(), "++|--", condEndToken); 
 633                         } 
 634                         if (bailout) 
 635                             break; 
 636                     } 
 637  
 638                     // Condition.. 
 639                     const Token *cond2 = tok->str() == "if" ? condStartToken->astOperand2() : condStartToken->astOperand1(); 
 640                     const bool isReturnVar = (tok->str() == "return" && !Token::Match(cond2, "%cop%")); 
 641  
 642                     ErrorPath errorPath; 
 643  
 644                     if (type == MULTICONDITIONTYPE::INNER) { 
 645                         std::stack<const Token *> tokens1; 
 646                         tokens1.push(cond1); 
 647                         while (!tokens1.empty()) { 
 648                             const Token *firstCondition = tokens1.top(); 
 649                             tokens1.pop(); 
 650                             if (!firstCondition) 
 651                                 continue; 
 652                             if (firstCondition->str() == "&&") { 
 653                                 tokens1.push(firstCondition->astOperand1()); 
 654                                 tokens1.push(firstCondition->astOperand2()); 
 655                             } else if (!firstCondition->hasKnownIntValue()) { 
 656                                 if (!isReturnVar && isOppositeCond(false, mTokenizer->isCPP(), firstCondition, cond2, mSettings->library, true, true, &errorPath)) { 
 657                                     if (!isAliased(vars)) 
 658                                         oppositeInnerConditionError(firstCondition, cond2, errorPath); 
 659                                 } else if (!isReturnVar && isSameExpression(mTokenizer->isCPP(), true, firstCondition, cond2, mSettings->library, true, true, &errorPath)) { 
 660                                     identicalInnerConditionError(firstCondition, cond2, errorPath); 
 661                                 } 
 662                             } 
 663                         } 
 664                     } else { 
 665                         std::stack<const Token *> tokens2; 
 666                         tokens2.push(cond2); 
 667                         while (!tokens2.empty()) { 
 668                             const Token *secondCondition = tokens2.top(); 
 669                             tokens2.pop(); 
 670                             if (!secondCondition) 
 671                                 continue; 
 672                             if (secondCondition->str() == "||" || secondCondition->str() == "&&") { 
 673                                 tokens2.push(secondCondition->astOperand1()); 
 674                                 tokens2.push(secondCondition->astOperand2()); 
 675                             } else if ((!cond1->hasKnownIntValue() || !secondCondition->hasKnownIntValue()) && 
 676                                        isSameExpression(mTokenizer->isCPP(), true, cond1, secondCondition, mSettings->library, true, true, &errorPath)) { 
 677                                 if (!isAliased(vars)) 
 678                                     identicalConditionAfterEarlyExitError(cond1, secondCondition, errorPath); 
 679                             } 
 680                         } 
 681                     } 
 682                 } 
 683                 if (Token::Match(tok, "%type% (") && nonlocal && isNonConstFunctionCall(tok, mSettings->library)) // non const function call -> bailout if there are nonlocal variables 
 684                     break; 
 685                 if (Token::Match(tok, "case|break|continue|return|throw") && tok->scope() == endToken->scope()) 
 686                     break; 
 687                 if (Token::Match(tok, "[;{}] %name% :")) 
 688                     break; 
 689                 // bailout if loop is seen. 
 690                 // TODO: handle loops better. 
 691                 if (Token::Match(tok, "for|while|do")) { 
 692                     const Token *tok1 = tok->next(); 
 693                     const Token *tok2; 
 694                     if (Token::simpleMatch(tok, "do {")) { 
 695                         if (!Token::simpleMatch(tok->linkAt(1), "} while (")) 
 696                             break; 
 697                         tok2 = tok->linkAt(1)->linkAt(2); 
 698                     } else if (Token::Match(tok, "if|while (")) { 
 699                         tok2 = tok->linkAt(1); 
 700                         if (Token::simpleMatch(tok2, ") {")) 
 701                             tok2 = tok2->linkAt(1); 
 702                         if (!tok2) 
 703                             break; 
 704                     } else { 
 705                         // Incomplete code 
 706                         break; 
 707                     } 
 708                     bool changed = false; 
 709                     for (unsigned int varid : vars) { 
 710                         if (isVariableChanged(tok1, tok2, varid, nonlocal, mSettings, mTokenizer->isCPP())) { 
 711                             changed = true; 
 712                             break; 
 713                         } 
 714                     } 
 715                     if (changed) 
 716                         break; 
 717                 } 
 718                 if ((tok->varId() && vars.find(tok->varId()) != vars.end()) || 
 719                     (!tok->varId() && nonlocal)) { 
 720                     if (Token::Match(tok, "%name% %assign%|++|--")) 
 721                         break; 
 722                     if (Token::Match(tok->astParent(), "*|.|[")) { 
 723                         const Token *parent = tok; 
 724                         while (Token::Match(parent->astParent(), ".|[") || (parent->astParent() && parent->astParent()->isUnaryOp("*"))) 
 725                             parent = parent->astParent(); 
 726                         if (Token::Match(parent->astParent(), "%assign%|++|--")) 
 727                             break; 
 728                     } 
 729                     if (mTokenizer->isCPP() && Token::Match(tok, "%name% <<") && (!tok->valueType() || !tok->valueType()->isIntegral())) 
 730                         break; 
 731                     if (isLikelyStreamRead(mTokenizer->isCPP(), tok->next()) || isLikelyStreamRead(mTokenizer->isCPP(), tok->previous())) 
 732                         break; 
 733                     if (Token::Match(tok, "%name% [")) { 
 734                         const Token *tok2 = tok->linkAt(1); 
 735                         while (Token::simpleMatch(tok2, "] [")) 
 736                             tok2 = tok2->linkAt(1); 
 737                         if (Token::Match(tok2, "] %assign%|++|--")) 
 738                             break; 
 739                     } 
 740                     if (Token::Match(tok->previous(), "++|--|& %name%")) 
 741                         break; 
 742                     if (tok->variable() && 
 743                         !tok->variable()->isConst() && 
 744                         Token::Match(tok, "%name% . %name% (")) { 
 745                         const Function* function = tok->tokAt(2)->function(); 
 746                         if (!function || !function->isConst()) 
 747                             break; 
 748                     } 
 749                     if (Token::Match(tok->previous(), "[(,] %name% [,)]") && isParameterChanged(tok)) 
 750                         break; 
 751                 } 
 752             } 
 753         } 
 754     } 
 755 }

 838 { 
 839     return (op == "==" && value1 == value2) || 
 840            (op == "!=" && value1 != value2) || 
 841            (op == ">"  && value1 >  value2) || 
 842            (op == ">=" && value1 >= value2) || 
 843            (op == "<"  && value1 <  value2) || 
 844            (op == "<=" && value1 <= value2);

 872 template<class T> 
 873 static inline T getvalue(const int test, const T value1, const T value2) 
 874 { 
 875     // test: 
 876     // 1 => return value that is less than both value1 and value2 
 877     // 2 => return value1 
 878     // 3 => return value that is between value1 and value2 
 879     // 4 => return value2 
 880     // 5 => return value that is larger than both value1 and value2 
 881     switch (test) { 
 882     case 1: 
 883         return std::numeric_limits<T>::lowest(); 
 884     case 2: 
 885         return value1; 
 886     case 3: 
 887         return getvalue3<T>(value1, value2); 
 888     case 4: 
 889         return value2; 
 890     case 5: 
 891         return std::numeric_limits<T>::max(); 
 892     }; 
 893     return 0; 
 894 }

 896 static bool parseComparison(const Token *comp, bool *not1, std::string *op, std::string *value, const Token **expr, bool* inconclusive) 
 897 { 
 898     *not1 = false; 
 899     while (comp && comp->str() == "!") { 
 900         *not1 = !(*not1); 
 901         comp = comp->astOperand1(); 
 902     } 
 903  
 904     if (!comp) 
 905         return false; 
 906  
 907     const Token* op1 = comp->astOperand1(); 
 908     const Token* op2 = comp->astOperand2(); 
 909     if (!comp->isComparisonOp() || !op1 || !op2) { 
 910         *op = "!="; 
 911         *value = "0"; 
 912         *expr = comp; 
 913     } else if (op1->isLiteral()) { 
 914         if (op1->isExpandedMacro()) 
 915             return false; 
 916         *op = invertOperatorForOperandSwap(comp->str()); 
 917         if (op1->enumerator() && op1->enumerator()->value_known) 
 918             *value = MathLib::toString(op1->enumerator()->value); 
 919         else 
 920             *value = op1->str(); 
 921         *expr = op2; 
 922     } else if (comp->astOperand2()->isLiteral()) { 
 923         if (op2->isExpandedMacro()) 
 924             return false; 
 925         *op = comp->str(); 
 926         if (op2->enumerator() && op2->enumerator()->value_known) 
 927             *value = MathLib::toString(op2->enumerator()->value); 
 928         else 
 929             *value = op2->str(); 
 930         *expr = op1; 
 931     } else { 
 932         *op = "!="; 
 933         *value = "0"; 
 934         *expr = comp; 
 935     } 
 936  
 937     *inconclusive = *inconclusive || ((*value)[0] == '\'' && !(*op == "!=" || *op == "==")); 
 938  
 939     // Only float and int values are currently handled 
 940     if (!MathLib::isInt(*value) && !MathLib::isFloat(*value) && (*value)[0] != '\'') 
 941         return false; 
 942  
 943     return true; 
 944 }

 946 static std::string conditionString(bool not1, const Token *expr1, const std::string &op, const std::string &value1) 
 947 { 
 948     if (expr1->astParent()->isComparisonOp()) 
 949         return std::string(not1 ? "!(" : "") + 
 950                (expr1->isName() ? expr1->str() : std::string("EXPR")) + 
 951                " " + 
 952                op + 
 953                " " + 
 954                value1 + 
 955                (not1 ? ")" : ""); 
 956  
 957     return std::string(not1 ? "!" : "") + 
 958            (expr1->isName() ? expr1->str() : std::string("EXPR")); 
 959 }

 961 static std::string conditionString(const Token * tok) 
 962 { 
 963     if (!tok) 
 964         return ""; 
 965     if (tok->isComparisonOp()) { 
 966         bool inconclusive = false; 
 967         bool not_; 
 968         std::string op, value; 
 969         const Token *expr; 
 970         if (parseComparison(tok, &not_, &op, &value, &expr, &inconclusive) && expr->isName()) { 
 971             return conditionString(not_, expr, op, value); 
 972         } 
 973     } 
 974     if (Token::Match(tok, "%cop%|&&|%oror%")) { 
 975         if (tok->astOperand2()) 
 976             return conditionString(tok->astOperand1()) + " " + tok->str() + " " + conditionString(tok->astOperand2()); 
 977         return tok->str() + "(" + conditionString(tok->astOperand1()) + ")"; 
 978  
 979     } 
 980     return tok->expressionString(); 
 981 }

 983 void CheckCondition::checkIncorrectLogicOperator() 
 984 { 
 985     const bool printStyle = mSettings->isEnabled(Settings::STYLE); 
 986     const bool printWarning = mSettings->isEnabled(Settings::WARNING); 
 987     if (!printWarning && !printStyle) 
 988         return; 
 989     const bool printInconclusive = mSettings->inconclusive; 
 990  
 991     const SymbolDatabase *symbolDatabase = mTokenizer->getSymbolDatabase(); 
 992     for (const Scope * scope : symbolDatabase->functionScopes) { 
 993  
 994         for (const Token* tok = scope->bodyStart->next(); tok != scope->bodyEnd; tok = tok->next()) { 
 995             if (!Token::Match(tok, "%oror%|&&") || !tok->astOperand1() || !tok->astOperand2()) 
 996                 continue; 
 997  
 998  
 999             // 'A && (!A || B)' is equivalent to 'A && B' 
1000             // 'A || (!A && B)' is equivalent to 'A || B' 
1001             if (printStyle && 
1002                 ((tok->str() == "||" && tok->astOperand2()->str() == "&&") || 
1003                  (tok->str() == "&&" && tok->astOperand2()->str() == "||"))) { 
1004                 const Token* tok2 = tok->astOperand2()->astOperand1(); 
1005                 if (isOppositeCond(true, mTokenizer->isCPP(), tok->astOperand1(), tok2, mSettings->library, true, false)) { 
1006                     std::string expr1(tok->astOperand1()->expressionString()); 
1007                     std::string expr2(tok->astOperand2()->astOperand1()->expressionString()); 
1008                     std::string expr3(tok->astOperand2()->astOperand2()->expressionString()); 
1009                     // make copy for later because the original string might get overwritten 
1010                     const std::string expr1VerboseMsg = expr1; 
1011                     const std::string expr2VerboseMsg = expr2; 
1012                     const std::string expr3VerboseMsg = expr3; 
1013  
1014                     if (expr1.length() + expr2.length() + expr3.length() > 50U) { 
1015                         if (expr1[0] == '!' && expr2[0] != '!') { 
1016                             expr1 = "!A"; 
1017                             expr2 = "A"; 
1018                         } else { 
1019                             expr1 = "A"; 
1020                             expr2 = "!A"; 
1021                         } 
1022  
1023                         expr3 = "B"; 
1024                     } 
1025  
1026                     const std::string cond1 = expr1 + " " + tok->str() + " (" + expr2 + " " + tok->astOperand2()->str() + " " + expr3 + ")"; 
1027                     const std::string cond2 = expr1 + " " + tok->str() + " " + expr3; 
1028  
1029                     const std::string cond1VerboseMsg = expr1VerboseMsg + " " + tok->str() + " " + expr2VerboseMsg + " " + tok->astOperand2()->str() + " " + expr3VerboseMsg; 
1030                     const std::string cond2VerboseMsg = expr1VerboseMsg + " " + tok->str() + " " + expr3VerboseMsg; 
1031                     // for the --verbose message, transform the actual condition and print it 
1032                     const std::string msg = tok2->expressionString() + ". '" + cond1 + "' is equivalent to '" + cond2 + "'\n" 
1033                                             "The condition '" + cond1VerboseMsg + "' is equivalent to '" + cond2VerboseMsg + "'."; 
1034                     redundantConditionError(tok, msg, false); 
1035                     continue; 
1036                 } 
1037             } 
1038  
1039             // Comparison #1 (LHS) 
1040             const Token *comp1 = tok->astOperand1(); 
1041             if (comp1 && comp1->str() == tok->str()) 
1042                 comp1 = comp1->astOperand2(); 
1043  
1044             // Comparison #2 (RHS) 
1045             const Token *comp2 = tok->astOperand2(); 
1046  
1047             bool inconclusive = false; 
1048             bool parseable = true; 
1049  
1050             // Parse LHS 
1051             bool not1; 
1052             std::string op1, value1; 
1053             const Token *expr1 = nullptr; 
1054             parseable &= (parseComparison(comp1, &not1, &op1, &value1, &expr1, &inconclusive)); 
1055  
1056             // Parse RHS 
1057             bool not2; 
1058             std::string op2, value2; 
1059             const Token *expr2 = nullptr; 
1060             parseable &= (parseComparison(comp2, &not2, &op2, &value2, &expr2, &inconclusive)); 
1061  
1062             if (inconclusive && !printInconclusive) 
1063                 continue; 
1064  
1065             const bool isfloat = astIsFloat(expr1, true) || MathLib::isFloat(value1) || astIsFloat(expr2, true) || MathLib::isFloat(value2); 
1066  
1067             ErrorPath errorPath; 
1068  
1069             // Opposite comparisons around || or && => always true or always false 
1070             if (!isfloat && isOppositeCond(tok->str() == "||", mTokenizer->isCPP(), tok->astOperand1(), tok->astOperand2(), mSettings->library, true, true, &errorPath)) { 
1071  
1072                 const bool alwaysTrue(tok->str() == "||"); 
1073                 incorrectLogicOperatorError(tok, conditionString(tok), alwaysTrue, inconclusive, errorPath); 
1074                 continue; 
1075             } 
1076  
1077             if (!parseable) 
1078                 continue; 
1079  
1080             if (isSameExpression(mTokenizer->isCPP(), true, comp1, comp2, mSettings->library, true, true)) 
1081                 continue; // same expressions => only report that there are same expressions 
1082             if (!isSameExpression(mTokenizer->isCPP(), true, expr1, expr2, mSettings->library, true, true)) 
1083                 continue; 
1084  
1085  
1086             // don't check floating point equality comparisons. that is bad 
1087             // and deserves different warnings. 
1088             if (isfloat && (op1 == "==" || op1 == "!=" || op2 == "==" || op2 == "!=")) 
1089                 continue; 
1090  
1091  
1092             const double d1 = (isfloat) ? MathLib::toDoubleNumber(value1) : 0; 
1093             const double d2 = (isfloat) ? MathLib::toDoubleNumber(value2) : 0; 
1094             const MathLib::bigint i1 = (isfloat) ? 0 : MathLib::toLongNumber(value1); 
1095             const MathLib::bigint i2 = (isfloat) ? 0 : MathLib::toLongNumber(value2); 
1096             const bool useUnsignedInt = (std::numeric_limits<MathLib::bigint>::max()==i1)||(std::numeric_limits<MathLib::bigint>::max()==i2); 
1097             const MathLib::biguint u1 = (useUnsignedInt) ? MathLib::toLongNumber(value1) : 0; 
1098             const MathLib::biguint u2 = (useUnsignedInt) ? MathLib::toLongNumber(value2) : 0; 
1099             // evaluate if expression is always true/false 
1100             bool alwaysTrue = true, alwaysFalse = true; 
1101             bool firstTrue = true, secondTrue = true; 
1102             for (int test = 1; test <= 5; ++test) { 
1103                 // test: 
1104                 // 1 => testvalue is less than both value1 and value2 
1105                 // 2 => testvalue is value1 
1106                 // 3 => testvalue is between value1 and value2 
1107                 // 4 => testvalue value2 
1108                 // 5 => testvalue is larger than both value1 and value2 
1109                 bool result1, result2; 
1110                 if (isfloat) { 
1111                     const double testvalue = getvalue<double>(test, d1, d2); 
1112                     result1 = checkFloatRelation(op1, testvalue, d1); 
1113                     result2 = checkFloatRelation(op2, testvalue, d2); 
1114                 } else if (useUnsignedInt) { 
1115                     const MathLib::biguint testvalue = getvalue<MathLib::biguint>(test, u1, u2); 
1116                     result1 = checkIntRelation(op1, testvalue, u1); 
1117                     result2 = checkIntRelation(op2, testvalue, u2); 
1118                 } else { 
1119                     const MathLib::bigint testvalue = getvalue<MathLib::bigint>(test, i1, i2); 
1120                     result1 = checkIntRelation(op1, testvalue, i1); 
1121                     result2 = checkIntRelation(op2, testvalue, i2); 
1122                 } 
1123                 if (not1)

1247                         tok2 = tok2->link();

1298 void CheckCondition::alwaysTrueFalse() 
1299 { 
1300     if (!mSettings->isEnabled(Settings::STYLE)) 
1301         return; 
1302  
1303     const SymbolDatabase *symbolDatabase = mTokenizer->getSymbolDatabase(); 
1304     for (const Scope * scope : symbolDatabase->functionScopes) { 
1305         for (const Token* tok = scope->bodyStart->next(); tok != scope->bodyEnd; tok = tok->next()) { 
1306  
1307             if (tok->link()) // don't write false positives when templates are used 
1308                 continue; 
1309             if (!tok->hasKnownIntValue()) 
1310                 continue; 
1311             // Skip already diagnosed values 
1312             if (diag(tok, false)) 
1313                 continue; 
1314             if (Token::Match(tok, "%num%|%bool%|%char%")) 
1315                 continue; 
1316             if (Token::Match(tok, "! %num%|%bool%|%char%")) 
1317                 continue; 
1318             if (Token::Match(tok, "%oror%|&&|:")) 
1319                 continue; 
1320             if (Token::Match(tok, "%comp%") && isSameExpression(mTokenizer->isCPP(), true, tok->astOperand1(), tok->astOperand2(), mSettings->library, true, true)) 
1321                 continue; 
1322  
1323             const bool constIfWhileExpression = 
1324                 tok->astParent() && Token::Match(tok->astTop()->astOperand1(), "if|while") && 
1325                 (Token::Match(tok->astParent(), "%oror%|&&") || Token::Match(tok->astParent()->astOperand1(), "if|while")); 
1326             const bool constValExpr = tok->isNumber() && Token::Match(tok->astParent(),"%oror%|&&|?"); // just one number in boolean expression 
1327             const bool compExpr = Token::Match(tok, "%comp%|!"); // a compare expression 
1328             const bool returnStatement = Token::simpleMatch(tok->astTop(), "return") && 
1329                                          Token::Match(tok->astParent(), "%oror%|&&|return"); 
1330  
1331             if (!(constIfWhileExpression || constValExpr || compExpr || returnStatement)) 
1332                 continue; 
1333  
1334             if (returnStatement && scope->function && !Token::simpleMatch(scope->function->retDef, "bool")) 
1335                 continue; 
1336  
1337             if (returnStatement && isConstVarExpression(tok)) 
1338                 continue; 
1339  
1340             if (returnStatement && Token::simpleMatch(tok->astParent(), "return") && tok->variable() && ( 
1341                     !tok->variable()->isLocal() || 
1342                     tok->variable()->isReference() || 
1343                     tok->variable()->isConst() || 
1344                     !isVariableChanged(tok->variable(), mSettings, mTokenizer->isCPP()))) 
1345                 continue; 
1346  
1347             // Don't warn in assertions. Condition is often 'always true' by intention. 
1348             // If platform,defines,etc cause 'always false' then that is not dangerous neither. 
1349             bool assertFound = false; 
1350             for (const Token * tok2 = tok->astParent(); tok2 ; tok2 = tok2->astParent()) { // move backwards and try to find "assert" 
1351                 if (tok2->str() == "(" && tok2->astOperand2()) { 
1352                     const std::string& str = tok2->previous()->str(); 
1353                     if ((str.find("assert")!=std::string::npos || str.find("ASSERT")!=std::string::npos)) 
1354                         assertFound = true; 
1355                     break; 
1356                 } 
1357             } 
1358             if (assertFound) 
1359                 continue; 
1360  
1361             // Don't warn when there are expanded macros.. 
1362             bool isExpandedMacro = false; 
1363             std::stack<const Token*> tokens; 
1364             tokens.push(tok); 
1365             while (!tokens.empty()) { 
1366                 const Token *tok2 = tokens.top(); 
1367                 tokens.pop(); 
1368                 if (!tok2) 
1369                     continue; 
1370                 tokens.push(tok2->astOperand1()); 
1371                 tokens.push(tok2->astOperand2()); 
1372                 if (tok2->isExpandedMacro()) { 
1373                     isExpandedMacro = true; 
1374                     break; 
1375                 } 
1376             } 
1377             if (isExpandedMacro) 
1378                 continue; 
1379             for (const Token *parent = tok; parent; parent = parent->astParent()) { 
1380                 if (parent->isExpandedMacro()) { 
1381                     isExpandedMacro = true; 
1382                     break; 
1383                 } 
1384             } 
1385             if (isExpandedMacro) 
1386                 continue; 
1387  
1388             // don't warn when condition checks sizeof result 
1389             bool hasSizeof = false; 
1390             tokens.push(tok); 
1391             while (!tokens.empty()) { 
1392                 const Token *tok2 = tokens.top(); 
1393                 tokens.pop(); 
1394                 if (!tok2) 
1395                     continue; 
1396                 if (tok2->isNumber()) 
1397                     continue; 
1398                 if (Token::simpleMatch(tok2->previous(), "sizeof (")) { 
1399                     hasSizeof = true; 
1400                     continue; 
1401                 } 
1402                 if (tok2->isComparisonOp() || tok2->isArithmeticalOp()) { 
1403                     tokens.push(tok2->astOperand1()); 
1404                     tokens.push(tok2->astOperand2()); 
1405                 } else 
1406                     break; 
1407             } 
1408             if (tokens.empty() && hasSizeof) 
1409                 continue; 
1410  
1411             alwaysTrueFalseError(tok, &tok->values().front()); 
1412         } 
1413     } 
1414 }

1430 { 
1431     if (!mSettings->isEnabled(Settings::WARNING)) 
1432         return; 
1433  
1434     const SymbolDatabase *symbolDatabase = mTokenizer->getSymbolDatabase(); 
1435     for (const Scope * scope : symbolDatabase->functionScopes) { 
1436  
1437         for (const Token* tok = scope->bodyStart; tok != scope->bodyEnd; tok = tok->next()) { 
1438             if (!tok->isComparisonOp() || !tok->astOperand1() || !tok->astOperand2()) 
1439                 continue; 
1440  
1441             const Token *calcToken, *exprToken; 
1442             bool result; 
1443             if (Token::Match(tok, "<|>=") && tok->astOperand1()->str() == "+") { 
1444                 calcToken = tok->astOperand1(); 
1445                 exprToken = tok->astOperand2(); 
1446                 result = (tok->str() == ">="); 
1447             } else if (Token::Match(tok, ">|<=") && tok->astOperand2()->str() == "+") { 
1448                 calcToken = tok->astOperand2(); 
1449                 exprToken = tok->astOperand1(); 
1450                 result = (tok->str() == "<="); 
1451             } else 
1452                 continue; 
1453  
1454             // Only warn for signed integer overflows and pointer overflows. 
1455             if (!(calcToken->valueType() && (calcToken->valueType()->pointer || calcToken->valueType()->sign == ValueType::Sign::SIGNED))) 
1456                 continue; 
1457             if (!(exprToken->valueType() && (exprToken->valueType()->pointer || exprToken->valueType()->sign == ValueType::Sign::SIGNED))) 
1458                 continue; 
1459  
1460             const Token *termToken; 
1461             if (isSameExpression(mTokenizer->isCPP(), true, exprToken, calcToken->astOperand1(), mSettings->library, true, false)) 
1462                 termToken = calcToken->astOperand2(); 
1463             else if (isSameExpression(mTokenizer->isCPP(), true, exprToken, calcToken->astOperand2(), mSettings->library, true, false)) 
1464                 termToken = calcToken->astOperand1(); 
1465             else 
1466                 continue; 
1467  
1468             if (!termToken) 
1469                 continue; 
1470  
1471             // Only warn when termToken is always positive 
1472             if (termToken->valueType() && termToken->valueType()->sign == ValueType::Sign::UNSIGNED) 
1473                 invalidTestForOverflow(tok, result); 
1474             else if (termToken->isNumber() && MathLib::isPositive(termToken->str())) 
1475                 invalidTestForOverflow(tok, result); 
1476         } 
1477     }

 120 { 
 121     bool addressOf = Token::Match(partok, "[(,] &");

 146 bool CheckCondition::assignIfParseScope(const Token * const assignTok, 
 147                                         const Token * const startTok, 
 148                                         const unsigned int varid, 
 149                                         const bool islocal, 
 150                                         const char bitop, 
 151                                         const MathLib::bigint num) 
 152 { 
 153     bool ret = false;

 155     for (const Token *tok2 = startTok; tok2; tok2 = tok2->next()) { 
 156         if ((bitop == '&') && Token::Match(tok2->tokAt(2), "%varid% %cop% %num% ;", varid) && tok2->strAt(3) == std::string(1U, bitop)) { 
 157             const MathLib::bigint num2 = MathLib::toLongNumber(tok2->strAt(4));

 162             return true; 
 163         } 
 164         if (Token::Match(tok2, "++|-- %varid%", varid) || Token::Match(tok2, "%varid% ++|--", varid)) 
 165             return true; 
 166         if (Token::Match(tok2, "[(,] &| %varid% [,)]", varid) && isParameterChanged(tok2)) 
 167             return true; 
 168         if (tok2->str() == "}") 
 169             return false; 
 170         if (Token::Match(tok2, "break|continue|return")) 
 171             ret = true; 
 172         if (ret && tok2->str() == ";") 
 173             return false; 
 174         if (!islocal && Token::Match(tok2, "%name% (") && !Token::simpleMatch(tok2->next()->link(), ") {")) 
 175             return true; 
 176         if (Token::Match(tok2, "if|while (")) { 
 177             if (!islocal && tok2->str() == "while") 
 178                 continue; 
 179             if (tok2->str() == "while") { 
 180                 // is variable changed in loop? 
 181                 const Token *bodyStart = tok2->linkAt(1)->next(); 
 182                 const Token *bodyEnd   = bodyStart ? bodyStart->link() : nullptr; 
 183                 if (!bodyEnd || bodyEnd->str() != "}" || isVariableChanged(bodyStart, bodyEnd, varid, !islocal, mSettings, mTokenizer->isCPP())) 
 184                     continue; 
 185             } 
 186  
 187             // parse condition 
 188             const Token * const end = tok2->next()->link(); 
 189             for (; tok2 != end; tok2 = tok2->next()) { 
 190                 if (Token::Match(tok2, "[(,] &| %varid% [,)]", varid)) { 
 191                     return true; 
 192                 } 
 193                 if (Token::Match(tok2,"&&|%oror%|( %varid% ==|!= %num% &&|%oror%|)", varid)) { 
 194                     const Token *vartok = tok2->next(); 
 195                     const MathLib::bigint num2 = MathLib::toLongNumber(vartok->strAt(2)); 
 196                     if ((num & num2) != ((bitop=='&') ? num2 : num)) { 
 197                         const std::string& op(vartok->strAt(1)); 
 198                         const bool alwaysTrue = op == "!="; 
 199                         const std::string condition(vartok->str() + op + vartok->strAt(2)); 
 200                         assignIfError(assignTok, tok2, condition, alwaysTrue); 
 201                     } 
 202                 } 
 203                 if (Token::Match(tok2, "%varid% %op%", varid) && tok2->next()->isAssignmentOp()) { 
 204                     return true; 
 205                 } 
 206             } 
 207  
 208             const bool ret1 = assignIfParseScope(assignTok, end->tokAt(2), varid, islocal, bitop, num); 
 209             bool ret2 = false; 
 210             if (Token::simpleMatch(end->next()->link(), "} else {")) 
 211                 ret2 = assignIfParseScope(assignTok, end->next()->link()->tokAt(3), varid, islocal, bitop, num); 
 212             if (ret1 || ret2) 
 213                 return true; 
 214         } 
 215     } 
 216     return false; 
 217 }

 301  
 302 void CheckCondition::comparison() 
 303 { 
 304     if (!mSettings->isEnabled(Settings::STYLE)) 
 305         return; 
 306  
 307     for (const Token *tok = mTokenizer->tokens(); tok; tok = tok->next()) { 
 308         if (!tok->isComparisonOp()) 
 309             continue; 
 310  
 311         const Token *expr1 = tok->astOperand1(); 
 312         const Token *expr2 = tok->astOperand2(); 
 313         if (!expr1 || !expr2) 
 314             continue; 
 315         if (expr1->isNumber()) 
 316             std::swap(expr1,expr2); 
 317         if (!expr2->isNumber()) 
 318             continue; 
 319         const MathLib::bigint num2 = MathLib::toLongNumber(expr2->str()); 
 320         if (num2 < 0) 
 321             continue; 
 322         if (!Token::Match(expr1,"[&|]")) 
 323             continue; 
 324         std::list<MathLib::bigint> numbers; 
 325         getnumchildren(expr1, numbers); 
 326         for (const MathLib::bigint num1 : numbers) { 
 327             if (num1 < 0) 
 328                 continue; 
 329             if (Token::Match(tok, "==|!=")) { 
 330                 if ((expr1->str() == "&" && (num1 & num2) != num2) || 
 331                     (expr1->str() == "|" && (num1 | num2) != num2)) { 
 332                     const std::string& op(tok->str()); 
 333                     comparisonError(expr1, expr1->str(), num1, op, num2, op=="==" ? false : true); 
 334                 } 
 335             } else if (expr1->str() == "&") { 
 336                 const bool or_equal = Token::Match(tok, ">=|<="); 
 337                 const std::string& op(tok->str()); 
 338                 if ((Token::Match(tok, ">=|<")) && (num1 < num2)) { 
 339                     comparisonError(expr1, expr1->str(), num1, op, num2, or_equal ? false : true); 
 340                 } else if ((Token::Match(tok, "<=|>")) && (num1 <= num2)) { 
 341                     comparisonError(expr1, expr1->str(), num1, op, num2, or_equal ? true : false); 
 342                 } 
 343             } else if (expr1->str() == "|") { 
 344                 if ((expr1->astOperand1()->valueType()) && 
 345                     (expr1->astOperand1()->valueType()->sign == ValueType::Sign::UNSIGNED)) { 
 346                     const bool or_equal = Token::Match(tok, ">=|<="); 
 347                     const std::string& op(tok->str()); 
 348                     if ((Token::Match(tok, ">=|<")) && (num1 >= num2)) { 
 349                         //"(a | 0x07) >= 7U" is always true for unsigned a 
 350                         //"(a | 0x07) < 7U" is always false for unsigned a 
 351                         comparisonError(expr1, expr1->str(), num1, op, num2, or_equal ? true : false); 
 352                     } else if ((Token::Match(tok, "<=|>")) && (num1 > num2)) { 
 353                         //"(a | 0x08) <= 7U" is always false for unsigned a 
 354                         //"(a | 0x07) > 6U" is always true for unsigned a 
 355                         comparisonError(expr1, expr1->str(), num1, op, num2, or_equal ? false : true); 
 356                     } 
 357                 } 
 358             } 
 359         } 
 360     } 
 361 }

 377 bool CheckCondition::isOverlappingCond(const Token * const cond1, const Token * const cond2, bool pure) const 
 378 { 
 379     if (!cond1 || !cond2) 
 380         return false; 
 381  
 382     // same expressions 
 383     if (isSameExpression(mTokenizer->isCPP(), true, cond1, cond2, mSettings->library, pure, false)) 
 384         return true; 
 385  
 386     // bitwise overlap for example 'x&7' and 'x==1' 
 387     if (cond1->str() == "&" && cond1->astOperand1() && cond2->astOperand2()) { 
 388         const Token *expr1 = cond1->astOperand1(); 
 389         const Token *num1  = cond1->astOperand2(); 
 390         if (!num1) // unary operator& 
 391             return false; 
 392         if (!num1->isNumber()) 
 393             std::swap(expr1,num1); 
 394         if (!num1->isNumber() || MathLib::isNegative(num1->str())) 
 395             return false; 
 396  
 397         if (!Token::Match(cond2, "&|==") || !cond2->astOperand1() || !cond2->astOperand2()) 
 398             return false; 
 399         const Token *expr2 = cond2->astOperand1(); 
 400         const Token *num2  = cond2->astOperand2(); 
 401         if (!num2->isNumber()) 
 402             std::swap(expr2,num2); 
 403         if (!num2->isNumber() || MathLib::isNegative(num2->str())) 
 404             return false; 
 405  
 406         if (!isSameExpression(mTokenizer->isCPP(), true, expr1, expr2, mSettings->library, pure, false)) 
 407             return false; 
 408  
 409         const MathLib::bigint value1 = MathLib::toLongNumber(num1->str()); 
 410         const MathLib::bigint value2 = MathLib::toLongNumber(num2->str()); 
 411         if (cond2->str() == "&") 
 412             return ((value1 & value2) == value2); 
 413         return ((value1 & value2) > 0); 
 414     } 
 415     return false; 
 416 }

 418 void CheckCondition::duplicateCondition() 
 419 { 
 420     if (!mSettings->isEnabled(Settings::STYLE)) 
 421         return; 
 422  
 423     const SymbolDatabase *const symbolDatabase = mTokenizer->getSymbolDatabase(); 
 424  
 425     for (const Scope &scope : symbolDatabase->scopeList) { 
 426         if (scope.type != Scope::eIf) 
 427             continue; 
 428  
 429         const Token *cond1 = scope.classDef->next()->astOperand2(); 
 430         if (!cond1) 
 431             continue; 
 432         if (cond1->hasKnownIntValue()) 
 433             continue; 
 434  
 435         const Token *tok2 = scope.classDef->next(); 
 436         if (!tok2) 
 437             continue; 
 438         tok2 = tok2->link(); 
 439         if (!Token::simpleMatch(tok2, ") {")) 
 440             continue; 
 441         tok2 = tok2->linkAt(1); 
 442         if (!Token::simpleMatch(tok2, "} if (")) 
 443             continue; 
 444         const Token *cond2 = tok2->tokAt(2)->astOperand2(); 
 445         if (!cond2) 
 446             continue; 
 447  
 448         bool modified = false; 
 449         visitAstNodes(cond1, [&](const Token *tok3) { 
 450             if (tok3->varId() > 0 && 
 451                 isVariableChanged(scope.classDef->next(), cond2, tok3->varId(), false, mSettings, mTokenizer->isCPP())) { 
 452                 modified = true; 
 453                 return ChildrenToVisit::done; 
 454             } 
 455             return ChildrenToVisit::op1_and_op2; 
 456         }); 
 457         ErrorPath errorPath; 
 458         if (!modified && 
 459             isSameExpression(mTokenizer->isCPP(), true, cond1, cond2, mSettings->library, true, true, &errorPath)) 
 460             duplicateConditionError(cond1, cond2, errorPath); 
 461     } 
 462 }

 479         return; 
 480

 482

 485             continue; 
 486

 539  
 540 void CheckCondition::multiCondition2() 
 541 { 
 542     if (!mSettings->isEnabled(Settings::WARNING)) 
 543         return; 
 544  
 545     const SymbolDatabase *symbolDatabase = mTokenizer->getSymbolDatabase(); 
 546  
 547     for (const Scope &scope : symbolDatabase->scopeList) { 
 548         const Token *condTok = nullptr; 
 549         if (scope.type == Scope::eIf || scope.type == Scope::eWhile) 
 550             condTok = scope.classDef->next()->astOperand2(); 
 551         else if (scope.type == Scope::eFor) { 
 552             condTok = scope.classDef->next()->astOperand2(); 
 553             if (!condTok || condTok->str() != ";") 
 554                 continue; 
 555             condTok = condTok->astOperand2(); 
 556             if (!condTok || condTok->str() != ";") 
 557                 continue; 
 558             condTok = condTok->astOperand1(); 
 559         } 
 560         if (!condTok) 
 561             continue; 
 562         const Token * const cond1 = condTok; 
 563  
 564         if (!Token::simpleMatch(scope.classDef->linkAt(1), ") {")) 
 565             continue; 
 566  
 567         bool nonConstFunctionCall = false; 
 568         bool nonlocal = false; // nonlocal variable used in condition 
 569         std::set<unsigned int> vars; // variables used in condition 
 570         visitAstNodes(condTok, 
 571         [&](const Token *cond) { 
 572             if (Token::Match(cond, "%name% (")) { 
 573                 nonConstFunctionCall = isNonConstFunctionCall(cond, mSettings->library); 
 574                 if (nonConstFunctionCall) 
 575                     return ChildrenToVisit::done; 
 576             } 
 577  
 578             if (cond->varId()) { 
 579                 vars.insert(cond->varId()); 
 580                 const Variable *var = cond->variable(); 
 581                 if (!nonlocal && var) { 
 582                     if (!(var->isLocal() || var->isArgument())) 
 583                         nonlocal = true; 
 584                     else if ((var->isPointer() || var->isReference()) && !Token::Match(cond->astParent(), "%oror%|&&|!")) 
 585                         // TODO: if var is pointer check what it points at 
 586                         nonlocal = true; 
 587                 } 
 588             } else if (!nonlocal && cond->isName()) { 
 589                 // varid is 0. this is possibly a nonlocal variable.. 
 590                 nonlocal = Token::Match(cond->astParent(), "%cop%|(|[") || Token::Match(cond, "%name% .") || (mTokenizer->isCPP() && cond->str() == "this"); 
 591             } else { 
 592                 return ChildrenToVisit::op1_and_op2; 
 593             } 
 594             return ChildrenToVisit::none; 
 595         }); 
 596  
 597         if (nonConstFunctionCall) 
 598             continue; 
 599  
 600         // parse until second condition is reached.. 
 601         enum MULTICONDITIONTYPE { INNER, AFTER }; 
 602         const Token *tok; 
 603  
 604         // Parse inner condition first and then early return condition 
 605         std::vector<MULTICONDITIONTYPE> types = {MULTICONDITIONTYPE::INNER}; 
 606         if (Token::Match(scope.bodyStart, "{ return|throw|continue|break")) 
 607             types.push_back(MULTICONDITIONTYPE::AFTER); 
 608         for (MULTICONDITIONTYPE type:types) { 
 609             if (type == MULTICONDITIONTYPE::AFTER) { 
 610                 tok = scope.bodyEnd->next(); 
 611             } else { 
 612                 tok = scope.bodyStart; 
 613             } 
 614             const Token * const endToken = tok->scope()->bodyEnd; 
 615  
 616             for (; tok && tok != endToken; tok = tok->next()) { 
 617                 if (Token::Match(tok, "if|return")) { 
 618                     const Token * condStartToken = tok->str() == "if" ? tok->next() : tok; 
 619                     const Token * condEndToken = tok->str() == "if" ? condStartToken->link() : Token::findsimplematch(condStartToken, ";"); 
 620                     // Does condition modify tracked variables? 
 621                     if (const Token *op = Token::findmatch(tok, "++|--", condEndToken)) { 
 622                         bool bailout = false; 
 623                         while (op) { 
 624                             if (vars.find(op->astOperand1()->varId()) != vars.end()) { 
 625                                 bailout = true; 
 626                                 break; 
 627                             } 
 628                             if (nonlocal && op->astOperand1()->varId() == 0) { 
 629                                 bailout = true; 
 630                                 break; 
 631                             } 
 632                             op = Token::findmatch(op->next(), "++|--", condEndToken); 
 633                         } 
 634                         if (bailout) 
 635                             break; 
 636                     } 
 637  
 638                     // Condition.. 
 639                     const Token *cond2 = tok->str() == "if" ? condStartToken->astOperand2() : condStartToken->astOperand1(); 
 640                     const bool isReturnVar = (tok->str() == "return" && !Token::Match(cond2, "%cop%")); 
 641  
 642                     ErrorPath errorPath; 
 643  
 644                     if (type == MULTICONDITIONTYPE::INNER) { 
 645                         std::stack<const Token *> tokens1; 
 646                         tokens1.push(cond1); 
 647                         while (!tokens1.empty()) { 
 648                             const Token *firstCondition = tokens1.top(); 
 649                             tokens1.pop(); 
 650                             if (!firstCondition) 
 651                                 continue; 
 652                             if (firstCondition->str() == "&&") { 
 653                                 tokens1.push(firstCondition->astOperand1()); 
 654                                 tokens1.push(firstCondition->astOperand2()); 
 655                             } else if (!firstCondition->hasKnownIntValue()) { 
 656                                 if (!isReturnVar && isOppositeCond(false, mTokenizer->isCPP(), firstCondition, cond2, mSettings->library, true, true, &errorPath)) { 
 657                                     if (!isAliased(vars)) 
 658                                         oppositeInnerConditionError(firstCondition, cond2, errorPath); 
 659                                 } else if (!isReturnVar && isSameExpression(mTokenizer->isCPP(), true, firstCondition, cond2, mSettings->library, true, true, &errorPath)) { 
 660                                     identicalInnerConditionError(firstCondition, cond2, errorPath); 
 661                                 } 
 662                             } 
 663                         } 
 664                     } else { 
 665                         std::stack<const Token *> tokens2; 
 666                         tokens2.push(cond2); 
 667                         while (!tokens2.empty()) { 
 668                             const Token *secondCondition = tokens2.top(); 
 669                             tokens2.pop(); 
 670                             if (!secondCondition) 
 671                                 continue; 
 672                             if (secondCondition->str() == "||" || secondCondition->str() == "&&") { 
 673                                 tokens2.push(secondCondition->astOperand1()); 
 674                                 tokens2.push(secondCondition->astOperand2()); 
 675                             } else if ((!cond1->hasKnownIntValue() || !secondCondition->hasKnownIntValue()) && 
 676                                        isSameExpression(mTokenizer->isCPP(), true, cond1, secondCondition, mSettings->library, true, true, &errorPath)) { 
 677                                 if (!isAliased(vars)) 
 678                                     identicalConditionAfterEarlyExitError(cond1, secondCondition, errorPath); 
 679                             } 
 680                         } 
 681                     } 
 682                 } 
 683                 if (Token::Match(tok, "%type% (") && nonlocal && isNonConstFunctionCall(tok, mSettings->library)) // non const function call -> bailout if there are nonlocal variables 
 684                     break; 
 685                 if (Token::Match(tok, "case|break|continue|return|throw") && tok->scope() == endToken->scope()) 
 686                     break; 
 687                 if (Token::Match(tok, "[;{}] %name% :")) 
 688                     break; 
 689                 // bailout if loop is seen. 
 690                 // TODO: handle loops better. 
 691                 if (Token::Match(tok, "for|while|do")) { 
 692                     const Token *tok1 = tok->next(); 
 693                     const Token *tok2; 
 694                     if (Token::simpleMatch(tok, "do {")) { 
 695                         if (!Token::simpleMatch(tok->linkAt(1), "} while (")) 
 696                             break; 
 697                         tok2 = tok->linkAt(1)->linkAt(2); 
 698                     } else if (Token::Match(tok, "if|while (")) { 
 699                         tok2 = tok->linkAt(1); 
 700                         if (Token::simpleMatch(tok2, ") {")) 
 701                             tok2 = tok2->linkAt(1); 
 702                         if (!tok2) 
 703                             break; 
 704                     } else { 
 705                         // Incomplete code 
 706                         break; 
 707                     } 
 708                     bool changed = false; 
 709                     for (unsigned int varid : vars) { 
 710                         if (isVariableChanged(tok1, tok2, varid, nonlocal, mSettings, mTokenizer->isCPP())) { 
 711                             changed = true; 
 712                             break; 
 713                         } 
 714                     } 
 715                     if (changed) 
 716                         break; 
 717                 } 
 718                 if ((tok->varId() && vars.find(tok->varId()) != vars.end()) || 
 719                     (!tok->varId() && nonlocal)) { 
 720                     if (Token::Match(tok, "%name% %assign%|++|--")) 
 721                         break; 
 722                     if (Token::Match(tok->astParent(), "*|.|[")) { 
 723                         const Token *parent = tok; 
 724                         while (Token::Match(parent->astParent(), ".|[") || (parent->astParent() && parent->astParent()->isUnaryOp("*"))) 
 725                             parent = parent->astParent(); 
 726                         if (Token::Match(parent->astParent(), "%assign%|++|--")) 
 727                             break; 
 728                     }

 873 static inline T getvalue(const int test, const T value1, const T value2) 
 874 { 
 875     // test: 
 876     // 1 => return value that is less than both value1 and value2 
 877     // 2 => return value1

 879     // 4 => return value2 
 880     // 5 => return value that is larger than both value1 and value2

 896 static bool parseComparison(const Token *comp, bool *not1, std::string *op, std::string *value, const Token **expr, bool* inconclusive) 
 897 { 
 898     *not1 = false; 
 899     while (comp && comp->str() == "!") { 
 900         *not1 = !(*not1); 
 901         comp = comp->astOperand1(); 
 902     } 
 903  
 904     if (!comp) 
 905         return false; 
 906  
 907     const Token* op1 = comp->astOperand1(); 
 908     const Token* op2 = comp->astOperand2(); 
 909     if (!comp->isComparisonOp() || !op1 || !op2) { 
 910         *op = "!="; 
 911         *value = "0"; 
 912         *expr = comp; 
 913     } else if (op1->isLiteral()) { 
 914         if (op1->isExpandedMacro()) 
 915             return false; 
 916         *op = invertOperatorForOperandSwap(comp->str()); 
 917         if (op1->enumerator() && op1->enumerator()->value_known) 
 918             *value = MathLib::toString(op1->enumerator()->value); 
 919         else 
 920             *value = op1->str(); 
 921         *expr = op2; 
 922     } else if (comp->astOperand2()->isLiteral()) { 
 923         if (op2->isExpandedMacro()) 
 924             return false; 
 925         *op = comp->str(); 
 926         if (op2->enumerator() && op2->enumerator()->value_known) 
 927             *value = MathLib::toString(op2->enumerator()->value); 
 928         else 
 929             *value = op2->str(); 
 930         *expr = op1; 
 931     } else { 
 932         *op = "!="; 
 933         *value = "0"; 
 934         *expr = comp; 
 935     } 
 936  
 937     *inconclusive = *inconclusive || ((*value)[0] == '\'' && !(*op == "!=" || *op == "==")); 
 938  
 939     // Only float and int values are currently handled 
 940     if (!MathLib::isInt(*value) && !MathLib::isFloat(*value) && (*value)[0] != '\'') 
 941         return false; 
 942  
 943     return true; 
 944 }

 993

1012                     const std::string expr3VerboseMsg = expr3;

1015                         if (expr1[0] == '!' && expr2[0] != '!') { 
1016                             expr1 = "!A"; 
1017                             expr2 = "A";

1019                             expr1 = "A"; 
1020                             expr2 = "!A"; 
1021                         } 
1022  
1023                         expr3 = "B"; 
1024                     }

1026                     const std::string cond1 = expr1 + " " + tok->str() + " (" + expr2 + " " + tok->astOperand2()->str() + " " + expr3 + ")";

1036                 } 
1037             } 
1038  
1039             // Comparison #1 (LHS) 
1040             const Token *comp1 = tok->astOperand1(); 
1041             if (comp1 && comp1->str() == tok->str()) 
1042                 comp1 = comp1->astOperand2(); 
1043  
1044             // Comparison #2 (RHS) 
1045             const Token *comp2 = tok->astOperand2(); 
1046

1048             bool parseable = true; 
1049

1051             bool not1;

1053             const Token *expr1 = nullptr; 
1054             parseable &= (parseComparison(comp1, &not1, &op1, &value1, &expr1, &inconclusive)); 
1055  
1056             // Parse RHS

1058             std::string op2, value2; 
1059             const Token *expr2 = nullptr;

1061  
1062             if (inconclusive && !printInconclusive) 
1063                 continue; 
1064  
1065             const bool isfloat = astIsFloat(expr1, true) || MathLib::isFloat(value1) || astIsFloat(expr2, true) || MathLib::isFloat(value2); 
1066  
1067             ErrorPath errorPath; 
1068

1070             if (!isfloat && isOppositeCond(tok->str() == "||", mTokenizer->isCPP(), tok->astOperand1(), tok->astOperand2(), mSettings->library, true, true, &errorPath)) { 
1071  
1072                 const bool alwaysTrue(tok->str() == "||"); 
1073                 incorrectLogicOperatorError(tok, conditionString(tok), alwaysTrue, inconclusive, errorPath);

1075             } 
1076  
1077             if (!parseable) 
1078                 continue; 
1079  
1080             if (isSameExpression(mTokenizer->isCPP(), true, comp1, comp2, mSettings->library, true, true)) 
1081                 continue; // same expressions => only report that there are same expressions 
1082             if (!isSameExpression(mTokenizer->isCPP(), true, expr1, expr2, mSettings->library, true, true)) 
1083                 continue; 
1084

1086             // don't check floating point equality comparisons. that is bad 
1087             // and deserves different warnings. 
1088             if (isfloat && (op1 == "==" || op1 == "!=" || op2 == "==" || op2 == "!=")) 
1089                 continue; 
1090  
1091  
1092             const double d1 = (isfloat) ? MathLib::toDoubleNumber(value1) : 0; 
1093             const double d2 = (isfloat) ? MathLib::toDoubleNumber(value2) : 0;

1112                     result1 = checkFloatRelation(op1, testvalue, d1); 
1113                     result2 = checkFloatRelation(op2, testvalue, d2); 
1114                 } else if (useUnsignedInt) { 
1115                     const MathLib::biguint testvalue = getvalue<MathLib::biguint>(test, u1, u2); 
1116                     result1 = checkIntRelation(op1, testvalue, u1); 
1117                     result2 = checkIntRelation(op2, testvalue, u2); 
1118                 } else { 
1119                     const MathLib::bigint testvalue = getvalue<MathLib::bigint>(test, i1, i2); 
1120                     result1 = checkIntRelation(op1, testvalue, i1); 
1121                     result2 = checkIntRelation(op2, testvalue, i2); 
1122                 } 
1123                 if (not1) 
1124                     result1 = !result1; 
1125                 if (not2) 
1126                     result2 = !result2; 
1127                 if (tok->str() == "&&") { 
1128                     alwaysTrue &= (result1 && result2); 
1129                     alwaysFalse &= !(result1 && result2); 
1130                 } else { 
1131                     alwaysTrue &= (result1 || result2); 
1132                     alwaysFalse &= !(result1 || result2); 
1133                 } 
1134                 firstTrue &= !(!result1 && result2); 
1135                 secondTrue &= !(result1 && !result2);

1137  
1138             const std::string cond1str = conditionString(not1, expr1, op1, value1);

1258                 } 
1259             } else if (tok->tokType() == Token::eBitOp && !tok->isUnaryOp("&")) { 
1260                 if (tok->astOperand2() && tok->astOperand2()->variable() && tok->astOperand2()->variable()->nameToken() == tok->astOperand2()) 
1261                     continue; 
1262  
1263                 // using boolean result in bitwise operation ! x [&|^] 
1264                 const ValueType* vt1 = tok->astOperand1() ? tok->astOperand1()->valueType() : nullptr; 
1265                 const ValueType* vt2 = tok->astOperand2() ? tok->astOperand2()->valueType() : nullptr; 
1266                 if (vt1 && vt1->type == ValueType::BOOL && !Token::Match(tok->astOperand1(), "%name%|(|[|::|.") && countPar(tok->astOperand1(), tok) == 0) 
1267                     clarifyConditionError(tok, false, true); 
1268                 else if (vt2 && vt2->type == ValueType::BOOL && !Token::Match(tok->astOperand2(), "%name%|(|[|::|.") && countPar(tok, tok->astOperand2()) == 0) 
1269                     clarifyConditionError(tok, false, true); 
1270             } 
1271         } 
1272     } 
1273 }

1298 void CheckCondition::alwaysTrueFalse() 
1299 { 
1300     if (!mSettings->isEnabled(Settings::STYLE)) 
1301         return; 
1302  
1303     const SymbolDatabase *symbolDatabase = mTokenizer->getSymbolDatabase(); 
1304     for (const Scope * scope : symbolDatabase->functionScopes) { 
1305         for (const Token* tok = scope->bodyStart->next(); tok != scope->bodyEnd; tok = tok->next()) { 
1306  
1307             if (tok->link()) // don't write false positives when templates are used 
1308                 continue; 
1309             if (!tok->hasKnownIntValue()) 
1310                 continue; 
1311             // Skip already diagnosed values 
1312             if (diag(tok, false)) 
1313                 continue; 
1314             if (Token::Match(tok, "%num%|%bool%|%char%")) 
1315                 continue; 
1316             if (Token::Match(tok, "! %num%|%bool%|%char%")) 
1317                 continue; 
1318             if (Token::Match(tok, "%oror%|&&|:")) 
1319                 continue; 
1320             if (Token::Match(tok, "%comp%") && isSameExpression(mTokenizer->isCPP(), true, tok->astOperand1(), tok->astOperand2(), mSettings->library, true, true)) 
1321                 continue; 
1322  
1323             const bool constIfWhileExpression = 
1324                 tok->astParent() && Token::Match(tok->astTop()->astOperand1(), "if|while") && 
1325                 (Token::Match(tok->astParent(), "%oror%|&&") || Token::Match(tok->astParent()->astOperand1(), "if|while")); 
1326             const bool constValExpr = tok->isNumber() && Token::Match(tok->astParent(),"%oror%|&&|?"); // just one number in boolean expression 
1327             const bool compExpr = Token::Match(tok, "%comp%|!"); // a compare expression 
1328             const bool returnStatement = Token::simpleMatch(tok->astTop(), "return") && 
1329                                          Token::Match(tok->astParent(), "%oror%|&&|return"); 
1330  
1331             if (!(constIfWhileExpression || constValExpr || compExpr || returnStatement)) 
1332                 continue; 
1333  
1334             if (returnStatement && (!scope->function || !Token::simpleMatch(scope->function->retDef, "bool"))) 
1335                 continue; 
1336  
1337             if (returnStatement && isConstVarExpression(tok)) 
1338                 continue; 
1339  
1340             if (returnStatement && Token::simpleMatch(tok->astParent(), "return") && tok->variable() && ( 
1341                     !tok->variable()->isLocal() || 
1342                     tok->variable()->isReference() || 
1343                     tok->variable()->isConst() || 
1344                     !isVariableChanged(tok->variable(), mSettings, mTokenizer->isCPP()))) 
1345                 continue; 
1346  
1347             // Don't warn in assertions. Condition is often 'always true' by intention. 
1348             // If platform,defines,etc cause 'always false' then that is not dangerous neither. 
1349             bool assertFound = false; 
1350             for (const Token * tok2 = tok->astParent(); tok2 ; tok2 = tok2->astParent()) { // move backwards and try to find "assert" 
1351                 if (tok2->str() == "(" && tok2->astOperand2()) { 
1352                     const std::string& str = tok2->previous()->str(); 
1353                     if ((str.find("assert")!=std::string::npos || str.find("ASSERT")!=std::string::npos)) 
1354                         assertFound = true; 
1355                     break; 
1356                 } 
1357             } 
1358             if (assertFound) 
1359                 continue; 
1360  
1361             // Don't warn when there are expanded macros.. 
1362             bool isExpandedMacro = false; 
1363             std::stack<const Token*> tokens; 
1364             tokens.push(tok); 
1365             while (!tokens.empty()) { 
1366                 const Token *tok2 = tokens.top(); 
1367                 tokens.pop(); 
1368                 if (!tok2) 
1369                     continue; 
1370                 tokens.push(tok2->astOperand1()); 
1371                 tokens.push(tok2->astOperand2()); 
1372                 if (tok2->isExpandedMacro()) { 
1373                     isExpandedMacro = true; 
1374                     break; 
1375                 } 
1376             } 
1377             if (isExpandedMacro) 
1378                 continue; 
1379             for (const Token *parent = tok; parent; parent = parent->astParent()) { 
1380                 if (parent->isExpandedMacro()) { 
1381                     isExpandedMacro = true; 
1382                     break; 
1383                 } 
1384             } 
1385             if (isExpandedMacro) 
1386                 continue; 
1387  
1388             // don't warn when condition checks sizeof result 
1389             bool hasSizeof = false; 
1390             tokens.push(tok); 
1391             while (!tokens.empty()) { 
1392                 const Token *tok2 = tokens.top(); 
1393                 tokens.pop(); 
1394                 if (!tok2) 
1395                     continue; 
1396                 if (tok2->isNumber()) 
1397                     continue; 
1398                 if (Token::simpleMatch(tok2->previous(), "sizeof (")) { 
1399                     hasSizeof = true; 
1400                     continue; 
1401                 } 
1402                 if (tok2->isComparisonOp() || tok2->isArithmeticalOp()) { 
1403                     tokens.push(tok2->astOperand1()); 
1404                     tokens.push(tok2->astOperand2()); 
1405                 } else 
1406                     break; 
1407             } 
1408             if (tokens.empty() && hasSizeof) 
1409                 continue; 
1410  
1411             alwaysTrueFalseError(tok, &tok->values().front()); 
1412         } 
1413     } 
1414 }

1429 void CheckCondition::checkInvalidTestForOverflow() 
1430 { 
1431     if (!mSettings->isEnabled(Settings::WARNING)) 
1432         return; 
1433  
1434     const SymbolDatabase *symbolDatabase = mTokenizer->getSymbolDatabase(); 
1435     for (const Scope * scope : symbolDatabase->functionScopes) { 
1436  
1437         for (const Token* tok = scope->bodyStart; tok != scope->bodyEnd; tok = tok->next()) { 
1438             if (!tok->isComparisonOp() || !tok->astOperand1() || !tok->astOperand2()) 
1439                 continue; 
1440  
1441             const Token *calcToken, *exprToken; 
1442             bool result; 
1443             if (Token::Match(tok, "<|>=") && tok->astOperand1()->str() == "+") { 
1444                 calcToken = tok->astOperand1(); 
1445                 exprToken = tok->astOperand2(); 
1446                 result = (tok->str() == ">="); 
1447             } else if (Token::Match(tok, ">|<=") && tok->astOperand2()->str() == "+") { 
1448                 calcToken = tok->astOperand2(); 
1449                 exprToken = tok->astOperand1(); 
1450                 result = (tok->str() == "<="); 
1451             } else 
1452                 continue; 
1453  
1454             // Only warn for signed integer overflows and pointer overflows. 
1455             if (!(calcToken->valueType() && (calcToken->valueType()->pointer || calcToken->valueType()->sign == ValueType::Sign::SIGNED))) 
1456                 continue; 
1457             if (!(exprToken->valueType() && (exprToken->valueType()->pointer || exprToken->valueType()->sign == ValueType::Sign::SIGNED))) 
1458                 continue; 
1459  
1460             const Token *termToken; 
1461             if (isSameExpression(mTokenizer->isCPP(), true, exprToken, calcToken->astOperand1(), mSettings->library, true, false)) 
1462                 termToken = calcToken->astOperand2(); 
1463             else if (isSameExpression(mTokenizer->isCPP(), true, exprToken, calcToken->astOperand2(), mSettings->library, true, false)) 
1464                 termToken = calcToken->astOperand1(); 
1465             else 
1466                 continue; 
1467  
1468             if (!termToken) 
1469                 continue; 
1470  
1471             // Only warn when termToken is always positive 
1472             if (termToken->valueType() && termToken->valueType()->sign == ValueType::Sign::UNSIGNED) 
1473                 invalidTestForOverflow(tok, result); 
1474             else if (termToken->isNumber() && MathLib::isPositive(termToken->str())) 
1475                 invalidTestForOverflow(tok, result); 
1476         } 
1477     } 
1478 }

1484                                "'. Condition is always " +
