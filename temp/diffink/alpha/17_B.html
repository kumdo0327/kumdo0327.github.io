 143 // Avoid Clang with Microsoft CodeGen's -Wunknown-pragmas warning. 
 144 #  ifndef __clang__ 
 145 #    pragma intrinsic(_BitScanReverse) 
 146 #  endif 
 147 inline uint32_t clz(uint32_t x) { 
 148   unsigned long r = 0; 
 149   _BitScanReverse(&r, x); 
 150  
 151   assert(x != 0); 
 152   // Static analysis complains about using uninitialized data 
 153   // "r", but the only way that can happen is if "x" is 0, 
 154   // which the callers guarantee to not happen. 
 155 #  pragma warning(suppress : 6102) 
 156   return 31 - r; 
 157 } 
 158 #  define FMT_BUILTIN_CLZ(n) fmt::internal::clz(n) 
 159  
 160 #  if defined(_WIN64) && !defined(__clang__) 
 161 #    pragma intrinsic(_BitScanReverse64) 
 162 #  endif 
 163  
 164 inline uint32_t clzll(uint64_t x) { 
 165   unsigned long r = 0; 
 166 #  ifdef _WIN64 
 167   _BitScanReverse64(&r, x); 
 168 #  else 
 169   // Scan the high 32 bits. 
 170   if (_BitScanReverse(&r, static_cast<uint32_t>(x >> 32))) return 63 - (r + 32); 
 171  
 172   // Scan the low 32 bits. 
 173   _BitScanReverse(&r, static_cast<uint32_t>(x)); 
 174 #  endif 
 175  
 176   assert(x != 0); 
 177   // Static analysis complains about using uninitialized data 
 178   // "r", but the only way that can happen is if "x" is 0, 
 179   // which the callers guarantee to not happen. 
 180 #  pragma warning(suppress : 6102) 
 181   return 63 - r; 
 182 }

 201 // An equivalent of `*reinterpret_cast<Dest*>(&source)` that doesn't produce 
 202 // undefined behavior (e.g. due to type aliasing). 
 203 // Example: uint64_t d = bit_cast<uint64_t>(2.718); 
 204 template <typename Dest, typename Source> 
 205 inline Dest bit_cast(const Source& source) { 
 206   static_assert(sizeof(Dest) == sizeof(Source), "size mismatch"); 
 207   Dest dest; 
 208   std::memcpy(&dest, &source, sizeof(dest)); 
 209   return dest;

 231 // Detect if *any* given type models the OutputIterator concept. 
 232 template <typename It> class is_output_iterator { 
 233   // Check for mutability because all iterator categories derived from 
 234   // std::input_iterator_tag *may* also meet the requirements of an 
 235   // OutputIterator, thereby falling into the category of 'mutable iterators' 
 236   // [iterator.requirements.general] clause 4. The compiler reveals this 
 237   // property only at the point of *actually dereferencing* the iterator! 
 238   template <typename U> 
 239   static decltype(*(std::declval<U>())) test(std::input_iterator_tag); 
 240   template <typename U> static char& test(std::output_iterator_tag); 
 241   template <typename U> static const char& test(...); 
 242  
 243   using type = decltype(test<It>(typename iterator_category<It>::type{})); 
 244  
 245  public: 
 246   static const bool value = !std::is_const<remove_reference_t<type>>::value; 
 247 };

 495 /** 
 496   \rst 
 497   A dynamically growing memory buffer for trivially copyable/constructible types 
 498   with the first ``SIZE`` elements stored in the object itself. 
 499  
 500   You can use one of the following type aliases for common character types: 
 501  
 502   +----------------+------------------------------+ 
 503   | Type           | Definition                   | 
 504   +================+==============================+ 
 505   | memory_buffer  | basic_memory_buffer<char>    | 
 506   +----------------+------------------------------+ 
 507   | wmemory_buffer | basic_memory_buffer<wchar_t> | 
 508   +----------------+------------------------------+ 
 509  
 510   **Example**:: 
 511  
 512      fmt::memory_buffer out; 
 513      format_to(out, "The answer is {}.", 42); 
 514  
 515   This will append the following output to the ``out`` object: 
 516  
 517   .. code-block:: none 
 518  
 519      The answer is 42. 
 520  
 521   The output can be converted to an ``std::string`` with ``to_string(out)``. 
 522   \endrst 
 523  */ 
 524 template <typename T, std::size_t SIZE = inline_buffer_size, 
 525           typename Allocator = std::allocator<T>> 
 526 class basic_memory_buffer : private Allocator, public internal::buffer<T> { 
 527  private: 
 528   T store_[SIZE]; 
 529  
 530   // Deallocate memory allocated by the buffer. 
 531   void deallocate() { 
 532     T* data = this->data(); 
 533     if (data != store_) Allocator::deallocate(data, this->capacity()); 
 534   } 
 535  
 536  protected: 
 537   void grow(std::size_t size) FMT_OVERRIDE; 
 538  
 539  public: 
 540   using value_type = T; 
 541   using const_reference = const T&; 
 542  
 543   explicit basic_memory_buffer(const Allocator& alloc = Allocator()) 
 544       : Allocator(alloc) { 
 545     this->set(store_, SIZE); 
 546   } 
 547   ~basic_memory_buffer() { deallocate(); } 
 548  
 549  private: 
 550   // Move data from other to this buffer. 
 551   void move(basic_memory_buffer& other) { 
 552     Allocator &this_alloc = *this, &other_alloc = other; 
 553     this_alloc = std::move(other_alloc); 
 554     T* data = other.data(); 
 555     std::size_t size = other.size(), capacity = other.capacity(); 
 556     if (data == other.store_) { 
 557       this->set(store_, capacity); 
 558       std::uninitialized_copy(other.store_, other.store_ + size, 
 559                               internal::make_checked(store_, capacity)); 
 560     } else { 
 561       this->set(data, capacity); 
 562       // Set pointer to the inline array so that delete is not called 
 563       // when deallocating. 
 564       other.set(other.store_, 0); 
 565     } 
 566     this->resize(size); 
 567   } 
 568  
 569  public: 
 570   /** 
 571     \rst 
 572     Constructs a :class:`fmt::basic_memory_buffer` object moving the content 
 573     of the other object to it. 
 574     \endrst 
 575    */ 
 576   basic_memory_buffer(basic_memory_buffer&& other) { move(other); } 
 577  
 578   /** 
 579     \rst 
 580     Moves the content of the other ``basic_memory_buffer`` object to this one. 
 581     \endrst 
 582    */ 
 583   basic_memory_buffer& operator=(basic_memory_buffer&& other) { 
 584     assert(this != &other); 
 585     deallocate(); 
 586     move(other); 
 587     return *this; 
 588   } 
 589  
 590   // Returns a copy of the allocator associated with this buffer. 
 591   Allocator get_allocator() const { return *this; } 
 592 };

 594 template <typename T, std::size_t SIZE, typename Allocator> 
 595 void basic_memory_buffer<T, SIZE, Allocator>::grow(std::size_t size) { 
 596 #ifdef FUZZING_BUILD_MODE_UNSAFE_FOR_PRODUCTION 
 597   if (size > 1000) throw std::runtime_error("fuzz mode - won't grow that much"); 
 598 #endif 
 599   std::size_t old_capacity = this->capacity(); 
 600   std::size_t new_capacity = old_capacity + old_capacity / 2; 
 601   if (size > new_capacity) new_capacity = size; 
 602   T* old_data = this->data(); 
 603   T* new_data = std::allocator_traits<Allocator>::allocate(*this, new_capacity); 
 604   // The following code doesn't throw, so the raw pointer above doesn't leak. 
 605   std::uninitialized_copy(old_data, old_data + this->size(), 
 606                           internal::make_checked(new_data, new_capacity)); 
 607   this->set(new_data, new_capacity); 
 608   // deallocate must not throw according to the standard, but even if it does, 
 609   // the buffer already uses the new storage and will deallocate it in 
 610   // destructor. 
 611   if (old_data != store_) Allocator::deallocate(old_data, old_capacity); 
 612 }

 744 // An lg handler that formats a decimal number. 
 745 // Usage: lg(n, decimal_formatter(buffer)); 
 746 class decimal_formatter { 
 747  private: 
 748   char* buffer_; 
 749  
 750   void write_pair(unsigned N, uint32_t index) { 
 751     std::memcpy(buffer_ + N, data::digits + index * 2, 2); 
 752   } 
 753  
 754  public: 
 755   explicit decimal_formatter(char* buf) : buffer_(buf) {} 
 756  
 757   template <unsigned N> char* on(uint32_t u) { 
 758     if (N == 0) { 
 759       *buffer_ = static_cast<char>(u) + '0'; 
 760     } else if (N == 1) { 
 761       write_pair(0, u); 
 762     } else { 
 763       // The idea of using 4.32 fixed-point numbers is based on 
 764       // https://github.com/jeaiii/itoa 
 765       unsigned n = N - 1; 
 766       unsigned a = n / 5 * n * 53 / 16; 
 767       uint64_t t = 
 768           ((1ULL << (32 + a)) / data::zero_or_powers_of_10_32[n] + 1 - n / 9); 
 769       t = ((t * u) >> a) + n / 5 * 4; 
 770       write_pair(0, t >> 32); 
 771       for (unsigned i = 2; i < N; i += 2) { 
 772         t = 100ULL * static_cast<uint32_t>(t); 
 773         write_pair(i, t >> 32); 
 774       } 
 775       if (N % 2 == 0) { 
 776         buffer_[N] = 
 777             static_cast<char>((10ULL * static_cast<uint32_t>(t)) >> 32) + '0'; 
 778       } 
 779     } 
 780     return buffer_ += N + 1; 
 781   } 
 782 };

 808 // Formats a decimal unsigned integer value writing into buffer. 
 809 // add_thousands_sep is called after writing each char to add a thousands 
 810 // separator if necessary. 
 811 template <typename UInt, typename Char, typename F> 
 812 inline Char* format_decimal(Char* buffer, UInt value, int num_digits, 
 813                             F add_thousands_sep) { 
 814   FMT_ASSERT(num_digits >= 0, "invalid digit count"); 
 815   buffer += num_digits; 
 816   Char* end = buffer; 
 817   while (value >= 100) { 
 818     // Integer division is slow so do it for a group of two digits instead 
 819     // of for every digit. The idea comes from the talk by Alexandrescu 
 820     // "Three Optimization Tips for C++". See speed-test for a comparison. 
 821     unsigned index = static_cast<unsigned>((value % 100) * 2); 
 822     value /= 100; 
 823     *--buffer = static_cast<Char>(data::digits[index + 1]); 
 824     add_thousands_sep(buffer); 
 825     *--buffer = static_cast<Char>(data::digits[index]); 
 826     add_thousands_sep(buffer); 
 827   } 
 828   if (value < 10) { 
 829     *--buffer = static_cast<Char>('0' + value); 
 830     return end; 
 831   } 
 832   unsigned index = static_cast<unsigned>(value * 2); 
 833   *--buffer = static_cast<Char>(data::digits[index + 1]); 
 834   add_thousands_sep(buffer); 
 835   *--buffer = static_cast<Char>(data::digits[index]); 
 836   return end; 
 837 }

 861 template <unsigned BASE_BITS, typename Char, typename UInt> 
 862 inline Char* format_uint(Char* buffer, UInt value, int num_digits, 
 863                          bool upper = false) { 
 864   buffer += num_digits; 
 865   Char* end = buffer; 
 866   do { 
 867     const char* digits = upper ? "0123456789ABCDEF" : data::hex_digits; 
 868     unsigned digit = (value & ((1 << BASE_BITS) - 1)); 
 869     *--buffer = static_cast<Char>(BASE_BITS < 4 ? static_cast<char>('0' + digit) 
 870                                                 : digits[digit]); 
 871   } while ((value >>= BASE_BITS) != 0); 
 872   return end; 
 873 }

 875 template <unsigned BASE_BITS, typename Char> 
 876 Char* format_uint(Char* buffer, internal::fallback_uintptr n, int num_digits, 
 877                   bool = false) { 
 878   auto char_digits = std::numeric_limits<unsigned char>::digits / 4; 
 879   int start = (num_digits + char_digits - 1) / char_digits - 1; 
 880   if (int start_digits = num_digits % char_digits) { 
 881     unsigned value = n.value[start--]; 
 882     buffer = format_uint<BASE_BITS>(buffer, value, start_digits); 
 883   } 
 884   for (; start >= 0; --start) { 
 885     unsigned value = n.value[start]; 
 886     buffer += char_digits; 
 887     auto p = buffer; 
 888     for (int i = 0; i < char_digits; ++i) { 
 889       unsigned digit = (value & ((1 << BASE_BITS) - 1)); 
 890       *--p = static_cast<Char>(data::hex_digits[digit]); 
 891       value >>= BASE_BITS; 
 892     } 
 893   } 
 894   return buffer; 
 895 }

1010 // Writes the exponent exp in the form "[+-]d{2,3}" to buffer. 
1011 template <typename Char, typename It> It write_exponent(int exp, It it) { 
1012   FMT_ASSERT(-1000 < exp && exp < 1000, "exponent out of range"); 
1013   if (exp < 0) { 
1014     *it++ = static_cast<Char>('-'); 
1015     exp = -exp; 
1016   } else { 
1017     *it++ = static_cast<Char>('+'); 
1018   } 
1019   if (exp >= 100) { 
1020     *it++ = static_cast<Char>(static_cast<char>('0' + exp / 100));

1022   } 
1023   const char* d = data::digits + exp * 2; 
1024   *it++ = static_cast<Char>(d[0]); 
1025   *it++ = static_cast<Char>(d[1]); 
1026   return it;

1036 // The number is given as v = digits * pow(10, exp). 
1037 template <typename Char, typename It> 
1038 It grisu_prettify(const char* digits, int size, int exp, It it, 
1039                   gen_digits_params params, Char decimal_point) { 
1040   // pow(10, full_exp - 1) <= v <= pow(10, full_exp). 
1041   int full_exp = size + exp; 
1042   if (!params.fixed) { 
1043     // Insert a decimal point after the first digit and add an exponent. 
1044     *it++ = static_cast<Char>(*digits); 
1045     if (size > 1) *it++ = decimal_point; 
1046     exp += size - 1; 
1047     it = copy_str<Char>(digits + 1, digits + size, it); 
1048     if (size < params.num_digits) 
1049       it = std::fill_n(it, params.num_digits - size, static_cast<Char>('0')); 
1050     *it++ = static_cast<Char>(params.upper ? 'E' : 'e'); 
1051     return write_exponent<Char>(exp, it); 
1052   } 
1053   if (size <= full_exp) { 
1054     // 1234e7 -> 12340000000[.0+] 
1055     it = copy_str<Char>(digits, digits + size, it); 
1056     it = std::fill_n(it, full_exp - size, static_cast<Char>('0')); 
1057     int num_zeros = (std::max)(params.num_digits - full_exp, 1); 
1058     if (params.trailing_zeros) { 
1059       *it++ = decimal_point; 
1060 #ifdef FUZZING_BUILD_MODE_UNSAFE_FOR_PRODUCTION 
1061       if (num_zeros > 1000) 
1062         throw std::runtime_error("fuzz mode - avoiding excessive cpu use"); 
1063 #endif 
1064       it = std::fill_n(it, num_zeros, static_cast<Char>('0')); 
1065     } 
1066   } else if (full_exp > 0) { 
1067     // 1234e-2 -> 12.34[0+] 
1068     it = copy_str<Char>(digits, digits + full_exp, it); 
1069     if (!params.trailing_zeros) { 
1070       // Remove trailing zeros. 
1071       while (size > full_exp && digits[size - 1] == '0') --size; 
1072       if (size != full_exp) *it++ = decimal_point; 
1073       return copy_str<Char>(digits + full_exp, digits + size, it); 
1074     } 
1075     *it++ = decimal_point; 
1076     it = copy_str<Char>(digits + full_exp, digits + size, it); 
1077     if (params.num_digits > size) { 
1078       // Add trailing zeros. 
1079       int num_zeros = params.num_digits - size; 
1080       it = std::fill_n(it, num_zeros, static_cast<Char>('0')); 
1081     } 
1082   } else { 
1083     // 1234e-6 -> 0.001234 
1084     *it++ = static_cast<Char>('0'); 
1085     int num_zeros = -full_exp; 
1086     if (params.num_digits >= 0 && params.num_digits < num_zeros) 
1087       num_zeros = params.num_digits; 
1088     if (!params.trailing_zeros) 
1089       while (size > 0 && digits[size - 1] == '0') --size; 
1090     if (num_zeros != 0 || size != 0) { 
1091       *it++ = decimal_point; 
1092       it = std::fill_n(it, num_zeros, static_cast<Char>('0')); 
1093       it = copy_str<Char>(digits, digits + size, it); 
1094     } 
1095   } 
1096   return it; 
1097 }

1327   //   <left-padding><prefix><numeric-padding><digits><right-padding> 
1328   // where <digits> are written by f(it). 
1329   template <typename F> 
1330   void write_int(int num_digits, string_view prefix, format_specs specs, F f) { 
1331     std::size_t size = prefix.size() + internal::to_unsigned(num_digits); 
1332     char_type fill = specs.fill[0]; 
1333     std::size_t padding = 0; 
1334     if (specs.align == align::numeric) { 
1335       auto unsiged_width = internal::to_unsigned(specs.width); 
1336       if (unsiged_width > size) { 
1337         padding = unsiged_width - size; 
1338         size = unsiged_width; 
1339       } 
1340     } else if (specs.precision > num_digits) { 
1341       size = prefix.size() + internal::to_unsigned(specs.precision); 
1342       padding = internal::to_unsigned(specs.precision - num_digits); 
1343       fill = static_cast<char_type>('0'); 
1344     } 
1345     if (specs.align == align::none) specs.align = align::right; 
1346     write_padded(specs, padded_int_writer<F>{size, prefix, fill, padding, f}); 
1347   }

1606   // Writes a value in the format 
1607   //   <left-padding><value><right-padding> 
1608   // where <value> is written by f(it). 
1609   template <typename F> void write_padded(const format_specs& specs, F&& f) { 
1610     // User-perceived width (in code points). 
1611     unsigned width = to_unsigned(specs.width); 
1612     size_t size = f.size();  // The number of code units. 
1613     size_t num_code_points = width != 0 ? f.width() : size; 
1614     if (width <= num_code_points) return f(reserve(size)); 
1615     auto&& it = reserve(width + (size - num_code_points)); 
1616     char_type fill = specs.fill[0]; 
1617     std::size_t padding = width - num_code_points; 
1618     if (specs.align == align::right) { 
1619       it = std::fill_n(it, padding, fill); 
1620       f(it); 
1621     } else if (specs.align == align::center) { 
1622       std::size_t left_padding = padding / 2; 
1623       it = std::fill_n(it, left_padding, fill); 
1624       f(it); 
1625       it = std::fill_n(it, padding - left_padding, fill); 
1626     } else { 
1627       f(it); 
1628       it = std::fill_n(it, padding, fill); 
1629     } 
1630   }

1875 // Parses the range [begin, end) as an unsigned integer. This function assumes 
1876 // that the range is non-empty and the first character is a digit. 
1877 template <typename Char, typename ErrorHandler> 
1878 FMT_CONSTEXPR int parse_nonnegative_int(const Char*& begin, const Char* end, 
1879                                         ErrorHandler&& eh) { 
1880   assert(begin != end && '0' <= *begin && *begin <= '9'); 
1881   if (*begin == '0') { 
1882     ++begin; 
1883     return 0; 
1884   } 
1885   unsigned value = 0; 
1886   // Convert to unsigned to prevent a warning. 
1887   constexpr unsigned max_int = (std::numeric_limits<int>::max)(); 
1888   unsigned big = max_int / 10; 
1889   do { 
1890     // Check for overflow. 
1891     if (value > big) { 
1892       value = max_int + 1; 
1893       break; 
1894     } 
1895     value = value * 10 + unsigned(*begin - '0'); 
1896     ++begin; 
1897   } while (begin != end && '0' <= *begin && *begin <= '9'); 
1898   if (value > max_int) eh.on_error("number is too big"); 
1899   return static_cast<int>(value); 
1900 }

2236   if (c >= '0' && c <= '9') { 
2237     int index = parse_nonnegative_int(begin, end, handler); 
2238     if (begin == end || (*begin != '}' && *begin != ':')) 
2239       return handler.on_error("invalid format string"), begin; 
2240     handler(index); 
2241     return begin;

2243   if (!is_name_start(c)) 
2244     return handler.on_error("invalid format string"), begin; 
2245   auto it = begin; 
2246   do { 
2247     ++it; 
2248   } while (it != end && (is_name_start(c = *it) || ('0' <= c && c <= '9'))); 
2249   handler(basic_string_view<Char>(begin, to_unsigned(it - begin))); 
2250   return it; 
2251 }

2287 // Parses fill and alignment. 
2288 template <typename Char, typename Handler> 
2289 FMT_CONSTEXPR const Char* parse_align(const Char* begin, const Char* end, 
2290                                       Handler&& handler) { 
2291   FMT_ASSERT(begin != end, ""); 
2292   auto align = align::none; 
2293   int i = 0; 
2294   if (begin + 1 != end) ++i; 
2295   do { 
2296     switch (static_cast<char>(begin[i])) { 
2297     case '<': 
2298       align = align::left; 
2299       break; 
2300     case '>': 
2301       align = align::right; 
2302       break; 
2303     case '=': 
2304       align = align::numeric; 
2305       break; 
2306     case '^': 
2307       align = align::center; 
2308       break; 
2309     } 
2310     if (align != align::none) { 
2311       if (i > 0) { 
2312         auto c = *begin; 
2313         if (c == '{') 
2314           return handler.on_error("invalid fill character '{'"), begin; 
2315         begin += 2; 
2316         handler.on_fill(c); 
2317       } else 
2318         ++begin; 
2319       handler.on_align(align); 
2320       break; 
2321     } 
2322   } while (i-- > 0); 
2323   return begin; 
2324 }

2326 template <typename Char, typename Handler> 
2327 FMT_CONSTEXPR const Char* parse_width(const Char* begin, const Char* end, 
2328                                       Handler&& handler) { 
2329   FMT_ASSERT(begin != end, ""); 
2330   if ('0' <= *begin && *begin <= '9') { 
2331     handler.on_width(parse_nonnegative_int(begin, end, handler)); 
2332   } else if (*begin == '{') { 
2333     ++begin; 
2334     if (begin != end) 
2335       begin = parse_arg_id(begin, end, width_adapter<Handler, Char>(handler)); 
2336     if (begin == end || *begin != '}') 
2337       return handler.on_error("invalid format string"), begin; 
2338     ++begin; 
2339   } 
2340   return begin; 
2341 }

2343 template <typename Char, typename Handler> 
2344 FMT_CONSTEXPR const Char* parse_precision(const Char* begin, const Char* end, 
2345                                           Handler&& handler) { 
2346   ++begin; 
2347   auto c = begin != end ? *begin : Char(); 
2348   if ('0' <= c && c <= '9') { 
2349     handler.on_precision(parse_nonnegative_int(begin, end, handler)); 
2350   } else if (c == '{') { 
2351     ++begin; 
2352     if (begin != end) { 
2353       begin = 
2354           parse_arg_id(begin, end, precision_adapter<Handler, Char>(handler)); 
2355     } 
2356     if (begin == end || *begin++ != '}') 
2357       return handler.on_error("invalid format string"), begin; 
2358   } else { 
2359     return handler.on_error("missing precision specifier"), begin; 
2360   } 
2361   handler.end_precision(); 
2362   return begin;

2365 // Parses standard format specifiers and sends notifications about parsed 
2366 // components to handler. 
2367 template <typename Char, typename SpecHandler> 
2368 FMT_CONSTEXPR const Char* parse_format_specs(const Char* begin, const Char* end, 
2369                                              SpecHandler&& handler) { 
2370   if (begin == end || *begin == '}') return begin; 
2371  
2372   begin = parse_align(begin, end, handler); 
2373   if (begin == end) return begin; 
2374  
2375   // Parse sign. 
2376   switch (static_cast<char>(*begin)) { 
2377   case '+': 
2378     handler.on_plus(); 
2379     ++begin; 
2380     break; 
2381   case '-': 
2382     handler.on_minus(); 
2383     ++begin; 
2384     break; 
2385   case ' ': 
2386     handler.on_space(); 
2387     ++begin; 
2388     break; 
2389   } 
2390   if (begin == end) return begin; 
2391  
2392   if (*begin == '#') { 
2393     handler.on_hash(); 
2394     if (++begin == end) return begin; 
2395   } 
2396  
2397   // Parse zero flag. 
2398   if (*begin == '0') { 
2399     handler.on_zero(); 
2400     if (++begin == end) return begin; 
2401   } 
2402  
2403   begin = parse_width(begin, end, handler); 
2404   if (begin == end) return begin; 
2405  
2406   // Parse precision. 
2407   if (*begin == '.') { 
2408     begin = parse_precision(begin, end, handler); 
2409   } 
2410  
2411   // Parse type. 
2412   if (begin != end && *begin != '}') handler.on_type(*begin++); 
2413   return begin; 
2414 }

2445 template <bool IS_CONSTEXPR, typename Char, typename Handler> 
2446 FMT_CONSTEXPR void parse_format_string(basic_string_view<Char> format_str, 
2447                                        Handler&& handler) { 
2448   struct pfs_writer { 
2449     FMT_CONSTEXPR void operator()(const Char* begin, const Char* end) { 
2450       if (begin == end) return; 
2451       for (;;) { 
2452         const Char* p = nullptr; 
2453         if (!find<IS_CONSTEXPR>(begin, end, '}', p)) 
2454           return handler_.on_text(begin, end); 
2455         ++p; 
2456         if (p == end || *p != '}') 
2457           return handler_.on_error("unmatched '}' in format string"); 
2458         handler_.on_text(begin, p); 
2459         begin = p + 1; 
2460       } 
2461     } 
2462     Handler& handler_; 
2463   } write{handler}; 
2464   auto begin = format_str.data(); 
2465   auto end = begin + format_str.size(); 
2466   while (begin != end) { 
2467     // Doing two passes with memchr (one for '{' and another for '}') is up to 
2468     // 2.5x faster than the naive one-pass implementation on big format strings. 
2469     const Char* p = begin; 
2470     if (*begin != '{' && !find<IS_CONSTEXPR>(begin, end, '{', p)) 
2471       return write(begin, end); 
2472     write(begin, p); 
2473     ++p; 
2474     if (p == end) return handler.on_error("invalid format string"); 
2475     if (static_cast<char>(*p) == '}') { 
2476       handler.on_arg_id(); 
2477       handler.on_replacement_field(p); 
2478     } else if (*p == '{') { 
2479       handler.on_text(p, p + 1); 
2480     } else { 
2481       p = parse_arg_id(p, end, id_adapter<Handler, Char>{handler}); 
2482       Char c = p != end ? *p : Char(); 
2483       if (c == '}') { 
2484         handler.on_replacement_field(p); 
2485       } else if (c == ':') { 
2486         p = handler.on_format_specs(p + 1, end); 
2487         if (p == end || *p != '}') 
2488           return handler.on_error("unknown format specifier"); 
2489       } else { 
2490         return handler.on_error("missing '}' in format string"); 
2491       } 
2492     } 
2493     begin = p + 1; 
2494   } 
2495 }

2751 template <typename Range> 
2752 template <typename T, bool USE_GRISU> 
2753 void internal::basic_writer<Range>::write_double(T value, 
2754                                                  const format_specs& specs) { 
2755   // Check type. 
2756   float_spec_handler handler(static_cast<char>(specs.type)); 
2757   internal::handle_float_type_spec(handler.type, handler); 
2758  
2759   char sign = 0; 
2760   // Use signbit instead of value < 0 since the latter is always false for NaN. 
2761   if (std::signbit(value)) { 
2762     sign = '-'; 
2763     value = -value; 
2764   } else if (specs.sign != sign::none) { 
2765     if (specs.sign == sign::plus) 
2766       sign = '+'; 
2767     else if (specs.sign == sign::space) 
2768       sign = ' '; 
2769   } 
2770  
2771   if (!std::isfinite(value)) { 
2772     // Format infinity and NaN ourselves because sprintf's output is not 
2773     // consistent across platforms. 
2774     const char* str = std::isinf(value) ? (handler.upper ? "INF" : "inf") 
2775                                         : (handler.upper ? "NAN" : "nan"); 
2776     return write_padded(specs, 
2777                         inf_or_nan_writer{sign, handler.as_percentage, str}); 
2778   } 
2779  
2780   if (handler.as_percentage) value *= 100; 
2781  
2782   memory_buffer buffer; 
2783   int exp = 0; 
2784   int precision = specs.precision >= 0 || !specs.type ? specs.precision : 6; 
2785   unsigned options = handler.fixed ? internal::grisu_options::fixed : 0; 
2786   bool use_grisu = USE_GRISU && 
2787                    (specs.type != 'a' && specs.type != 'A' && 
2788                     specs.type != 'e' && specs.type != 'E') && 
2789                    internal::grisu_format(static_cast<double>(value), buffer, 
2790                                           precision, options, exp); 
2791   char* decimal_point_pos = nullptr; 
2792   if (!use_grisu) 
2793     decimal_point_pos = internal::sprintf_format(value, buffer, specs); 
2794  
2795   if (handler.as_percentage) { 
2796     buffer.push_back('%'); 
2797     --exp;  // Adjust decimal place position. 
2798   } 
2799   format_specs as = specs; 
2800   if (specs.align == align::numeric) { 
2801     if (sign) { 
2802       auto&& it = reserve(1); 
2803       *it++ = static_cast<char_type>(sign); 
2804       sign = 0; 
2805       if (as.width) --as.width; 
2806     } 
2807     as.align = align::right; 
2808   } else if (specs.align == align::none) { 
2809     as.align = align::right; 
2810   } 
2811   char_type decimal_point = handler.use_locale 
2812                                 ? internal::decimal_point<char_type>(locale_) 
2813                                 : static_cast<char_type>('.'); 
2814   if (use_grisu) { 
2815     auto params = internal::gen_digits_params(); 
2816     params.fixed = handler.fixed; 
2817     params.num_digits = precision; 
2818     params.trailing_zeros = 
2819         (precision != 0 && (handler.fixed || !specs.type)) || specs.alt; 
2820     write_padded(as, grisu_writer(sign, buffer, exp, params, decimal_point)); 
2821   } else { 
2822     write_padded(as, 
2823                  double_writer{sign, buffer, decimal_point_pos, decimal_point}); 
2824   } 
2825 }

2881 /** Fast integer formatter. */ 
2882 class format_int { 
2883  private: 
2884   // Buffer should be large enough to hold all digits (digits10 + 1), 
2885   // a sign and a null character. 
2886   enum { buffer_size = std::numeric_limits<unsigned long long>::digits10 + 3 }; 
2887   mutable char buffer_[buffer_size]; 
2888   char* str_; 
2889  
2890   // Formats value in reverse and returns a pointer to the beginning. 
2891   char* format_decimal(unsigned long long value) { 
2892     char* ptr = buffer_ + (buffer_size - 1);  // Parens to workaround MSVC bug. 
2893     while (value >= 100) { 
2894       // Integer division is slow so do it for a group of two digits instead 
2895       // of for every digit. The idea comes from the talk by Alexandrescu 
2896       // "Three Optimization Tips for C++". See speed-test for a comparison. 
2897       unsigned index = static_cast<unsigned>((value % 100) * 2); 
2898       value /= 100; 
2899       *--ptr = internal::data::digits[index + 1]; 
2900       *--ptr = internal::data::digits[index]; 
2901     } 
2902     if (value < 10) { 
2903       *--ptr = static_cast<char>('0' + value); 
2904       return ptr; 
2905     } 
2906     unsigned index = static_cast<unsigned>(value * 2); 
2907     *--ptr = internal::data::digits[index + 1]; 
2908     *--ptr = internal::data::digits[index]; 
2909     return ptr; 
2910   } 
2911  
2912   void format_signed(long long value) { 
2913     unsigned long long abs_value = static_cast<unsigned long long>(value); 
2914     bool negative = value < 0; 
2915     if (negative) abs_value = 0 - abs_value; 
2916     str_ = format_decimal(abs_value); 
2917     if (negative) *--str_ = '-'; 
2918   } 
2919  
2920  public: 
2921   explicit format_int(int value) { format_signed(value); } 
2922   explicit format_int(long value) { format_signed(value); } 
2923   explicit format_int(long long value) { format_signed(value); } 
2924   explicit format_int(unsigned value) : str_(format_decimal(value)) {} 
2925   explicit format_int(unsigned long value) : str_(format_decimal(value)) {} 
2926   explicit format_int(unsigned long long value) : str_(format_decimal(value)) {} 
2927  
2928   /** Returns the number of characters written to the output buffer. */ 
2929   std::size_t size() const { 
2930     return internal::to_unsigned(buffer_ - str_ + buffer_size - 1); 
2931   } 
2932  
2933   /** 
2934     Returns a pointer to the output buffer content. No terminating null 
2935     character is appended. 
2936    */ 
2937   const char* data() const { return str_; } 
2938  
2939   /** 
2940     Returns a pointer to the output buffer content with terminating null 
2941     character appended. 
2942    */ 
2943   const char* c_str() const { 
2944     buffer_[buffer_size - 1] = '\0'; 
2945     return str_; 
2946   } 
2947  
2948   /** 
2949     \rst 
2950     Returns the content of the output buffer as an ``std::string``. 
2951     \endrst 
2952    */ 
2953   std::string str() const { return std::string(str_, size()); } 
2954 };

2965   // terminating '}'. 
2966   template <typename ParseContext> 
2967   FMT_CONSTEXPR auto parse(ParseContext& ctx) -> decltype(ctx.begin()) { 
2968     using handler_type = internal::dynamic_specs_handler<ParseContext>; 
2969     auto type = internal::type_constant<T, Char>::value; 
2970     internal::specs_checker<handler_type> handler(handler_type(specs_, ctx), 
2971                                                   type); 
2972     auto it = parse_format_specs(ctx.begin(), ctx.end(), handler); 
2973     auto eh = ctx.error_handler(); 
2974     switch (type) { 
2975     case internal::none_type: 
2976     case internal::named_arg_type: 
2977       FMT_ASSERT(false, "invalid argument type"); 
2978       break; 
2979     case internal::int_type: 
2980     case internal::uint_type: 
2981     case internal::long_long_type: 
2982     case internal::ulong_long_type: 
2983     case internal::int128_type: 
2984     case internal::uint128_type: 
2985     case internal::bool_type:

2987                            internal::int_type_checker<decltype(eh)>(eh));

2991           &specs_, internal::char_specs_checker<decltype(eh)>(specs_.type, eh)); 
2992       break; 
2993     case internal::double_type:

3081  private: 
3082   struct null_handler : internal::error_handler {

3084     void on_plus() {}

3088   };

3091   template <typename ParseContext> 
3092   auto parse(ParseContext& ctx) -> decltype(ctx.begin()) { 
3093     format_str_ = ctx.begin(); 
3094     // Checks are deferred to formatting time when the argument type is known. 
3095     internal::dynamic_specs_handler<ParseContext> handler(specs_, ctx); 
3096     return parse_format_specs(ctx.begin(), ctx.end(), handler); 
3097   }

3099   template <typename T, typename FormatContext> 
3100   auto format(const T& val, FormatContext& ctx) -> decltype(ctx.out()) { 
3101     handle_specs(ctx); 
3102     internal::specs_checker<null_handler> checker( 
3103         null_handler(), 
3104         internal::mapped_type_constant<T, FormatContext>::value); 
3105     checker.on_align(specs_.align); 
3106     switch (specs_.sign) { 
3107     case sign::none: 
3108       break; 
3109     case sign::plus: 
3110       checker.on_plus(); 
3111       break; 
3112     case sign::minus: 
3113       checker.on_minus(); 
3114       break; 
3115     case sign::space: 
3116       checker.on_space(); 
3117       break; 
3118     } 
3119     if (specs_.alt) checker.on_hash(); 
3120     if (specs_.precision >= 0) checker.end_precision(); 
3121     using range = internal::output_range<typename FormatContext::iterator, 
3122                                          typename FormatContext::char_type>; 
3123     visit_format_arg(arg_formatter<range>(ctx, nullptr, &specs_), 
3124                      internal::make_arg<FormatContext>(val)); 
3125     return ctx.out(); 
3126   }

3165     auto size = internal::to_unsigned(end - begin);

3192     if (visit_format_arg(f, arg)) return parse_context.begin();

3200     if (begin == end || *begin != '}') on_error("missing '}' in format string"); 
3201     advance_to(parse_context, begin); 
3202     context.advance_to( 
3203         visit_format_arg(ArgFormatter(context, &parse_context, &specs), arg)); 
3204     return begin; 
3205   }

 144 #  ifndef __clang__ 
 145 #    pragma intrinsic(_BitScanReverse)

 147 inline uint32_t clz(uint32_t x) { 
 148   unsigned long r = 0; 
 149   _BitScanReverse(&r, x); 
 150  
 151   assert(x != 0); 
 152   // Static analysis complains about using uninitialized data 
 153   // "r", but the only way that can happen is if "x" is 0, 
 154   // which the callers guarantee to not happen. 
 155 #  pragma warning(suppress : 6102) 
 156   return 31 - r; 
 157 } 
 158 #  define FMT_BUILTIN_CLZ(n) fmt::internal::clz(n) 
 159  
 160 #  if defined(_WIN64) && !defined(__clang__) 
 161 #    pragma intrinsic(_BitScanReverse64) 
 162 #  endif 
 163  
 164 inline uint32_t clzll(uint64_t x) { 
 165   unsigned long r = 0; 
 166 #  ifdef _WIN64 
 167   _BitScanReverse64(&r, x); 
 168 #  else 
 169   // Scan the high 32 bits. 
 170   if (_BitScanReverse(&r, static_cast<uint32_t>(x >> 32))) return 63 - (r + 32); 
 171  
 172   // Scan the low 32 bits. 
 173   _BitScanReverse(&r, static_cast<uint32_t>(x)); 
 174 #  endif 
 175  
 176   assert(x != 0); 
 177   // Static analysis complains about using uninitialized data 
 178   // "r", but the only way that can happen is if "x" is 0, 
 179   // which the callers guarantee to not happen. 
 180 #  pragma warning(suppress : 6102) 
 181   return 63 - r; 
 182 }

 231 // Detect if *any* given type models the OutputIterator concept. 
 232 template <typename It> class is_output_iterator { 
 233   // Check for mutability because all iterator categories derived from 
 234   // std::input_iterator_tag *may* also meet the requirements of an 
 235   // OutputIterator, thereby falling into the category of 'mutable iterators' 
 236   // [iterator.requirements.general] clause 4. The compiler reveals this 
 237   // property only at the point of *actually dereferencing* the iterator! 
 238   template <typename U> 
 239   static decltype(*(std::declval<U>())) test(std::input_iterator_tag); 
 240   template <typename U> static char& test(std::output_iterator_tag); 
 241   template <typename U> static const char& test(...);

 246   static const bool value = !std::is_const<remove_reference_t<type>>::value;

 495 /** 
 496   \rst 
 497   A dynamically growing memory buffer for trivially copyable/constructible types 
 498   with the first ``SIZE`` elements stored in the object itself. 
 499  
 500   You can use one of the following type aliases for common character types: 
 501  
 502   +----------------+------------------------------+ 
 503   | Type           | Definition                   | 
 504   +================+==============================+ 
 505   | memory_buffer  | basic_memory_buffer<char>    | 
 506   +----------------+------------------------------+ 
 507   | wmemory_buffer | basic_memory_buffer<wchar_t> | 
 508   +----------------+------------------------------+ 
 509  
 510   **Example**:: 
 511  
 512      fmt::memory_buffer out; 
 513      format_to(out, "The answer is {}.", 42); 
 514  
 515   This will append the following output to the ``out`` object: 
 516  
 517   .. code-block:: none 
 518  
 519      The answer is 42. 
 520  
 521   The output can be converted to an ``std::string`` with ``to_string(out)``. 
 522   \endrst 
 523  */ 
 524 template <typename T, std::size_t SIZE = inline_buffer_size, 
 525           typename Allocator = std::allocator<T>> 
 526 class basic_memory_buffer : private Allocator, public internal::buffer<T> { 
 527  private: 
 528   T store_[SIZE]; 
 529  
 530   // Deallocate memory allocated by the buffer. 
 531   void deallocate() { 
 532     T* data = this->data(); 
 533     if (data != store_) Allocator::deallocate(data, this->capacity()); 
 534   } 
 535  
 536  protected: 
 537   void grow(std::size_t size) FMT_OVERRIDE; 
 538  
 539  public: 
 540   using value_type = T; 
 541   using const_reference = const T&; 
 542  
 543   explicit basic_memory_buffer(const Allocator& alloc = Allocator()) 
 544       : Allocator(alloc) { 
 545     this->set(store_, SIZE); 
 546   } 
 547   ~basic_memory_buffer() { deallocate(); } 
 548  
 549  private: 
 550   // Move data from other to this buffer. 
 551   void move(basic_memory_buffer& other) { 
 552     Allocator &this_alloc = *this, &other_alloc = other; 
 553     this_alloc = std::move(other_alloc); 
 554     T* data = other.data(); 
 555     std::size_t size = other.size(), capacity = other.capacity(); 
 556     if (data == other.store_) { 
 557       this->set(store_, capacity); 
 558       std::uninitialized_copy(other.store_, other.store_ + size, 
 559                               internal::make_checked(store_, capacity)); 
 560     } else { 
 561       this->set(data, capacity); 
 562       // Set pointer to the inline array so that delete is not called 
 563       // when deallocating. 
 564       other.set(other.store_, 0); 
 565     } 
 566     this->resize(size); 
 567   } 
 568  
 569  public: 
 570   /** 
 571     \rst 
 572     Constructs a :class:`fmt::basic_memory_buffer` object moving the content 
 573     of the other object to it. 
 574     \endrst 
 575    */ 
 576   basic_memory_buffer(basic_memory_buffer&& other) { move(other); } 
 577  
 578   /** 
 579     \rst 
 580     Moves the content of the other ``basic_memory_buffer`` object to this one. 
 581     \endrst 
 582    */ 
 583   basic_memory_buffer& operator=(basic_memory_buffer&& other) { 
 584     assert(this != &other); 
 585     deallocate(); 
 586     move(other); 
 587     return *this; 
 588   } 
 589  
 590   // Returns a copy of the allocator associated with this buffer. 
 591   Allocator get_allocator() const { return *this; } 
 592 }; 
 593  
 594 template <typename T, std::size_t SIZE, typename Allocator> 
 595 void basic_memory_buffer<T, SIZE, Allocator>::grow(std::size_t size) { 
 596 #ifdef FUZZING_BUILD_MODE_UNSAFE_FOR_PRODUCTION 
 597   if (size > 1000) throw std::runtime_error("fuzz mode - won't grow that much"); 
 598 #endif 
 599   std::size_t old_capacity = this->capacity(); 
 600   std::size_t new_capacity = old_capacity + old_capacity / 2; 
 601   if (size > new_capacity) new_capacity = size; 
 602   T* old_data = this->data(); 
 603   T* new_data = std::allocator_traits<Allocator>::allocate(*this, new_capacity); 
 604   // The following code doesn't throw, so the raw pointer above doesn't leak. 
 605   std::uninitialized_copy(old_data, old_data + this->size(), 
 606                           internal::make_checked(new_data, new_capacity)); 
 607   this->set(new_data, new_capacity); 
 608   // deallocate must not throw according to the standard, but even if it does, 
 609   // the buffer already uses the new storage and will deallocate it in 
 610   // destructor. 
 611   if (old_data != store_) Allocator::deallocate(old_data, old_capacity); 
 612 }

 744 // An lg handler that formats a decimal number. 
 745 // Usage: lg(n, decimal_formatter(buffer)); 
 746 class decimal_formatter { 
 747  private: 
 748   char* buffer_; 
 749  
 750   void write_pair(unsigned N, uint32_t index) { 
 751     std::memcpy(buffer_ + N, data::digits + index * 2, 2); 
 752   } 
 753  
 754  public: 
 755   explicit decimal_formatter(char* buf) : buffer_(buf) {} 
 756  
 757   template <unsigned N> char* on(uint32_t u) { 
 758     if (N == 0) { 
 759       *buffer_ = static_cast<char>(u) + '0'; 
 760     } else if (N == 1) { 
 761       write_pair(0, u); 
 762     } else { 
 763       // The idea of using 4.32 fixed-point numbers is based on 
 764       // https://github.com/jeaiii/itoa 
 765       unsigned n = N - 1; 
 766       unsigned a = n / 5 * n * 53 / 16; 
 767       uint64_t t = 
 768           ((1ULL << (32 + a)) / data::zero_or_powers_of_10_32[n] + 1 - n / 9); 
 769       t = ((t * u) >> a) + n / 5 * 4; 
 770       write_pair(0, t >> 32); 
 771       for (unsigned i = 2; i < N; i += 2) { 
 772         t = 100ULL * static_cast<uint32_t>(t); 
 773         write_pair(i, t >> 32); 
 774       } 
 775       if (N % 2 == 0) { 
 776         buffer_[N] = 
 777             static_cast<char>((10ULL * static_cast<uint32_t>(t)) >> 32) + '0'; 
 778       } 
 779     } 
 780     return buffer_ += N + 1; 
 781   } 
 782 };

 808 // Formats a decimal unsigned integer value writing into buffer. 
 809 // add_thousands_sep is called after writing each char to add a thousands 
 810 // separator if necessary. 
 811 template <typename UInt, typename Char, typename F> 
 812 inline Char* format_decimal(Char* buffer, UInt value, int num_digits, 
 813                             F add_thousands_sep) { 
 814   FMT_ASSERT(num_digits >= 0, "invalid digit count"); 
 815   buffer += num_digits; 
 816   Char* end = buffer; 
 817   while (value >= 100) { 
 818     // Integer division is slow so do it for a group of two digits instead 
 819     // of for every digit. The idea comes from the talk by Alexandrescu 
 820     // "Three Optimization Tips for C++". See speed-test for a comparison. 
 821     unsigned index = static_cast<unsigned>((value % 100) * 2); 
 822     value /= 100; 
 823     *--buffer = static_cast<Char>(data::digits[index + 1]); 
 824     add_thousands_sep(buffer); 
 825     *--buffer = static_cast<Char>(data::digits[index]); 
 826     add_thousands_sep(buffer); 
 827   } 
 828   if (value < 10) { 
 829     *--buffer = static_cast<Char>('0' + value); 
 830     return end; 
 831   } 
 832   unsigned index = static_cast<unsigned>(value * 2); 
 833   *--buffer = static_cast<Char>(data::digits[index + 1]); 
 834   add_thousands_sep(buffer); 
 835   *--buffer = static_cast<Char>(data::digits[index]); 
 836   return end; 
 837 }

 861 template <unsigned BASE_BITS, typename Char, typename UInt> 
 862 inline Char* format_uint(Char* buffer, UInt value, int num_digits, 
 863                          bool upper = false) { 
 864   buffer += num_digits; 
 865   Char* end = buffer; 
 866   do { 
 867     const char* digits = upper ? "0123456789ABCDEF" : data::hex_digits; 
 868     unsigned digit = (value & ((1 << BASE_BITS) - 1)); 
 869     *--buffer = static_cast<Char>(BASE_BITS < 4 ? static_cast<char>('0' + digit) 
 870                                                 : digits[digit]); 
 871   } while ((value >>= BASE_BITS) != 0); 
 872   return end; 
 873 }

 875 template <unsigned BASE_BITS, typename Char> 
 876 Char* format_uint(Char* buffer, internal::fallback_uintptr n, int num_digits, 
 877                   bool = false) { 
 878   auto char_digits = std::numeric_limits<unsigned char>::digits / 4; 
 879   int start = (num_digits + char_digits - 1) / char_digits - 1; 
 880   if (int start_digits = num_digits % char_digits) { 
 881     unsigned value = n.value[start--]; 
 882     buffer = format_uint<BASE_BITS>(buffer, value, start_digits); 
 883   } 
 884   for (; start >= 0; --start) { 
 885     unsigned value = n.value[start]; 
 886     buffer += char_digits; 
 887     auto p = buffer; 
 888     for (int i = 0; i < char_digits; ++i) { 
 889       unsigned digit = (value & ((1 << BASE_BITS) - 1)); 
 890       *--p = static_cast<Char>(data::hex_digits[digit]); 
 891       value >>= BASE_BITS; 
 892     } 
 893   } 
 894   return buffer; 
 895 }

1010 // Writes the exponent exp in the form "[+-]d{2,3}" to buffer. 
1011 template <typename Char, typename It> It write_exponent(int exp, It it) { 
1012   FMT_ASSERT(-1000 < exp && exp < 1000, "exponent out of range"); 
1013   if (exp < 0) { 
1014     *it++ = static_cast<Char>('-'); 
1015     exp = -exp; 
1016   } else { 
1017     *it++ = static_cast<Char>('+'); 
1018   } 
1019   if (exp >= 100) { 
1020     *it++ = static_cast<Char>(static_cast<char>('0' + exp / 100)); 
1021     exp %= 100; 
1022   } 
1023   const char* d = data::digits + exp * 2; 
1024   *it++ = static_cast<Char>(d[0]); 
1025   *it++ = static_cast<Char>(d[1]); 
1026   return it;

1036 // The number is given as v = digits * pow(10, exp). 
1037 template <typename Char, typename It> 
1038 It grisu_prettify(const char* digits, int size, int exp, It it, 
1039                   gen_digits_params params, Char decimal_point) { 
1040   // pow(10, full_exp - 1) <= v <= pow(10, full_exp). 
1041   int full_exp = size + exp; 
1042   if (!params.fixed) { 
1043     // Insert a decimal point after the first digit and add an exponent. 
1044     *it++ = static_cast<Char>(*digits); 
1045     if (size > 1) *it++ = decimal_point; 
1046     exp += size - 1; 
1047     it = copy_str<Char>(digits + 1, digits + size, it); 
1048     if (size < params.num_digits) 
1049       it = std::fill_n(it, params.num_digits - size, static_cast<Char>('0')); 
1050     *it++ = static_cast<Char>(params.upper ? 'E' : 'e'); 
1051     return write_exponent<Char>(exp, it); 
1052   } 
1053   if (size <= full_exp) { 
1054     // 1234e7 -> 12340000000[.0+] 
1055     it = copy_str<Char>(digits, digits + size, it); 
1056     it = std::fill_n(it, full_exp - size, static_cast<Char>('0')); 
1057     int num_zeros = (std::max)(params.num_digits - full_exp, 1); 
1058     if (params.trailing_zeros) { 
1059       *it++ = decimal_point; 
1060 #ifdef FUZZING_BUILD_MODE_UNSAFE_FOR_PRODUCTION 
1061       if (num_zeros > 1000) 
1062         throw std::runtime_error("fuzz mode - avoiding excessive cpu use"); 
1063 #endif 
1064       it = std::fill_n(it, num_zeros, static_cast<Char>('0')); 
1065     } 
1066   } else if (full_exp > 0) { 
1067     // 1234e-2 -> 12.34[0+] 
1068     it = copy_str<Char>(digits, digits + full_exp, it); 
1069     if (!params.trailing_zeros) { 
1070       // Remove trailing zeros. 
1071       while (size > full_exp && digits[size - 1] == '0') --size; 
1072       if (size != full_exp) *it++ = decimal_point; 
1073       return copy_str<Char>(digits + full_exp, digits + size, it); 
1074     } 
1075     *it++ = decimal_point; 
1076     it = copy_str<Char>(digits + full_exp, digits + size, it); 
1077     if (params.num_digits > size) { 
1078       // Add trailing zeros. 
1079       int num_zeros = params.num_digits - size; 
1080       it = std::fill_n(it, num_zeros, static_cast<Char>('0')); 
1081     } 
1082   } else { 
1083     // 1234e-6 -> 0.001234 
1084     *it++ = static_cast<Char>('0'); 
1085     int num_zeros = -full_exp; 
1086     if (params.num_digits >= 0 && params.num_digits < num_zeros) 
1087       num_zeros = params.num_digits; 
1088     if (!params.trailing_zeros) 
1089       while (size > 0 && digits[size - 1] == '0') --size; 
1090     if (num_zeros != 0 || size != 0) { 
1091       *it++ = decimal_point; 
1092       it = std::fill_n(it, num_zeros, static_cast<Char>('0')); 
1093       it = copy_str<Char>(digits, digits + size, it); 
1094     } 
1095   } 
1096   return it; 
1097 }

1327   //   <left-padding><prefix><numeric-padding><digits><right-padding> 
1328   // where <digits> are written by f(it). 
1329   template <typename F> 
1330   void write_int(int num_digits, string_view prefix, format_specs specs, F f) { 
1331     std::size_t size = prefix.size() + internal::to_unsigned(num_digits); 
1332     char_type fill = specs.fill[0]; 
1333     std::size_t padding = 0; 
1334     if (specs.align == align::numeric) { 
1335       auto unsiged_width = internal::to_unsigned(specs.width); 
1336       if (unsiged_width > size) { 
1337         padding = unsiged_width - size; 
1338         size = unsiged_width; 
1339       } 
1340     } else if (specs.precision > num_digits) { 
1341       size = prefix.size() + internal::to_unsigned(specs.precision); 
1342       padding = internal::to_unsigned(specs.precision - num_digits); 
1343       fill = static_cast<char_type>('0'); 
1344     } 
1345     if (specs.align == align::none) specs.align = align::right; 
1346     write_padded(specs, padded_int_writer<F>{size, prefix, fill, padding, f}); 
1347   }

1606   // Writes a value in the format 
1607   //   <left-padding><value><right-padding> 
1608   // where <value> is written by f(it). 
1609   template <typename F> void write_padded(const format_specs& specs, F&& f) { 
1610     // User-perceived width (in code points). 
1611     unsigned width = to_unsigned(specs.width); 
1612     size_t size = f.size();  // The number of code units. 
1613     size_t num_code_points = width != 0 ? f.width() : size; 
1614     if (width <= num_code_points) return f(reserve(size)); 
1615     auto&& it = reserve(width + (size - num_code_points)); 
1616     char_type fill = specs.fill[0]; 
1617     std::size_t padding = width - num_code_points; 
1618     if (specs.align == align::right) { 
1619       it = std::fill_n(it, padding, fill); 
1620       f(it); 
1621     } else if (specs.align == align::center) { 
1622       std::size_t left_padding = padding / 2; 
1623       it = std::fill_n(it, left_padding, fill); 
1624       f(it); 
1625       it = std::fill_n(it, padding - left_padding, fill); 
1626     } else { 
1627       f(it); 
1628       it = std::fill_n(it, padding, fill); 
1629     } 
1630   }

1875 // Parses the range [begin, end) as an unsigned integer. This function assumes 
1876 // that the range is non-empty and the first character is a digit. 
1877 template <typename Char, typename ErrorHandler> 
1878 FMT_CONSTEXPR int parse_nonnegative_int(const Char*& begin, const Char* end, 
1879                                         ErrorHandler&& eh) { 
1880   assert(begin != end && '0' <= *begin && *begin <= '9'); 
1881   if (*begin == '0') { 
1882     ++begin; 
1883     return 0; 
1884   } 
1885   unsigned value = 0; 
1886   // Convert to unsigned to prevent a warning. 
1887   constexpr unsigned max_int = (std::numeric_limits<int>::max)(); 
1888   unsigned big = max_int / 10; 
1889   do { 
1890     // Check for overflow. 
1891     if (value > big) { 
1892       value = max_int + 1; 
1893       break; 
1894     } 
1895     value = value * 10 + unsigned(*begin - '0'); 
1896     ++begin; 
1897   } while (begin != end && '0' <= *begin && *begin <= '9'); 
1898   if (value > max_int) eh.on_error("number is too big"); 
1899   return static_cast<int>(value); 
1900 }

2232                                        IDHandler&& handler) {

2236   if (c >= '0' && c <= '9') { 
2237     int index = parse_nonnegative_int(begin, end, handler); 
2238     if (begin == end || (*begin != '}' && *begin != ':')) 
2239       return handler.on_error("invalid format string"), begin; 
2240     handler(index); 
2241     return begin; 
2242   } 
2243   if (!is_name_start(c)) 
2244     return handler.on_error("invalid format string"), begin; 
2245   auto it = begin; 
2246   do { 
2247     ++it; 
2248   } while (it != end && (is_name_start(c = *it) || ('0' <= c && c <= '9'))); 
2249   handler(basic_string_view<Char>(begin, to_unsigned(it - begin))); 
2250   return it; 
2251 }

2287 // Parses fill and alignment. 
2288 template <typename Char, typename Handler> 
2289 FMT_CONSTEXPR const Char* parse_align(const Char* begin, const Char* end, 
2290                                       Handler&& handler) { 
2291   FMT_ASSERT(begin != end, ""); 
2292   auto align = align::none; 
2293   int i = 0; 
2294   if (begin + 1 != end) ++i; 
2295   do { 
2296     switch (static_cast<char>(begin[i])) { 
2297     case '<': 
2298       align = align::left; 
2299       break; 
2300     case '>': 
2301       align = align::right; 
2302       break; 
2303     case '=': 
2304       align = align::numeric; 
2305       break; 
2306     case '^': 
2307       align = align::center; 
2308       break; 
2309     } 
2310     if (align != align::none) { 
2311       if (i > 0) { 
2312         auto c = *begin; 
2313         if (c == '{') 
2314           return handler.on_error("invalid fill character '{'"), begin; 
2315         begin += 2; 
2316         handler.on_fill(c); 
2317       } else 
2318         ++begin; 
2319       handler.on_align(align); 
2320       break; 
2321     } 
2322   } while (i-- > 0); 
2323   return begin; 
2324 }

2326 template <typename Char, typename Handler> 
2327 FMT_CONSTEXPR const Char* parse_width(const Char* begin, const Char* end, 
2328                                       Handler&& handler) { 
2329   FMT_ASSERT(begin != end, ""); 
2330   if ('0' <= *begin && *begin <= '9') { 
2331     handler.on_width(parse_nonnegative_int(begin, end, handler)); 
2332   } else if (*begin == '{') { 
2333     ++begin; 
2334     if (begin != end) 
2335       begin = parse_arg_id(begin, end, width_adapter<Handler, Char>(handler)); 
2336     if (begin == end || *begin != '}') 
2337       return handler.on_error("invalid format string"), begin; 
2338     ++begin; 
2339   } 
2340   return begin; 
2341 }

2343 template <typename Char, typename Handler> 
2344 FMT_CONSTEXPR const Char* parse_precision(const Char* begin, const Char* end, 
2345                                           Handler&& handler) { 
2346   ++begin; 
2347   auto c = begin != end ? *begin : Char(); 
2348   if ('0' <= c && c <= '9') { 
2349     handler.on_precision(parse_nonnegative_int(begin, end, handler)); 
2350   } else if (c == '{') { 
2351     ++begin; 
2352     if (begin != end) { 
2353       begin = 
2354           parse_arg_id(begin, end, precision_adapter<Handler, Char>(handler)); 
2355     } 
2356     if (begin == end || *begin++ != '}') 
2357       return handler.on_error("invalid format string"), begin; 
2358   } else { 
2359     return handler.on_error("missing precision specifier"), begin; 
2360   } 
2361   handler.end_precision(); 
2362   return begin;

2365 // Parses standard format specifiers and sends notifications about parsed 
2366 // components to handler. 
2367 template <typename Char, typename SpecHandler> 
2368 FMT_CONSTEXPR const Char* parse_format_specs(const Char* begin, const Char* end, 
2369                                              SpecHandler&& handler) { 
2370   if (begin == end || *begin == '}') return begin; 
2371  
2372   begin = parse_align(begin, end, handler); 
2373   if (begin == end) return begin; 
2374  
2375   // Parse sign. 
2376   switch (static_cast<char>(*begin)) { 
2377   case '+': 
2378     handler.on_plus(); 
2379     ++begin; 
2380     break; 
2381   case '-': 
2382     handler.on_minus(); 
2383     ++begin; 
2384     break; 
2385   case ' ': 
2386     handler.on_space(); 
2387     ++begin; 
2388     break; 
2389   } 
2390   if (begin == end) return begin; 
2391  
2392   if (*begin == '#') { 
2393     handler.on_hash(); 
2394     if (++begin == end) return begin; 
2395   } 
2396  
2397   // Parse zero flag. 
2398   if (*begin == '0') { 
2399     handler.on_zero(); 
2400     if (++begin == end) return begin; 
2401   } 
2402  
2403   begin = parse_width(begin, end, handler); 
2404   if (begin == end) return begin; 
2405  
2406   // Parse precision. 
2407   if (*begin == '.') { 
2408     begin = parse_precision(begin, end, handler); 
2409   } 
2410  
2411   // Parse type. 
2412   if (begin != end && *begin != '}') handler.on_type(*begin++); 
2413   return begin; 
2414 }

2455         ++p; 
2456         if (p == end || *p != '}') 
2457           return handler_.on_error("unmatched '}' in format string"); 
2458         handler_.on_text(begin, p); 
2459         begin = p + 1; 
2460       } 
2461     } 
2462     Handler& handler_; 
2463   } write{handler}; 
2464   auto begin = format_str.data(); 
2465   auto end = begin + format_str.size(); 
2466   while (begin != end) { 
2467     // Doing two passes with memchr (one for '{' and another for '}') is up to 
2468     // 2.5x faster than the naive one-pass implementation on big format strings. 
2469     const Char* p = begin; 
2470     if (*begin != '{' && !find<IS_CONSTEXPR>(begin, end, '{', p)) 
2471       return write(begin, end); 
2472     write(begin, p); 
2473     ++p; 
2474     if (p == end) return handler.on_error("invalid format string"); 
2475     if (static_cast<char>(*p) == '}') { 
2476       handler.on_arg_id(); 
2477       handler.on_replacement_field(p); 
2478     } else if (*p == '{') { 
2479       handler.on_text(p, p + 1); 
2480     } else { 
2481       p = parse_arg_id(p, end, id_adapter<Handler, Char>{handler}); 
2482       Char c = p != end ? *p : Char(); 
2483       if (c == '}') { 
2484         handler.on_replacement_field(p); 
2485       } else if (c == ':') { 
2486         p = handler.on_format_specs(p + 1, end); 
2487         if (p == end || *p != '}') 
2488           return handler.on_error("unknown format specifier"); 
2489       } else { 
2490         return handler.on_error("missing '}' in format string"); 
2491       } 
2492     } 
2493     begin = p + 1; 
2494   } 
2495 }

2751 template <typename Range> 
2752 template <typename T, bool USE_GRISU> 
2753 void internal::basic_writer<Range>::write_double(T value, 
2754                                                  const format_specs& specs) { 
2755   // Check type. 
2756   float_spec_handler handler(static_cast<char>(specs.type)); 
2757   internal::handle_float_type_spec(handler.type, handler); 
2758  
2759   char sign = 0; 
2760   // Use signbit instead of value < 0 since the latter is always false for NaN. 
2761   if (std::signbit(value)) { 
2762     sign = '-'; 
2763     value = -value; 
2764   } else if (specs.sign != sign::none) { 
2765     if (specs.sign == sign::plus) 
2766       sign = '+'; 
2767     else if (specs.sign == sign::space) 
2768       sign = ' '; 
2769   } 
2770  
2771   if (!std::isfinite(value)) { 
2772     // Format infinity and NaN ourselves because sprintf's output is not 
2773     // consistent across platforms. 
2774     const char* str = std::isinf(value) ? (handler.upper ? "INF" : "inf") 
2775                                         : (handler.upper ? "NAN" : "nan"); 
2776     return write_padded(specs, 
2777                         inf_or_nan_writer{sign, handler.as_percentage, str}); 
2778   } 
2779  
2780   if (handler.as_percentage) value *= 100; 
2781  
2782   memory_buffer buffer; 
2783   int exp = 0; 
2784   int precision = specs.precision >= 0 || !specs.type ? specs.precision : 6; 
2785   unsigned options = handler.fixed ? internal::grisu_options::fixed : 0; 
2786   bool use_grisu = USE_GRISU && 
2787                    (specs.type != 'a' && specs.type != 'A' && 
2788                     specs.type != 'e' && specs.type != 'E') && 
2789                    internal::grisu_format(static_cast<double>(value), buffer, 
2790                                           precision, options, exp); 
2791   char* decimal_point_pos = nullptr; 
2792   if (!use_grisu) 
2793     decimal_point_pos = internal::sprintf_format(value, buffer, specs); 
2794  
2795   if (handler.as_percentage) { 
2796     buffer.push_back('%'); 
2797     --exp;  // Adjust decimal place position. 
2798   } 
2799   format_specs as = specs; 
2800   if (specs.align == align::numeric) { 
2801     if (sign) { 
2802       auto&& it = reserve(1); 
2803       *it++ = static_cast<char_type>(sign); 
2804       sign = 0; 
2805       if (as.width) --as.width; 
2806     } 
2807     as.align = align::right; 
2808   } else if (specs.align == align::none) { 
2809     as.align = align::right; 
2810   } 
2811   char_type decimal_point = handler.use_locale 
2812                                 ? internal::decimal_point<char_type>(locale_) 
2813                                 : static_cast<char_type>('.'); 
2814   if (use_grisu) { 
2815     auto params = internal::gen_digits_params(); 
2816     params.fixed = handler.fixed; 
2817     params.num_digits = precision; 
2818     params.trailing_zeros = 
2819         (precision != 0 && (handler.fixed || !specs.type)) || specs.alt; 
2820     write_padded(as, grisu_writer(sign, buffer, exp, params, decimal_point)); 
2821   } else { 
2822     write_padded(as, 
2823                  double_writer{sign, buffer, decimal_point_pos, decimal_point}); 
2824   } 
2825 }

2881 /** Fast integer formatter. */ 
2882 class format_int { 
2883  private: 
2884   // Buffer should be large enough to hold all digits (digits10 + 1), 
2885   // a sign and a null character. 
2886   enum { buffer_size = std::numeric_limits<unsigned long long>::digits10 + 3 }; 
2887   mutable char buffer_[buffer_size]; 
2888   char* str_; 
2889  
2890   // Formats value in reverse and returns a pointer to the beginning. 
2891   char* format_decimal(unsigned long long value) { 
2892     char* ptr = buffer_ + (buffer_size - 1);  // Parens to workaround MSVC bug. 
2893     while (value >= 100) { 
2894       // Integer division is slow so do it for a group of two digits instead 
2895       // of for every digit. The idea comes from the talk by Alexandrescu 
2896       // "Three Optimization Tips for C++". See speed-test for a comparison. 
2897       unsigned index = static_cast<unsigned>((value % 100) * 2); 
2898       value /= 100; 
2899       *--ptr = internal::data::digits[index + 1]; 
2900       *--ptr = internal::data::digits[index]; 
2901     } 
2902     if (value < 10) { 
2903       *--ptr = static_cast<char>('0' + value); 
2904       return ptr; 
2905     } 
2906     unsigned index = static_cast<unsigned>(value * 2); 
2907     *--ptr = internal::data::digits[index + 1]; 
2908     *--ptr = internal::data::digits[index]; 
2909     return ptr; 
2910   } 
2911  
2912   void format_signed(long long value) { 
2913     unsigned long long abs_value = static_cast<unsigned long long>(value); 
2914     bool negative = value < 0; 
2915     if (negative) abs_value = 0 - abs_value; 
2916     str_ = format_decimal(abs_value); 
2917     if (negative) *--str_ = '-'; 
2918   } 
2919  
2920  public: 
2921   explicit format_int(int value) { format_signed(value); } 
2922   explicit format_int(long value) { format_signed(value); } 
2923   explicit format_int(long long value) { format_signed(value); } 
2924   explicit format_int(unsigned value) : str_(format_decimal(value)) {} 
2925   explicit format_int(unsigned long value) : str_(format_decimal(value)) {} 
2926   explicit format_int(unsigned long long value) : str_(format_decimal(value)) {} 
2927  
2928   /** Returns the number of characters written to the output buffer. */ 
2929   std::size_t size() const { 
2930     return internal::to_unsigned(buffer_ - str_ + buffer_size - 1); 
2931   } 
2932  
2933   /** 
2934     Returns a pointer to the output buffer content. No terminating null 
2935     character is appended. 
2936    */ 
2937   const char* data() const { return str_; } 
2938  
2939   /** 
2940     Returns a pointer to the output buffer content with terminating null 
2941     character appended. 
2942    */ 
2943   const char* c_str() const { 
2944     buffer_[buffer_size - 1] = '\0'; 
2945     return str_; 
2946   } 
2947  
2948   /** 
2949     \rst 
2950     Returns the content of the output buffer as an ``std::string``. 
2951     \endrst 
2952    */ 
2953   std::string str() const { return std::string(str_, size()); } 
2954 };

3091   template <typename ParseContext> 
3092   auto parse(ParseContext& ctx) -> decltype(ctx.begin()) { 
3093     format_str_ = ctx.begin(); 
3094     // Checks are deferred to formatting time when the argument type is known. 
3095     internal::dynamic_specs_handler<ParseContext> handler(specs_, ctx); 
3096     return parse_format_specs(ctx.begin(), ctx.end(), handler);

3109     case sign::plus:

3111       break; 
3112     case sign::minus: 
3113       checker.on_minus(); 
3114       break; 
3115     case sign::space: 
3116       checker.on_space(); 
3117       break; 
3118     } 
3119     if (specs_.alt) checker.on_hash(); 
3120     if (specs_.precision >= 0) checker.end_precision(); 
3121     using range = internal::output_range<typename FormatContext::iterator, 
3122                                          typename FormatContext::char_type>; 
3123     visit_format_arg(arg_formatter<range>(ctx, nullptr, &specs_), 
3124                      internal::make_arg<FormatContext>(val));

3126   }
