 37 def cut( 
 38     x, 
 39     bins, 
 40     right=True, 
 41     labels=None, 
 42     retbins=False, 
 43     precision=3, 
 44     include_lowest=False, 
 45     duplicates="raise", 
 46 ): 
 47     """ 
 48     Bin values into discrete intervals. 
 49  
 50     Use `cut` when you need to segment and sort data values into bins. This 
 51     function is also useful for going from a continuous variable to a 
 52     categorical variable. For example, `cut` could convert ages to groups of 
 53     age ranges. Supports binning into an equal number of bins, or a 
 54     pre-specified array of bins. 
 55  
 56     Parameters 
 57     ---------- 
 58     x : array-like 
 59         The input array to be binned. Must be 1-dimensional. 
 60     bins : int, sequence of scalars, or IntervalIndex 
 61         The criteria to bin by. 
 62  
 63         * int : Defines the number of equal-width bins in the range of `x`. The 
 64           range of `x` is extended by .1% on each side to include the minimum 
 65           and maximum values of `x`. 
 66         * sequence of scalars : Defines the bin edges allowing for non-uniform 
 67           width. No extension of the range of `x` is done. 
 68         * IntervalIndex : Defines the exact bins to be used. Note that 
 69           IntervalIndex for `bins` must be non-overlapping. 
 70  
 71     right : bool, default True 
 72         Indicates whether `bins` includes the rightmost edge or not. If 
 73         ``right == True`` (the default), then the `bins` ``[1, 2, 3, 4]`` 
 74         indicate (1,2], (2,3], (3,4]. This argument is ignored when 
 75         `bins` is an IntervalIndex. 
 76     labels : array or bool, optional 
 77         Specifies the labels for the returned bins. Must be the same length as 
 78         the resulting bins. If False, returns only integer indicators of the 
 79         bins. This affects the type of the output container (see below). 
 80         This argument is ignored when `bins` is an IntervalIndex. 
 81     retbins : bool, default False 
 82         Whether to return the bins or not. Useful when bins is provided 
 83         as a scalar. 
 84     precision : int, default 3 
 85         The precision at which to store and display the bins labels. 
 86     include_lowest : bool, default False 
 87         Whether the first interval should be left-inclusive or not. 
 88     duplicates : {default 'raise', 'drop'}, optional 
 89         If bin edges are not unique, raise ValueError or drop non-uniques. 
 90  
 91         .. versionadded:: 0.23.0 
 92  
 93     Returns 
 94     ------- 
 95     out : Categorical, Series, or ndarray 
 96         An array-like object representing the respective bin for each value 
 97         of `x`. The type depends on the value of `labels`. 
 98  
 99         * True (default) : returns a Series for Series `x` or a 
100           Categorical for all other inputs. The values stored within 
101           are Interval dtype. 
102  
103         * sequence of scalars : returns a Series for Series `x` or a 
104           Categorical for all other inputs. The values stored within 
105           are whatever the type in the sequence is. 
106  
107         * False : returns an ndarray of integers. 
108  
109     bins : numpy.ndarray or IntervalIndex. 
110         The computed or specified bins. Only returned when `retbins=True`. 
111         For scalar or sequence `bins`, this is an ndarray with the computed 
112         bins. If set `duplicates=drop`, `bins` will drop non-unique bin. For 
113         an IntervalIndex `bins`, this is equal to `bins`. 
114  
115     See Also 
116     -------- 
117     qcut : Discretize variable into equal-sized buckets based on rank 
118         or based on sample quantiles. 
119     Categorical : Array type for storing data that come from a 
120         fixed set of values. 
121     Series : One-dimensional array with axis labels (including time series). 
122     IntervalIndex : Immutable Index implementing an ordered, sliceable set. 
123  
124     Notes 
125     ----- 
126     Any NA values will be NA in the result. Out of bounds values will be NA in 
127     the resulting Series or Categorical object. 
128  
129     Examples 
130     -------- 
131     Discretize into three equal-sized bins. 
132  
133     >>> pd.cut(np.array([1, 7, 5, 4, 6, 3]), 3) 
134     ... # doctest: +ELLIPSIS 
135     [(0.994, 3.0], (5.0, 7.0], (3.0, 5.0], (3.0, 5.0], (5.0, 7.0], ... 
136     Categories (3, interval[float64]): [(0.994, 3.0] < (3.0, 5.0] ... 
137  
138     >>> pd.cut(np.array([1, 7, 5, 4, 6, 3]), 3, retbins=True) 
139     ... # doctest: +ELLIPSIS 
140     ([(0.994, 3.0], (5.0, 7.0], (3.0, 5.0], (3.0, 5.0], (5.0, 7.0], ... 
141     Categories (3, interval[float64]): [(0.994, 3.0] < (3.0, 5.0] ... 
142     array([0.994, 3.   , 5.   , 7.   ])) 
143  
144     Discovers the same bins, but assign them specific labels. Notice that 
145     the returned Categorical's categories are `labels` and is ordered. 
146  
147     >>> pd.cut(np.array([1, 7, 5, 4, 6, 3]), 
148     ...        3, labels=["bad", "medium", "good"]) 
149     [bad, good, medium, medium, good, bad] 
150     Categories (3, object): [bad < medium < good] 
151  
152     ``labels=False`` implies you just want the bins back. 
153  
154     >>> pd.cut([0, 1, 1, 2], bins=4, labels=False) 
155     array([0, 1, 1, 3]) 
156  
157     Passing a Series as an input returns a Series with categorical dtype: 
158  
159     >>> s = pd.Series(np.array([2, 4, 6, 8, 10]), 
160     ...               index=['a', 'b', 'c', 'd', 'e']) 
161     >>> pd.cut(s, 3) 
162     ... # doctest: +ELLIPSIS 
163     a    (1.992, 4.667] 
164     b    (1.992, 4.667] 
165     c    (4.667, 7.333] 
166     d     (7.333, 10.0] 
167     e     (7.333, 10.0] 
168     dtype: category 
169     Categories (3, interval[float64]): [(1.992, 4.667] < (4.667, ... 
170  
171     Passing a Series as an input returns a Series with mapping value. 
172     It is used to map numerically to intervals based on bins. 
173  
174     >>> s = pd.Series(np.array([2, 4, 6, 8, 10]), 
175     ...               index=['a', 'b', 'c', 'd', 'e']) 
176     >>> pd.cut(s, [0, 2, 4, 6, 8, 10], labels=False, retbins=True, right=False) 
177     ... # doctest: +ELLIPSIS 
178     (a    0.0 
179      b    1.0 
180      c    2.0 
181      d    3.0 
182      e    4.0 
183      dtype: float64, array([0, 2, 4, 6, 8])) 
184  
185     Use `drop` optional when bins is not unique 
186  
187     >>> pd.cut(s, [0, 2, 4, 6, 10, 10], labels=False, retbins=True, 
188     ...        right=False, duplicates='drop') 
189     ... # doctest: +ELLIPSIS 
190     (a    0.0 
191      b    1.0 
192      c    2.0 
193      d    3.0 
194      e    3.0 
195      dtype: float64, array([0, 2, 4, 6, 8])) 
196  
197     Passing an IntervalIndex for `bins` results in those categories exactly. 
198     Notice that values not covered by the IntervalIndex are set to NaN. 0 
199     is to the left of the first bin (which is closed on the right), and 1.5 
200     falls between two bins. 
201  
202     >>> bins = pd.IntervalIndex.from_tuples([(0, 1), (2, 3), (4, 5)]) 
203     >>> pd.cut([0, 0.5, 1.5, 2.5, 4.5], bins) 
204     [NaN, (0, 1], NaN, (2, 3], (4, 5]] 
205     Categories (3, interval[int64]): [(0, 1] < (2, 3] < (4, 5]] 
206     """ 
207     # NOTE: this binning code is changed a bit from histogram for var(x) == 0 
208  
209     # for handling the cut for datetime and timedelta objects 
210     x_is_series, series_index, name, x = _preprocess_for_cut(x) 
211     x, dtype = _coerce_to_type(x) 
212  
213     if not np.iterable(bins): 
214         if is_scalar(bins) and bins < 1: 
215             raise ValueError("`bins` should be a positive integer.") 
216  
217         try:  # for array-like 
218             sz = x.size 
219         except AttributeError: 
220             x = np.asarray(x) 
221             sz = x.size 
222  
223         if sz == 0: 
224             raise ValueError("Cannot cut empty array") 
225  
226         rng = (nanops.nanmin(x), nanops.nanmax(x)) 
227         mn, mx = [mi + 0.0 for mi in rng] 
228  
229         if np.isinf(mn) or np.isinf(mx): 
230             # GH 24314 
231             raise ValueError( 
232                 "cannot specify integer `bins` when input data contains infinity" 
233             ) 
234         elif mn == mx:  # adjust end points before binning 
235             mn -= 0.001 * abs(mn) if mn != 0 else 0.001 
236             mx += 0.001 * abs(mx) if mx != 0 else 0.001 
237             bins = np.linspace(mn, mx, bins + 1, endpoint=True) 
238         else:  # adjust end points after binning 
239             bins = np.linspace(mn, mx, bins + 1, endpoint=True) 
240             adj = (mx - mn) * 0.001  # 0.1% of the range 
241             if right: 
242                 bins[0] -= adj 
243             else: 
244                 bins[-1] += adj 
245  
246     elif isinstance(bins, IntervalIndex): 
247         if bins.is_overlapping: 
248             raise ValueError("Overlapping IntervalIndex is not accepted.") 
249  
250     else: 
251         if is_datetime64tz_dtype(bins): 
252             bins = np.asarray(bins, dtype=_NS_DTYPE) 
253         else: 
254             bins = np.asarray(bins) 
255         bins = _convert_bin_to_numeric_type(bins, dtype) 
256  
257         # GH 26045: cast to float64 to avoid an overflow 
258         if (np.diff(bins.astype("float64")) < 0).any(): 
259             raise ValueError("bins must increase monotonically.") 
260  
261     fac, bins = _bins_to_cuts( 
262         x, 
263         bins, 
264         right=right, 
265         labels=labels, 
266         precision=precision, 
267         include_lowest=include_lowest, 
268         dtype=dtype, 
269         duplicates=duplicates, 
270     ) 
271  
272     return _postprocess_for_cut( 
273         fac, bins, retbins, x_is_series, series_index, name, dtype 
274     )

277 def qcut(x, q, labels=None, retbins=False, precision=3, duplicates="raise"): 
278     """ 
279     Quantile-based discretization function. Discretize variable into 
280     equal-sized buckets based on rank or based on sample quantiles. For example 
281     1000 values for 10 quantiles would produce a Categorical object indicating 
282     quantile membership for each data point. 
283  
284     Parameters 
285     ---------- 
286     x : 1d ndarray or Series 
287     q : integer or array of quantiles 
288         Number of quantiles. 10 for deciles, 4 for quartiles, etc. Alternately 
289         array of quantiles, e.g. [0, .25, .5, .75, 1.] for quartiles 
290     labels : array or boolean, default None 
291         Used as labels for the resulting bins. Must be of the same length as 
292         the resulting bins. If False, return only integer indicators of the 
293         bins. 
294     retbins : bool, optional 
295         Whether to return the (bins, labels) or not. Can be useful if bins 
296         is given as a scalar. 
297     precision : int, optional 
298         The precision at which to store and display the bins labels 
299     duplicates : {default 'raise', 'drop'}, optional 
300         If bin edges are not unique, raise ValueError or drop non-uniques. 
301  
302         .. versionadded:: 0.20.0 
303  
304     Returns 
305     ------- 
306     out : Categorical or Series or array of integers if labels is False 
307         The return type (Categorical or Series) depends on the input: a Series 
308         of type category if input is a Series else Categorical. Bins are 
309         represented as categories when categorical data is returned. 
310     bins : ndarray of floats 
311         Returned only if `retbins` is True. 
312  
313     Notes 
314     ----- 
315     Out of bounds values will be NA in the resulting Categorical object 
316  
317     Examples 
318     -------- 
319     >>> pd.qcut(range(5), 4) 
320     ... # doctest: +ELLIPSIS 
321     [(-0.001, 1.0], (-0.001, 1.0], (1.0, 2.0], (2.0, 3.0], (3.0, 4.0]] 
322     Categories (4, interval[float64]): [(-0.001, 1.0] < (1.0, 2.0] ... 
323  
324     >>> pd.qcut(range(5), 3, labels=["good", "medium", "bad"]) 
325     ... # doctest: +SKIP 
326     [good, good, medium, bad, bad] 
327     Categories (3, object): [good < medium < bad] 
328  
329     >>> pd.qcut(range(5), 4, labels=False) 
330     array([0, 0, 1, 2, 3]) 
331     """ 
332     x_is_series, series_index, name, x = _preprocess_for_cut(x) 
333  
334     x, dtype = _coerce_to_type(x) 
335  
336     if is_integer(q): 
337         quantiles = np.linspace(0, 1, q + 1) 
338     else: 
339         quantiles = q 
340     bins = algos.quantile(x, quantiles) 
341     fac, bins = _bins_to_cuts( 
342         x, 
343         bins, 
344         labels=labels, 
345         precision=precision, 
346         include_lowest=True, 
347         dtype=dtype, 
348         duplicates=duplicates, 
349     ) 
350  
351     return _postprocess_for_cut( 
352         fac, bins, retbins, x_is_series, series_index, name, dtype 
353     )

356 def _bins_to_cuts( 
357     x, 
358     bins, 
359     right=True, 
360     labels=None, 
361     precision=3, 
362     include_lowest=False, 
363     dtype=None, 
364     duplicates="raise", 
365 ): 
366  
367     if duplicates not in ["raise", "drop"]: 
368         raise ValueError( 
369             "invalid value for 'duplicates' parameter, " 
370             "valid options are: raise, drop" 
371         ) 
372  
373     if isinstance(bins, IntervalIndex): 
374         # we have a fast-path here 
375         ids = bins.get_indexer(x) 
376         result = Categorical.from_codes(ids, categories=bins, ordered=True) 
377         return result, bins 
378  
379     unique_bins = algos.unique(bins) 
380     if len(unique_bins) < len(bins) and len(bins) != 2: 
381         if duplicates == "raise": 
382             raise ValueError( 
383                 "Bin edges must be unique: {bins!r}.\nYou " 
384                 "can drop duplicate edges by setting " 
385                 "the 'duplicates' kwarg".format(bins=bins) 
386             ) 
387         else: 
388             bins = unique_bins 
389  
390     side = "left" if right else "right" 
391     ids = ensure_int64(bins.searchsorted(x, side=side)) 
392  
393     if include_lowest: 
394         ids[x == bins[0]] = 1 
395  
396     na_mask = isna(x) | (ids == len(bins)) | (ids == 0) 
397     has_nas = na_mask.any() 
398  
399     if labels is not False: 
400         if labels is None: 
401             labels = _format_labels( 
402                 bins, precision, right=right, include_lowest=include_lowest, dtype=dtype 
403             ) 
404         else: 
405             if len(labels) != len(bins) - 1: 
406                 raise ValueError( 
407                     "Bin labels must be one fewer than the number of bin edges" 
408                 ) 
409         if not is_categorical_dtype(labels): 
410             labels = Categorical(labels, categories=labels, ordered=True) 
411  
412         np.putmask(ids, na_mask, 0) 
413         result = algos.take_nd(labels, ids - 1) 
414  
415     else: 
416         result = ids - 1 
417         if has_nas: 
418             result = result.astype(np.float64) 
419             np.putmask(result, na_mask, np.nan) 
420  
421     return result, bins

424 def _coerce_to_type(x): 
425     """ 
426     if the passed data is of datetime/timedelta type, 
427     this method converts it to numeric so that cut method can 
428     handle it 
429     """ 
430     dtype = None 
431  
432     if is_datetime64tz_dtype(x): 
433         dtype = x.dtype 
434     elif is_datetime64_dtype(x): 
435         x = to_datetime(x) 
436         dtype = np.dtype("datetime64[ns]") 
437     elif is_timedelta64_dtype(x): 
438         x = to_timedelta(x) 
439         dtype = np.dtype("timedelta64[ns]") 
440  
441     if dtype is not None: 
442         # GH 19768: force NaT to NaN during integer conversion 
443         x = np.where(x.notna(), x.view(np.int64), np.nan) 
444  
445     return x, dtype

448 def _convert_bin_to_numeric_type(bins, dtype): 
449     """ 
450     if the passed bin is of datetime/timedelta type, 
451     this method converts it to integer 
452  
453     Parameters 
454     ---------- 
455     bins : list-like of bins 
456     dtype : dtype of data 
457  
458     Raises 
459     ------ 
460     ValueError if bins are not of a compat dtype to dtype 
461     """ 
462     bins_dtype = infer_dtype(bins, skipna=False) 
463     if is_timedelta64_dtype(dtype): 
464         if bins_dtype in ["timedelta", "timedelta64"]: 
465             bins = to_timedelta(bins).view(np.int64) 
466         else: 
467             raise ValueError("bins must be of timedelta64 dtype") 
468     elif is_datetime64_dtype(dtype) or is_datetime64tz_dtype(dtype): 
469         if bins_dtype in ["datetime", "datetime64"]: 
470             bins = to_datetime(bins).view(np.int64) 
471         else: 
472             raise ValueError("bins must be of datetime64 dtype") 
473  
474     return bins

477 def _convert_bin_to_datelike_type(bins, dtype): 
478     """ 
479     Convert bins to a DatetimeIndex or TimedeltaIndex if the original dtype is 
480     datelike 
481  
482     Parameters 
483     ---------- 
484     bins : list-like of bins 
485     dtype : dtype of data 
486  
487     Returns 
488     ------- 
489     bins : Array-like of bins, DatetimeIndex or TimedeltaIndex if dtype is 
490            datelike 
491     """ 
492     if is_datetime64tz_dtype(dtype): 
493         bins = to_datetime(bins.astype(np.int64), utc=True).tz_convert(dtype.tz) 
494     elif is_datetime_or_timedelta_dtype(dtype): 
495         bins = Index(bins.astype(np.int64), dtype=dtype) 
496     return bins

499 def _format_labels(bins, precision, right=True, include_lowest=False, dtype=None): 
500     """ based on the dtype, return our labels """ 
501  
502     closed = "right" if right else "left" 
503  
504     if is_datetime64tz_dtype(dtype): 
505         formatter = partial(Timestamp, tz=dtype.tz) 
506         adjust = lambda x: x - Timedelta("1ns") 
507     elif is_datetime64_dtype(dtype): 
508         formatter = Timestamp 
509         adjust = lambda x: x - Timedelta("1ns") 
510     elif is_timedelta64_dtype(dtype): 
511         formatter = Timedelta 
512         adjust = lambda x: x - Timedelta("1ns") 
513     else: 
514         precision = _infer_precision(precision, bins) 
515         formatter = lambda x: _round_frac(x, precision) 
516         adjust = lambda x: x - 10 ** (-precision) 
517  
518     breaks = [formatter(b) for b in bins] 
519     labels = IntervalIndex.from_breaks(breaks, closed=closed) 
520  
521     if right and include_lowest: 
522         # we will adjust the left hand side by precision to 
523         # account that we are all right closed 
524         v = adjust(labels[0].left) 
525  
526         i = IntervalIndex([Interval(v, labels[0].right, closed="right")]) 
527         labels = i.append(labels[1:]) 
528  
529     return labels

532 def _preprocess_for_cut(x): 
533     """ 
534     handles preprocessing for cut where we convert passed 
535     input to array, strip the index information and store it 
536     separately 
537     """ 
538     x_is_series = isinstance(x, Series) 
539     series_index = None 
540     name = None 
541  
542     if x_is_series: 
543         series_index = x.index 
544         name = x.name 
545  
546     # Check that the passed array is a Pandas or Numpy object 
547     # We don't want to strip away a Pandas data-type here (e.g. datetimetz) 
548     ndim = getattr(x, "ndim", None) 
549     if ndim is None: 
550         x = np.asarray(x) 
551     if x.ndim != 1: 
552         raise ValueError("Input array must be 1 dimensional") 
553  
554     return x_is_series, series_index, name, x

557 def _postprocess_for_cut(fac, bins, retbins, x_is_series, series_index, name, dtype): 
558     """ 
559     handles post processing for the cut method where 
560     we combine the index information if the originally passed 
561     datatype was a series 
562     """ 
563     if x_is_series: 
564         fac = Series(fac, index=series_index, name=name) 
565  
566     if not retbins: 
567         return fac 
568  
569     bins = _convert_bin_to_datelike_type(bins, dtype)

571     return fac, bins

574 def _round_frac(x, precision): 
575     """ 
576     Round the fractional part of the given number 
577     """ 
578     if not np.isfinite(x) or x == 0: 
579         return x 
580     else: 
581         frac, whole = np.modf(x) 
582         if whole == 0: 
583             digits = -int(np.floor(np.log10(abs(frac)))) - 1 + precision 
584         else: 
585             digits = precision 
586         return np.around(x, digits)

589 def _infer_precision(base_precision, bins): 
590     """Infer an appropriate precision for _round_frac 
591     """ 
592     for precision in range(base_precision, 20): 
593         levels = [_round_frac(b, precision) for b in bins] 
594         if algos.unique(levels).size == bins.size: 
595             return precision 
596     return base_precision  # default

 38 def cut( 
 39     x, 
 40     bins, 
 41     right=True, 
 42     labels=None, 
 43     retbins=False, 
 44     precision=3, 
 45     include_lowest=False, 
 46     duplicates="raise", 
 47 ): 
 48     """ 
 49     Bin values into discrete intervals. 
 50  
 51     Use `cut` when you need to segment and sort data values into bins. This 
 52     function is also useful for going from a continuous variable to a 
 53     categorical variable. For example, `cut` could convert ages to groups of 
 54     age ranges. Supports binning into an equal number of bins, or a 
 55     pre-specified array of bins. 
 56  
 57     Parameters 
 58     ---------- 
 59     x : array-like 
 60         The input array to be binned. Must be 1-dimensional. 
 61     bins : int, sequence of scalars, or IntervalIndex 
 62         The criteria to bin by. 
 63  
 64         * int : Defines the number of equal-width bins in the range of `x`. The 
 65           range of `x` is extended by .1% on each side to include the minimum 
 66           and maximum values of `x`. 
 67         * sequence of scalars : Defines the bin edges allowing for non-uniform 
 68           width. No extension of the range of `x` is done. 
 69         * IntervalIndex : Defines the exact bins to be used. Note that 
 70           IntervalIndex for `bins` must be non-overlapping. 
 71  
 72     right : bool, default True 
 73         Indicates whether `bins` includes the rightmost edge or not. If 
 74         ``right == True`` (the default), then the `bins` ``[1, 2, 3, 4]`` 
 75         indicate (1,2], (2,3], (3,4]. This argument is ignored when 
 76         `bins` is an IntervalIndex. 
 77     labels : array or bool, optional 
 78         Specifies the labels for the returned bins. Must be the same length as 
 79         the resulting bins. If False, returns only integer indicators of the 
 80         bins. This affects the type of the output container (see below). 
 81         This argument is ignored when `bins` is an IntervalIndex. 
 82     retbins : bool, default False 
 83         Whether to return the bins or not. Useful when bins is provided 
 84         as a scalar. 
 85     precision : int, default 3 
 86         The precision at which to store and display the bins labels. 
 87     include_lowest : bool, default False 
 88         Whether the first interval should be left-inclusive or not. 
 89     duplicates : {default 'raise', 'drop'}, optional 
 90         If bin edges are not unique, raise ValueError or drop non-uniques. 
 91  
 92         .. versionadded:: 0.23.0 
 93  
 94     Returns 
 95     ------- 
 96     out : Categorical, Series, or ndarray 
 97         An array-like object representing the respective bin for each value 
 98         of `x`. The type depends on the value of `labels`. 
 99  
100         * True (default) : returns a Series for Series `x` or a 
101           Categorical for all other inputs. The values stored within 
102           are Interval dtype. 
103  
104         * sequence of scalars : returns a Series for Series `x` or a 
105           Categorical for all other inputs. The values stored within 
106           are whatever the type in the sequence is. 
107  
108         * False : returns an ndarray of integers. 
109  
110     bins : numpy.ndarray or IntervalIndex. 
111         The computed or specified bins. Only returned when `retbins=True`. 
112         For scalar or sequence `bins`, this is an ndarray with the computed 
113         bins. If set `duplicates=drop`, `bins` will drop non-unique bin. For 
114         an IntervalIndex `bins`, this is equal to `bins`. 
115  
116     See Also 
117     -------- 
118     qcut : Discretize variable into equal-sized buckets based on rank 
119         or based on sample quantiles. 
120     Categorical : Array type for storing data that come from a 
121         fixed set of values. 
122     Series : One-dimensional array with axis labels (including time series). 
123     IntervalIndex : Immutable Index implementing an ordered, sliceable set. 
124  
125     Notes 
126     ----- 
127     Any NA values will be NA in the result. Out of bounds values will be NA in 
128     the resulting Series or Categorical object. 
129  
130     Examples 
131     -------- 
132     Discretize into three equal-sized bins. 
133  
134     >>> pd.cut(np.array([1, 7, 5, 4, 6, 3]), 3) 
135     ... # doctest: +ELLIPSIS 
136     [(0.994, 3.0], (5.0, 7.0], (3.0, 5.0], (3.0, 5.0], (5.0, 7.0], ... 
137     Categories (3, interval[float64]): [(0.994, 3.0] < (3.0, 5.0] ... 
138  
139     >>> pd.cut(np.array([1, 7, 5, 4, 6, 3]), 3, retbins=True) 
140     ... # doctest: +ELLIPSIS 
141     ([(0.994, 3.0], (5.0, 7.0], (3.0, 5.0], (3.0, 5.0], (5.0, 7.0], ... 
142     Categories (3, interval[float64]): [(0.994, 3.0] < (3.0, 5.0] ... 
143     array([0.994, 3.   , 5.   , 7.   ])) 
144  
145     Discovers the same bins, but assign them specific labels. Notice that 
146     the returned Categorical's categories are `labels` and is ordered. 
147  
148     >>> pd.cut(np.array([1, 7, 5, 4, 6, 3]), 
149     ...        3, labels=["bad", "medium", "good"]) 
150     [bad, good, medium, medium, good, bad] 
151     Categories (3, object): [bad < medium < good] 
152  
153     ``labels=False`` implies you just want the bins back. 
154  
155     >>> pd.cut([0, 1, 1, 2], bins=4, labels=False) 
156     array([0, 1, 1, 3]) 
157  
158     Passing a Series as an input returns a Series with categorical dtype: 
159  
160     >>> s = pd.Series(np.array([2, 4, 6, 8, 10]), 
161     ...               index=['a', 'b', 'c', 'd', 'e']) 
162     >>> pd.cut(s, 3) 
163     ... # doctest: +ELLIPSIS 
164     a    (1.992, 4.667] 
165     b    (1.992, 4.667] 
166     c    (4.667, 7.333] 
167     d     (7.333, 10.0] 
168     e     (7.333, 10.0] 
169     dtype: category 
170     Categories (3, interval[float64]): [(1.992, 4.667] < (4.667, ... 
171  
172     Passing a Series as an input returns a Series with mapping value. 
173     It is used to map numerically to intervals based on bins. 
174  
175     >>> s = pd.Series(np.array([2, 4, 6, 8, 10]), 
176     ...               index=['a', 'b', 'c', 'd', 'e']) 
177     >>> pd.cut(s, [0, 2, 4, 6, 8, 10], labels=False, retbins=True, right=False) 
178     ... # doctest: +ELLIPSIS 
179     (a    0.0 
180      b    1.0 
181      c    2.0 
182      d    3.0 
183      e    4.0 
184      dtype: float64, array([0, 2, 4, 6, 8])) 
185  
186     Use `drop` optional when bins is not unique 
187  
188     >>> pd.cut(s, [0, 2, 4, 6, 10, 10], labels=False, retbins=True, 
189     ...        right=False, duplicates='drop') 
190     ... # doctest: +ELLIPSIS 
191     (a    0.0 
192      b    1.0 
193      c    2.0 
194      d    3.0 
195      e    3.0 
196      dtype: float64, array([0, 2, 4, 6, 8])) 
197  
198     Passing an IntervalIndex for `bins` results in those categories exactly. 
199     Notice that values not covered by the IntervalIndex are set to NaN. 0 
200     is to the left of the first bin (which is closed on the right), and 1.5 
201     falls between two bins. 
202  
203     >>> bins = pd.IntervalIndex.from_tuples([(0, 1), (2, 3), (4, 5)]) 
204     >>> pd.cut([0, 0.5, 1.5, 2.5, 4.5], bins) 
205     [NaN, (0, 1], NaN, (2, 3], (4, 5]] 
206     Categories (3, interval[int64]): [(0, 1] < (2, 3] < (4, 5]] 
207     """ 
208     # NOTE: this binning code is changed a bit from histogram for var(x) == 0 
209  
210     # for handling the cut for datetime and timedelta objects 
211     x_is_series, series_index, name, x = _preprocess_for_cut(x) 
212     x, dtype = _coerce_to_type(x) 
213  
214     if not np.iterable(bins): 
215         if is_scalar(bins) and bins < 1: 
216             raise ValueError("`bins` should be a positive integer.") 
217  
218         try:  # for array-like 
219             sz = x.size 
220         except AttributeError: 
221             x = np.asarray(x) 
222             sz = x.size 
223  
224         if sz == 0: 
225             raise ValueError("Cannot cut empty array") 
226  
227         rng = (nanops.nanmin(x), nanops.nanmax(x)) 
228         mn, mx = [mi + 0.0 for mi in rng] 
229  
230         if np.isinf(mn) or np.isinf(mx): 
231             # GH 24314 
232             raise ValueError( 
233                 "cannot specify integer `bins` when input data contains infinity" 
234             ) 
235         elif mn == mx:  # adjust end points before binning 
236             mn -= 0.001 * abs(mn) if mn != 0 else 0.001 
237             mx += 0.001 * abs(mx) if mx != 0 else 0.001 
238             bins = np.linspace(mn, mx, bins + 1, endpoint=True) 
239         else:  # adjust end points after binning 
240             bins = np.linspace(mn, mx, bins + 1, endpoint=True) 
241             adj = (mx - mn) * 0.001  # 0.1% of the range 
242             if right: 
243                 bins[0] -= adj 
244             else: 
245                 bins[-1] += adj 
246  
247     elif isinstance(bins, IntervalIndex): 
248         if bins.is_overlapping: 
249             raise ValueError("Overlapping IntervalIndex is not accepted.") 
250  
251     else: 
252         if is_datetime64tz_dtype(bins): 
253             bins = np.asarray(bins, dtype=_NS_DTYPE) 
254         else: 
255             bins = np.asarray(bins) 
256         bins = _convert_bin_to_numeric_type(bins, dtype) 
257  
258         # GH 26045: cast to float64 to avoid an overflow 
259         if (np.diff(bins.astype("float64")) < 0).any(): 
260             raise ValueError("bins must increase monotonically.") 
261  
262     fac, bins = _bins_to_cuts( 
263         x, 
264         bins, 
265         right=right, 
266         labels=labels, 
267         precision=precision, 
268         include_lowest=include_lowest, 
269         dtype=dtype, 
270         duplicates=duplicates, 
271     ) 
272  
273     return _postprocess_for_cut( 
274         fac, bins, retbins, x_is_series, series_index, name, dtype 
275     )

278 def qcut(x, q, labels=None, retbins=False, precision=3, duplicates="raise"): 
279     """ 
280     Quantile-based discretization function. Discretize variable into 
281     equal-sized buckets based on rank or based on sample quantiles. For example 
282     1000 values for 10 quantiles would produce a Categorical object indicating 
283     quantile membership for each data point. 
284  
285     Parameters 
286     ---------- 
287     x : 1d ndarray or Series 
288     q : integer or array of quantiles 
289         Number of quantiles. 10 for deciles, 4 for quartiles, etc. Alternately 
290         array of quantiles, e.g. [0, .25, .5, .75, 1.] for quartiles 
291     labels : array or boolean, default None 
292         Used as labels for the resulting bins. Must be of the same length as 
293         the resulting bins. If False, return only integer indicators of the 
294         bins. 
295     retbins : bool, optional 
296         Whether to return the (bins, labels) or not. Can be useful if bins 
297         is given as a scalar. 
298     precision : int, optional 
299         The precision at which to store and display the bins labels 
300     duplicates : {default 'raise', 'drop'}, optional 
301         If bin edges are not unique, raise ValueError or drop non-uniques. 
302  
303         .. versionadded:: 0.20.0 
304  
305     Returns 
306     ------- 
307     out : Categorical or Series or array of integers if labels is False 
308         The return type (Categorical or Series) depends on the input: a Series 
309         of type category if input is a Series else Categorical. Bins are 
310         represented as categories when categorical data is returned. 
311     bins : ndarray of floats 
312         Returned only if `retbins` is True. 
313  
314     Notes 
315     ----- 
316     Out of bounds values will be NA in the resulting Categorical object 
317  
318     Examples 
319     -------- 
320     >>> pd.qcut(range(5), 4) 
321     ... # doctest: +ELLIPSIS 
322     [(-0.001, 1.0], (-0.001, 1.0], (1.0, 2.0], (2.0, 3.0], (3.0, 4.0]] 
323     Categories (4, interval[float64]): [(-0.001, 1.0] < (1.0, 2.0] ... 
324  
325     >>> pd.qcut(range(5), 3, labels=["good", "medium", "bad"]) 
326     ... # doctest: +SKIP 
327     [good, good, medium, bad, bad] 
328     Categories (3, object): [good < medium < bad] 
329  
330     >>> pd.qcut(range(5), 4, labels=False) 
331     array([0, 0, 1, 2, 3]) 
332     """ 
333     x_is_series, series_index, name, x = _preprocess_for_cut(x) 
334  
335     x, dtype = _coerce_to_type(x) 
336  
337     if is_integer(q): 
338         quantiles = np.linspace(0, 1, q + 1) 
339     else: 
340         quantiles = q 
341     bins = algos.quantile(x, quantiles) 
342     fac, bins = _bins_to_cuts( 
343         x, 
344         bins, 
345         labels=labels, 
346         precision=precision, 
347         include_lowest=True, 
348         dtype=dtype, 
349         duplicates=duplicates, 
350     ) 
351  
352     return _postprocess_for_cut( 
353         fac, bins, retbins, x_is_series, series_index, name, dtype 
354     )

357 def _bins_to_cuts( 
358     x, 
359     bins, 
360     right=True, 
361     labels=None, 
362     precision=3, 
363     include_lowest=False, 
364     dtype=None, 
365     duplicates="raise", 
366 ): 
367  
368     if duplicates not in ["raise", "drop"]: 
369         raise ValueError( 
370             "invalid value for 'duplicates' parameter, " 
371             "valid options are: raise, drop" 
372         ) 
373  
374     if isinstance(bins, IntervalIndex): 
375         # we have a fast-path here 
376         ids = bins.get_indexer(x) 
377         result = Categorical.from_codes(ids, categories=bins, ordered=True) 
378         return result, bins 
379  
380     unique_bins = algos.unique(bins) 
381     if len(unique_bins) < len(bins) and len(bins) != 2: 
382         if duplicates == "raise": 
383             raise ValueError( 
384                 "Bin edges must be unique: {bins!r}.\nYou " 
385                 "can drop duplicate edges by setting " 
386                 "the 'duplicates' kwarg".format(bins=bins) 
387             ) 
388         else: 
389             bins = unique_bins 
390  
391     side = "left" if right else "right" 
392     ids = ensure_int64(bins.searchsorted(x, side=side)) 
393  
394     if include_lowest: 
395         ids[x == bins[0]] = 1 
396  
397     na_mask = isna(x) | (ids == len(bins)) | (ids == 0) 
398     has_nas = na_mask.any() 
399  
400     if labels is not False: 
401         if labels is None: 
402             labels = _format_labels( 
403                 bins, precision, right=right, include_lowest=include_lowest, dtype=dtype 
404             ) 
405         else: 
406             if len(labels) != len(bins) - 1: 
407                 raise ValueError( 
408                     "Bin labels must be one fewer than the number of bin edges" 
409                 ) 
410         if not is_categorical_dtype(labels): 
411             labels = Categorical(labels, categories=labels, ordered=True) 
412  
413         np.putmask(ids, na_mask, 0) 
414         result = algos.take_nd(labels, ids - 1) 
415  
416     else: 
417         result = ids - 1 
418         if has_nas: 
419             result = result.astype(np.float64) 
420             np.putmask(result, na_mask, np.nan) 
421  
422     return result, bins 
423

425 def _coerce_to_type(x): 
426     """ 
427     if the passed data is of datetime/timedelta or bool type, 
428     this method converts it to numeric so that cut or qcut method can 
429     handle it 
430     """ 
431     dtype = None 
432  
433     if is_datetime64tz_dtype(x): 
434         dtype = x.dtype 
435     elif is_datetime64_dtype(x): 
436         x = to_datetime(x) 
437         dtype = np.dtype("datetime64[ns]") 
438     elif is_timedelta64_dtype(x): 
439         x = to_timedelta(x) 
440         dtype = np.dtype("timedelta64[ns]") 
441     elif is_bool_dtype(x): 
442         # GH 20303 
443         x = x.astype(np.int64) 
444  
445     if dtype is not None: 
446         # GH 19768: force NaT to NaN during integer conversion 
447         x = np.where(x.notna(), x.view(np.int64), np.nan) 
448  
449     return x, dtype

452 def _convert_bin_to_numeric_type(bins, dtype): 
453     """ 
454     if the passed bin is of datetime/timedelta type, 
455     this method converts it to integer 
456  
457     Parameters 
458     ---------- 
459     bins : list-like of bins 
460     dtype : dtype of data 
461  
462     Raises 
463     ------ 
464     ValueError if bins are not of a compat dtype to dtype 
465     """ 
466     bins_dtype = infer_dtype(bins, skipna=False) 
467     if is_timedelta64_dtype(dtype): 
468         if bins_dtype in ["timedelta", "timedelta64"]: 
469             bins = to_timedelta(bins).view(np.int64) 
470         else: 
471             raise ValueError("bins must be of timedelta64 dtype") 
472     elif is_datetime64_dtype(dtype) or is_datetime64tz_dtype(dtype): 
473         if bins_dtype in ["datetime", "datetime64"]: 
474             bins = to_datetime(bins).view(np.int64) 
475         else: 
476             raise ValueError("bins must be of datetime64 dtype") 
477  
478     return bins

481 def _convert_bin_to_datelike_type(bins, dtype): 
482     """ 
483     Convert bins to a DatetimeIndex or TimedeltaIndex if the original dtype is 
484     datelike 
485  
486     Parameters 
487     ---------- 
488     bins : list-like of bins 
489     dtype : dtype of data 
490  
491     Returns 
492     ------- 
493     bins : Array-like of bins, DatetimeIndex or TimedeltaIndex if dtype is 
494            datelike 
495     """ 
496     if is_datetime64tz_dtype(dtype): 
497         bins = to_datetime(bins.astype(np.int64), utc=True).tz_convert(dtype.tz) 
498     elif is_datetime_or_timedelta_dtype(dtype): 
499         bins = Index(bins.astype(np.int64), dtype=dtype) 
500     return bins

503 def _format_labels(bins, precision, right=True, include_lowest=False, dtype=None): 
504     """ based on the dtype, return our labels """ 
505  
506     closed = "right" if right else "left" 
507  
508     if is_datetime64tz_dtype(dtype): 
509         formatter = partial(Timestamp, tz=dtype.tz) 
510         adjust = lambda x: x - Timedelta("1ns") 
511     elif is_datetime64_dtype(dtype): 
512         formatter = Timestamp 
513         adjust = lambda x: x - Timedelta("1ns") 
514     elif is_timedelta64_dtype(dtype): 
515         formatter = Timedelta 
516         adjust = lambda x: x - Timedelta("1ns") 
517     else: 
518         precision = _infer_precision(precision, bins) 
519         formatter = lambda x: _round_frac(x, precision) 
520         adjust = lambda x: x - 10 ** (-precision) 
521  
522     breaks = [formatter(b) for b in bins] 
523     labels = IntervalIndex.from_breaks(breaks, closed=closed) 
524  
525     if right and include_lowest: 
526         # we will adjust the left hand side by precision to 
527         # account that we are all right closed 
528         v = adjust(labels[0].left) 
529  
530         i = IntervalIndex([Interval(v, labels[0].right, closed="right")]) 
531         labels = i.append(labels[1:]) 
532  
533     return labels

536 def _preprocess_for_cut(x): 
537     """ 
538     handles preprocessing for cut where we convert passed 
539     input to array, strip the index information and store it 
540     separately 
541     """ 
542     x_is_series = isinstance(x, Series) 
543     series_index = None 
544     name = None 
545  
546     if x_is_series: 
547         series_index = x.index 
548         name = x.name 
549  
550     # Check that the passed array is a Pandas or Numpy object 
551     # We don't want to strip away a Pandas data-type here (e.g. datetimetz) 
552     ndim = getattr(x, "ndim", None) 
553     if ndim is None: 
554         x = np.asarray(x) 
555     if x.ndim != 1: 
556         raise ValueError("Input array must be 1 dimensional") 
557  
558     return x_is_series, series_index, name, x

561 def _postprocess_for_cut(fac, bins, retbins, x_is_series, series_index, name, dtype): 
562     """ 
563     handles post processing for the cut method where 
564     we combine the index information if the originally passed 
565     datatype was a series 
566     """ 
567     if x_is_series: 
568         fac = Series(fac, index=series_index, name=name) 
569  
570     if not retbins: 
571         return fac 
572  
573     bins = _convert_bin_to_datelike_type(bins, dtype) 
574  
575     return fac, bins

578 def _round_frac(x, precision): 
579     """ 
580     Round the fractional part of the given number 
581     """ 
582     if not np.isfinite(x) or x == 0: 
583         return x 
584     else: 
585         frac, whole = np.modf(x) 
586         if whole == 0: 
587             digits = -int(np.floor(np.log10(abs(frac)))) - 1 + precision 
588         else: 
589             digits = precision 
590         return np.around(x, digits) 
591

593 def _infer_precision(base_precision, bins): 
594     """Infer an appropriate precision for _round_frac 
595     """ 
596     for precision in range(base_precision, 20): 
597         levels = [_round_frac(b, precision) for b in bins] 
598         if algos.unique(levels).size == bins.size: 
599             return precision 
600     return base_precision  # default
