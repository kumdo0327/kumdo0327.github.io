 585             case NJS_VMCODE_REGEXP: 
 586                 ret = njs_vmcode_regexp(vm, pc); 
 587                 break; 
 588  
 589             case NJS_VMCODE_INSTANCE_OF: 
 590                 ret = njs_vmcode_instance_of(vm, value1, value2); 
 591                 break; 
 592  
 593             case NJS_VMCODE_TYPEOF: 
 594                 ret = njs_vmcode_typeof(vm, value1, value2); 
 595                 break; 
 596  
 597             case NJS_VMCODE_VOID: 
 598                 njs_set_undefined(&vm->retval); 
 599  
 600                 ret = sizeof(njs_vmcode_2addr_t); 
 601                 break; 
 602  
 603             case NJS_VMCODE_DELETE: 
 604                 njs_release(vm, value1); 
 605                 vm->retval = njs_value_true; 
 606  
 607                 ret = sizeof(njs_vmcode_2addr_t); 
 608                 break; 
 609  
 610             case NJS_VMCODE_DEBUGGER: 
 611                 ret = njs_vmcode_debugger(vm); 
 612                 break; 
 613  
 614             default: 
 615                 njs_internal_error(vm, "%d has retval", op); 
 616                 goto error; 
 617             } 
 618  
 619             if (njs_slow_path(ret < 0 && ret >= NJS_PREEMPT)) { 
 620                 break; 
 621             } 
 622  
 623             njs_vmcode_operand(vm, vmcode->operand1, retval); 
 624             njs_release(vm, retval); 
 625             *retval = vm->retval; 
 626  
 627         } else { 
 628  
 629             switch (op) { 
 630             case NJS_VMCODE_MOVE_ARG: 
 631                 move_arg = (njs_vmcode_move_arg_t *) pc; 
 632                 native = vm->top_frame; 
 633  
 634                 hint = move_arg->dst; 
 635  
 636                 value1 = &native->arguments_offset[hint]; 
 637                 njs_vmcode_operand(vm, move_arg->src, value2); 
 638  
 639                 *value1 = *value2; 
 640  
 641                 ret = sizeof(njs_vmcode_move_arg_t); 
 642                 break; 
 643  
 644             case NJS_VMCODE_STOP: 
 645                 njs_vmcode_operand(vm, (njs_index_t) value2, value2); 
 646                 vm->retval = *value2; 
 647  
 648                 return NJS_OK; 
 649  
 650             case NJS_VMCODE_JUMP: 
 651                 ret = (njs_jump_off_t) value2; 
 652                 break; 
 653  
 654             case NJS_VMCODE_PROPERTY_SET: 
 655                 set = (njs_vmcode_prop_set_t *) pc; 
 656                 njs_vmcode_operand(vm, set->value, retval); 
 657  
 658                 ret = njs_value_property_set(vm, value1, value2, retval); 
 659                 if (njs_slow_path(ret == NJS_ERROR)) { 
 660                     goto error; 
 661                 } 
 662  
 663                 ret = sizeof(njs_vmcode_prop_set_t); 
 664                 break; 
 665  
 666             case NJS_VMCODE_PROPERTY_ACCESSOR: 
 667                 accessor = (njs_vmcode_prop_accessor_t *) pc; 
 668                 njs_vmcode_operand(vm, accessor->value, function); 
 669  
 670                 ret = njs_value_to_key(vm, &name, value2); 
 671                 if (njs_slow_path(ret != NJS_OK)) { 
 672                     njs_internal_error(vm, "failed conversion of type \"%s\" " 
 673                                        "to string while property define", 
 674                                        njs_type_string(value2->type)); 
 675                     goto error; 
 676                 } 
 677  
 678                 ret = njs_object_prop_define(vm, value1, &name, function, 
 679                                              accessor->type); 
 680                 if (njs_slow_path(ret != NJS_OK)) { 
 681                     return NJS_ERROR; 
 682                 } 
 683  
 684                 ret = sizeof(njs_vmcode_prop_accessor_t); 
 685                 break; 
 686  
 687             case NJS_VMCODE_IF_TRUE_JUMP: 
 688             case NJS_VMCODE_IF_FALSE_JUMP: 
 689                 ret = njs_is_true(value1); 
 690  
 691                 ret ^= op - NJS_VMCODE_IF_TRUE_JUMP; 
 692  
 693                 ret = ret ? (njs_jump_off_t) value2 
 694                           : (njs_jump_off_t) sizeof(njs_vmcode_cond_jump_t); 
 695  
 696                 break; 
 697  
 698             case NJS_VMCODE_IF_EQUAL_JUMP: 
 699                 if (njs_values_strict_equal(value1, value2)) { 
 700                     equal = (njs_vmcode_equal_jump_t *) pc; 
 701                     ret = equal->offset; 
 702  
 703                 } else { 
 704                     ret = sizeof(njs_vmcode_3addr_t); 
 705                 } 
 706  
 707                 break; 
 708  
 709             case NJS_VMCODE_PROPERTY_INIT: 
 710                 set = (njs_vmcode_prop_set_t *) pc; 
 711                 njs_vmcode_operand(vm, set->value, retval); 
 712                 ret = njs_vmcode_property_init(vm, value1, value2, retval); 
 713                 if (njs_slow_path(ret == NJS_ERROR)) { 
 714                     goto error; 
 715                 } 
 716  
 717                 break; 
 718  
 719             case NJS_VMCODE_RETURN: 
 720                 njs_vmcode_operand(vm, (njs_index_t) value2, value2); 
 721                 return njs_vmcode_return(vm, NULL, value2); 
 722  
 723             case NJS_VMCODE_FUNCTION_COPY: 
 724                 fcopy = (njs_vmcode_function_copy_t *) pc; 
 725                 ret = njs_vmcode_function_copy(vm, fcopy->function, 
 726                                                fcopy->retval); 
 727                 break; 
 728  
 729             case NJS_VMCODE_FUNCTION_FRAME: 
 730                 function_frame = (njs_vmcode_function_frame_t *) pc; 
 731  
 732                 /* TODO: external object instead of void this. */ 
 733  
 734                 ret = njs_function_frame_create(vm, value1, 
 735                                                 &njs_value_undefined, 
 736                                                 (uintptr_t) value2, 
 737                                                 function_frame->ctor); 
 738  
 739                 if (njs_slow_path(ret != NJS_OK)) { 
 740                     goto error; 
 741                 } 
 742  
 743                 ret = sizeof(njs_vmcode_function_frame_t); 
 744                 break; 
 745  
 746             case NJS_VMCODE_METHOD_FRAME: 
 747                 method_frame = (njs_vmcode_method_frame_t *) pc; 
 748  
 749                 ret = njs_value_property(vm, value1, value2, &dst); 
 750                 if (njs_slow_path(ret == NJS_ERROR)) { 
 751                     goto error; 
 752                 } 
 753  
 754                 if (njs_slow_path(!njs_is_function(&dst))) { 
 755                     ret = njs_value_to_key(vm, value2, value2); 
 756                     if (njs_slow_path(ret != NJS_OK)) { 
 757                         return NJS_ERROR; 
 758                     } 
 759  
 760                     njs_key_string_get(vm, value2, &string); 
 761                     njs_type_error(vm, 
 762                                "(intermediate value)[\"%V\"] is not a function", 
 763                                &string); 
 764                     goto error; 
 765                 } 
 766  
 767                 ret = njs_function_frame_create(vm, &dst, value1, 
 768                                                 method_frame->nargs, 
 769                                                 method_frame->ctor); 
 770  
 771                 if (njs_slow_path(ret != NJS_OK)) { 
 772                     goto error; 
 773                 } 
 774  
 775                 ret = sizeof(njs_vmcode_method_frame_t); 
 776                 break; 
 777  
 778             case NJS_VMCODE_FUNCTION_CALL: 
 779                 vm->active_frame->native.pc = pc; 
 780  
 781                 njs_vmcode_operand(vm, (njs_index_t) value2, value2); 
 782  
 783                 ret = njs_function_frame_invoke(vm, value2); 
 784                 if (njs_slow_path(ret == NJS_ERROR)) { 
 785                     goto error; 
 786                 } 
 787  
 788                 ret = sizeof(njs_vmcode_function_call_t); 
 789                 break; 
 790  
 791             case NJS_VMCODE_PROPERTY_NEXT: 
 792                 pnext = (njs_vmcode_prop_next_t *) pc; 
 793                 retval = njs_scope_value(vm, pnext->retval); 
 794  
 795                 next = value2->data.u.next; 
 796  
 797                 if (next->index < next->array->length) { 
 798                     *retval = next->array->start[next->index++]; 
 799  
 800                     ret = pnext->offset; 
 801                     break; 
 802                 } 
 803  
 804                 njs_mp_free(vm->mem_pool, next); 
 805  
 806                 ret = sizeof(njs_vmcode_prop_next_t); 
 807                 break; 
 808  
 809             case NJS_VMCODE_ARGUMENTS: 
 810                 ret = njs_vmcode_arguments(vm, pc); 
 811                 if (njs_slow_path(ret == NJS_ERROR)) { 
 812                     goto error; 
 813                 } 
 814  
 815                 break; 
 816  
 817             case NJS_VMCODE_PROTO_INIT: 
 818                 set = (njs_vmcode_prop_set_t *) pc; 
 819                 njs_vmcode_operand(vm, set->value, retval); 
 820                 ret = njs_vmcode_proto_init(vm, value1, value2, retval); 
 821                 if (njs_slow_path(ret == NJS_ERROR)) { 
 822                     goto error; 
 823                 } 
 824  
 825                 break; 
 826  
 827             case NJS_VMCODE_AWAIT: 
 828                 await = (njs_vmcode_await_t *) pc; 
 829                 return njs_vmcode_await(vm, await); 
 830  
 831             case NJS_VMCODE_TRY_START: 
 832                 ret = njs_vmcode_try_start(vm, value1, value2, pc); 
 833                 if (njs_slow_path(ret == NJS_ERROR)) { 
 834                     goto error; 
 835                 } 
 836  
 837                 break; 
 838  
 839             case NJS_VMCODE_THROW: 
 840                 njs_vmcode_operand(vm, (njs_index_t) value2, value2); 
 841                 vm->retval = *value2; 
 842                 goto error; 
 843  
 844             case NJS_VMCODE_TRY_BREAK: 
 845                 try_trampoline = (njs_vmcode_try_trampoline_t *) pc; 
 846                 value1 = njs_scope_value(vm, try_trampoline->exit_value); 
 847  
 848                 ret = njs_vmcode_try_break(vm, value1, value2); 
 849                 break; 
 850  
 851             case NJS_VMCODE_TRY_CONTINUE: 
 852                 try_trampoline = (njs_vmcode_try_trampoline_t *) pc; 
 853                 value1 = njs_scope_value(vm, try_trampoline->exit_value); 
 854  
 855                 ret = njs_vmcode_try_continue(vm, value1, value2); 
 856                 break; 
 857  
 858             case NJS_VMCODE_TRY_END: 
 859                 ret = njs_vmcode_try_end(vm, value1, value2); 
 860                 break; 
 861  
 862             /* 
 863              * njs_vmcode_catch() is set on the start of a "catch" block to 
 864              * store exception and to remove a "try" block if there is no 
 865              * "finally" block or to update a catch address to the start of 
 866              * a "finally" block. 
 867              * njs_vmcode_catch() is set on the start of a "finally" block 
 868              * to store uncaught exception and to remove a "try" block. 
 869              */ 
 870             case NJS_VMCODE_CATCH: 
 871                 *value1 = vm->retval; 
 872  
 873                 if ((njs_jump_off_t) value2 == sizeof(njs_vmcode_catch_t)) { 
 874                     ret = njs_vmcode_try_end(vm, value1, value2); 
 875  
 876                 } else { 
 877                     frame = (njs_frame_t *) vm->top_frame; 
 878                     frame->exception.catch = pc + (njs_jump_off_t) value2; 
 879                     ret = sizeof(njs_vmcode_catch_t); 
 880                 } 
 881  
 882                 break; 
 883  
 884             case NJS_VMCODE_FINALLY: 
 885                 finally = (njs_vmcode_finally_t *) pc; 
 886                 value1 = njs_scope_value(vm, finally->exit_value); 
 887  
 888                 ret = njs_vmcode_finally(vm, value1, value2, pc); 
 889  
 890                 switch (ret) { 
 891                 case NJS_OK: 
 892                     return NJS_OK; 
 893                 case NJS_ERROR: 
 894                     goto error; 
 895                 } 
 896  
 897                 break; 
 898  
 899             case NJS_VMCODE_LET: 
 900                 var = (njs_vmcode_variable_t *) pc; 
 901                 value1 = njs_scope_value(vm, var->dst); 
 902  
 903                 if (njs_is_valid(value1)) { 
 904                     value1 = njs_mp_alloc(vm->mem_pool, sizeof(njs_value_t)); 
 905                     if (njs_slow_path(value1 == NULL)) { 
 906                         return NJS_ERROR; 
 907                     } 
 908  
 909                     njs_scope_value_set(vm, var->dst, value1); 
 910                 } 
 911  
 912                 njs_set_undefined(value1); 
 913  
 914                 ret = sizeof(njs_vmcode_variable_t); 
 915                 break; 
 916  
 917             case NJS_VMCODE_LET_UPDATE: 
 918                 var = (njs_vmcode_variable_t *) pc; 
 919                 value2 = njs_scope_value(vm, var->dst); 
 920  
 921                 value1 = njs_mp_alloc(vm->mem_pool, sizeof(njs_value_t)); 
 922                 if (njs_slow_path(value1 == NULL)) { 
 923                     return NJS_ERROR; 
 924                 } 
 925  
 926                 *value1 = *value2; 
 927  
 928                 njs_scope_value_set(vm, var->dst, value1); 
 929  
 930                 ret = sizeof(njs_vmcode_variable_t); 
 931                 break; 
 932  
 933             case NJS_VMCODE_INITIALIZATION_TEST: 
 934                 var = (njs_vmcode_variable_t *) pc; 
 935                 value1 = njs_scope_value(vm, var->dst); 
 936  
 937                 if (njs_is_valid(value1)) { 
 938                     ret = sizeof(njs_vmcode_variable_t); 
 939                     break; 
 940                 } 
 941  
 942                 /* Fall through. */ 
 943  
 944             case NJS_VMCODE_NOT_INITIALIZED: 
 945                 njs_reference_error(vm, "cannot access variable " 
 946                                         "before initialization"); 
 947                 goto error; 
 948  
 949             case NJS_VMCODE_ERROR: 
 950                 njs_vmcode_error(vm, pc); 
 951                 goto error; 
 952  
 953             case NJS_VMCODE_ASSIGNMENT_ERROR: 
 954                 njs_type_error(vm, "assignment to constant variable"); 
 955                 goto error; 
 956  
 957             default: 
 958                 njs_internal_error(vm, "%d has NO retval", op); 
 959                 goto error; 
 960             } 
 961         } 
 962  
 963         pc += ret; 
 964     } 
 965  
 966 error: 
 967  
 968     if (njs_is_error(&vm->retval)) { 
 969         vm->active_frame->native.pc = pc; 
 970         (void) njs_error_stack_attach(vm, &vm->retval); 
 971     } 
 972  
 973     for ( ;; ) { 
 974         native = vm->top_frame; 
 975  
 976         if (!native->native) { 
 977             frame = (njs_frame_t *) native; 
 978             catch = frame->exception.catch; 
 979  
 980             if (catch != NULL) { 
 981                 pc = catch; 
 982  
 983                 goto next; 
 984             } 
 985         } 
 986  
 987         previous = native->previous; 
 988         if (previous == NULL) { 
 989             break; 
 990         } 
 991  
 992         lambda_call = (native == &vm->active_frame->native); 
 993  
 994         njs_vm_scopes_restore(vm, native, previous); 
 995  
 996         if (native->size != 0) { 
 997             vm->stack_size -= native->size; 
 998             njs_mp_free(vm->mem_pool, native); 
 999         } 
1000  
1001         if (lambda_call) { 
1002             break; 
1003         } 
1004     } 
1005  
1006     return NJS_ERROR; 
1007 }

1590 njs_values_equal(njs_vm_t *vm, njs_value_t *val1, njs_value_t *val2) 
1591 {

1593     njs_bool_t   nv1, nv2; 
1594     njs_value_t  primitive; 
1595     njs_value_t  *hv, *lv; 
1596  
1597 again: 
1598  
1599     nv1 = njs_is_null_or_undefined(val1); 
1600     nv2 = njs_is_null_or_undefined(val2); 
1601  
1602     /* Void and null are equal and not comparable with anything else. */ 
1603     if (nv1 || nv2) { 
1604         return (nv1 && nv2); 
1605     } 
1606  
1607     if (njs_is_numeric(val1) && njs_is_numeric(val2)) { 
1608         /* NaNs and Infinities are handled correctly by comparision. */ 
1609         return (njs_number(val1) == njs_number(val2)); 
1610     } 
1611  
1612     if (val1->type == val2->type) { 
1613  
1614         if (njs_is_string(val1)) { 
1615             return njs_string_eq(val1, val2); 
1616         } 
1617  
1618         if (njs_is_symbol(val1)) { 
1619             return njs_symbol_eq(val1, val2); 
1620         } 
1621  
1622         return (njs_object(val1) == njs_object(val2)); 
1623     } 
1624  
1625     /* Sort values as: numeric < symbol < string < objects. */ 
1626  
1627     if (val1->type > val2->type) { 
1628         hv = val1; 
1629         lv = val2; 
1630  
1631     } else { 
1632         hv = val2; 
1633         lv = val1; 
1634     } 
1635  
1636     /* If "lv" is an object then "hv" can only be another object. */ 
1637     if (njs_is_object(lv)) { 
1638         return 0; 
1639     } 
1640  
1641     /* If "hv" is a symbol then "lv" can only be a numeric. */ 
1642     if (njs_is_symbol(hv)) { 
1643         return 0; 
1644     } 
1645  
1646     /* If "hv" is a string then "lv" can be a numeric or symbol. */ 
1647     if (njs_is_string(hv)) { 
1648         return !njs_is_symbol(lv) 
1649             && (njs_number(lv) == njs_string_to_number(hv, 0)); 
1650     } 
1651  
1652     /* "hv" is an object and "lv" is either a string or a symbol or a numeric. */ 
1653  
1654     ret = njs_value_to_primitive(vm, &primitive, hv, 0); 
1655     if (ret != NJS_OK) { 
1656         return ret; 
1657     } 
1658  
1659     val1 = &primitive; 
1660     val2 = lv; 
1661  
1662     goto again; 
1663 }

1667  * ECMAScript 5.1: 11.8.5 
1668  * njs_primitive_values_compare() returns 
1669  *   1 if val1 is less than val2, 
1670  *   0 if val1 is greater than or equal to val2, 
1671  *  -1 if the values are not comparable. 
1672  */

1674 static njs_jump_off_t 
1675 njs_primitive_values_compare(njs_vm_t *vm, njs_value_t *val1, njs_value_t *val2) 
1676 { 
1677     double   num1, num2; 
1678  
1679     if (njs_fast_path(njs_is_numeric(val1))) { 
1680         num1 = njs_number(val1); 
1681  
1682         if (njs_fast_path(njs_is_numeric(val2))) { 
1683             num2 = njs_number(val2); 
1684  
1685         } else { 
1686             num2 = njs_string_to_number(val2, 0); 
1687         } 
1688  
1689     } else if (njs_is_numeric(val2)) { 
1690         num1 = njs_string_to_number(val1, 0); 
1691         num2 = njs_number(val2); 
1692  
1693     } else { 
1694         return (njs_string_cmp(val1, val2) < 0) ? 1 : 0; 
1695     } 
1696  
1697     /* NaN and void values are not comparable with anything. */ 
1698     if (isnan(num1) || isnan(num2)) { 
1699         return -1; 
1700     } 
1701  
1702     /* Infinities are handled correctly by comparision. */ 
1703     return (num1 < num2); 
1704 }

 415                     break; 
 416  
 417                 case NJS_VMCODE_REMAINDER: 
 418                     num = fmod(num, njs_number(value2)); 
 419                     break; 
 420  
 421                 case NJS_VMCODE_BITWISE_AND: 
 422                 case NJS_VMCODE_BITWISE_OR: 
 423                 case NJS_VMCODE_BITWISE_XOR: 
 424                     i32 = njs_number_to_int32(njs_number(value2)); 
 425  
 426                     switch (op) { 
 427                     case NJS_VMCODE_BITWISE_AND: 
 428                         i32 &= njs_number_to_int32(num); 
 429                         break; 
 430  
 431                     case NJS_VMCODE_BITWISE_OR: 
 432                         i32 |= njs_number_to_int32(num); 
 433                         break; 
 434  
 435                     case NJS_VMCODE_BITWISE_XOR: 
 436                         i32 ^= njs_number_to_int32(num); 
 437                         break; 
 438                     } 
 439  
 440                     njs_set_int32(retval, i32); 
 441                     goto next; 
 442  
 443                 default: 
 444                     u32 = njs_number_to_uint32(njs_number(value2)) & 0x1f; 
 445  
 446                     switch (op) { 
 447                     case NJS_VMCODE_LEFT_SHIFT: 
 448                     case NJS_VMCODE_RIGHT_SHIFT: 
 449                         i32 = njs_number_to_int32(num); 
 450  
 451                         if (op == NJS_VMCODE_LEFT_SHIFT) { 
 452                             /* Shifting of negative numbers is undefined. */ 
 453                             i32 = (uint32_t) i32 << u32; 
 454                         } else { 
 455                             i32 >>= u32; 
 456                         } 
 457  
 458                         njs_set_int32(retval, i32); 
 459                         break; 
 460  
 461                     default: /* NJS_VMCODE_UNSIGNED_RIGHT_SHIFT */ 
 462                         njs_set_uint32(retval, 
 463                                        njs_number_to_uint32(num) >> u32); 
 464                     } 
 465  
 466                     goto next; 
 467                 } 
 468  
 469                 njs_set_number(retval, num); 
 470                 goto next; 
 471  
 472             case NJS_VMCODE_OBJECT_COPY: 
 473                 ret = njs_vmcode_object_copy(vm, value1, value2); 
 474                 break; 
 475  
 476             case NJS_VMCODE_TEMPLATE_LITERAL: 
 477                 ret = njs_vmcode_template_literal(vm, value1, value2); 
 478                 break; 
 479  
 480             case NJS_VMCODE_PROPERTY_IN: 
 481                 ret = njs_vmcode_property_in(vm, value1, value2); 
 482                 break; 
 483  
 484             case NJS_VMCODE_PROPERTY_DELETE: 
 485                 ret = njs_value_property_delete(vm, value1, value2, NULL); 
 486                 if (njs_fast_path(ret != NJS_ERROR)) { 
 487                     vm->retval = njs_value_true; 
 488  
 489                     ret = sizeof(njs_vmcode_3addr_t); 
 490                 } 
 491  
 492                 break; 
 493  
 494             case NJS_VMCODE_PROPERTY_FOREACH: 
 495                 ret = njs_vmcode_property_foreach(vm, value1, value2, pc); 
 496                 break; 
 497  
 498             case NJS_VMCODE_STRICT_EQUAL: 
 499             case NJS_VMCODE_STRICT_NOT_EQUAL: 
 500                 ret = njs_values_strict_equal(value1, value2); 
 501  
 502                 ret ^= op - NJS_VMCODE_STRICT_EQUAL; 
 503  
 504                 njs_vmcode_operand(vm, vmcode->operand1, retval); 
 505                 njs_set_boolean(retval, ret); 
 506  
 507                 pc += sizeof(njs_vmcode_3addr_t); 
 508                 goto next; 
 509  
 510             case NJS_VMCODE_TEST_IF_TRUE: 
 511             case NJS_VMCODE_TEST_IF_FALSE: 
 512             case NJS_VMCODE_COALESCE: 
 513                 if (op == NJS_VMCODE_COALESCE) { 
 514                     ret = !njs_is_null_or_undefined(value1); 
 515  
 516                 } else { 
 517                     ret = njs_is_true(value1); 
 518                     ret ^= op - NJS_VMCODE_TEST_IF_TRUE; 
 519                 } 
 520  
 521                 if (ret) { 
 522                     test_jump = (njs_vmcode_test_jump_t *) pc; 
 523                     ret = test_jump->offset; 
 524  
 525                 } else { 
 526                     ret = sizeof(njs_vmcode_3addr_t); 
 527                 } 
 528  
 529                 njs_vmcode_operand(vm, vmcode->operand1, retval); 
 530                 *retval = *value1; 
 531  
 532                 pc += ret; 
 533                 goto next; 
 534  
 535             case NJS_VMCODE_UNARY_PLUS: 
 536             case NJS_VMCODE_UNARY_NEGATION: 
 537             case NJS_VMCODE_BITWISE_NOT: 
 538                 if (njs_slow_path(!njs_is_numeric(value1))) { 
 539                     ret = njs_value_to_numeric(vm, value1, &numeric1); 
 540                     if (njs_slow_path(ret != NJS_OK)) { 
 541                         goto error; 
 542                     } 
 543  
 544                     value1 = &numeric1; 
 545                 } 
 546  
 547                 num = njs_number(value1); 
 548                 njs_vmcode_operand(vm, vmcode->operand1, retval); 
 549  
 550                 switch (op) { 
 551                 case NJS_VMCODE_UNARY_NEGATION: 
 552                     num = -num; 
 553  
 554                     /* Fall through. */ 
 555                 case NJS_VMCODE_UNARY_PLUS: 
 556                     njs_set_number(retval, num); 
 557                     break; 
 558  
 559                 case NJS_VMCODE_BITWISE_NOT: 
 560                     njs_set_int32(retval, ~njs_number_to_uint32(num)); 
 561                 } 
 562  
 563                 pc += sizeof(njs_vmcode_2addr_t); 
 564                 goto next; 
 565  
 566             case NJS_VMCODE_LOGICAL_NOT: 
 567                 njs_vmcode_operand(vm, vmcode->operand1, retval); 
 568                 njs_set_boolean(retval, !njs_is_true(value1)); 
 569  
 570                 pc += sizeof(njs_vmcode_2addr_t); 
 571                 goto next; 
 572  
 573             case NJS_VMCODE_OBJECT: 
 574                 ret = njs_vmcode_object(vm); 
 575                 break; 
 576  
 577             case NJS_VMCODE_ARRAY: 
 578                 ret = njs_vmcode_array(vm, pc); 
 579                 break; 
 580  
 581             case NJS_VMCODE_FUNCTION: 
 582                 ret = njs_vmcode_function(vm, pc); 
 583                 break; 
 584  
 585             case NJS_VMCODE_REGEXP: 
 586                 ret = njs_vmcode_regexp(vm, pc); 
 587                 break; 
 588  
 589             case NJS_VMCODE_INSTANCE_OF: 
 590                 ret = njs_vmcode_instance_of(vm, value1, value2); 
 591                 break; 
 592  
 593             case NJS_VMCODE_TYPEOF: 
 594                 ret = njs_vmcode_typeof(vm, value1, value2); 
 595                 break; 
 596  
 597             case NJS_VMCODE_VOID: 
 598                 njs_set_undefined(&vm->retval); 
 599  
 600                 ret = sizeof(njs_vmcode_2addr_t); 
 601                 break; 
 602  
 603             case NJS_VMCODE_DELETE: 
 604                 njs_release(vm, value1); 
 605                 vm->retval = njs_value_true; 
 606  
 607                 ret = sizeof(njs_vmcode_2addr_t); 
 608                 break; 
 609  
 610             case NJS_VMCODE_DEBUGGER: 
 611                 ret = njs_vmcode_debugger(vm); 
 612                 break; 
 613  
 614             default: 
 615                 njs_internal_error(vm, "%d has retval", op); 
 616                 goto error; 
 617             } 
 618  
 619             if (njs_slow_path(ret < 0 && ret >= NJS_PREEMPT)) { 
 620                 break; 
 621             } 
 622  
 623             njs_vmcode_operand(vm, vmcode->operand1, retval); 
 624             njs_release(vm, retval); 
 625             *retval = vm->retval; 
 626  
 627         } else { 
 628  
 629             switch (op) { 
 630             case NJS_VMCODE_MOVE_ARG: 
 631                 move_arg = (njs_vmcode_move_arg_t *) pc; 
 632                 native = vm->top_frame; 
 633  
 634                 hint = move_arg->dst; 
 635  
 636                 value1 = &native->arguments_offset[hint]; 
 637                 njs_vmcode_operand(vm, move_arg->src, value2); 
 638  
 639                 *value1 = *value2; 
 640  
 641                 ret = sizeof(njs_vmcode_move_arg_t); 
 642                 break; 
 643  
 644             case NJS_VMCODE_STOP: 
 645                 njs_vmcode_operand(vm, (njs_index_t) value2, value2); 
 646                 vm->retval = *value2; 
 647  
 648                 return NJS_OK; 
 649  
 650             case NJS_VMCODE_JUMP: 
 651                 ret = (njs_jump_off_t) value2; 
 652                 break; 
 653  
 654             case NJS_VMCODE_PROPERTY_SET: 
 655                 set = (njs_vmcode_prop_set_t *) pc; 
 656                 njs_vmcode_operand(vm, set->value, retval); 
 657  
 658                 ret = njs_value_property_set(vm, value1, value2, retval); 
 659                 if (njs_slow_path(ret == NJS_ERROR)) { 
 660                     goto error; 
 661                 } 
 662  
 663                 ret = sizeof(njs_vmcode_prop_set_t); 
 664                 break; 
 665  
 666             case NJS_VMCODE_PROPERTY_ACCESSOR: 
 667                 accessor = (njs_vmcode_prop_accessor_t *) pc; 
 668                 njs_vmcode_operand(vm, accessor->value, function); 
 669  
 670                 ret = njs_value_to_key(vm, &name, value2); 
 671                 if (njs_slow_path(ret != NJS_OK)) { 
 672                     njs_internal_error(vm, "failed conversion of type \"%s\" " 
 673                                        "to string while property define", 
 674                                        njs_type_string(value2->type)); 
 675                     goto error; 
 676                 } 
 677  
 678                 ret = njs_object_prop_define(vm, value1, &name, function, 
 679                                              accessor->type); 
 680                 if (njs_slow_path(ret != NJS_OK)) { 
 681                     return NJS_ERROR; 
 682                 } 
 683  
 684                 ret = sizeof(njs_vmcode_prop_accessor_t); 
 685                 break; 
 686  
 687             case NJS_VMCODE_IF_TRUE_JUMP: 
 688             case NJS_VMCODE_IF_FALSE_JUMP: 
 689                 ret = njs_is_true(value1); 
 690  
 691                 ret ^= op - NJS_VMCODE_IF_TRUE_JUMP; 
 692  
 693                 ret = ret ? (njs_jump_off_t) value2 
 694                           : (njs_jump_off_t) sizeof(njs_vmcode_cond_jump_t); 
 695  
 696                 break; 
 697  
 698             case NJS_VMCODE_IF_EQUAL_JUMP: 
 699                 if (njs_values_strict_equal(value1, value2)) { 
 700                     equal = (njs_vmcode_equal_jump_t *) pc; 
 701                     ret = equal->offset; 
 702  
 703                 } else { 
 704                     ret = sizeof(njs_vmcode_3addr_t); 
 705                 } 
 706  
 707                 break; 
 708  
 709             case NJS_VMCODE_PROPERTY_INIT: 
 710                 set = (njs_vmcode_prop_set_t *) pc; 
 711                 njs_vmcode_operand(vm, set->value, retval); 
 712                 ret = njs_vmcode_property_init(vm, value1, value2, retval); 
 713                 if (njs_slow_path(ret == NJS_ERROR)) { 
 714                     goto error; 
 715                 } 
 716  
 717                 break; 
 718  
 719             case NJS_VMCODE_RETURN: 
 720                 njs_vmcode_operand(vm, (njs_index_t) value2, value2); 
 721                 return njs_vmcode_return(vm, NULL, value2); 
 722  
 723             case NJS_VMCODE_FUNCTION_COPY: 
 724                 fcopy = (njs_vmcode_function_copy_t *) pc; 
 725                 ret = njs_vmcode_function_copy(vm, fcopy->function, 
 726                                                fcopy->retval); 
 727                 break; 
 728  
 729             case NJS_VMCODE_FUNCTION_FRAME: 
 730                 function_frame = (njs_vmcode_function_frame_t *) pc; 
 731  
 732                 /* TODO: external object instead of void this. */ 
 733  
 734                 ret = njs_function_frame_create(vm, value1, 
 735                                                 &njs_value_undefined, 
 736                                                 (uintptr_t) value2, 
 737                                                 function_frame->ctor); 
 738  
 739                 if (njs_slow_path(ret != NJS_OK)) { 
 740                     goto error; 
 741                 } 
 742  
 743                 ret = sizeof(njs_vmcode_function_frame_t); 
 744                 break; 
 745  
 746             case NJS_VMCODE_METHOD_FRAME: 
 747                 method_frame = (njs_vmcode_method_frame_t *) pc; 
 748  
 749                 ret = njs_value_property(vm, value1, value2, &dst); 
 750                 if (njs_slow_path(ret == NJS_ERROR)) { 
 751                     goto error; 
 752                 } 
 753  
 754                 if (njs_slow_path(!njs_is_function(&dst))) { 
 755                     ret = njs_value_to_key(vm, value2, value2); 
 756                     if (njs_slow_path(ret != NJS_OK)) { 
 757                         return NJS_ERROR; 
 758                     } 
 759  
 760                     njs_key_string_get(vm, value2, &string); 
 761                     njs_type_error(vm, 
 762                                "(intermediate value)[\"%V\"] is not a function", 
 763                                &string); 
 764                     goto error; 
 765                 } 
 766  
 767                 ret = njs_function_frame_create(vm, &dst, value1, 
 768                                                 method_frame->nargs, 
 769                                                 method_frame->ctor); 
 770  
 771                 if (njs_slow_path(ret != NJS_OK)) { 
 772                     goto error; 
 773                 } 
 774  
 775                 ret = sizeof(njs_vmcode_method_frame_t); 
 776                 break; 
 777  
 778             case NJS_VMCODE_FUNCTION_CALL: 
 779                 vm->active_frame->native.pc = pc; 
 780  
 781                 njs_vmcode_operand(vm, (njs_index_t) value2, value2); 
 782  
 783                 ret = njs_function_frame_invoke(vm, value2); 
 784                 if (njs_slow_path(ret == NJS_ERROR)) { 
 785                     goto error; 
 786                 } 
 787  
 788                 ret = sizeof(njs_vmcode_function_call_t); 
 789                 break; 
 790  
 791             case NJS_VMCODE_PROPERTY_NEXT: 
 792                 pnext = (njs_vmcode_prop_next_t *) pc; 
 793                 retval = njs_scope_value(vm, pnext->retval); 
 794  
 795                 next = value2->data.u.next; 
 796  
 797                 if (next->index < next->array->length) { 
 798                     *retval = next->array->start[next->index++]; 
 799  
 800                     ret = pnext->offset; 
 801                     break; 
 802                 } 
 803  
 804                 njs_mp_free(vm->mem_pool, next); 
 805  
 806                 ret = sizeof(njs_vmcode_prop_next_t); 
 807                 break; 
 808  
 809             case NJS_VMCODE_ARGUMENTS: 
 810                 ret = njs_vmcode_arguments(vm, pc); 
 811                 if (njs_slow_path(ret == NJS_ERROR)) { 
 812                     goto error; 
 813                 } 
 814  
 815                 break; 
 816  
 817             case NJS_VMCODE_PROTO_INIT: 
 818                 set = (njs_vmcode_prop_set_t *) pc; 
 819                 njs_vmcode_operand(vm, set->value, retval); 
 820                 ret = njs_vmcode_proto_init(vm, value1, value2, retval); 
 821                 if (njs_slow_path(ret == NJS_ERROR)) { 
 822                     goto error; 
 823                 } 
 824  
 825                 break; 
 826  
 827             case NJS_VMCODE_AWAIT: 
 828                 await = (njs_vmcode_await_t *) pc; 
 829                 return njs_vmcode_await(vm, await); 
 830  
 831             case NJS_VMCODE_TRY_START: 
 832                 ret = njs_vmcode_try_start(vm, value1, value2, pc); 
 833                 if (njs_slow_path(ret == NJS_ERROR)) { 
 834                     goto error; 
 835                 } 
 836  
 837                 break; 
 838  
 839             case NJS_VMCODE_THROW: 
 840                 njs_vmcode_operand(vm, (njs_index_t) value2, value2); 
 841                 vm->retval = *value2; 
 842                 goto error; 
 843  
 844             case NJS_VMCODE_TRY_BREAK: 
 845                 try_trampoline = (njs_vmcode_try_trampoline_t *) pc; 
 846                 value1 = njs_scope_value(vm, try_trampoline->exit_value); 
 847  
 848                 ret = njs_vmcode_try_break(vm, value1, value2); 
 849                 break; 
 850  
 851             case NJS_VMCODE_TRY_CONTINUE: 
 852                 try_trampoline = (njs_vmcode_try_trampoline_t *) pc; 
 853                 value1 = njs_scope_value(vm, try_trampoline->exit_value); 
 854  
 855                 ret = njs_vmcode_try_continue(vm, value1, value2); 
 856                 break; 
 857  
 858             case NJS_VMCODE_TRY_END: 
 859                 ret = njs_vmcode_try_end(vm, value1, value2); 
 860                 break; 
 861  
 862             /* 
 863              * njs_vmcode_catch() is set on the start of a "catch" block to 
 864              * store exception and to remove a "try" block if there is no 
 865              * "finally" block or to update a catch address to the start of 
 866              * a "finally" block. 
 867              * njs_vmcode_catch() is set on the start of a "finally" block 
 868              * to store uncaught exception and to remove a "try" block. 
 869              */ 
 870             case NJS_VMCODE_CATCH: 
 871                 *value1 = vm->retval; 
 872  
 873                 if ((njs_jump_off_t) value2 == sizeof(njs_vmcode_catch_t)) { 
 874                     ret = njs_vmcode_try_end(vm, value1, value2); 
 875  
 876                 } else { 
 877                     frame = (njs_frame_t *) vm->top_frame; 
 878                     frame->exception.catch = pc + (njs_jump_off_t) value2; 
 879                     ret = sizeof(njs_vmcode_catch_t); 
 880                 } 
 881  
 882                 break; 
 883  
 884             case NJS_VMCODE_FINALLY: 
 885                 finally = (njs_vmcode_finally_t *) pc; 
 886                 value1 = njs_scope_value(vm, finally->exit_value); 
 887  
 888                 ret = njs_vmcode_finally(vm, value1, value2, pc); 
 889  
 890                 switch (ret) { 
 891                 case NJS_OK: 
 892                     return NJS_OK; 
 893                 case NJS_ERROR: 
 894                     goto error; 
 895                 } 
 896  
 897                 break; 
 898  
 899             case NJS_VMCODE_LET: 
 900                 var = (njs_vmcode_variable_t *) pc; 
 901                 value1 = njs_scope_value(vm, var->dst); 
 902  
 903                 if (njs_is_valid(value1)) { 
 904                     value1 = njs_mp_alloc(vm->mem_pool, sizeof(njs_value_t)); 
 905                     if (njs_slow_path(value1 == NULL)) { 
 906                         return NJS_ERROR; 
 907                     } 
 908  
 909                     njs_scope_value_set(vm, var->dst, value1); 
 910                 } 
 911  
 912                 njs_set_undefined(value1); 
 913  
 914                 ret = sizeof(njs_vmcode_variable_t); 
 915                 break; 
 916  
 917             case NJS_VMCODE_LET_UPDATE: 
 918                 var = (njs_vmcode_variable_t *) pc; 
 919                 value2 = njs_scope_value(vm, var->dst); 
 920  
 921                 value1 = njs_mp_alloc(vm->mem_pool, sizeof(njs_value_t)); 
 922                 if (njs_slow_path(value1 == NULL)) { 
 923                     return NJS_ERROR; 
 924                 } 
 925  
 926                 *value1 = *value2; 
 927  
 928                 njs_scope_value_set(vm, var->dst, value1); 
 929  
 930                 ret = sizeof(njs_vmcode_variable_t); 
 931                 break; 
 932  
 933             case NJS_VMCODE_INITIALIZATION_TEST: 
 934                 var = (njs_vmcode_variable_t *) pc; 
 935                 value1 = njs_scope_value(vm, var->dst); 
 936  
 937                 if (njs_is_valid(value1)) { 
 938                     ret = sizeof(njs_vmcode_variable_t); 
 939                     break; 
 940                 } 
 941  
 942                 /* Fall through. */ 
 943  
 944             case NJS_VMCODE_NOT_INITIALIZED: 
 945                 njs_reference_error(vm, "cannot access variable " 
 946                                         "before initialization"); 
 947                 goto error; 
 948  
 949             case NJS_VMCODE_ERROR: 
 950                 njs_vmcode_error(vm, pc); 
 951                 goto error; 
 952  
 953             case NJS_VMCODE_ASSIGNMENT_ERROR: 
 954                 njs_type_error(vm, "assignment to constant variable"); 
 955                 goto error; 
 956  
 957             default: 
 958                 njs_internal_error(vm, "%d has NO retval", op); 
 959                 goto error; 
 960             } 
 961         } 
 962  
 963         pc += ret; 
 964     } 
 965  
 966 error: 
 967  
 968     if (njs_is_error(&vm->retval)) { 
 969         vm->active_frame->native.pc = pc; 
 970         (void) njs_error_stack_attach(vm, &vm->retval); 
 971     } 
 972  
 973     for ( ;; ) { 
 974         native = vm->top_frame; 
 975  
 976         if (!native->native) { 
 977             frame = (njs_frame_t *) native; 
 978             catch = frame->exception.catch; 
 979  
 980             if (catch != NULL) { 
 981                 pc = catch; 
 982  
 983                 goto next; 
 984             } 
 985         } 
 986  
 987         previous = native->previous; 
 988         if (previous == NULL) { 
 989             break; 
 990         } 
 991  
 992         lambda_call = (native == &vm->active_frame->native); 
 993  
 994         njs_vm_scopes_restore(vm, native, previous); 
 995  
 996         if (native->size != 0) { 
 997             vm->stack_size -= native->size; 
 998             njs_mp_free(vm->mem_pool, native); 
 999         } 
1000  
1001         if (lambda_call) { 
1002             break; 
1003         } 
1004     } 
1005  
1006     return NJS_ERROR; 
1007 }

1028 njs_vmcode_array(njs_vm_t *vm, u_char *pc) 
1029 { 
1030     uint32_t            length; 
1031     njs_array_t         *array; 
1032     njs_value_t         *value; 
1033     njs_vmcode_array_t  *code; 
1034  
1035     code = (njs_vmcode_array_t *) pc; 
1036  
1037     array = njs_array_alloc(vm, 0, code->length, NJS_ARRAY_SPARE); 
1038  
1039     if (njs_fast_path(array != NULL)) { 
1040  
1041         if (code->ctor) { 
1042             /* Array of the form [,,,], [1,,]. */ 
1043             value = array->start; 
1044             length = array->length; 
1045  
1046             do { 
1047                 njs_set_invalid(value); 
1048                 value++; 
1049                 length--; 
1050             } while (length != 0); 
1051  
1052         } else { 
1053             /* Array of the form [], [,,1], [1,2,3]. */ 
1054             array->length = 0; 
1055         } 
1056  
1057         njs_set_array(&vm->retval, array); 
1058  
1059         return sizeof(njs_vmcode_array_t); 
1060     } 
1061  
1062     return NJS_ERROR; 
1063 }

1066 static njs_jump_off_t 
1067 njs_vmcode_function(njs_vm_t *vm, u_char *pc) 
1068 { 
1069     njs_function_t         *function; 
1070     njs_vmcode_function_t  *code; 
1071     njs_function_lambda_t  *lambda; 
1072  
1073     code = (njs_vmcode_function_t *) pc; 
1074     lambda = code->lambda; 
1075  
1076     function = njs_function_alloc(vm, lambda, code->async); 
1077     if (njs_slow_path(function == NULL)) { 
1078         return NJS_ERROR; 
1079     } 
1080  
1081     if (njs_function_capture_closure(vm, function, lambda) != NJS_OK) { 
1082         return NJS_ERROR; 
1083     } 
1084  
1085     function->args_count = lambda->nargs - lambda->rest_parameters;

1610         return (njs_number(val1) == njs_number(val2));

1624     }

1670  *   1 if val1 is less than val2,

1760     } 
1761  
1762     function = njs_function(constructor); 
1763

1765         function = function->u.bound_target; 
1766         njs_set_function(&bound, function);

1772

1774         return NULL; 
1775     } 
1776  
1777     if (njs_fast_path(njs_is_object(&proto))) { 
1778         object->__proto__ = njs_object(&proto); 
1779     } 
1780  
1781     return object;

1786 njs_vmcode_return(njs_vm_t *vm, njs_value_t *invld, njs_value_t *retval) 
1787 {

1789     njs_native_frame_t  *previous; 
1790  
1791     frame = (njs_frame_t *) vm->top_frame; 
1792  
1793     if (frame->native.ctor) { 
1794         if (njs_is_object(retval)) { 
1795             njs_release(vm, frame->native.local[0]); 
1796  
1797         } else { 
1798             retval = frame->native.local[0]; 
1799         } 
1800     } 
1801  
1802     previous = njs_function_previous_frame(&frame->native); 
1803  
1804     njs_vm_scopes_restore(vm, &frame->native, previous); 
1805  
1806     *frame->native.retval = *retval; 
1807  
1808     njs_function_frame_free(vm, &frame->native); 
1809  
1810     return NJS_OK;

1814 static njs_jump_off_t 
1815 njs_vmcode_await(njs_vm_t *vm, njs_vmcode_await_t *await) 
1816 { 
1817     size_t              size; 
1818     njs_int_t           ret; 
1819     njs_frame_t         *frame; 
1820     njs_value_t         ctor, val, on_fulfilled, on_rejected, *value; 
1821     njs_promise_t       *promise; 
1822     njs_function_t      *fulfilled, *rejected; 
1823     njs_async_ctx_t     *ctx; 
1824     njs_native_frame_t  *active; 
1825  
1826     active = &vm->active_frame->native; 
1827  
1828     value = njs_scope_valid_value(vm, await->retval); 
1829     if (njs_slow_path(value == NULL)) { 
1830         return NJS_ERROR; 
1831     } 
1832  
1833     njs_set_function(&ctor, &vm->constructors[NJS_OBJ_TYPE_PROMISE]); 
1834  
1835     promise = njs_promise_resolve(vm, &ctor, value); 
1836     if (njs_slow_path(promise == NULL)) { 
1837         return NJS_ERROR; 
1838     } 
1839  
1840     ctx = active->function->await; 
1841  
1842     if (ctx == NULL) { 
1843         ctx = njs_mp_alloc(vm->mem_pool, sizeof(njs_async_ctx_t)); 
1844         if (njs_slow_path(ctx == NULL)) { 
1845             njs_memory_error(vm); 
1846             return NJS_ERROR; 
1847         } 
1848  
1849         size = njs_function_frame_size(active); 
1850  
1851         fulfilled = njs_promise_create_function(vm, size); 
1852         if (njs_slow_path(fulfilled == NULL)) { 
1853             return NJS_ERROR; 
1854         } 
1855  
1856         ctx->await = fulfilled->context; 
1857         ctx->capability = active->function->context; 
1858  
1859         active->function->context = NULL; 
1860  
1861         ret = njs_function_frame_save(vm, ctx->await, NULL); 
1862         if (njs_slow_path(ret != NJS_OK)) { 
1863             return NJS_ERROR; 
1864         } 
1865  
1866     } else { 
1867         fulfilled = njs_promise_create_function(vm, 0); 
1868         if (njs_slow_path(fulfilled == NULL)) { 
1869             return NJS_ERROR; 
1870         } 
1871     } 
1872  
1873     ctx->pc = (u_char *) await + sizeof(njs_vmcode_await_t); 
1874     ctx->index = await->retval; 
1875  
1876     frame = (njs_frame_t *) active; 
1877  
1878     if (frame->exception.catch != NULL) { 
1879         ctx->await->native.pc = frame->exception.catch; 
1880  
1881     } else { 
1882         ctx->await->native.pc = ctx->pc; 
1883     } 
1884  
1885     fulfilled->context = ctx; 
1886     fulfilled->args_count = 1; 
1887     fulfilled->u.native = njs_await_fulfilled; 
1888  
1889     rejected = njs_promise_create_function(vm, 0); 
1890     if (njs_slow_path(rejected == NULL)) { 
1891         return NJS_ERROR; 
1892     } 
1893  
1894     rejected->context = ctx; 
1895     rejected->args_count = 1; 
1896     rejected->u.native = njs_await_rejected; 
1897  
1898     njs_set_object(&val, &promise->object); 
1899     njs_set_function(&on_fulfilled, fulfilled); 
1900     njs_set_function(&on_rejected, rejected); 
1901  
1902     ret = njs_promise_perform_then(vm, &val, &on_fulfilled, &on_rejected, NULL); 
1903     if (njs_slow_path(ret != NJS_OK)) { 
1904         return NJS_ERROR; 
1905     } 
1906  
1907     (void) njs_vmcode_return(vm, NULL, &vm->retval); 
1908  
1909     return NJS_AGAIN; 
1910 }

1913 /* 
1914  * njs_vmcode_try_start() is set on the start of a "try" block to create 
1915  * a "try" block, to set a catch address to the start of a "catch" or 
1916  * "finally" blocks and to initialize a value to track uncaught exception. 
1917  */ 
1918  
1919 static njs_jump_off_t 
1920 njs_vmcode_try_start(njs_vm_t *vm, njs_value_t *exception_value, 
1921     njs_value_t *offset, u_char *pc) 
1922 { 
1923     njs_value_t             *exit_value; 
1924     njs_frame_t             *frame; 
1925     njs_exception_t         *e; 
1926     njs_vmcode_try_start_t  *try_start; 
1927  
1928     frame = (njs_frame_t *) vm->top_frame; 
1929  
1930     if (frame->exception.catch != NULL) { 
1931         e = njs_mp_alloc(vm->mem_pool, sizeof(njs_exception_t)); 
1932         if (njs_slow_path(e == NULL)) { 
1933             njs_memory_error(vm); 
1934             return NJS_ERROR; 
1935         } 
1936  
1937         *e = frame->exception; 
1938         frame->exception.next = e; 
1939     } 
1940  
1941     frame->exception.catch = pc + (njs_jump_off_t) offset; 
1942  
1943     njs_set_invalid(exception_value); 
1944  
1945     try_start = (njs_vmcode_try_start_t *) pc; 
1946     exit_value = njs_scope_value(vm, try_start->exit_value); 
1947  
1948     njs_set_invalid(exit_value); 
1949     njs_number(exit_value) = 0; 
1950  
1951     return sizeof(njs_vmcode_try_start_t); 
1952 }

1957  * the nearest try_end block. The exit_value is checked by njs_vmcode_finally().

1988 /* 
1989  * njs_vmcode_try_end() is set on the end of a "try" block to remove the block. 
1990  * It is also set on the end of a "catch" block followed by a "finally" block. 
1991  */ 
1992  
1993 static njs_jump_off_t 
1994 njs_vmcode_try_end(njs_vm_t *vm, njs_value_t *invld, njs_value_t *offset) 
1995 { 
1996     njs_frame_t      *frame; 
1997     njs_exception_t  *e; 
1998  
1999     frame = (njs_frame_t *) vm->top_frame; 
2000     e = frame->exception.next; 
2001  
2002     if (e == NULL) { 
2003         frame->exception.catch = NULL; 
2004  
2005     } else { 
2006         frame->exception = *e; 
2007         njs_mp_free(vm->mem_pool, e); 
2008     } 
2009  
2010     return (njs_jump_off_t) offset; 
2011 }

2014 /* 
2015  * njs_vmcode_finally() is set on the end of a "finally" or a "catch" block. 
2016  *   1) to throw uncaught exception. 
2017  *   2) to make a jump to an enslosing loop exit if "continue" or "break" 
2018  *      statement was used inside try block. 
2019  *   3) to finalize "return" instruction from "try" block. 
2020  */ 
2021  
2022 static njs_jump_off_t 
2023 njs_vmcode_finally(njs_vm_t *vm, njs_value_t *invld, njs_value_t *retval, 
2024     u_char *pc) 
2025 { 
2026     njs_value_t           *exception_value, *exit_value; 
2027     njs_vmcode_finally_t  *finally; 
2028  
2029     exception_value = njs_scope_value(vm, (njs_index_t) retval); 
2030  
2031     if (njs_is_valid(exception_value)) { 
2032         vm->retval = *exception_value; 
2033  
2034         return NJS_ERROR; 
2035     } 
2036  
2037     finally = (njs_vmcode_finally_t *) pc; 
2038  
2039     exit_value = njs_scope_value(vm, finally->exit_value); 
2040  
2041     /* 
2042      * exit_value is set by: 
2043      *   vmcode_try_start to INVALID 0 
2044      *   vmcode_try_break to INVALID 1 
2045      *   vmcode_try_continue to INVALID -1 
2046      *   vmcode_try_return to a valid return value 
2047      */ 
2048  
2049     if (njs_is_valid(exit_value)) { 
2050         return njs_vmcode_return(vm, NULL, exit_value); 
2051  
2052     } else if (njs_number(exit_value) != 0) { 
2053         return (njs_jump_off_t) (njs_number(exit_value) > 0) 
2054                                 ? finally->break_offset 
2055                                 : finally->continue_offset; 
2056     } 
2057  
2058     return sizeof(njs_vmcode_finally_t); 
2059 }
