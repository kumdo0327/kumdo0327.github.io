 91

143 def get_filepath_or_buffer( 
144     filepath_or_buffer: FilePathOrBuffer, 
145     encoding: Optional[str] = None, 
146     compression: Optional[str] = None, 
147     mode: Optional[str] = None, 
148 ): 
149     """ 
150     If the filepath_or_buffer is a url, translate and return the buffer. 
151     Otherwise passthrough. 
152  
153     Parameters 
154     ---------- 
155     filepath_or_buffer : a url, filepath (str, py.path.local or pathlib.Path), 
156                          or buffer 
157     compression : {{'gzip', 'bz2', 'zip', 'xz', None}}, optional 
158     encoding : the encoding to use to decode bytes, default is 'utf-8' 
159     mode : str, optional 
160  
161     Returns 
162     ------- 
163     Tuple[FilePathOrBuffer, str, str, bool] 
164         Tuple containing the filepath or buffer, the encoding, the compression 
165         and should_close. 
166     """ 
167     filepath_or_buffer = stringify_path(filepath_or_buffer) 
168  
169     if isinstance(filepath_or_buffer, str) and is_url(filepath_or_buffer): 
170         req = urlopen(filepath_or_buffer) 
171         content_encoding = req.headers.get("Content-Encoding", None) 
172         if content_encoding == "gzip": 
173             # Override compression based on Content-Encoding header 
174             compression = "gzip" 
175         reader = BytesIO(req.read()) 
176         req.close() 
177         return reader, encoding, compression, True 
178  
179     if is_s3_url(filepath_or_buffer): 
180         from pandas.io import s3 
181  
182         return s3.get_filepath_or_buffer( 
183             filepath_or_buffer, encoding=encoding, compression=compression, mode=mode 
184         ) 
185  
186     if is_gcs_url(filepath_or_buffer): 
187         from pandas.io import gcs 
188  
189         return gcs.get_filepath_or_buffer( 
190             filepath_or_buffer, encoding=encoding, compression=compression, mode=mode 
191         ) 
192  
193     if isinstance(filepath_or_buffer, (str, bytes, mmap.mmap)): 
194         return _expand_user(filepath_or_buffer), None, compression, False 
195  
196     if not is_file_like(filepath_or_buffer): 
197         msg = f"Invalid file path or buffer object type: {type(filepath_or_buffer)}" 
198         raise ValueError(msg) 
199  
200     return filepath_or_buffer, None, compression, False

224 def get_compression_method( 
225     compression: Optional[Union[str, Mapping[str, str]]] 
226 ) -> Tuple[Optional[str], Dict[str, str]]: 
227     """ 
228     Simplifies a compression argument to a compression method string and 
229     a mapping containing additional arguments. 
230  
231     Parameters 
232     ---------- 
233     compression : str or mapping 
234         If string, specifies the compression method. If mapping, value at key 
235         'method' specifies compression method. 
236  
237     Returns 
238     ------- 
239     tuple of ({compression method}, Optional[str] 
240               {compression arguments}, Dict[str, str]) 
241  
242     Raises 
243     ------ 
244     ValueError on mapping missing 'method' key 
245     """ 
246     if isinstance(compression, Mapping): 
247         compression_args = dict(compression) 
248         try: 
249             compression = compression_args.pop("method") 
250         except KeyError: 
251             raise ValueError("If mapping, compression must have key 'method'") 
252     else: 
253         compression_args = {} 
254     return compression, compression_args

257 def infer_compression( 
258     filepath_or_buffer: FilePathOrBuffer, compression: Optional[str] 
259 ) -> Optional[str]: 
260     """ 
261     Get the compression method for filepath_or_buffer. If compression='infer', 
262     the inferred compression method is returned. Otherwise, the input 
263     compression method is returned unchanged, unless it's invalid, in which 
264     case an error is raised. 
265  
266     Parameters 
267     ---------- 
268     filepath_or_buffer : str or file handle 
269         File path or object. 
270     compression : {'infer', 'gzip', 'bz2', 'zip', 'xz', None} 
271         If 'infer' and `filepath_or_buffer` is path-like, then detect 
272         compression from the following extensions: '.gz', '.bz2', '.zip', 
273         or '.xz' (otherwise no compression). 
274  
275     Returns 
276     ------- 
277     string or None 
278  
279     Raises 
280     ------ 
281     ValueError on invalid compression specified. 
282     """ 
283  
284     # No compression has been explicitly specified 
285     if compression is None: 
286         return None 
287  
288     # Infer compression 
289     if compression == "infer": 
290         # Convert all path types (e.g. pathlib.Path) to strings 
291         filepath_or_buffer = stringify_path(filepath_or_buffer) 
292         if not isinstance(filepath_or_buffer, str): 
293             # Cannot infer compression of a buffer, assume no compression 
294             return None 
295  
296         # Infer compression from the filename/URL extension 
297         for compression, extension in _compression_to_extension.items(): 
298             if filepath_or_buffer.endswith(extension): 
299                 return compression 
300         return None 
301  
302     # Compression has been specified. Check that it's valid 
303     if compression in _compression_to_extension: 
304         return compression 
305  
306     msg = f"Unrecognized compression type: {compression}" 
307     valid = ["infer", None] + sorted(_compression_to_extension) 
308     msg += f"\nValid compression types are {valid}" 
309     raise ValueError(msg)

312 def get_handle( 
313     path_or_buf, 
314     mode: str, 
315     encoding=None, 
316     compression: Optional[Union[str, Mapping[str, Any]]] = None, 
317     memory_map: bool = False, 
318     is_text: bool = True, 
319 ): 
320     """ 
321     Get file handle for given path/buffer and mode. 
322  
323     Parameters 
324     ---------- 
325     path_or_buf : str or file handle 
326         File path or object. 
327     mode : str 
328         Mode to open path_or_buf with. 
329     encoding : str or None 
330         Encoding to use. 
331     compression : str or dict, default None 
332         If string, specifies compression mode. If dict, value at key 'method' 
333         specifies compression mode. Compression mode must be one of {'infer', 
334         'gzip', 'bz2', 'zip', 'xz', None}. If compression mode is 'infer' 
335         and `filepath_or_buffer` is path-like, then detect compression from 
336         the following extensions: '.gz', '.bz2', '.zip', or '.xz' (otherwise 
337         no compression). If dict and compression mode is 'zip' or inferred as 
338         'zip', other entries passed as additional compression options. 
339  
340         .. versionchanged:: 1.0.0 
341  
342            May now be a dict with key 'method' as compression mode 
343            and other keys as compression options if compression 
344            mode is 'zip'. 
345  
346     memory_map : boolean, default False 
347         See parsers._parser_params for more information. 
348     is_text : boolean, default True 
349         whether file/buffer is in text format (csv, json, etc.), or in binary 
350         mode (pickle, etc.). 
351  
352     Returns 
353     ------- 
354     f : file-like 
355         A file-like object. 
356     handles : list of file-like objects 
357         A list of file-like object that were opened in this function. 
358     """ 
359     try: 
360         from s3fs import S3File 
361  
362         need_text_wrapping = (BufferedIOBase, S3File) 
363     except ImportError: 
364         need_text_wrapping = BufferedIOBase  # type: ignore 
365  
366     handles: List[IO] = list() 
367     f = path_or_buf 
368  
369     # Convert pathlib.Path/py.path.local or string 
370     path_or_buf = stringify_path(path_or_buf) 
371     is_path = isinstance(path_or_buf, str) 
372  
373     compression, compression_args = get_compression_method(compression) 
374     if is_path: 
375         compression = infer_compression(path_or_buf, compression) 
376  
377     if compression: 
378  
379         # GZ Compression 
380         if compression == "gzip": 
381             if is_path: 
382                 f = gzip.open(path_or_buf, mode) 
383             else: 
384                 f = gzip.GzipFile(fileobj=path_or_buf) 
385  
386         # BZ Compression 
387         elif compression == "bz2": 
388             if is_path: 
389                 f = bz2.BZ2File(path_or_buf, mode) 
390             else: 
391                 f = bz2.BZ2File(path_or_buf) 
392  
393         # ZIP Compression 
394         elif compression == "zip": 
395             zf = _BytesZipFile(path_or_buf, mode, **compression_args) 
396             # Ensure the container is closed as well. 
397             handles.append(zf) 
398             if zf.mode == "w": 
399                 f = zf 
400             elif zf.mode == "r": 
401                 zip_names = zf.namelist() 
402                 if len(zip_names) == 1: 
403                     f = zf.open(zip_names.pop()) 
404                 elif len(zip_names) == 0: 
405                     raise ValueError(f"Zero files found in ZIP file {path_or_buf}") 
406                 else: 
407                     raise ValueError( 
408                         "Multiple files found in ZIP file. " 
409                         f"Only one file per ZIP: {zip_names}" 
410                     ) 
411  
412         # XZ Compression 
413         elif compression == "xz": 
414             f = _get_lzma_file(lzma)(path_or_buf, mode) 
415  
416         # Unrecognized Compression 
417         else: 
418             msg = f"Unrecognized compression type: {compression}" 
419             raise ValueError(msg) 
420  
421         handles.append(f) 
422  
423     elif is_path: 
424         if encoding: 
425             # Encoding 
426             f = open(path_or_buf, mode, encoding=encoding, newline="") 
427         elif is_text: 
428             # No explicit encoding 
429             f = open(path_or_buf, mode, errors="replace", newline="") 
430         else: 
431             # Binary mode 
432             f = open(path_or_buf, mode) 
433         handles.append(f) 
434  
435     # Convert BytesIO or file objects passed with an encoding 
436     if is_text and (compression or isinstance(f, need_text_wrapping)): 
437         from io import TextIOWrapper 
438  
439         g = TextIOWrapper(f, encoding=encoding, newline="") 
440         if not isinstance(f, BufferedIOBase): 
441             handles.append(g) 
442         f = g 
443  
444     if memory_map and hasattr(f, "fileno"): 
445         try: 
446             wrapped = _MMapWrapper(f) 
447             f.close() 
448             f = wrapped 
449         except Exception: 
450             # we catch any errors that may have occurred 
451             # because that is consistent with the lower-level 
452             # functionality of the C engine (pd.read_csv), so 
453             # leave the file handler as is then 
454             pass 
455  
456     return f, handles

459 class _BytesZipFile(zipfile.ZipFile, BytesIO):  # type: ignore 
460     """ 
461     Wrapper for standard library class ZipFile and allow the returned file-like 
462     handle to accept byte strings via `write` method. 
463  
464     BytesIO provides attributes of file-like object and ZipFile.writestr writes 
465     bytes strings into a member of the archive. 
466     """ 
467  
468     # GH 17778 
469     def __init__( 
470         self, 
471         file: FilePathOrBuffer, 
472         mode: str, 
473         archive_name: Optional[str] = None, 
474         **kwargs, 
475     ): 
476         if mode in ["wb", "rb"]: 
477             mode = mode.replace("b", "") 
478         self.archive_name = archive_name 
479         super().__init__(file, mode, zipfile.ZIP_DEFLATED, **kwargs) 
480  
481     def write(self, data): 
482         archive_name = self.filename 
483         if self.archive_name is not None: 
484             archive_name = self.archive_name 
485         super().writestr(archive_name, data) 
486  
487     @property 
488     def closed(self): 
489         return self.fp is None

492 class _MMapWrapper(abc.Iterator): 
493     """ 
494     Wrapper for the Python's mmap class so that it can be properly read in 
495     by Python's csv.reader class. 
496  
497     Parameters 
498     ---------- 
499     f : file object 
500         File object to be mapped onto memory. Must support the 'fileno' 
501         method or have an equivalent attribute 
502  
503     """ 
504  
505     def __init__(self, f: IO): 
506         self.mmap = mmap.mmap(f.fileno(), 0, access=mmap.ACCESS_READ) 
507  
508     def __getattr__(self, name: str): 
509         return getattr(self.mmap, name) 
510  
511     def __iter__(self) -> "_MMapWrapper": 
512         return self 
513  
514     def __next__(self) -> str: 
515         newbytes = self.mmap.readline() 
516  
517         # readline returns bytes, not str, but Python's CSV reader 
518         # expects str, so convert the output to str before continuing 
519         newline = newbytes.decode("utf-8") 
520  
521         # mmap doesn't raise if reading past the allocated 
522         # data but instead returns an empty string, so raise 
523         # if that is returned 
524         if newline == "": 
525             raise StopIteration 
526         return newline

221 _compression_to_extension = {"gzip": ".gz", "bz2": ".bz2", "zip": ".zip", "xz": ".xz"} 
222  
223  
224 def get_compression_method( 
225     compression: Optional[Union[str, Mapping[str, str]]] 
226 ) -> Tuple[Optional[str], Dict[str, str]]: 
227     """ 
228     Simplifies a compression argument to a compression method string and 
229     a mapping containing additional arguments. 
230  
231     Parameters 
232     ---------- 
233     compression : str or mapping 
234         If string, specifies the compression method. If mapping, value at key 
235         'method' specifies compression method. 
236  
237     Returns 
238     ------- 
239     tuple of ({compression method}, Optional[str] 
240               {compression arguments}, Dict[str, str]) 
241  
242     Raises 
243     ------ 
244     ValueError on mapping missing 'method' key 
245     """ 
246     if isinstance(compression, Mapping): 
247         compression_args = dict(compression) 
248         try: 
249             compression = compression_args.pop("method") 
250         except KeyError: 
251             raise ValueError("If mapping, compression must have key 'method'") 
252     else: 
253         compression_args = {} 
254     return compression, compression_args

257 def infer_compression( 
258     filepath_or_buffer: FilePathOrBuffer, compression: Optional[str] 
259 ) -> Optional[str]: 
260     """ 
261     Get the compression method for filepath_or_buffer. If compression='infer', 
262     the inferred compression method is returned. Otherwise, the input 
263     compression method is returned unchanged, unless it's invalid, in which 
264     case an error is raised. 
265  
266     Parameters 
267     ---------- 
268     filepath_or_buffer : str or file handle 
269         File path or object. 
270     compression : {'infer', 'gzip', 'bz2', 'zip', 'xz', None} 
271         If 'infer' and `filepath_or_buffer` is path-like, then detect 
272         compression from the following extensions: '.gz', '.bz2', '.zip', 
273         or '.xz' (otherwise no compression). 
274  
275     Returns 
276     ------- 
277     string or None 
278  
279     Raises 
280     ------ 
281     ValueError on invalid compression specified. 
282     """ 
283  
284     # No compression has been explicitly specified 
285     if compression is None: 
286         return None 
287  
288     # Infer compression 
289     if compression == "infer": 
290         # Convert all path types (e.g. pathlib.Path) to strings 
291         filepath_or_buffer = stringify_path(filepath_or_buffer) 
292         if not isinstance(filepath_or_buffer, str): 
293             # Cannot infer compression of a buffer, assume no compression 
294             return None 
295  
296         # Infer compression from the filename/URL extension 
297         for compression, extension in _compression_to_extension.items(): 
298             if filepath_or_buffer.endswith(extension): 
299                 return compression 
300         return None 
301  
302     # Compression has been specified. Check that it's valid 
303     if compression in _compression_to_extension: 
304         return compression 
305  
306     msg = f"Unrecognized compression type: {compression}" 
307     valid = ["infer", None] + sorted(_compression_to_extension) 
308     msg += f"\nValid compression types are {valid}" 
309     raise ValueError(msg) 
310

312 def get_handle( 
313     path_or_buf, 
314     mode: str, 
315     encoding=None, 
316     compression: Optional[Union[str, Mapping[str, Any]]] = None, 
317     memory_map: bool = False, 
318     is_text: bool = True, 
319 ): 
320     """ 
321     Get file handle for given path/buffer and mode. 
322  
323     Parameters 
324     ---------- 
325     path_or_buf : str or file handle 
326         File path or object. 
327     mode : str 
328         Mode to open path_or_buf with. 
329     encoding : str or None 
330         Encoding to use. 
331     compression : str or dict, default None 
332         If string, specifies compression mode. If dict, value at key 'method' 
333         specifies compression mode. Compression mode must be one of {'infer', 
334         'gzip', 'bz2', 'zip', 'xz', None}. If compression mode is 'infer' 
335         and `filepath_or_buffer` is path-like, then detect compression from 
336         the following extensions: '.gz', '.bz2', '.zip', or '.xz' (otherwise 
337         no compression). If dict and compression mode is 'zip' or inferred as 
338         'zip', other entries passed as additional compression options. 
339  
340         .. versionchanged:: 1.0.0 
341  
342            May now be a dict with key 'method' as compression mode 
343            and other keys as compression options if compression 
344            mode is 'zip'. 
345  
346     memory_map : boolean, default False 
347         See parsers._parser_params for more information. 
348     is_text : boolean, default True 
349         whether file/buffer is in text format (csv, json, etc.), or in binary 
350         mode (pickle, etc.). 
351  
352     Returns 
353     ------- 
354     f : file-like 
355         A file-like object. 
356     handles : list of file-like objects 
357         A list of file-like object that were opened in this function. 
358     """ 
359     try: 
360         from s3fs import S3File 
361  
362         need_text_wrapping = (BufferedIOBase, RawIOBase, S3File) 
363     except ImportError: 
364         need_text_wrapping = (BufferedIOBase, RawIOBase)  # type: ignore 
365  
366     handles: List[IO] = list() 
367     f = path_or_buf 
368  
369     # Convert pathlib.Path/py.path.local or string 
370     path_or_buf = stringify_path(path_or_buf) 
371     is_path = isinstance(path_or_buf, str) 
372  
373     compression, compression_args = get_compression_method(compression) 
374     if is_path: 
375         compression = infer_compression(path_or_buf, compression) 
376  
377     if compression: 
378  
379         # GZ Compression 
380         if compression == "gzip": 
381             if is_path: 
382                 f = gzip.open(path_or_buf, mode) 
383             else: 
384                 f = gzip.GzipFile(fileobj=path_or_buf) 
385  
386         # BZ Compression 
387         elif compression == "bz2": 
388             if is_path: 
389                 f = bz2.BZ2File(path_or_buf, mode) 
390             else: 
391                 f = bz2.BZ2File(path_or_buf) 
392  
393         # ZIP Compression 
394         elif compression == "zip": 
395             zf = _BytesZipFile(path_or_buf, mode, **compression_args) 
396             # Ensure the container is closed as well. 
397             handles.append(zf) 
398             if zf.mode == "w": 
399                 f = zf 
400             elif zf.mode == "r": 
401                 zip_names = zf.namelist() 
402                 if len(zip_names) == 1: 
403                     f = zf.open(zip_names.pop()) 
404                 elif len(zip_names) == 0: 
405                     raise ValueError(f"Zero files found in ZIP file {path_or_buf}") 
406                 else: 
407                     raise ValueError( 
408                         "Multiple files found in ZIP file. " 
409                         f"Only one file per ZIP: {zip_names}" 
410                     ) 
411  
412         # XZ Compression 
413         elif compression == "xz": 
414             f = _get_lzma_file(lzma)(path_or_buf, mode) 
415  
416         # Unrecognized Compression 
417         else: 
418             msg = f"Unrecognized compression type: {compression}" 
419             raise ValueError(msg) 
420  
421         handles.append(f) 
422  
423     elif is_path: 
424         if encoding: 
425             # Encoding 
426             f = open(path_or_buf, mode, encoding=encoding, newline="") 
427         elif is_text: 
428             # No explicit encoding 
429             f = open(path_or_buf, mode, errors="replace", newline="") 
430         else: 
431             # Binary mode 
432             f = open(path_or_buf, mode) 
433         handles.append(f) 
434  
435     # Convert BytesIO or file objects passed with an encoding 
436     if is_text and (compression or isinstance(f, need_text_wrapping)): 
437         from io import TextIOWrapper 
438  
439         g = TextIOWrapper(f, encoding=encoding, newline="") 
440         if not isinstance(f, (BufferedIOBase, RawIOBase)): 
441             handles.append(g) 
442         f = g 
443  
444     if memory_map and hasattr(f, "fileno"): 
445         try: 
446             wrapped = _MMapWrapper(f) 
447             f.close() 
448             f = wrapped 
449         except Exception: 
450             # we catch any errors that may have occurred 
451             # because that is consistent with the lower-level 
452             # functionality of the C engine (pd.read_csv), so 
453             # leave the file handler as is then 
454             pass 
455  
456     return f, handles

459 class _BytesZipFile(zipfile.ZipFile, BytesIO):  # type: ignore 
460     """ 
461     Wrapper for standard library class ZipFile and allow the returned file-like 
462     handle to accept byte strings via `write` method. 
463  
464     BytesIO provides attributes of file-like object and ZipFile.writestr writes 
465     bytes strings into a member of the archive. 
466     """ 
467  
468     # GH 17778 
469     def __init__( 
470         self, 
471         file: FilePathOrBuffer, 
472         mode: str, 
473         archive_name: Optional[str] = None, 
474         **kwargs, 
475     ): 
476         if mode in ["wb", "rb"]: 
477             mode = mode.replace("b", "") 
478         self.archive_name = archive_name 
479         super().__init__(file, mode, zipfile.ZIP_DEFLATED, **kwargs) 
480  
481     def write(self, data): 
482         archive_name = self.filename 
483         if self.archive_name is not None: 
484             archive_name = self.archive_name 
485         super().writestr(archive_name, data) 
486  
487     @property 
488     def closed(self): 
489         return self.fp is None

492 class _MMapWrapper(abc.Iterator): 
493     """ 
494     Wrapper for the Python's mmap class so that it can be properly read in 
495     by Python's csv.reader class. 
496  
497     Parameters 
498     ---------- 
499     f : file object 
500         File object to be mapped onto memory. Must support the 'fileno' 
501         method or have an equivalent attribute 
502  
503     """ 
504  
505     def __init__(self, f: IO): 
506         self.mmap = mmap.mmap(f.fileno(), 0, access=mmap.ACCESS_READ) 
507  
508     def __getattr__(self, name: str): 
509         return getattr(self.mmap, name) 
510  
511     def __iter__(self) -> "_MMapWrapper": 
512         return self 
513  
514     def __next__(self) -> str: 
515         newbytes = self.mmap.readline() 
516  
517         # readline returns bytes, not str, but Python's CSV reader 
518         # expects str, so convert the output to str before continuing 
519         newline = newbytes.decode("utf-8") 
520  
521         # mmap doesn't raise if reading past the allocated 
522         # data but instead returns an empty string, so raise 
523         # if that is returned 
524         if newline == "": 
525             raise StopIteration 
526         return newline
