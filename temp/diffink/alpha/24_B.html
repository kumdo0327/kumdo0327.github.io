 3 Colormap Normalizations Symlognorm

 5  
 6 Demonstration of using norm to map colormaps onto data in non-linear ways. 
 7 """ 
 8  
 9 import numpy as np 
10 import matplotlib.pyplot as plt 
11 import matplotlib.colors as colors 
12  
13 """ 
14 SymLogNorm: two humps, one negative and one positive, The positive 
15 with 5-times the amplitude. Linearly, you cannot see detail in the 
16 negative hump.  Here we logarithmically scale the positive and 
17 negative data separately. 
18  
19 Note that colorbar labels do not come out looking very good. 
20 """ 
21  
22 N = 100 
23 X, Y = np.mgrid[-3:3:complex(0, N), -2:2:complex(0, N)] 
24 Z1 = np.exp(-X**2 - Y**2) 
25 Z2 = np.exp(-(X - 1)**2 - (Y - 1)**2) 
26 Z = (Z1 - Z2) * 2 
27  
28 fig, ax = plt.subplots(2, 1) 
29  
30 pcm = ax[0].pcolormesh(X, Y, Z, 
31                        norm=colors.SymLogNorm(linthresh=0.03, linscale=0.03, 
32                                               vmin=-1.0, vmax=1.0), 
33                        cmap='RdBu_r') 
34 fig.colorbar(pcm, ax=ax[0], extend='both') 
35  
36 pcm = ax[1].pcolormesh(X, Y, Z, cmap='RdBu_r', vmin=-np.max(Z)) 
37 fig.colorbar(pcm, ax=ax[1], extend='both') 
38  
39 plt.show()
